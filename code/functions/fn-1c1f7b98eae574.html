inline Mat3<Scalar> add3(
    const Mat3<Scalar>& A,
    const Mat3<Scalar>& B
) {
    auto C = zero3<Scalar>();
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) C[i][j] = A[i][j] + B[i][j];
    }
    return C;
}
