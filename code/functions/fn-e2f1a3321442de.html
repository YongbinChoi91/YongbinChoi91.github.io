static std::vector<int> candidate_owner_elements_by_node_ids(
    const FieldMesh& mesh,
    const BoundaryEntity& ent
){
    const auto& adj = boundary_owner_node_adjacency(mesh);
    std::unordered_map<int,int> hit_count;
    int valid_nodes=0;
    for(int node:ent.node_ids){
        if(node<0 || node>=static_cast<int>(adj.node_to_elements.size())) return {};
        ++valid_nodes;
        for(int e:adj.node_to_elements[node]) ++hit_count[e];
    }
    std::vector<int> out;
    out.reserve(hit_count.size());
    for(const auto& kv:hit_count){
        if(kv.second==valid_nodes) out.push_back(kv.first);
    }
    std::sort(out.begin(),out.end());
    return out;
}
