static void apply_torsion_dirichlet_bcs_to_domain(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    double time,
    BoundaryData& bc
){
    if(params.torsion_dirichlet_bcs.empty()) return;
    if(mesh.dim != 3){
        throw std::runtime_error("TORSION_DIRICHLET requires a 3D mesh.");
    }
 
    const DofLayout& gmesh = get_cached_dof_layout(mesh);
    for(const auto& rule: params.torsion_dirichlet_bcs){
        const int fid = find_field_index(gmesh, rule.field, false);
        if(fid < 0) continue;
        const FieldInfo& field = gmesh.fields[fid];
        if(field.components < 3){
            throw std::runtime_error("TORSION_DIRICHLET requires a vector field with at least 3 components: " + rule.field);
        }
        if(rule.axis < 0 || rule.axis > 2){
            throw std::runtime_error("TORSION_DIRICHLET axis must be x, y, or z.");
        }
 
        BCSelector selector = rule.selector;
        if(selector.kind == BCSelectorKind::BoundaryGroup && selector.group.empty()) selector.group = rule.group;
        const std::set<int> nodes = resolve_dirichlet_nodes_from_selector(field, selector, mesh.dim, rule.field);
        const std::array<double,3> center = rule.auto_center
            ? average_selected_field_node_coordinates(field, nodes, rule.field)
            : rule.center;
 
        for(int node: nodes){
            if(node<0 || node>=static_cast<int>(field.nodes.size())){
                throw std::runtime_error("Torsion Dirichlet boundary node id out of range for field '" + rule.field + "'.");
            }
            const auto& X = field.nodes[node].x;
            const double theta = eval_scalar_function(rule.angle, X[0], X[1], X[2], time);
            const std::array<double,3> u = torsion_rotation_displacement(X, center, rule.axis, theta);
            for(int c=0; c<3; ++c){
                add_dirichlet_bc(mesh, params, bc, global_field_dof(gmesh, fid, node, c), u[c],
                       "TORSION_DIRICHLET field=" + rule.field + " group=" + rule.group);
            }
        }
    }
}
