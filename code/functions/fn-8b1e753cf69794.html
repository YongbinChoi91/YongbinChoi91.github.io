inline Mat2T<Scalar> quadratic_projection_gradient_basis(
    const Coeff6x2T<Scalar>& a,
    const std::array<std::array<double,2>,6>& dp
) {
    auto grad = zero_grad2<Scalar>();
    for (int b = 0; b < 6; ++b) {
        for (int c = 0; c < 2; ++c) {
            grad[c][0] += Scalar(dp[b][0]) * a[b][c];
            grad[c][1] += Scalar(dp[b][1]) * a[b][c];
        }
    }
    return grad;
}
