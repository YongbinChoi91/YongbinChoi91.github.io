static void add_existing_csr(CSRMatrix& K,int row,int col,double value){
    const int idx=find_csr_index_optional(K,row,col);
    if(idx<0){
        std::ostringstream os;
        os << "Pressure Schur preconditioner entry missing in CSR sparsity: row=" << row << ", col=" << col;
        throw std::runtime_error(os.str());
    }
    K.values[idx] += value;
}
