static CSRMatrix build_pressure_schur_preconditioner_matrix(
    const CSRMatrix& Kff,
    const std::vector<int>& free_dofs,
    const FieldMesh& mesh,
    const ProblemParameters& params,
    const LinearSolverSettings& settings,
    double dt_for_schur
){
    const int nf=Kff.n_rows;
    if(Kff.n_cols!=nf || static_cast<int>(free_dofs.size())!=nf) throw std::runtime_error("Bad reduced matrix for pressure Schur preconditioner.");
 
    CSRMatrix P=Kff;
    const int n_full=total_dof_count(mesh);
    std::vector<int> full_to_red(n_full,-1);
    for(int i=0;i<nf;++i){
        const int full=free_dofs[i];
        if(full>=0 && full<n_full) full_to_red[full]=i;
    }
 
    auto is_pressure_full=[&](int full){ return is_pressure_full_dof_for_mesh(mesh,full); };
 
    // Replace only the pressure-pressure block of P.  The mechanical block and
    // coupling blocks stay equal to the Newton tangent, so FieldSplit still uses
    // the same A00 structure while A11 is our explicit Schur approximation.
    for(int i=0;i<nf;++i){
        if(!is_pressure_full(free_dofs[i])) continue;
        for(int q=P.row_ptr[i]; q<P.row_ptr[i+1]; ++q){
            const int j=P.col_idx[q];
            if(j>=0 && j<nf && is_pressure_full(free_dofs[j])) P.values[q]=0.0;
        }
    }
 
    const bool use_mass=pressure_schur_uses_mass(settings);
    const bool use_lap=pressure_schur_uses_laplacian(settings);
    const double mass_w = use_mass ? settings.schur_pressure_mass_weight : 0.0;
    double lap_w = 0.0;
    if(use_lap){
        lap_w = settings.schur_pressure_laplacian_weight;
        if(lap_w < 0.0){
            // For the dt-multiplied incompressible saddle-point system,
            // S = A11 - A10*A00^{-1}*A01 is negative definite in the
            // pressure block.  A negative pressure Laplacian/mass sign
            // matches PETSc Schur USER preconditioning much better.
            lap_w = -std::max(dt_for_schur,1.0e-30) / std::max(params.material.rho,1.0e-30);
        }
    }
 
    if(mesh.pressure_elements.empty()) throw std::runtime_error("Cannot build pressure Schur matrix: no pressure elements.");
 
    for(int e=0;e<static_cast<int>(mesh.pressure_elements.size());++e){
        const auto& pe=mesh.pressure_elements[e];
        const ElementGeometry geom=make_velocity_element_geometry(mesh,e);
        for(const auto& gp: gauss_rule(mesh.velocity_type, params.solver.gauss_order)){
            const ShapeData sv=compute_shape(mesh.velocity_type,gp.xi,gp.eta,gp.zeta);
            const JacobianData J=compute_jacobian_data(geom,sv);
            if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while assembling pressure Schur preconditioner.");
 
            const ShapeData sp=compute_shape(mesh.pressure_type,gp.xi,gp.eta,gp.zeta);
            const auto dNp=compute_dN_dx(sp,J);
            const double w=J.detJ*gp.weight;
            const int np_loc=std::min(static_cast<int>(pe.size()),static_cast<int>(sp.N.size()));
 
            for(int a=0;a<np_loc;++a){
                const int Ia_full=pressure_dof(mesh,pe[a]);
                const int Ia= (Ia_full>=0 && Ia_full<n_full) ? full_to_red[Ia_full] : -1;
                if(Ia<0) continue; // pressure pin / Dirichlet eliminated
                for(int b=0;b<np_loc;++b){
                    const int Ib_full=pressure_dof(mesh,pe[b]);
                    const int Ib= (Ib_full>=0 && Ib_full<n_full) ? full_to_red[Ib_full] : -1;
                    if(Ib<0) continue;
                    double val=0.0;
                    if(use_mass) val += mass_w * sp.N[a] * sp.N[b] * w;
                    if(use_lap){
                        double gd=0.0;
                        for(int d=0; d<mesh.dim; ++d) gd += dNp[a][d]*dNp[b][d];
                        val += lap_w * gd * w;
                    }
                    if(val!=0.0) add_existing_csr(P,Ia,Ib,val);
                }
            }
        }
    }
 
    if(settings.schur_pressure_shift>0.0){
        for(int i=0;i<nf;++i){
            if(is_pressure_full(free_dofs[i])) add_existing_csr(P,i,i,settings.schur_pressure_shift);
        }
    }
 
 
    return P;
}
