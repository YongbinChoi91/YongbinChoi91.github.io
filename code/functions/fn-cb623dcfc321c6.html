static double point_segment_distance_squared(
    const std::array<double,3>& X,
    const std::array<double,3>& A,
    const std::array<double,3>& B,
    int dim
){
    double ab2=0.0, ax=0.0;
    for(int d=0; d<dim; ++d){
        const double ab=B[d]-A[d];
        ab2 += ab*ab;
        ax += (X[d]-A[d])*ab;
    }
    if(ab2<=0.0){
        double d2=0.0;
        for(int d=0; d<dim; ++d){
            const double diff=X[d]-A[d];
            d2 += diff*diff;
        }
        return d2;
    }
    const double t=std::max(0.0,std::min(1.0,ax/ab2));
    double d2=0.0;
    for(int d=0; d<dim; ++d){
        const double P=A[d]+t*(B[d]-A[d]);
        const double diff=X[d]-P;
        d2 += diff*diff;
    }
    return d2;
}
