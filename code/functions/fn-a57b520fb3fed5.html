static bool save_data_requested(const ProblemParameters& params,const std::string& name){
    for(std::string item:params.solver.save_data){
        std::replace(item.begin(), item.end(), '-', '_');
        if(item==name) return true;
        if(name=="history" && (item=="h" || item=="hg" || item=="material_history")) return true;
    }
    return false;
}
