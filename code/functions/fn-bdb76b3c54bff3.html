Vector gather_local_vector(const Vector& global,const std::vector<int>& dofs){
    Vector local(dofs.size(),0.0);
    for(int i=0;i<static_cast<int>(dofs.size());++i){
        const int I=dofs[i];
        if(I<0 || I>=static_cast<int>(global.size())) throw std::runtime_error("Local extraction DOF out of range.");
        local[i]=global[I];
    }
    return local;
}
