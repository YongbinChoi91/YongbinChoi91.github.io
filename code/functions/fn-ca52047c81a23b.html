void validate_boundary_integral_or_throw(
    const FieldMesh& mesh,
    const std::string& boundary_name,
    double tolerance
){
    const BoundaryIntegralCheck chk=compute_boundary_integral_check(mesh,boundary_name);
    std::cout << "Boundary line-integral check for '" << boundary_name << "'"
              << ": entities=" << chk.boundary_entities
              << ", integrated=" << chk.integrated_entities
              << ", length=" << chk.measure
              << ", sum_normal=(" << chk.nx << ", " << chk.ny << ", " << chk.nz << ")"
              << ", norm=" << chk.norm
              << ", tol=" << tolerance << "\n";
    if(!(chk.norm <= tolerance)){
        std::ostringstream os;
        os << "Boundary line-integral check failed for '" << boundary_name
           << "': ||int n dGamma||=" << chk.norm << " > " << tolerance
           << ". Check owner elements, edge ordering, or whether the boundary is closed.";
        throw std::runtime_error(os.str());
    }
}
