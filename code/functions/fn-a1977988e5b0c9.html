RveMacroResponse make_rve_macro_response(
    const ConstitutiveMultiscaleResponse& response
) {
    if (response.volume <= 0.0) {
        throw std::runtime_error("RVE multiscale response volume is non-positive.");
    }
 
    RveMacroResponse out;
    out.volume = response.volume;
    for (int i = 0; i < 3; ++i) {
        for (int A = 0; A < 3; ++A) {
            out.intP[i][A] = response.intP[i][A];
            out.Pbar[i][A] = response.intP[i][A] / response.volume;
        }
    }
    return out;
}
