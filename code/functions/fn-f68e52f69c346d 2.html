inline ElementType history_element_type_for_mesh(const FieldMesh& mesh) {
    if (!mesh.displacement_elements.empty()) return mesh.displacement_type;
    if (!mesh.velocity_elements.empty()) return mesh.velocity_type;
    if (!mesh.pressure_elements.empty()) return mesh.pressure_type;
    return ElementType::Unknown;
}
