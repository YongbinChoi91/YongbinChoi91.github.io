RveSolveOptions make_multiscale_rve_solve_options() {
    RveSolveOptions opt = multiscale_rve_options();
    opt.verbose = false;
    opt.vtk_output = "none";
    opt.csv_output = "none";
    opt.initial_displacement = nullptr;
    opt.initial_lambda = nullptr;
    opt.initial_history = nullptr;
    opt.initial_Fmacro = nullptr;
    return opt;
}
