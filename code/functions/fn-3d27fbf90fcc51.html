bool self_check_neo_hookean_2d() {
    const std::array<std::array<double, 2>, 2> F{{{{1.1, 0.2}}, {{0.1, 0.95}}}};
    const double mu = 2.0;
    const double lambda = 5.0;
    const auto P = grad([&](const auto& F_ad) {
        return neo_hookean_energy_lame(F_ad, mu, lambda);
    }, F);
 
    const double J = F[0][0] * F[1][1] - F[0][1] * F[1][0];
    const std::array<std::array<double, 2>, 2> FinvT{{{{ F[1][1] / J, -F[1][0] / J }},
                                                     {{-F[0][1] / J,  F[0][0] / J }}}};
    bool ok = true;
    for (int i = 0; i < 2; ++i) {
        for (int j = 0; j < 2; ++j) {
            const double exact = mu * (F[i][j] - FinvT[i][j])
                               + 0.5 * lambda * (J * J - 1.0) * FinvT[i][j];
            ok = ok && close(P[i][j], exact, 1e-11);
        }
    }
    return ok;
}
