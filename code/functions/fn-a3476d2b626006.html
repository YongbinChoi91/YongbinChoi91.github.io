static int parse_vem_order_token(const std::string& token){
    const std::string t=lower_copy(token);
    if(t=="1" || t=="linear" || t=="lin" || t=="first" || t=="p1") return 1;
    if(t=="2" || t=="quadratic" || t=="quad" || t=="second" || t=="p2") return 2;
    throw std::runtime_error("Unknown VEM order in DOFS_info.txt: "+token);
}
