static int count_active_elements(
    const DofLayout& gmesh,
    const std::vector<std::string>& active_fields
){
    if(active_fields.empty()) throw std::runtime_error("No active constitutive fields were provided.");
 
    int nelem=-1;
    std::string owner;
    for(const std::string& name: active_fields){
        const int fid=find_field_index(gmesh,name,true);
        const FieldInfo& f=gmesh.fields[fid];
        if(f.elements.empty()){
            throw std::runtime_error("Active constitutive field has no elements: "+name);
        }
        const int n=static_cast<int>(f.elements.size());
        if(nelem<0){
            nelem=n;
            owner=name;
        } else if(nelem!=n){
            std::ostringstream os;
            os << "Active constitutive fields have inconsistent element counts: "
               << owner << " has " << nelem << ", " << name << " has " << n
               << ". Displacement-only, thermal-only, and mixed problems require all active fields "
               << "to provide one element row per assembled element.";
            throw std::runtime_error(os.str());
        }
    }
    return nelem;
}
