std::string make_solution_vtk_path(
    const ProblemParameters& params,
    const Problem& problem,
    const Domain& domain,
    int step
){
    std::ostringstream os;
    if(problem.domains.size()==1){
        os << path_join_simple(params.solver.vtk_dir, "solution_");
    } else {
        os << path_join_simple(params.solver.vtk_dir, "solution_" + domain.name + "_");
    }
    os << std::setw(4) << std::setfill('0') << step << ".vtk";
    return os.str();
}
