std::set<int> resolve_dirichlet_nodes_from_selector(
    const FieldInfo& field,
    const BCSelector& selector,
    int dim,
    const std::string& field_name
){
    std::set<int> nodes;
    if(selector.kind==BCSelectorKind::BoundaryGroup){
        auto it=field.boundaries.find(selector.group);
        if(it==field.boundaries.end()){
            throw std::runtime_error("Unknown Dirichlet boundary group for field '"+field_name+"': "+selector.group);
        }
        for(const auto& ent: it->second){
            for(int node: ent.node_ids) nodes.insert(node);
        }
        return nodes;
    }
 
    for(int node=0; node<static_cast<int>(field.nodes.size()); ++node){
        if(node_matches_selector(field.nodes[node],selector,dim)) nodes.insert(node);
    }
    if(nodes.empty()){
        throw std::runtime_error("No Dirichlet nodes matched coordinate selector for field '"+field_name+"': "+selector.group);
    }
    return nodes;
}
