static std::array<double,3> signed_surface_vec_area_from_J_3d(ElementType type,int face,const JacobianData& J){
    std::array<double,3> a{{0.0,0.0,0.0}};
    std::array<double,3> b{{0.0,0.0,0.0}};
    double sign=1.0;
 
    if(is_hex_element_type_post(type)){
        if(face==0 || face==1){
            a=jacobian_row3_post(J,0);
            b=jacobian_row3_post(J,1);
            sign=(face==0) ? -1.0 : 1.0;
        } else if(face==2 || face==4){
            a=jacobian_row3_post(J,0);
            b=jacobian_row3_post(J,2);
            sign=(face==2) ? 1.0 : -1.0;
        } else if(face==3 || face==5){
            a=jacobian_row3_post(J,1);
            b=jacobian_row3_post(J,2);
            sign=(face==3) ? 1.0 : -1.0;
        } else {
            throw std::runtime_error("Unsupported hex face id in boundary force surface vector.");
        }
    } else if(is_tet_element_type_post(type)){
        if(face==0){
            a=jacobian_row3_post(J,0);
            b=jacobian_row3_post(J,1);
            sign=-1.0;
        } else if(face==1){
            a=jacobian_row3_post(J,0);
            b=jacobian_row3_post(J,2);
            sign=1.0;
        } else if(face==2){
            a=jacobian_row3_post(J,1);
            b=jacobian_row3_post(J,2);
            sign=-1.0;
        } else if(face==3){
            const auto j0=jacobian_row3_post(J,0);
            const auto j1=jacobian_row3_post(J,1);
            const auto j2=jacobian_row3_post(J,2);
            for(int d=0; d<3; ++d){
                a[d]=j1[d]-j0[d];
                b[d]=j2[d]-j0[d];
            }
            sign=1.0;
        } else {
            throw std::runtime_error("Unsupported tet face id in boundary force surface vector.");
        }
    } else {
        throw std::runtime_error("3D boundary force supports only Tet/Hex elements.");
    }
 
    const auto v=scale3_post(cross3_post(a,b),sign);
    if(norm3_post(v)<=0.0) throw std::runtime_error("Degenerate 3D boundary face: zero surface measure.");
    return v;
}
