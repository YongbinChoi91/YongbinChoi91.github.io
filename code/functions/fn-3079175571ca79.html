int choose_pressure_pin_node(const FieldMesh& mesh){
    if(mesh.pressure_nodes.empty()) throw std::runtime_error("No pressure nodes for pressure pin.");
    int best=0; double bx=mesh.pressure_nodes[0].x[0], by=mesh.pressure_nodes[0].x[1];
    const double eps=1.0e-12;
    for(int i=1;i<static_cast<int>(mesh.pressure_nodes.size());++i){
        const double x=mesh.pressure_nodes[i].x[0], y=mesh.pressure_nodes[i].x[1];
        if(x>bx+eps || (std::abs(x-bx)<=eps && y<by)){ best=i; bx=x; by=y; }
    }
    return best;
}
