bool need_deferred_snapshot(const ProblemParameters& params,int accepted_steps,double time){
    if(!params.solver.postprocess_deferred) return false;
    const bool final = time >= params.solver.total_time - 1.0e-14;
    const bool vtk_step = params.solver.output_every>0 && (accepted_steps%params.solver.output_every==0 || final);
    // Force and minJ CSVs are still every accepted step, so keep all snapshots
    // when either one is enabled. VTK-only runs keep only output_every snapshots.
    return vtk_step || params.solver.force_output || params.solver.minJ_output;
}
