inline std::vector<std::vector<Scalar>> field_grad_as(
    const LocalFieldView& L,
    const std::string& field_name,
    const Vec& x_elem,
    const std::vector<std::array<double,3>>& dNdx
) {
    const auto& Fld = L.field(field_name);
    if (static_cast<int>(dNdx.size()) < Fld.n_nodes) {
        throw std::runtime_error("Gradient shape-function size is smaller than field node count for " + field_name + ".");
    }
 
    std::vector<std::vector<Scalar>> grad(
        Fld.components,
        std::vector<Scalar>(L.dim, Scalar(0.0))
    );
 
    for (int a = 0; a < Fld.n_nodes; ++a) {
        const int base = Fld.local_offset + Fld.components * a;
        for (int i = 0; i < Fld.components; ++i) {
            const Scalar qa = Scalar(x_elem[base + i]);
            for (int A = 0; A < L.dim; ++A) {
                grad[i][A] += qa * Scalar(dNdx[a][A]);
            }
        }
    }
    return grad;
}
