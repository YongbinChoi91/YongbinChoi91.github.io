void reset_hdf5_solid_gauss_point_snapshots(
    const ProblemParameters& params,
    const Problem& problem,
    const MaterialHistoryStore* history
){
    if(!params.solver.hdf5_output) return;
    if(!any_solid_gauss_save_data_requested(params)) return;
    const auto constitutive_specs = normalized_constitutive_gauss_output_specs();
    validate_solid_gauss_save_data(params,constitutive_specs);
#ifdef HIPECS_USE_HDF5
    hid_t file=h5_open_file_rw(params);
    h5_ensure_group(file,"/gauss");
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain = problem.domains[d];
        const auto field_names = active_constitutive_field_names(domain.mesh);
        const FieldInfo* primary_field = primary_postprocess_field(domain.mesh,field_names);
        if(!primary_field) continue;
        const hsize_t nelem = static_cast<hsize_t>(primary_field->elements.size());
        const std::string base = "/gauss/domain_" + std::to_string(d);
        h5_ensure_group(file,base);
#ifdef HIPECS_ENABLE_VEM
        const VemGaussLayoutInfo vem_info = make_vem_gauss_layout_info(domain.mesh,*primary_field,params);
        if(vem_info.enabled){
            if(save_data_requested(params,"history")){
                throw std::runtime_error("VEM Gauss HDF5 output does not support save_data history.");
            }
            write_vem_gauss_layout_metadata(file,base,vem_info,domain.mesh.vem_order);
            for(const std::string& item:params.solver.save_data){
                const std::string name = normalize_save_data_name(item);
                if(name == "history") continue;
                int components = 0;
                if(name == "f" || name == "c") components = 9;
                else if(const ConstitutivePostprocessOutputSpec* spec = find_constitutive_gauss_spec(constitutive_specs,name)){
                    components = spec->components;
                }
                if(components > 0){
                    h5_create_resizable_double_nd(
                        file,
                        base + "/" + hdf5_gauss_dataset_name(name),
                        {nelem,vem_info.max_tri,vem_info.npg,static_cast<hsize_t>(components)}
                    );
                }
            }
            hid_t group = H5Gopen2(file,base.c_str(),H5P_DEFAULT);
            if(group>=0){
                h5_write_string_attr(group,"domain_name",domain.name);
                h5_write_string_attr(group,"element_field",primary_field->name);
                H5Gclose(group);
            }
            continue;
        }
#endif
        const auto gps = gauss_rule(primary_field->element_type, params.solver.gauss_order);
        const hsize_t ngp = static_cast<hsize_t>(gps.size());
        if(domain.type == DomainType::Solid && has_displacement_field(domain.mesh)){
            if(save_data_requested(params,"f")) h5_create_resizable_double_nd(file,base + "/F",{nelem,ngp,3,3});
            if(save_data_requested(params,"c")) h5_create_resizable_double_nd(file,base + "/C",{nelem,ngp,3,3});
        }
        hid_t group = H5Gopen2(file,base.c_str(),H5P_DEFAULT);
        if(group>=0){
            h5_write_string_attr(group,"domain_name",domain.name);
            h5_write_string_attr(group,"element_field",primary_field->name);
            H5Gclose(group);
        }
        if(domain.type == DomainType::Solid && save_data_requested(params,"history") &&
           history && d<static_cast<int>(history->domains.size()) && !history->domains[d].elements.empty()){
            int nvars = primary_history_nvars(history->domains[d].specs);
            if(nvars>0) h5_create_resizable_double_nd(file,base + "/history",{nelem,ngp,static_cast<hsize_t>(nvars)});
        }
        for(const std::string& item:params.solver.save_data){
            const std::string name = normalize_save_data_name(item);
            if(is_generic_solid_gauss_output(name)) continue;
            const ConstitutivePostprocessOutputSpec* spec = find_constitutive_gauss_spec(constitutive_specs,name);
            if(spec && spec->components>0){
                h5_create_resizable_double_nd(file,base + "/" + name,{nelem,ngp,static_cast<hsize_t>(spec->components)});
            }
        }
    }
    H5Fclose(file);
#else
    (void)problem; (void)history;
#endif
}
