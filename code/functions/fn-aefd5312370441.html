inline int history_element_count_for_mesh(const FieldMesh& mesh) {
    if (!mesh.displacement_elements.empty()) return static_cast<int>(mesh.displacement_elements.size());
    if (!mesh.velocity_elements.empty()) return static_cast<int>(mesh.velocity_elements.size());
    if (!mesh.pressure_elements.empty()) return static_cast<int>(mesh.pressure_elements.size());
    return 0;
}
