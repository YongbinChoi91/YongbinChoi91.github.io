inline Mat3<Scalar> dev3(const Mat3<Scalar>& A) {
    auto D = A;
    const Scalar mean = tr3<Scalar>(A) / Scalar(3.0);
    for (int i = 0; i < 3; ++i) D[i][i] -= mean;
    return D;
}
