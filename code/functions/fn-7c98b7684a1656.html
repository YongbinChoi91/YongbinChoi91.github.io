static std::vector<ConstitutivePostprocessOutputSpec> normalized_constitutive_gauss_output_specs(){
    std::vector<ConstitutivePostprocessOutputSpec> specs = active_constitutive_postprocess_outputs();
    specs.erase(
        std::remove_if(specs.begin(), specs.end(), [](const ConstitutivePostprocessOutputSpec& spec){
            return spec.location != ConstitutivePostprocessLocation::GaussPoint;
        }),
        specs.end()
    );
    for(auto& spec:specs) spec.name = normalize_save_data_name(spec.name);
    return specs;
}
