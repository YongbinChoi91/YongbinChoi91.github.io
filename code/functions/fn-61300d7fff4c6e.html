BoundaryIntegralCheck compute_boundary_integral_check(
    const FieldMesh& mesh,
    const std::string& boundary_name
){
    auto it=mesh.velocity_boundaries.find(boundary_name);
    if(it==mesh.velocity_boundaries.end()){
        throw std::runtime_error("Unknown velocity boundary group for force/check: "+boundary_name);
    }
 
    BoundaryIntegralCheck out;
    out.boundary_name=boundary_name;
    out.boundary_entities=static_cast<int>(it->second.size());
 
    if(mesh.dim==2){
        out.integrated_entities = integrate_boundary_edges_2d(
            mesh,
            boundary_name,
            [&](const BoundaryQuadraturePoint2D& qp){
                const double w=qp.weight;
                out.nx += qp.normal_measure[0]*w;
                out.ny += qp.normal_measure[1]*w;
                out.nz += qp.normal_measure[2]*w;
                out.measure += qp.measure*w;
            }
        );
    } else if(mesh.dim==3){
        out.integrated_entities = integrate_boundary_faces_3d(
            mesh,
            boundary_name,
            [&](const BoundaryQuadraturePoint3D& qp){
                const double w=qp.weight;
                out.nx += qp.normal_measure[0]*w;
                out.ny += qp.normal_measure[1]*w;
                out.nz += qp.normal_measure[2]*w;
                out.measure += qp.measure*w;
            }
        );
    } else {
        throw std::runtime_error("Boundary integration supports only 2D and 3D meshes.");
    }
 
    out.norm=std::sqrt(out.nx*out.nx+out.ny*out.ny+out.nz*out.nz);
    return out;
}
