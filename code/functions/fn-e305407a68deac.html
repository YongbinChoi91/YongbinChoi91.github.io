static void destroy_petsc_outer_cache(PetscOuterSolveCache& cache){
    if(cache.is_mech) ISDestroy(&cache.is_mech);
    if(cache.is_p) ISDestroy(&cache.is_p);
    if(cache.x) VecDestroy(&cache.x);
    if(cache.b) VecDestroy(&cache.b);
    if(cache.P && cache.P!=cache.K) MatDestroy(&cache.P);
    if(cache.K) MatDestroy(&cache.K);
    cache=PetscOuterSolveCache();
}
