std::array<double,2> fixed_axis_vector_for_edge_2d(int edge){
    if(edge==0) return { 0.0,-1.0};
    if(edge==1) return { 1.0, 0.0};
    if(edge==2) return { 0.0, 1.0};
    if(edge==3) return {-1.0, 0.0};
    throw std::runtime_error("Unsupported 2D edge for fixed-axis vector.");
}
