void append_hdf5_table_row(const ProblemParameters& params,const std::string& dataset_path,const std::vector<double>& row){
    if(!params.solver.hdf5_output) return;
#ifdef HIPECS_USE_HDF5
    hid_t file=h5_open_file_rw(params);
    hid_t dset=H5Dopen2(file,dataset_path.c_str(),H5P_DEFAULT);
    if(dset<0) throw std::runtime_error("Could not open HDF5 dataset: " + dataset_path);
    hid_t space=H5Dget_space(dset);
    hsize_t dims[2]={0,0};
    H5Sget_simple_extent_dims(space,dims,nullptr);
    H5Sclose(space);
    if(dims[1] != static_cast<hsize_t>(row.size())) throw std::runtime_error("HDF5 row width mismatch for " + dataset_path);
    hsize_t newdims[2]={dims[0]+1,dims[1]};
    h5_check(H5Dset_extent(dset,newdims),"extending dataset " + dataset_path);
    space=H5Dget_space(dset);
    hsize_t start[2]={dims[0],0};
    hsize_t count[2]={1,dims[1]};
    h5_check(H5Sselect_hyperslab(space,H5S_SELECT_SET,start,nullptr,count,nullptr),"selecting append row");
    hid_t mem=H5Screate_simple(2,count,nullptr);
    h5_check(H5Dwrite(dset,H5T_NATIVE_DOUBLE,mem,space,H5P_DEFAULT,row.data()),"appending row to " + dataset_path);
    H5Sclose(mem); H5Sclose(space); H5Dclose(dset); H5Fclose(file);
#else
    (void)dataset_path; (void)row;
#endif
}
