inline ElementSystem scalar_weak_form_element_system_impl(
    int n,
    const Vector& x_elem,
    Eval&& eval
) {
    std::vector<OuterFad> x_ad(n);
    std::vector<OuterFad> test_ad(n);
 
    for (int j = 0; j < n; ++j) {
        const InnerFad x_value(x_elem[j]);
        x_ad[j] = OuterFad(n, j, x_value);
    }
    for (int i = 0; i < n; ++i) {
        test_ad[i] = OuterFad(InnerFad(n, i, 0.0));
    }
 
    const OuterFad G = std::forward<Eval>(eval)(x_ad, test_ad);
 
    ElementSystem sys;
    sys.residual.assign(n, 0.0);
    sys.tangent.assign(n * n, 0.0);
 
    const InnerFad G_value = G.val();
    for (int i = 0; i < n; ++i) {
        sys.residual[i] = weak_inner_dx(G_value, i);
        const int row = i * n;
        for (int j = 0; j < n; ++j) {
            const InnerFad dG_dxj = (j >= 0 && j < G.size()) ? G.dx(j) : InnerFad(0.0);
            sys.tangent[row + j] = weak_inner_dx(dG_dxj, i);
        }
    }
 
    return sys;
}
