HysteresisMonitor make_hysteresis_monitor(const Problem& problem,const ProblemParameters& params){
    HysteresisMonitor h;
    if(!params.solver.hysteresis_output) return h;
    h.enabled=true;
    auto xs = params.solver.hysteresis_x_axis.enabled ? params.solver.hysteresis_x_axis : default_hysteresis_x_axis(params);
    auto ys = params.solver.hysteresis_y_axis.enabled ? params.solver.hysteresis_y_axis : default_hysteresis_y_axis(params);
    h.x_axis = make_hysteresis_axis_runtime_multi(problem,params,xs);
    h.y_axis = make_hysteresis_axis_runtime_multi(problem,params,ys);
    h.label = h.y_axis.label + " vs " + h.x_axis.label;
    return h;
}
