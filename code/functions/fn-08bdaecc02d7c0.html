LocalFieldView make_local_field_view(
    const FieldMesh& mesh,
    int elem_id
) {
    std::vector<std::string> order;
    order.push_back("velocity");
    if (has_displacement_field(mesh)) order.push_back("displacement");
    order.push_back("pressure");
    return make_local_field_view(mesh, elem_id, order);
}
