static HysteresisAxisRuntime make_hysteresis_axis_runtime_multi(
    const Problem& problem,
    const ProblemParameters& params,
    TimeSolverSettings::HysteresisAxisSpec spec
){
    HysteresisAxisRuntime axis;
    axis.spec=spec;
    if(axis.spec.label.empty()){
        if(axis.spec.mode=="time") axis.spec.label="time";
        else if(axis.spec.mode=="function") axis.spec.label=axis.spec.function.expression;
        else axis.spec.label=axis.spec.mode + " " + hysteresis_label(axis.spec.target);
    }
    axis.label=axis.spec.label;
 
    if(axis.spec.mode=="prescribed"){
        axis.prescribed_function=find_hysteresis_x_function(params,axis.spec.target);
    } else if(hysteresis_axis_needs_dofs(axis.spec)){
        std::set<int> dofs;
        if(axis.spec.mode=="value" && axis.spec.node>=0){
            if(problem.domains.empty()) throw std::runtime_error("No domain for hysteresis point value axis.");
            if(problem.shared_dofs){
                collect_hysteresis_node_dof_from_mesh(problem.domains.front().mesh,axis.spec,0,dofs);
            } else {
                bool found=false;
                for(const auto& domain: problem.domains){
                    if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
                    try{
                        collect_hysteresis_node_dof_from_mesh(domain.mesh,axis.spec,domain.dof_offset,dofs);
                        found=true;
                        break;
                    } catch(const std::exception&) {}
                }
                if(!found) throw std::runtime_error("Could not resolve hysteresis point value axis: "+axis.label);
            }
        } else if(axis.spec.mode=="value" && axis.spec.has_point){
            if(problem.domains.empty()) throw std::runtime_error("No domain for hysteresis coordinate point value axis.");
            if(problem.shared_dofs){
                collect_hysteresis_point_dof_from_mesh(problem.domains.front().mesh,axis.spec,0,dofs);
            } else {
                for(const auto& domain: problem.domains){
                    if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
                    try{
                        collect_hysteresis_point_dof_from_mesh(domain.mesh,axis.spec,domain.dof_offset,dofs);
                        if(!dofs.empty()) break;
                    } catch(const std::exception&) {}
                }
            }
        } else if(problem.shared_dofs){
            bool found=false;
            for(const auto& domain: problem.domains){
                if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
                std::string group;
                const bool qualified=strip_hysteresis_domain_group_prefix(axis.spec.target.group,domain.name,group);
                if(!qualified) group=axis.spec.target.group;
                const DofLayout& gmesh=get_cached_dof_layout(domain.mesh);
                const int fid=find_field_index(gmesh,axis.spec.target.field,false);
                if(fid<0) continue;
                if(qualified || gmesh.fields[fid].boundaries.count(group)>0){
                    FieldComponentTarget local=axis.spec.target; local.group=group;
                    const std::size_t before=dofs.size();
                    collect_hysteresis_dofs_from_mesh(domain.mesh,local,0,dofs);
                    if(dofs.size()!=before || gmesh.fields[fid].boundaries.count(group)>0) found=true;
                }
            }
            if(!found) throw std::runtime_error("Unknown hysteresis axis boundary group in shared mesh: "+axis.spec.target.group);
        } else {
            const bool multiple=problem.domains.size()>1;
            bool found=false;
            for(const auto& domain: problem.domains){
                if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
                std::string group;
                const bool qualified=strip_hysteresis_domain_group_prefix(axis.spec.target.group,domain.name,group);
                if(!qualified) group=axis.spec.target.group;
                const DofLayout& gmesh=get_cached_dof_layout(domain.mesh);
                const int fid=find_field_index(gmesh,axis.spec.target.field,false);
                const bool has=(fid>=0 && gmesh.fields[fid].boundaries.count(group)>0);
                if((qualified || !multiple || has) && has){
                    FieldComponentTarget local=axis.spec.target; local.group=group;
                    collect_hysteresis_dofs_from_mesh(domain.mesh,local,domain.dof_offset,dofs);
                    found=true;
                }
            }
            if(!found) throw std::runtime_error("Unknown hysteresis axis boundary group in multidomain mesh: "+axis.spec.target.group);
        }
        axis.dofs.assign(dofs.begin(),dofs.end());
        if(axis.dofs.empty() && !axis.spec.has_point) throw std::runtime_error("hysteresis axis produced no DOFs: "+axis.label);
    }
    return axis;
}
