static ElementGeometry make_owner_geometry_element(const FieldMesh& mesh,const ResolvedBoundaryEdge& r){
    ElementGeometry elem;
    elem.dim = mesh.dim;
    const auto& conn = owner_connectivity(mesh,r);
    elem.type = element_type_from_connectivity(mesh,conn);
    elem.coords.resize(conn.size());
    for(int a=0;a<static_cast<int>(conn.size());++a){
        const int node=conn[a];
        if(node<0 || node>=static_cast<int>(mesh.velocity_nodes.size()))
            throw std::runtime_error("Surface owner element node id is out of velocity-node range.");
        elem.coords[a] = mesh.velocity_nodes[node].x;
    }
    return elem;
}
