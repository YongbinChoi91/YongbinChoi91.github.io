static std::vector<HistorySpec> history_specs_for_domain(const DofsInfo& info,const Domain& domain){
    const std::string name_key=normalize_domain_key(domain.name);
    auto it=info.domain_histories.find(name_key);
    if(it!=info.domain_histories.end()) return it->second;
 
    const std::string type_key=normalize_domain_key(domain_type_to_string(domain.type));
    it=info.domain_histories.find(type_key);
    if(it!=info.domain_histories.end()) return it->second;
 
    return info.histories;
}
