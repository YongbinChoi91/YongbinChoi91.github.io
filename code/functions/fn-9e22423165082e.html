static bool read_on_off_bool(std::stringstream& ss, const std::string& key){
    std::string v;
    ss >> v;
    if(v.empty()) throw std::runtime_error("Missing on/off value for key: " + key);
    v = lower_copy(v);
    if(v=="on" || v=="true" || v=="1" || v=="yes") return true;
    if(v=="off" || v=="false" || v=="0" || v=="no") return false;
    throw std::runtime_error("Expected on/off, true/false, or 1/0 for key: " + key);
}
