std::vector<RveBatchResult> solve_rve_pbc_augmented_batch(
    const Problem& problem,
    const ProblemParameters& params,
    const RveSolveOptions& base_options,
    const std::vector<RveBatchTask>& tasks,
    int requested_threads
) {
    std::vector<RveBatchResult> results(tasks.size());
 
    auto solve_one = [&](int i) {
        try {
            RveSolveOptions opt = base_options;
            opt.Fmacro = tasks[i].Fmacro;
            opt.dt = tasks[i].dt;
            opt.dt_previous = tasks[i].dt_previous;
            opt.step = tasks[i].step;
            opt.time = tasks[i].time;
            opt.verbose = false;
            opt.vtk_output = "none";
            opt.csv_output = "none";
            if (tasks[i].has_initial_displacement) opt.initial_displacement = &tasks[i].initial_displacement;
            else opt.initial_displacement = nullptr;
            if (tasks[i].has_initial_lambda) opt.initial_lambda = &tasks[i].initial_lambda;
            else opt.initial_lambda = nullptr;
            if (tasks[i].has_initial_history) opt.initial_history = &tasks[i].initial_history;
            else opt.initial_history = nullptr;
            if (tasks[i].has_initial_Fmacro) opt.initial_Fmacro = &tasks[i].initial_Fmacro;
            else opt.initial_Fmacro = nullptr;
            results[i].rve = solve_rve_pbc_augmented(problem, params, opt);
        } catch (const std::exception& e) {
            results[i].error = e.what();
        } catch (...) {
            results[i].error = "Unknown exception during batched RVE solve.";
        }
    };
 
#ifdef USE_OPENMP
    if (requested_threads > 0) {
        #pragma omp parallel for schedule(dynamic) num_threads(requested_threads)
        for (int i = 0; i < static_cast<int>(tasks.size()); ++i) solve_one(i);
    } else {
        #pragma omp parallel for schedule(dynamic)
        for (int i = 0; i < static_cast<int>(tasks.size()); ++i) solve_one(i);
    }
#else
    (void)requested_threads;
    for (int i = 0; i < static_cast<int>(tasks.size()); ++i) solve_one(i);
#endif
 
    return results;
}
