inline Scalar von_mises(
    const std::vector<std::vector<Scalar>>& sigma
) {
    if (static_cast<int>(sigma.size()) != 3) {
        throw std::runtime_error("von Mises helper expects a 3x3 Cauchy stress.");
    }
    for (const auto& row : sigma) {
        if (static_cast<int>(row.size()) != 3) {
            throw std::runtime_error("von Mises helper expects a 3x3 Cauchy stress.");
        }
    }
    const Scalar sxx = sigma[0][0];
    const Scalar syy = sigma[1][1];
    const Scalar szz = sigma[2][2];
    const Scalar sxy = Scalar(0.5) * (sigma[0][1] + sigma[1][0]);
    const Scalar syz = Scalar(0.5) * (sigma[1][2] + sigma[2][1]);
    const Scalar szx = Scalar(0.5) * (sigma[2][0] + sigma[0][2]);
    return std::sqrt(
        Scalar(0.5) * ((sxx - syy) * (sxx - syy) +
                       (syy - szz) * (syy - szz) +
                       (szz - sxx) * (szz - sxx)) +
        Scalar(3.0) * (sxy * sxy + syz * syz + szx * szx)
    );
}
