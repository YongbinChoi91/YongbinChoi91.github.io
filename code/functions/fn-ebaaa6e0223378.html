std::array<double,3> surface_area_vector_from_jacobian_2d(
    const JacobianData& J,
    const std::array<double,2>& reference_normal
){
    const std::array<double,2> a{{J.J[0][0], J.J[0][1]}};
    const std::array<double,2> b{{J.J[1][0], J.J[1][1]}};
    const std::array<double,2> cof0{{ b[1], -b[0] }};
    const std::array<double,2> cof1{{ a[1], -a[0] }};
    return {
        reference_normal[0]*cof0[0] - reference_normal[1]*cof1[0],
        reference_normal[0]*cof0[1] - reference_normal[1]*cof1[1],
        0.0
    };
}
