inline Vector solve_local_dense_lu(const LocalDenseLU& lu, Vector b) {
    const int n = static_cast<int>(b.size());
    if (static_cast<int>(lu.lu.size()) != n || static_cast<int>(lu.pivots.size()) != n) {
        throw std::runtime_error("Local dense LU solve received incompatible factor/vector size.");
    }
 
    for (int k = 0; k < n; ++k) {
        const int piv = lu.pivots[k];
        if (piv != k) std::swap(b[piv], b[k]);
    }
 
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < i; ++j) b[i] -= lu.lu[i][j] * b[j];
    }
 
    for (int i = n - 1; i >= 0; --i) {
        for (int j = i + 1; j < n; ++j) b[i] -= lu.lu[i][j] * b[j];
        b[i] /= lu.lu[i][i];
    }
 
    return b;
}
