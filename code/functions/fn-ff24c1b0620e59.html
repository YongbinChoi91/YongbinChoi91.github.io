    double parse_primary(){
        skip();
        if(eat('(')){ double v=parse_expression(); if(!eat(')')) throw std::runtime_error("Missing ')'"); return v; }
        if(p_<e_.size() && (std::isdigit(static_cast<unsigned char>(e_[p_])) || e_[p_]=='.')) return number();
        if(p_<e_.size() && (std::isalpha(static_cast<unsigned char>(e_[p_])) || e_[p_]=='_')){
            const std::string k=id();
            if(k=="x") return x_;
            if(k=="y") return y_;
            if(k=="z") return z_;
            if(k=="t") return t_;
            if(k=="parabolic_profile") return parabolic_profile_;
            if(k=="u_bar") return u_bar_;
            if(k=="pi") return std::acos(-1.0);
            if(!eat('(')) throw std::runtime_error("Unknown variable or function: "+k);
            std::vector<double> args; args.push_back(parse_expression());
            while(eat(',')) args.push_back(parse_expression());
            if(!eat(')')) throw std::runtime_error("Missing ')' after function.");
            auto req=[&](int n){ if(static_cast<int>(args.size())!=n) throw std::runtime_error("Wrong number of args for "+k); };
            if(k=="sin"){req(1);return std::sin(args[0]);}
            if(k=="cos"){req(1);return std::cos(args[0]);}
            if(k=="tan"){req(1);return std::tan(args[0]);}
            if(k=="exp"){req(1);return std::exp(args[0]);}
            if(k=="sqrt"){req(1);return std::sqrt(args[0]);}
            if(k=="abs"){req(1);return std::abs(args[0]);}
            if(k=="log"){req(1);return std::log(args[0]);}
            if(k=="min"){req(2);return std::min(args[0],args[1]);}
            if(k=="max"){req(2);return std::max(args[0],args[1]);}
            throw std::runtime_error("Unknown function: "+k);
        }
        throw std::runtime_error("Unexpected expression token.");
    }
