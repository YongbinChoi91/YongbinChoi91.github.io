void assemble_global_jacobian_diagonal(
    ConstitutiveElementKernel kernel,
    const Problem& problem,
    const ProblemParameters& params,
    const Vector& current_dofs,
    const PastDofs& past_dofs,
    double dt,
    double dt_previous,
    int step,
    double time,
    Vector& diag,
    const ProblemElementDofMap& dof_maps,
    const MaterialHistoryStore* history
){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in matrix-free diagonal assembly.");
    const int ndofs=problem.total_dofs;
    validate_global_dofs_size(current_dofs,past_dofs,ndofs,"matrix-free diagonal assembly");
 
    MaterialHistoryStore history_copy;
    MaterialHistoryStore* mutable_history=nullptr;
    if(history){ history_copy=*history; mutable_history=&history_copy; }
 
    AssemblyMpiInfo mpi = get_assembly_mpi_info();
    if(params.solver.multiscale_rve_assembly_context){ mpi.active=false; mpi.rank=0; mpi.size=1; }
    diag.assign(ndofs,0.0);
 
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain=problem.domains[d];
        if(!is_supported_continuum_domain(domain.type)) throw std::runtime_error("Unsupported domain type in matrix-free diagonal.");
        const FieldMesh& mesh=domain.mesh;
        const int nelem=static_cast<int>(dof_maps[d].size());
        std::vector<Vector> local_diag(nelem);
        std::vector<std::string> element_errors(nelem);
 
        auto compute_local_diagonal = [&](int e){
            if(!assembly_rank_owns_element(e,mpi)) return;
            try{
                const auto& global_dofs=dof_maps[d][e];
                Vector xe;
                PastDofs local_past_dofs;
                gather_local_vector_into(current_dofs,global_dofs,xe);
                gather_local_past_dofs_into(past_dofs,global_dofs,local_past_dofs);
                ElementState st;
                st.past_dofs=std::move(local_past_dofs);
                st.dt=dt;
                st.dt_previous=dt_previous;
                st.step=step;
                st.domain_name=domain.name;
                st.domain_type=domain.type;
                st.domain_id=d;
                if(mutable_history && d<static_cast<int>(mutable_history->domains.size()) && e<static_cast<int>(mutable_history->domains[d].elements.size())){
                    st.material_history=&mutable_history->domains[d].elements[e];
                }
#ifdef HIPECS_ENABLE_MATFREE
                if(lower_copy(params.solver.linear.petsc_matfree_jvp_mode)=="exact_ad") {
                    // Extract one diagonal entry at a time. Only vectors are
                    // retained; no dense element tangent is formed or cached.
                    const int n=static_cast<int>(xe.size());
                    local_diag[e].assign(n,0.0);
                    Vector direction(n,0.0);
                    for(int a=0; a<n; ++a) {
                        direction[a]=1.0;
                        const Vector action=compute_constitutive_matfree_jvp(mesh,e,xe,direction,params,st);
                        if(static_cast<int>(action.size())!=n) throw std::runtime_error("Bad matrix-free diagonal action size.");
                        local_diag[e][a]=action[a];
                        direction[a]=0.0;
                    }
                    return;
                }
#endif
                ElementSystem es=kernel(mesh,e,xe,params,st);
                const int n=static_cast<int>(global_dofs.size());
                if(static_cast<int>(es.tangent.size())!=n*n) throw std::runtime_error("Bad element tangent size in matrix-free diagonal.");
                local_diag[e].assign(n,0.0);
                for(int a=0;a<n;++a) local_diag[e][a]=es.tangent[a*n+a];
            } catch(const std::exception& ex){ element_errors[e]=ex.what(); }
              catch(...){ element_errors[e]="Unknown exception during matrix-free diagonal assembly."; }
        };
 
        const bool parallel_element_loop = use_parallel_element_loop(params.solver);
        const int requested_threads = element_loop_thread_count(params.solver);
#ifdef USE_OPENMP
        if(parallel_element_loop && requested_threads>0){
            #pragma omp parallel for schedule(dynamic) num_threads(requested_threads)
            for(int e=0;e<nelem;++e) compute_local_diagonal(e);
        } else if(parallel_element_loop){
            #pragma omp parallel for schedule(dynamic)
            for(int e=0;e<nelem;++e) compute_local_diagonal(e);
        } else
#endif
        { for(int e=0;e<nelem;++e) compute_local_diagonal(e); }
 
        int local_error=0;
        std::string first_local_error;
        for(int e=0;e<nelem;++e){
            if(assembly_rank_owns_element(e,mpi) && !element_errors[e].empty()){
                local_error=1;
                if(first_local_error.empty()){
                    std::ostringstream os;
                    os << "Matrix-free diagonal failed in domain " << d << ", element " << e << ": " << element_errors[e];
                    first_local_error=os.str();
                }
            }
        }
        if(any_rank_reported_assembly_error(local_error,mpi)){
            throw std::runtime_error(first_local_error.empty() ? std::string("Matrix-free diagonal failed on another MPI rank.") : first_local_error);
        }
 
        for(int e=0;e<nelem;++e){
            if(!assembly_rank_owns_element(e,mpi)) continue;
            const auto& dofs=dof_maps[d][e];
            const Vector& de=local_diag[e];
            if(de.size()!=dofs.size()) throw std::runtime_error("Bad local diagonal size.");
            for(int a=0;a<static_cast<int>(dofs.size());++a) diag[dofs[a]] += de[a];
        }
        (void)time;
    }
    CSRMatrix dummy;
    reduce_assembled_system_if_parallel(diag,dummy,mpi);
}
