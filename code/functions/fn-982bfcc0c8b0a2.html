void write_hysteresis_svg_from_csv(
    const std::string& csv_filename,
    const std::string& svg_filename,
    const std::string& title,
    const std::string& x_label,
    const std::string& y_label
){
    std::ifstream in(csv_filename);
    if(!in) return;
    std::vector<double> xs, ys;
    std::string line;
    while(std::getline(in,line)){
        if(line.empty() || line[0]=='#') continue;
        if(line.find("time,")==0) continue;
        std::replace(line.begin(),line.end(),',',' ');
        std::stringstream ss(line);
        double t,xv,yv; ss>>t>>xv>>yv;
        if(ss && std::isfinite(xv) && std::isfinite(yv)){ xs.push_back(xv); ys.push_back(yv); }
    }
    if(xs.empty()) return;
    write_xy_svg_plot(svg_filename,"Hysteresis: "+title,x_label,y_label,xs,{
        {y_label,"black",ys}
    });
}
