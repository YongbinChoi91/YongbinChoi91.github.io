Vector solve_with_pardiso_backend(const CSRMatrix& A,const Vector& rhs,const LinearSolverSettings& settings){
    (void)settings;
    if(A.n_rows!=A.n_cols || static_cast<int>(rhs.size())!=A.n_rows) throw std::runtime_error("PARDISO dimension mismatch.");
    const MKL_INT n=A.n_rows, nrhs=1, mtype=11;
    void* pt[64]={nullptr}; MKL_INT iparm[64]={0};
    MKL_INT maxfct=1,mnum=1,phase=13,error=0,msglvl=0;
    iparm[0]=1;   // use custom iparm values
    iparm[1]=2;   // METIS reordering
    iparm[7]=4;   // max iterative refinement steps
    iparm[9]=13;  // pivot perturbation: 1e-13
    iparm[10]=1;  // nonsymmetric scaling, important for fluid/FSI blocks
    iparm[12]=1;  // weighted matching, important for saddle-point systems
    iparm[17]=-1; // report factor nonzeros
    iparm[18]=-1; // report factorization MFLOPS
    iparm[34]=1;  // zero-based CSR indexing
    std::vector<MKL_INT> ia(A.row_ptr.begin(),A.row_ptr.end());
    std::vector<MKL_INT> ja(A.col_idx.begin(),A.col_idx.end());
    std::vector<double> a=A.values, b=rhs, x(n,0.0);
    pardiso(pt,&maxfct,&mnum,const_cast<MKL_INT*>(&mtype),&phase,const_cast<MKL_INT*>(&n),
            a.data(),ia.data(),ja.data(),nullptr,&nrhs,iparm,&msglvl,b.data(),x.data(),&error);
    if(error!=0) throw std::runtime_error("PARDISO solve failed.");
    phase=-1;
    pardiso(pt,&maxfct,&mnum,const_cast<MKL_INT*>(&mtype),&phase,const_cast<MKL_INT*>(&n),
            a.data(),ia.data(),ja.data(),nullptr,&nrhs,iparm,&msglvl,b.data(),x.data(),&error);
    return x;
}
