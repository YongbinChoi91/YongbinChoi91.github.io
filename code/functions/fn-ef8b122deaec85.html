static void collect_hysteresis_node_dof_from_mesh(
    const FieldMesh& mesh,
    const TimeSolverSettings::HysteresisAxisSpec& spec,
    int global_offset,
    std::set<int>& dofs
){
    const DofLayout& gmesh=get_cached_dof_layout(mesh);
    const int fid=find_field_index(gmesh,spec.target.field,true);
    const FieldInfo& field=gmesh.fields[fid];
    if(spec.node<0 || spec.node>=static_cast<int>(field.nodes.size())){
        throw std::runtime_error("hysteresis value node is out of range for field '"+spec.target.field+"'.");
    }
    if(spec.target.component<0 || spec.target.component>=field.components){
        throw std::runtime_error("hysteresis value component is incompatible with field '"+spec.target.field+"'.");
    }
    dofs.insert(global_offset + global_field_dof(gmesh,fid,spec.node,spec.target.component));
}
