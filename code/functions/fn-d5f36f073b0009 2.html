void assemble_neumann_conditions(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    double time,
    int global_offset,
    Vector& R
){
    if(params.neumann_bcs.empty()) return;
    if(mesh.dim!=2 && mesh.dim!=3) throw std::runtime_error("Generic Neumann surface assembly supports only 2D boundary edges and 3D boundary faces.");
 
    const DofLayout& gmesh = get_cached_dof_layout(mesh);
 
    for(const FieldNeumannBC& bc: params.neumann_bcs){
        const int fid = find_field_index(gmesh, bc.field, false);
        if(fid<0) continue; // Field not present in this mesh/domain.
        const FieldInfo& field = gmesh.fields[fid];
        if(bc.component<0 || bc.component>=field.components){
            std::ostringstream os;
            os << "Neumann component " << bc.component << " is incompatible with field '"
               << bc.field << "' which has " << field.components << " component(s).";
            throw std::runtime_error(os.str());
        }
 
        const std::vector<BoundaryEntity> boundary_entities =
            resolve_neumann_boundary_entities(field,bc,mesh.dim);
        if(boundary_entities.empty()) continue;
 
        if(mesh.dim==3){
            const bool hex = is_hex_element_type(field.element_type);
            const bool tet = is_tet_element_type(field.element_type);
            if(!hex && !tet){
                throw std::runtime_error("3D Neumann surface assembly supports only Tet4/Tet10/H8/H20/H27 for field '"+bc.field+"'.");
            }
 
            const auto qface = hex ? gauss_rule_quad_face_assembly(3) : gauss_rule(ElementType::T3,3);
 
            for(const BoundaryEntity& ent: boundary_entities){
                const int elem_id = recover_neumann_owner_element(field, ent);
                if(elem_id<0){
                    throw std::runtime_error("Could not recover 3D Neumann owner element for field '"+bc.field+"', group '"+bc.group+"'.");
                }
                const int face = recover_neumann_local_face_3d(field, elem_id, ent);
                if(face<0){
                    throw std::runtime_error("Could not recover 3D Neumann local face for field '"+bc.field+"', group '"+bc.group+"'.");
                }
 
                const ElementGeometry geom = make_field_geometry_element(field, mesh.dim, elem_id);
                const auto faces = reference_faces_3d(field.element_type);
                const auto& conn = field.elements[elem_id];
 
                for(const auto& q: qface){
                    const auto rp = hex
                        ? hex_face_reference_point(face,q.xi,q.eta)
                        : tet_face_reference_point(face,q.xi,q.eta);
                    const ShapeData shape = compute_shape(field.element_type, rp[0], rp[1], rp[2]);
                    const JacobianData J = compute_jacobian_data(geom, shape);
                    if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while assembling 3D Neumann boundary term.");
                    const double dGamma = surface_measure_from_J_3d(field.element_type, face, J) * q.weight;
 
                    std::array<double,3> Xq{{0.0,0.0,0.0}};
                    const int nloc = std::min(static_cast<int>(conn.size()), static_cast<int>(shape.N.size()));
                    for(int a=0; a<nloc; ++a){
                        const int node = conn[a];
                        for(int d=0; d<3; ++d) Xq[d] += shape.N[a] * field.nodes[node].x[d];
                    }
                    const double traction = eval_scalar_function(bc.value, Xq[0], Xq[1], Xq[2], time);
 
                    // Residual convention: R = internal - external.
                    // Dead traction/flux is independent of the unknown, so K is unchanged.
                    for(int a_local: faces[face]){
                        if(a_local<0 || a_local>=nloc) continue;
                        const int node = conn[a_local];
                        const int I = global_offset + global_field_dof(gmesh, fid, node, bc.component);
                        if(I<0 || I>=static_cast<int>(R.size()))
                            throw std::runtime_error("Global 3D Neumann DOF out of range.");
                        R[I] -= shape.N[a_local] * traction * dGamma;
                    }
                }
            }
            continue;
        }
 
        const auto qline = gauss_rule_line(3);
        for(const BoundaryEntity& ent: boundary_entities){
            if(field.element_type==ElementType::Polygon){
                assemble_polygon_lobatto_neumann_2d(
                    gmesh, field, fid, bc, ent, time, global_offset, R
                );
                continue;
            }
 
            const int elem_id = recover_neumann_owner_element(field, ent);
            if(elem_id<0){
                throw std::runtime_error("Could not recover Neumann owner element for field '"+bc.field+"', group '"+bc.group+"'.");
            }
            const int edge = recover_neumann_local_edge_2d(field, elem_id, ent);
            if(edge<0){
                throw std::runtime_error("Could not recover Neumann local edge for field '"+bc.field+"', group '"+bc.group+"'.");
            }
 
            const ElementGeometry geom = make_field_geometry_element(field, mesh.dim, elem_id);
            const auto edges = reference_edges(field.element_type);
            const auto fixed_axis = edge_reference_normal_2d(field.element_type, edge);
            const auto& conn = field.elements[elem_id];
 
            for(const auto& q: qline){
                const auto rp = edge_reference_point(field.element_type, edge, q.xi);
                const ShapeData shape = compute_shape(field.element_type, rp[0], rp[1], rp[2]);
                const JacobianData J = compute_jacobian_data(geom, shape);
                if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while assembling Neumann boundary term.");
                const double dGamma = surface_measure_from_jacobian_2d(J, fixed_axis) * q.weight;
 
                std::array<double,3> Xq{{0.0,0.0,0.0}};
                const int nloc = std::min(static_cast<int>(conn.size()), static_cast<int>(shape.N.size()));
                for(int a=0; a<nloc; ++a){
                    const int node = conn[a];
                    for(int d=0; d<3; ++d) Xq[d] += shape.N[a] * field.nodes[node].x[d];
                }
                const double traction = eval_scalar_function(bc.value, Xq[0], Xq[1], Xq[2], time);
 
                // Residual convention: R = internal - external.
                // Dead traction/flux is independent of the unknown, so K is unchanged.
                for(int a_local: edges[edge]){
                    if(a_local<0 || a_local>=nloc) continue;
                    const int node = conn[a_local];
                    const int I = global_offset + global_field_dof(gmesh, fid, node, bc.component);
                    if(I<0 || I>=static_cast<int>(R.size()))
                        throw std::runtime_error("Global Neumann DOF out of range.");
                    R[I] -= shape.N[a_local] * traction * dGamma;
                }
            }
        }
    }
}
