inline AD1<Scalar, N> cos(const AD1<Scalar, N>& a) {
    using std::sin; using std::cos;
    AD1<Scalar, N> c(cos(a.val));
    const Scalar sp = sin(a.val);
    for (std::size_t i = 0; i < N; ++i) c.grad[i] = -sp * a.grad[i];
    return c;
}
