static void validate_solid_gauss_save_data(
    const ProblemParameters& params,
    const std::vector<ConstitutivePostprocessOutputSpec>& specs
){
    for(const std::string& item:params.solver.save_data){
        const std::string name = normalize_save_data_name(item);
        if(is_generic_solid_gauss_output(name)) continue;
        if(find_constitutive_gauss_spec(specs,name)) continue;
        std::vector<std::string> supported = {"F", "C", "history"};
        for(const auto& spec:specs) supported.push_back(spec.name);
        throw std::runtime_error("Unsupported save_data item '" + item +
                                 "'. Supported data: " + join_strings(supported,","));
    }
}
