void print_run_overview(
    const Problem& problem,
    const ProblemParameters& params,
    const RunOverviewOptions& options
) {
    if (problem.domains.empty()) return;
 
    const FieldMesh& mesh = problem.domains.front().mesh;
    const DofLayout& g = get_cached_dof_layout(mesh);
    const std::vector<std::string> active = active_constitutive_field_names(mesh);
 
    int active_nodes = 0;
    int active_elements = 0;
    int active_boundary_groups = 0;
    int active_dofs = 0;
    std::string active_type = "Unknown";
    for (const std::string& name : active) {
        auto it = std::find_if(g.fields.begin(), g.fields.end(), [&](const FieldInfo& f) {
            return f.name == name;
        });
        if (it == g.fields.end()) continue;
        active_nodes += static_cast<int>(it->nodes.size());
        active_elements = std::max(active_elements, static_cast<int>(it->elements.size()));
        active_boundary_groups += static_cast<int>(it->boundaries.size());
        active_dofs += it->ndofs;
        if (active_type == "Unknown") active_type = element_type_to_string(it->element_type);
    }
 
    std::string mesh_label = active_type;
    if (mesh.vem_order > 0) {
        mesh_label = std::string("VEM ") + (mesh.vem_order == 1 ? "linear " : "quadratic ") + active_type;
    }
 
    const std::string type = options.problem_type.empty()
        ? format_problem_type_label(problem)
        : options.problem_type;
 
    std::cout << "\nProblem\n"
              << "  type        : " << type << "\n"
              << "  fields      : " << join_strings(active, ", ") << "\n"
              << "  mesh        : " << mesh_label << "\n"
              << "  size        : " << active_nodes << " nodes, "
              << active_elements << " elements, " << active_dofs << " dofs\n"
              << "  boundaries  : " << active_boundary_groups << " groups\n";
    if (problem.domains.size() > 1) {
        std::vector<std::string> domains;
        domains.reserve(problem.domains.size());
        for (const Domain& domain : problem.domains) {
            domains.push_back(domain.name.empty() ? "domain" : domain.name);
        }
        std::cout << "  domains     : " << join_strings(domains, ", ") << "\n";
    }
    for (const auto& [name, value] : options.problem_extra) {
        std::cout << "  " << name << "  : " << value << "\n";
    }
    const std::vector<HistorySpec> history_specs = problem_history_specs_summary(problem);
    if (options.include_history && !history_specs.empty()) {
        std::cout << "  history     :";
        for (const auto& h : history_specs) std::cout << " " << h.name << "(" << h.nvars << ")";
        std::cout << "\n";
    }
 
    std::cout << "\nModel\n";
    if (options.model_lines.empty()) {
        std::cout << "  name                      : " << active_constitutive_model_name() << "\n";
    } else {
        for (const auto& [name, value] : options.model_lines) {
            std::cout << "  " << std::left << std::setw(25) << name << std::right << " : " << value << "\n";
        }
    }
    if (options.include_constitutive_parameters) {
        print_active_constitutive_parameters(std::cout, params);
    }
 
    std::cout << "\nSolver\n"
              << "  nonlinear   : Newton max=" << params.solver.max_iter
              << ", tol=" << params.solver.tol << "\n"
              << "  linear      : " << params.solver.linear.backend << " "
              << params.solver.linear.linear_solver
              << ", pc=" << params.solver.linear.preconditioner << "\n"
              << "  time        : dt=" << params.solver.dt
              << ", total=" << params.solver.total_time
              << ", adaptive=" << (params.solver.adaptive_time_step ? "on" : "off") << "\n"
              << "  quadrature  : " << (params.solver.gauss_order > 0 ? std::to_string(params.solver.gauss_order) : std::string("auto")) << "\n";
    for (const auto& [name, value] : options.solver_extra) {
        std::cout << "  " << name << "    : " << value << "\n";
    }
 
    std::cout << "\nOutput\n"
              << "  hdf5        : " << (params.solver.hdf5_output ? params.solver.hdf5_file : std::string("off")) << "\n"
              << "  dirs        : vtk=" << params.solver.vtk_dir
              << ", csv=" << params.solver.csv_dir
              << ", figures=" << params.solver.figure_dir
              << ", dofs=" << params.solver.dof_dir << "\n";
    if (options.include_hysteresis && params.solver.hysteresis_output) {
        std::cout << "  hysteresis  : " << params.solver.hysteresis_csv << "\n";
    }
}
