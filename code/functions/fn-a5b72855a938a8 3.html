static std::vector<GaussPoint> tensor_product_rule(ElementType type,int order){
    std::vector<double> x,w;
    gauss_legendre_1d_interval(order,-1.0,1.0,x,w);
    std::vector<GaussPoint> gps;
    if(type==ElementType::Q4 || type==ElementType::Q9){
        gps.reserve(order*order);
        for(int i=0;i<order;++i) for(int j=0;j<order;++j)
            gps.push_back({x[i],x[j],0.0,w[i]*w[j]});
        return gps;
    }
    if(type==ElementType::H8 || type==ElementType::H20 || type==ElementType::H27){
        gps.reserve(order*order*order);
        for(int i=0;i<order;++i) for(int j=0;j<order;++j) for(int k=0;k<order;++k)
            gps.push_back({x[i],x[j],x[k],w[i]*w[j]*w[k]});
        return gps;
    }
    throw std::runtime_error("tensor_product_rule called for non tensor-product element.");
}
