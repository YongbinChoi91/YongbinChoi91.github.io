inline Mat3<Scalar> deformation_gradient_array(
    int dim,
    const std::vector<std::array<double,3>>& dNdx,
    const std::vector<Scalar>& u_local_interleaved
) {
    if (dim != 2 && dim != 3) {
        throw std::runtime_error("Array deformation gradient supports only 2D/3D.");
    }
    const int nnode = static_cast<int>(dNdx.size());
    if (static_cast<int>(u_local_interleaved.size()) < dim * nnode) {
        throw std::runtime_error("Deformation gradient received too few displacement DOFs.");
    }
 
    auto F = eye3<Scalar>();
    for (int a = 0; a < nnode; ++a) {
        for (int i = 0; i < dim; ++i) {
            const Scalar ui = u_local_interleaved[dim * a + i];
            for (int A = 0; A < dim; ++A) F[i][A] += ui * Scalar(dNdx[a][A]);
        }
    }
    return F;
}
