FieldRole field_role_from_string(const std::string& role){
    const std::string k=lower_copy(role);
    if(k=="mechanical" || k=="mech") return FieldRole::Mechanical;
    if(k=="constraint" || k=="lagrange" || k=="pressure") return FieldRole::Constraint;
    if(k=="thermal" || k=="temperature" || k=="heat") return FieldRole::Thermal;
    if(k=="internal") return FieldRole::Internal;
    if(k=="auxiliary" || k=="aux") return FieldRole::Auxiliary;
    return FieldRole::Unknown;
}
