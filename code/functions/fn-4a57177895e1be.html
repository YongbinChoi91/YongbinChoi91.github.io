static std::vector<std::array<std::array<double,3>,3>> ad2_reference_hessian(
    ElementType type,
    double xi0,
    double eta0,
    double zeta0
) {
    const std::vector<AD2> Nad = evaluate_shape_ad2(type, xi0, eta0, zeta0);
 
    std::vector<std::array<std::array<double,3>,3>> H(Nad.size());
    for(std::size_t a=0; a<Nad.size(); ++a){
        for(int i=0;i<3;++i) for(int j=0;j<3;++j) H[a][i][j] = Nad[a].h[i][j];
    }
    return H;
}
