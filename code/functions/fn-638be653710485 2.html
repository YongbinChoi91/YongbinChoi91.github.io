static bool any_rank_reported_assembly_error(int local_error, const AssemblyMpiInfo& mpi) {
#ifdef USE_MPI
    if (!mpi.active) return local_error != 0;
    int global_error = 0;
    MPI_Allreduce(&local_error, &global_error, 1, MPI_INT, MPI_MAX, MPI_COMM_WORLD);
    return global_error != 0;
#else
    (void)mpi;
    return local_error != 0;
#endif
}
