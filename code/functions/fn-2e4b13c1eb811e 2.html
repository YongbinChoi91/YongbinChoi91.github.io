static double average_gp_history_value(
    const ElementMaterialHistory& element_history,
    int index
) {
    double sum = 0.0;
    int count = 0;
    for (const MaterialPointHistory& gp : element_history.gp) {
        const Vector& h = gp.h_n.empty() ? gp.h_trial : gp.h_n;
        if (index < static_cast<int>(h.size())) {
            sum += h[index];
            ++count;
        }
    }
    return count > 0 ? sum / static_cast<double>(count) : 0.0;
}
