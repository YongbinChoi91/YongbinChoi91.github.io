static Vector solve_reduced_free_system(const CSRMatrix& K,const Vector& R,const BoundaryData& bc,
                                        ReducedSystemWorkspace& reduced,
                                        LinearSolverSettings settings,
                                        const FieldMesh* schur_mesh,
                                        const Problem* schur_problem,
                                        const ProblemParameters* params,
                                        double dt_for_schur,
                                        const Vector& x_current){
    const int ndofs=K.n_rows;
    if(K.n_cols!=ndofs || static_cast<int>(R.size())!=ndofs) throw std::runtime_error("Reduced solve dimension mismatch.");
    update_reduced_matrix_values(K,reduced);
    const std::vector<int>& free_dofs = reduced.free_dofs;
    CSRMatrix& Kff = reduced.Kff;
 
    // Preserve the reduced-row -> original-global-DOF map for PETSc FieldSplit.
    // Without this map the backend sees only Kff and cannot know which rows are
    // velocity/displacement/pressure after Dirichlet elimination.
    settings.reduced_to_full_dofs = free_dofs;
 
    // Important reduced-DOF convention:
    // x_trial already contains the current Dirichlet/pressure-pin values before
    // residual and tangent assembly. Therefore R_f is already evaluated with
    // the fixed values. Do NOT add/subtract a K_fc*x_fixed correction here.
    Vector rhs_f(free_dofs.size(),0.0);
    for(int i=0;i<static_cast<int>(free_dofs.size());++i){
        rhs_f[i] = -R[free_dofs[i]];
    }
 
    CSRMatrix Pff;
    const CSRMatrix* Pptr=nullptr;
    const PetscPCDData* pcd_ptr=nullptr;
    if(params && lower_copy(settings.backend)=="petsc" && uses_custom_pressure_schur(settings)){
        if(schur_problem && schur_problem->shared_dofs){
            Pff = build_pressure_schur_preconditioner_matrix_shared_problem(Kff,free_dofs,*schur_problem,*params,settings,dt_for_schur);
            Pptr = &Pff;
        } else if(schur_mesh){
            Pff = build_pressure_schur_preconditioner_matrix(Kff,free_dofs,*schur_mesh,*params,settings,dt_for_schur);
            Pptr = &Pff;
        }
    }
    if(params && lower_copy(settings.backend)=="petsc" && wants_pcd_pressure_pc(settings)){
        if(schur_problem && schur_problem->shared_dofs){
            // Shared multi-domain uses a representative global pressure mesh in the first domain.
            pcd_ptr=&update_pressure_pcd_data_cached(reduced.pcd_cache, free_dofs, schur_problem->domains.front().mesh, *params, settings, dt_for_schur, x_current);
        } else if(schur_mesh){
            pcd_ptr=&update_pressure_pcd_data_cached(reduced.pcd_cache, free_dofs, *schur_mesh, *params, settings, dt_for_schur, x_current);
        }
    }
 
    Vector dx_f=solve_linear_system_backend(Kff,rhs_f,settings,Pptr,pcd_ptr);
    if(dx_f.size()!=free_dofs.size()) throw std::runtime_error("Reduced solve returned wrong vector size.");
 
    Vector dx(ndofs,0.0);
    for(int i=0;i<static_cast<int>(free_dofs.size());++i){
        if(!std::isfinite(dx_f[i])) throw std::runtime_error("Reduced solve produced non-finite increment.");
        dx[free_dofs[i]]=dx_f[i];
    }
 
    // Fixed DOFs are increments, not values. They must remain exactly zero.
    for(int d:bc.dofs){
        if(d>=0 && d<ndofs) dx[d]=0.0;
    }
    return dx;
}
