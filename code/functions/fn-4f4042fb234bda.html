std::vector<int> make_element_dofs_for_fields(const DofLayout& mesh,int elem_id,const std::vector<std::string>& field_names){
    std::vector<int> dofs;
    for(const std::string& name: field_names){
        const int fid=find_field_index(mesh,name,true);
        const FieldInfo& f=mesh.fields[fid];
        if(elem_id<0 || elem_id>=static_cast<int>(f.elements.size())){
            throw std::runtime_error("Element id out of range for generic field "+name+".");
        }
        const auto& conn=f.elements[elem_id];
        for(int node:conn){
            for(int c=0;c<f.components;++c) dofs.push_back(global_field_dof(mesh,fid,node,c));
        }
    }
    return dofs;
}
