void append_hdf5_solid_gauss_point_snapshot(
    const ProblemParameters& params,
    const Problem& problem,
    int step,
    double time,
    const Vector& x,
    const MaterialHistoryStore* history
){
    if(!params.solver.hdf5_output) return;
    if(!any_solid_gauss_save_data_requested(params)) return;
    const auto constitutive_specs = normalized_constitutive_gauss_output_specs();
    validate_solid_gauss_save_data(params,constitutive_specs);
    (void)step;
    (void)time;
#ifdef HIPECS_USE_HDF5
    hid_t file=h5_open_file_rw(params);
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain = problem.domains[d];
        const auto field_names = active_constitutive_field_names(domain.mesh);
        const FieldInfo* primary_field = primary_postprocess_field(domain.mesh,field_names);
        if(!primary_field) continue;
        const Vector xd = extract_domain_vector(x,problem,domain);
        const hsize_t nelem = static_cast<hsize_t>(primary_field->elements.size());
        const std::string base = "/gauss/domain_" + std::to_string(d);
#ifdef HIPECS_ENABLE_VEM
        const VemGaussLayoutInfo vem_info = make_vem_gauss_layout_info(domain.mesh,*primary_field,params);
        if(vem_info.enabled){
            std::vector<std::string> requested_constitutive;
            for(const std::string& item:params.solver.save_data){
                const std::string name = normalize_save_data_name(item);
                if(name != "history") requested_constitutive.push_back(name);
            }
            if(!requested_constitutive.empty()){
                std::map<std::string,std::vector<std::vector<double>>> by_name;
                std::map<std::string,int> components_by_name;
                const DofLayout& layout = get_cached_dof_layout(domain.mesh);
                for(int e=0; e<static_cast<int>(primary_field->elements.size()); ++e){
                    const auto dofs = make_element_dofs_for_fields(layout,e,field_names);
                    const Vector x_elem = gather_local_vector(xd,dofs);
                    ElementState state;
                    state.domain_id = d;
                    state.domain_name = domain.name;
                    state.domain_type = domain.type;
                    std::vector<ConstitutivePostprocessOutputValues> element_outputs;
                    compute_constitutive_postprocess(domain.mesh,e,x_elem,params,state,requested_constitutive,element_outputs);
                    for(auto& out:element_outputs){
                        if(out.location != ConstitutivePostprocessLocation::GaussPoint) continue;
                        out.name = normalize_save_data_name(out.name);
                        by_name[out.name].resize(primary_field->elements.size());
                        by_name[out.name][e] = std::move(out.values);
                        components_by_name[out.name] = out.components;
                    }
                }
                for(const std::string& item:requested_constitutive){
                    const std::string name = normalize_save_data_name(item);
                    const auto comp_it = components_by_name.find(name);
                    if(comp_it == components_by_name.end()) continue;
                    append_vem_nan_padded_output(file,base + "/" + hdf5_gauss_dataset_name(name),vem_info,comp_it->second,by_name[name]);
                }
            }
            continue;
        }
#endif
        const auto gps = gauss_rule(primary_field->element_type, params.solver.gauss_order);
        const hsize_t ngp = static_cast<hsize_t>(gps.size());
        std::vector<double> F_data;
        if(domain.type == DomainType::Solid && has_displacement_field(domain.mesh) &&
           (save_data_requested(params,"f") || save_data_requested(params,"c"))){
            F_data = collect_solid_deformation_gradients(domain.mesh,params,xd);
        }
        if(domain.type == DomainType::Solid && has_displacement_field(domain.mesh) && save_data_requested(params,"f")){
            h5_append_double_nd(file,base + "/F",{nelem,ngp,3,3},F_data);
        }
        if(domain.type == DomainType::Solid && has_displacement_field(domain.mesh) && save_data_requested(params,"c")){
            h5_append_double_nd(file,base + "/C",{nelem,ngp,3,3},collect_right_cauchy_green_from_F(F_data));
        }
        if(domain.type == DomainType::Solid && save_data_requested(params,"history") &&
           history && d<static_cast<int>(history->domains.size()) && H5Lexists(file,(base + "/history").c_str(),H5P_DEFAULT)>0){
            const int nvars = primary_history_nvars(history->domains[d].specs);
            h5_append_double_nd(file,base + "/history",{nelem,ngp,static_cast<hsize_t>(nvars)},collect_history_gauss_points(history->domains[d]));
        }
        std::vector<std::string> requested_constitutive;
        for(const std::string& item:params.solver.save_data){
            const std::string name = normalize_save_data_name(item);
            if(!is_generic_solid_gauss_output(name)) requested_constitutive.push_back(name);
        }
        if(!requested_constitutive.empty()){
            std::map<std::string,std::vector<double>> by_name;
            std::map<std::string,int> components_by_name;
            const DofLayout& layout = get_cached_dof_layout(domain.mesh);
            for(int e=0; e<static_cast<int>(primary_field->elements.size()); ++e){
                const auto dofs = make_element_dofs_for_fields(layout,e,field_names);
                const Vector x_elem = gather_local_vector(xd,dofs);
                ElementState state;
                if(history && d<static_cast<int>(history->domains.size()) &&
                   e<static_cast<int>(history->domains[d].elements.size())){
                    state.material_history = const_cast<ElementMaterialHistory*>(&history->domains[d].elements[e]);
                }
                state.domain_id = d;
                state.domain_name = domain.name;
                state.domain_type = domain.type;
                std::vector<ConstitutivePostprocessOutputValues> element_outputs;
                compute_constitutive_postprocess(domain.mesh,e,x_elem,params,state,requested_constitutive,element_outputs);
                for(auto& out:element_outputs){
                    if(out.location != ConstitutivePostprocessLocation::GaussPoint) continue;
                    out.name = normalize_save_data_name(out.name);
                    by_name[out.name].insert(by_name[out.name].end(),out.values.begin(),out.values.end());
                    components_by_name[out.name] = out.components;
                }
            }
            for(const std::string& name:requested_constitutive){
                const ConstitutivePostprocessOutputSpec* spec = find_constitutive_gauss_spec(constitutive_specs,name);
                if(!spec) continue;
                h5_append_double_nd(file,base + "/" + name,
                                    {nelem,ngp,static_cast<hsize_t>(spec->components)},
                                    by_name[name]);
            }
        }
    }
    H5Fclose(file);
#else
    (void)problem; (void)step; (void)time; (void)x; (void)history;
#endif
}
