static int find_csr_entry_index(const CSRMatrix& K,int row,int col){
    if(row<0 || row>=K.n_rows) throw std::runtime_error("CSR row out of range.");
    auto b=K.col_idx.begin()+K.row_ptr[row];
    auto e=K.col_idx.begin()+K.row_ptr[row+1];
    auto it=std::lower_bound(b,e,col);
    if(it==e || *it!=col) throw std::runtime_error("CSR entry missing.");
    return static_cast<int>(it-K.col_idx.begin());
}
