Problem read_multi_domain_problem(const std::string& filename, bool vem_taylor_hood){
    std::ifstream in(filename);
    if(!in) throw std::runtime_error("Could not open multi-domain mesh file: "+filename);
 
    std::vector<std::string> lines;
    std::string line;
    bool has_domain=false;
    int global_dim=0;
    while(std::getline(in,line)){
        if(!line.empty() && line.back()=='\r') line.pop_back();
        std::string raw=line;
        const auto comment=raw.find('#');
        if(comment!=std::string::npos) raw=raw.substr(0,comment);
        raw=trim_copy(raw);
        if(raw.empty()) { lines.push_back(line); continue; }
        std::stringstream ss(raw);
        std::string key; ss>>key;
        if(key=="DOMAIN") has_domain=true;
        if(key=="DIM") ss>>global_dim;
        lines.push_back(line);
    }
 
    if(mesh_file_uses_shared_multi_domain(lines)){
        return read_shared_multi_domain_problem_from_lines(lines,filename,vem_taylor_hood);
    }
 
    Problem problem;
    if(!has_domain){
        Domain d;
        d.name="domain";
        d.mesh=read_mixed_mesh(filename,vem_taylor_hood);
        if(!d.mesh.velocity_nodes.empty() || !d.mesh.pressure_nodes.empty()) {
            d.type=DomainType::Fluid;
        } else if(!d.mesh.displacement_nodes.empty()) {
            d.type=DomainType::Solid;
        } else {
            d.type=DomainType::Unknown;
        }
        d.dof_offset=0;
        d.n_dofs=total_dof_count(d.mesh);
        problem.dim=d.mesh.dim;
        problem.total_dofs=d.n_dofs;
        problem.domains.push_back(std::move(d));
        return problem;
    }
 
    struct RawDomainBlock {
        std::string name;
        DomainType type=DomainType::Unknown;
        std::vector<std::string> body;
        bool has_dim=false;
        bool has_mesh_type=false;
    };
 
    std::vector<RawDomainBlock> blocks;
    RawDomainBlock cur;
    bool inside=false;
    for(const std::string& original:lines){
        std::string clean=original;
        const auto comment=clean.find('#');
        if(comment!=std::string::npos) clean=clean.substr(0,comment);
        clean=trim_copy(clean);
        if(clean.empty()){
            if(inside) cur.body.push_back(original);
            continue;
        }
        std::stringstream ss(clean);
        std::string key; ss>>key;
        if(key=="DIM" && !inside){
            if(global_dim==0) ss>>global_dim;
            continue;
        }
        if(key=="MESH_TYPE" && !inside){
            continue;
        }
        if(key=="DOMAIN"){
            if(inside) throw std::runtime_error("Nested DOMAIN blocks are not allowed.");
            inside=true;
            cur=RawDomainBlock{};
            ss>>cur.name;
            if(cur.name.empty()) cur.name="domain"+std::to_string(blocks.size());
            cur.type=DomainType::Unknown;
            continue;
        }
        if(key=="END_DOMAIN"){
            if(!inside) throw std::runtime_error("END_DOMAIN without DOMAIN.");
            if(cur.type==DomainType::Unknown){
                // Default: a domain called fluid is fluid; otherwise require TYPE.
                cur.type = (lower_copy(cur.name).find("fluid")!=std::string::npos) ? DomainType::Fluid : DomainType::Unknown;
            }
            if(cur.type==DomainType::Unknown) throw std::runtime_error("DOMAIN "+cur.name+" has unknown TYPE. Use TYPE fluid/solid/thermal.");
            blocks.push_back(cur);
            inside=false;
            continue;
        }
        if(!inside) continue;
        if(key=="TYPE"){
            std::string t; ss>>t;
            cur.type=domain_type_from_string(t);
            if(cur.type==DomainType::Unknown) throw std::runtime_error("Unknown DOMAIN TYPE: "+t);
            continue;
        }
        if(key=="DIM") cur.has_dim=true;
        if(key=="MESH_TYPE") cur.has_mesh_type=true;
        cur.body.push_back(original);
    }
    if(inside) throw std::runtime_error("Unclosed DOMAIN block in multi-domain mesh.");
    if(blocks.empty()) throw std::runtime_error("No DOMAIN blocks found in multi-domain mesh.");
 
    int offset=0;
    for(int i=0;i<static_cast<int>(blocks.size());++i){
        const RawDomainBlock& b=blocks[i];
        Domain d;
        d.name=b.name;
        d.type=b.type;
        d.dof_offset=offset;
 
        if(d.type==DomainType::Fluid || d.type==DomainType::Solid){
            const std::string tmp=safe_domain_tmp_name(filename,i);
            {
                std::ofstream out(tmp);
                if(!out) throw std::runtime_error("Could not create temporary domain mesh file: "+tmp);
                if(!b.has_dim){
                    if(global_dim==0) throw std::runtime_error("DIM must be provided globally or inside each DOMAIN.");
                    out << "DIM " << global_dim << "\n";
                }
                if(!b.has_mesh_type) out << "MESH_TYPE GENERAL_MIXED\n";
                for(const std::string& l:b.body) out << l << "\n";
            }
            d.mesh=read_mixed_mesh(tmp,vem_taylor_hood && d.type==DomainType::Fluid);
            std::remove(tmp.c_str());
            d.n_dofs=total_dof_count(d.mesh);
            if(problem.dim==0 || problem.domains.empty()) problem.dim=d.mesh.dim;
            else if(problem.dim!=d.mesh.dim) throw std::runtime_error("All domains must have the same dimension.");
        } else {
            throw std::runtime_error("DOMAIN '"+d.name+"' has TYPE "+domain_type_to_string(d.type)+", but current continuum assembly supports only fluid/solid domains.");
        }
 
        offset += d.n_dofs;
        problem.domains.push_back(std::move(d));
    }
    problem.total_dofs=offset;
 
    std::cout << "Multi-domain problem\n"
              << "  dim       : " << problem.dim << "D\n"
              << "  domains   : " << problem.domains.size() << "\n"
              << "  total dofs: " << problem.total_dofs << "\n";
    for(const auto& d:problem.domains){
        std::cout << "  - " << d.name
                  << " type=" << domain_type_to_string(d.type)
                  << " offset=" << d.dof_offset
                  << " ndofs=" << d.n_dofs << "\n";
    }
    return problem;
}
