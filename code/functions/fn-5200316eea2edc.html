static std::vector<AD2> evaluate_shape_ad2(
    ElementType type,
    const AD2& xi,
    const AD2& eta,
    const AD2& zeta
) {
    switch(type){
        case ElementType::T3:  return shape_values_T3_ad2(xi,eta,zeta);
        case ElementType::T6:  return shape_values_T6_ad2(xi,eta,zeta);
        case ElementType::Q4:  return shape_values_Q4_ad2(xi,eta,zeta);
        case ElementType::Q9:  return shape_values_Q9_ad2(xi,eta,zeta);
        case ElementType::Tet4: return shape_values_Tet4_ad2(xi,eta,zeta);
        case ElementType::Tet10:return shape_values_Tet10_ad2(xi,eta,zeta);
        case ElementType::H8:  return shape_values_H8_ad2(xi,eta,zeta);
        case ElementType::H20: return shape_values_H20_ad2(xi,eta,zeta);
        case ElementType::H27: return shape_values_H27_ad2(xi,eta,zeta);
        default: throw std::runtime_error("Unsupported element type in evaluate_shape_ad2.");
    }
}
