std::vector<std::string> active_dof_field_names(const FieldMesh& mesh){
    if(!mesh.dof_field_order.empty()) return mesh.dof_field_order;
    std::vector<std::string> names;
    if(!mesh.velocity_nodes.empty() && !mesh.velocity_elements.empty()) names.push_back("velocity");
    if(has_displacement_field(mesh)) names.push_back("displacement");
    if(!mesh.pressure_nodes.empty() && !mesh.pressure_elements.empty()) names.push_back("pressure");
    return names;
}
