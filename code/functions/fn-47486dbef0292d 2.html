ProblemElementTangentIndexMap precompute_global_tangent_indices(
    const ProblemElementDofMap& dof_maps,
    const CSRMatrix& K
){
    ProblemElementTangentIndexMap maps(dof_maps.size());
    for(int d=0; d<static_cast<int>(dof_maps.size()); ++d){
        maps[d].resize(dof_maps[d].size());
        for(int e=0;e<static_cast<int>(dof_maps[d].size());++e){
            const auto& dofs=dof_maps[d][e];
            const int m=static_cast<int>(dofs.size());
            maps[d][e].assign(m*m,-1);
            for(int a=0;a<m;++a) for(int b=0;b<m;++b) maps[d][e][a*m+b]=find_csr_entry_index(K,dofs[a],dofs[b]);
        }
    }
    return maps;
}
