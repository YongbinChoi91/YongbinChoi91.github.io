inline Vec2T<Scalar> quadratic_projection_value_basis(
    const Coeff6x2T<Scalar>& a,
    const std::array<double,6>& p
) {
    Vec2T<Scalar> u{{Scalar(0.0), Scalar(0.0)}};
    for (int b = 0; b < 6; ++b) {
        for (int c = 0; c < 2; ++c) u[c] += Scalar(p[b]) * a[b][c];
    }
    return u;
}
