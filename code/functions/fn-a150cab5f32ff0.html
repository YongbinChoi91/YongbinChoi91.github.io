static void validate_and_finalize_mixed_mesh_no_compaction(FieldMesh& mesh,const std::string& label,bool vem_taylor_hood){
    if(mesh.dim!=2 && mesh.dim!=3) throw std::runtime_error(label+": DIM must be 2 or 3.");
    if(mesh.velocity_nodes.empty()) throw std::runtime_error(label+": no velocity nodes.");
    if(mesh.pressure_nodes.empty()) throw std::runtime_error(label+": no pressure nodes.");
 
    // Strict consistency: pressure connectivity must reference only node ids
    // explicitly present in PRESSURE_NODES.  Do not silently recover missing
    // pressure nodes from the high-order/velocity node table; that hides mesh
    // writer errors and can create a wrong pressure DOF layout.
    if(mesh.velocity_elements.empty() || mesh.pressure_elements.empty()) throw std::runtime_error(label+": velocity/pressure elements are missing.");
    if(mesh.velocity_elements.size()!=mesh.pressure_elements.size()) throw std::runtime_error(label+": velocity/pressure element count mismatch.");
    if(mesh.displacement_elements.empty()) throw std::runtime_error(label+": shared multi-domain mesh requires displacement elements.");
    if(mesh.displacement_nodes.empty()) throw std::runtime_error(label+": shared multi-domain mesh requires displacement nodes.");
    if(mesh.displacement_elements.size()!=mesh.velocity_elements.size()) throw std::runtime_error(label+": velocity/displacement element count mismatch.");
 
    for(const auto& e:mesh.velocity_elements) if(e.empty()) throw std::runtime_error(label+": empty velocity element connectivity.");
    for(const auto& e:mesh.displacement_elements) if(e.empty()) throw std::runtime_error(label+": empty displacement element connectivity.");
    for(const auto& e:mesh.pressure_elements) if(e.empty()) throw std::runtime_error(label+": empty pressure element connectivity.");
 
    mesh.velocity_type=vem_taylor_hood ? ElementType::Polygon : element_type_from_node_count(static_cast<int>(mesh.velocity_elements.front().size()),mesh.dim);
    mesh.displacement_type=vem_taylor_hood ? ElementType::Polygon : element_type_from_node_count(static_cast<int>(mesh.displacement_elements.front().size()),mesh.dim);
    mesh.pressure_type=vem_taylor_hood ? ElementType::Polygon : element_type_from_node_count(static_cast<int>(mesh.pressure_elements.front().size()),mesh.dim);
    if(mesh.velocity_type==ElementType::Unknown || mesh.displacement_type==ElementType::Unknown || mesh.pressure_type==ElementType::Unknown)
        throw std::runtime_error(label+": unsupported element type.");
 
    auto check_elems=[&](const std::vector<std::vector<int>>& elems,int n_nodes,std::size_t nref,const std::string& what,ElementType type){
        for(const auto& e:elems){
            if(type==ElementType::Polygon){
                const bool quadratic_field=(what=="velocity" || what=="displacement");
                if((quadratic_field && (e.size()<6 || (e.size()%2)!=0)) ||
                   (!quadratic_field && e.size()<3)){
                    throw std::runtime_error(label+": invalid polygonal VEM "+what+" connectivity.");
                }
            } else if(e.size()!=nref){
                throw std::runtime_error(label+": mixed "+what+" element types are not supported.");
            }
            for(int nid:e){
                if(nid<0 || nid>=n_nodes){
                    std::ostringstream os;
                    os << label << ": " << what << " node id " << nid
                       << " out of range in element connectivity; "
                       << what << " node table size = " << n_nodes
                       << ". Check the corresponding NODES block in the mesh file.";
                    throw std::runtime_error(os.str());
                }
            }
        }
    };
    check_elems(mesh.velocity_elements,static_cast<int>(mesh.velocity_nodes.size()),mesh.velocity_elements.front().size(),"velocity",mesh.velocity_type);
    check_elems(mesh.displacement_elements,static_cast<int>(mesh.displacement_nodes.size()),mesh.displacement_elements.front().size(),"displacement",mesh.displacement_type);
    check_elems(mesh.pressure_elements,static_cast<int>(mesh.pressure_nodes.size()),mesh.pressure_elements.front().size(),"pressure",mesh.pressure_type);
 
    if(vem_taylor_hood){
        for(std::size_t e=0;e<mesh.velocity_elements.size();++e){
            if(mesh.displacement_elements[e].size()!=mesh.velocity_elements[e].size() ||
               mesh.pressure_elements[e].size()*2!=mesh.velocity_elements[e].size()){
                throw std::runtime_error(
                    label+": invalid VEM Taylor-Hood connectivity at element "+std::to_string(e)
                    +"; expected equal quadratic velocity/displacement slots and one pressure node per polygon vertex."
                );
            }
        }
    }
 
    auto check_boundaries=[&](const std::map<std::string,std::vector<BoundaryEntity>>& bmap,int n_nodes,const std::string& what){
        for(const auto& kv:bmap){
            for(const auto& ent:kv.second){
                for(int nid:ent.node_ids){
                    if(nid<0 || nid>=n_nodes){
                        std::ostringstream os;
                        os << label << ": " << what << " boundary node id " << nid
                           << " out of range in group " << kv.first
                           << "; " << what << " node table size = " << n_nodes
                           << ". Check the corresponding NODES block in the mesh file.";
                        throw std::runtime_error(os.str());
                    }
                }
            }
        }
    };
    check_boundaries(mesh.velocity_boundaries,static_cast<int>(mesh.velocity_nodes.size()),"velocity");
    check_boundaries(mesh.displacement_boundaries,static_cast<int>(mesh.displacement_nodes.size()),"displacement");
    check_boundaries(mesh.pressure_boundaries,static_cast<int>(mesh.pressure_nodes.size()),"pressure");
}
