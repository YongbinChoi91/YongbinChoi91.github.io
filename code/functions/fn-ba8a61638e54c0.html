inline int primary_history_nvars(const std::vector<HistorySpec>& specs) {
    const int plasticity = history_nvars_or_zero(specs, "plasticity");
    if (plasticity > 0) return plasticity;
    return specs.empty() ? 0 : specs.front().nvars;
}
