static void remap_element_connectivity(std::vector<std::vector<int>>& elements,const std::vector<int>& map,const std::string& block){
    for(auto& e:elements){
        for(int& old:e){
            if(old<0 || old>=static_cast<int>(map.size()) || map[old]<0)
                throw std::runtime_error("Inactive or invalid node id in "+block+" element connectivity while compacting mixed mesh.");
            old=map[old];
        }
    }
}
