int global_field_dof(const DofLayout& mesh,int field_id,int node_id,int component){
    if(field_id<0 || field_id>=static_cast<int>(mesh.fields.size())) throw std::runtime_error("Generic field id out of range.");
    const FieldInfo& f=mesh.fields[field_id];
    if(node_id<0 || node_id>=static_cast<int>(f.nodes.size())) throw std::runtime_error("Generic field node id out of range for field "+f.name+".");
    if(component<0 || component>=f.components) throw std::runtime_error("Generic field component out of range for field "+f.name+".");
    return f.offset + f.components*node_id + component;
}
