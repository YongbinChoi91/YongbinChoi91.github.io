inline void merge_history_specs_into(std::vector<HistorySpec>& dst, const std::vector<HistorySpec>& src) {
    for (const auto& h : src) {
        bool found = false;
        for (auto& existing : dst) {
            if (existing.name == h.name) {
                existing.nvars = std::max(existing.nvars, h.nvars);
                found = true;
                break;
            }
        }
        if (!found) dst.push_back(h);
    }
}
