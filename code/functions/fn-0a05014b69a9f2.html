void add_inactive_shared_constitutive_field_dof_constraints(
    BoundaryData& bc,
    const Problem& problem,
    const ProblemParameters& params,
    const std::vector<std::string>& active_fields
){
    if(problem.domains.empty()) return;
    const FieldMesh& rep=problem.domains.front().mesh;
 
    if(!contains_string(active_fields,"velocity")){
        for(int n=0;n<static_cast<int>(rep.velocity_nodes.size());++n)
            for(int c=0;c<rep.dim;++c)
                add_dirichlet_bc(rep,params,bc,velocity_dof(rep,n,c),0.0,"INACTIVE_SHARED_MODEL_VELOCITY_DOF");
    }
    if(has_displacement_field(rep) && !contains_string(active_fields,"displacement")){
        for(int n=0;n<static_cast<int>(rep.displacement_nodes.size());++n)
            for(int c=0;c<rep.dim;++c)
                add_dirichlet_bc(rep,params,bc,displacement_dof(rep,n,c),0.0,"INACTIVE_SHARED_MODEL_DISPLACEMENT_DOF");
    }
    if(!contains_string(active_fields,"pressure")){
        for(int n=0;n<static_cast<int>(rep.pressure_nodes.size());++n)
            add_dirichlet_bc(rep,params,bc,pressure_dof(rep,n),0.0,"INACTIVE_SHARED_MODEL_PRESSURE_DOF");
    }
}
