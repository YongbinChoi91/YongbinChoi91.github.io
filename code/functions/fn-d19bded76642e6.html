std::vector<std::array<std::array<double,3>,3>> compute_d2N_dx2(
    const ElementGeometry& elem,
    const ShapeData& s,
    const JacobianData& J,
    double xi,
    double eta,
    double zeta
) {
    const int dim = elem.dim;
    const auto H_ref = compute_d2N_dxi2_reference(elem.type, xi, eta, zeta);
    if (H_ref.size() != s.N.size()) throw std::runtime_error("Second derivative shape size mismatch.");
 
    std::array<std::array<std::array<double,3>,3>,3> x_xixi{};
    for (size_t a=0; a<elem.coords.size(); ++a) {
        for (int m=0; m<dim; ++m) {
            for (int A=0; A<dim; ++A) {
                for (int B=0; B<dim; ++B) {
                    x_xixi[m][A][B] += elem.coords[a][m] * H_ref[a][A][B];
                }
            }
        }
    }
 
    std::array<std::array<std::array<double,3>,3>,3> xi_xx{};
    for (int C=0; C<dim; ++C) {
        for (int i=0; i<dim; ++i) {
            for (int j=0; j<dim; ++j) {
                double rhs = 0.0;
                for (int m=0; m<dim; ++m) {
                    double curv = 0.0;
                    for (int A=0; A<dim; ++A) {
                        for (int B=0; B<dim; ++B) {
                            curv += x_xixi[m][A][B] * J.J_inv[i][A] * J.J_inv[j][B];
                        }
                    }
                    rhs += J.J_inv[m][C] * curv;
                }
                xi_xx[C][i][j] = -rhs;
            }
        }
    }
 
    std::vector<std::array<std::array<double,3>,3>> out(s.N.size());
    for (size_t a=0; a<s.N.size(); ++a) {
        for (int i=0; i<dim; ++i) {
            for (int j=0; j<dim; ++j) {
                double val = 0.0;
                for (int A=0; A<dim; ++A) {
                    for (int B=0; B<dim; ++B) {
                        val += H_ref[a][A][B] * J.J_inv[i][A] * J.J_inv[j][B];
                    }
                    val += s.dN_dxi[a][A] * xi_xx[A][i][j];
                }
                out[a][i][j] = val;
            }
        }
    }
    return out;
}
