static std::string bracket_body_or_throw(const std::string& token,const std::string& selector){
    const auto lb=token.find('[');
    const auto rb=token.rfind(']');
    if(lb==std::string::npos || rb==std::string::npos || rb<=lb+1){
        throw std::runtime_error("Bad "+selector+" selector. Use "+selector+"[...].");
    }
    return token.substr(lb+1,rb-lb-1);
}
