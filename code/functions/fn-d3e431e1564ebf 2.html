inline void add_element_system_into(ElementSystem& dst, const ElementSystem& src) {
    if (dst.residual.size() != src.residual.size() || dst.tangent.size() != src.tangent.size()) {
        throw std::runtime_error("Element system size mismatch while accumulating macro RVE contribution.");
    }
    for (int i = 0; i < static_cast<int>(dst.residual.size()); ++i) dst.residual[i] += src.residual[i];
    for (int i = 0; i < static_cast<int>(dst.tangent.size()); ++i) dst.tangent[i] += src.tangent[i];
}
