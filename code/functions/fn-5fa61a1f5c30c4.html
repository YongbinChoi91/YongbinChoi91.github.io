BoundaryForcePostprocessState initialize_boundary_force_postprocess(
    const Problem& problem,
    const ProblemParameters& params,
    const Vector& initial_solution
){
    BoundaryForcePostprocessState state;
    state.boundary_names = params.solver.force_boundaries.empty()
        ? std::vector<std::string>{params.solver.force_boundary}
        : params.solver.force_boundaries;
    for(int i=0;i<static_cast<int>(state.boundary_names.size());++i){
        if(i) state.boundary_name += "+";
        state.boundary_name += state.boundary_names[i];
    }
    state.force_csv = resolve_force_output_path(params.solver.force_csv, make_force_csv_path(params));
    state.force_svg = resolve_force_output_path(params.solver.force_svg, make_force_svg_path(params));
    state.csv_enabled = params.solver.debug_csv_output && !force_output_path_disabled(params.solver.force_csv);
    state.hdf5_enabled = params.solver.hdf5_output && params.solver.force_output;
    state.svg_enabled = params.solver.svg_output;
    state.deferred = params.solver.postprocess_deferred;
 
    if(!params.solver.force_output){
        return state;
    }
 
    for(const std::string& boundary:state.boundary_names){
        const auto ids=multidomain_force_domain_ids(problem,boundary);
        if(ids.empty() && state.boundary_names.size()>1){
            throw std::runtime_error(
                "Composite drag/lift boundary '"+boundary+"' was not found on a fluid domain."
            );
        }
        for(int id:ids){
            state.domain_ids.push_back(id);
            state.domain_precompute.push_back(
                precompute_boundary_force(problem.domains[id].mesh,boundary)
            );
        }
    }
    state.enabled = !state.domain_ids.empty();
    if(!state.enabled){
        std::cout << "  drag/lift output   : skipped; boundary '" << state.boundary_name
                  << "' was not found on any fluid domain.\n";
        return state;
    }
 
    validate_multidomain_boundary_integral_or_throw(
        problem,state.domain_ids,state.domain_precompute,state.boundary_name,
        params.solver.force_boundary_check_tol
    );
    state.summed_check = summed_multidomain_boundary_check(
        state.domain_ids,state.domain_precompute,state.boundary_name
    );
 
    if(state.csv_enabled){
        ensure_output_parent_directory(state.force_csv);
        reset_force_csv(state.force_csv);
    }
    if(state.hdf5_enabled){
        reset_hdf5_table(params,"/history/force",{"time","Drag","Lift"});
    }
 
    BoundaryForceResult force0 = append_force_sample(problem,initial_solution,params,state,0.0);
    (void)force0;
 
    std::cout << "  drag/lift mode     : weak-form boundary traction integral with legacy stress fallback\n";
    std::cout << "  drag/lift boundary : " << state.boundary_name
              << " on " << state.domain_ids.size() << " domain piece(s)\n";
    std::cout << "  drag/lift output   : "
              << (state.hdf5_enabled ? params.solver.hdf5_file + ":/history/force" : state.force_csv) << "\n";
    return state;
}
