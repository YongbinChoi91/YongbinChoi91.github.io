inline void initialize_material_history_store(
    const Problem& problem,
    const ProblemParameters& params,
    MaterialHistoryStore& history
) {
    history.domains.clear();
 
    history.domains.resize(problem.domains.size());
    for (int d = 0; d < static_cast<int>(problem.domains.size()); ++d) {
        const FieldMesh& mesh = problem.domains[d].mesh;
        const std::vector<HistorySpec>& specs = mesh.history_specs;
        history.domains[d].specs = specs;
        if (specs.empty()) continue;
        const int nvars = primary_history_nvars(specs);
        if (nvars <= 0) continue;
        const int nelem = history_element_count_for_mesh(mesh);
        const ElementType etype = history_element_type_for_mesh(mesh);
        if (nelem <= 0 || etype == ElementType::Unknown) continue;
        const auto gps = gauss_rule(etype, params.solver.gauss_order);
        history.domains[d].elements.resize(nelem);
        for (auto& eh : history.domains[d].elements) {
            eh.gp.resize(gps.size());
            for (auto& gh : eh.gp) {
                gh.h_n.assign(nvars, 0.0);
                gh.h_trial.assign(nvars, 0.0);
                gh.state_n = 0.0;
                gh.state_trial = 0.0;
            }
        }
    }
}
