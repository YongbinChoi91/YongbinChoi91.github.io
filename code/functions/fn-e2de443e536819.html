inline Scalar trace(const std::array<std::array<Scalar, C>, R>& A) {
    static_assert(R == C, "trace requires a square tensor.");
    static_assert(R > 0, "trace requires a non-empty tensor.");
    Scalar t = A[0][0];
    for (std::size_t i = 1; i < R; ++i) t = t + A[i][i];
    return t;
}
