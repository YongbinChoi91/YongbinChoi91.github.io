void store_multiscale_rve_warm_start(int elem_id, int gp_index, const RveSolveResult& rve) {
    if (!rve.converged) return;
    RveWarmStartState state;
    state.displacement = rve.displacement;
    state.lambda = rve.lambda;
    state.history = rve.history;
    state.Fmacro = rve.Fmacro;
    state.has_history = rve.has_history;
    state.valid = true;
 
    std::lock_guard<std::mutex> lock(warm_start_mutex());
    multiscale_context().trial_rve_cache[{elem_id, gp_index}] = std::move(state);
}
