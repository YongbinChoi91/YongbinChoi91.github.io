void reset_minj_csv(const std::string& filename){
    ensure_output_parent_directory(filename);
    std::ofstream out(filename);
    if(!out) throw std::runtime_error("Could not open minJ CSV for writing: "+filename);
    out << "time,minJ\n";
}
