inline AD1<Scalar, N> pow(const AD1<Scalar, N>& a, double p) {
    using std::pow;
    AD1<Scalar, N> c(pow(a.val, p));
    const Scalar fp = Scalar(p) * pow(a.val, p - 1.0);
    for (std::size_t i = 0; i < N; ++i) c.grad[i] = fp * a.grad[i];
    return c;
}
