static std::vector<int> pressure_full_dofs_for_mesh(const FieldMesh& mesh){
    std::vector<int> ids;
    ids.reserve(n_pressure_dofs(mesh));
    for(int p=0;p<n_pressure_dofs(mesh);++p) ids.push_back(pressure_dof(mesh,p));
    return ids;
}
