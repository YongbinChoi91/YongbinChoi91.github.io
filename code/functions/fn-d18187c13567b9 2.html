SolveResult solve_time_stepping(
    ConstitutiveElementKernel kernel,
    const char* model_name,
    const Problem& problem,
    const ProblemParameters& params
){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in time-step solver.");
    const char* output_model_name = (model_name && model_name[0] != '\0') ? model_name : "constitutive_model";
    if(problem.domains.empty()) throw std::runtime_error("Cannot solve empty Problem.");
 
    const auto total_wall_start = std::chrono::steady_clock::now();
    const std::string timing_csv = make_timing_csv_path(params);
    reset_hdf5_results_file(params,problem,output_model_name);
    if(params.solver.timing_diagnostics){
        reset_hdf5_table(params,"/diagnostics/timing",{"step","time","dt","newton_iterations","assembly_time","linear_solver_time","postprocess_time","step_wall_time"});
    }
    if(params.solver.timing_diagnostics && params.solver.timing_csv && params.solver.debug_csv_output) reset_timing_csv(timing_csv);
 
    const int ndofs=problem.total_dofs;
    SolveResult result;
    result.x.assign(ndofs,0.0);
    result.residual.assign(ndofs,0.0);
    const bool petsc_matfree_full = params.solver.linear.petsc_matfree_full;
#ifndef HIPECS_ENABLE_MATFREE
    if(petsc_matfree_full){
        throw std::runtime_error("petsc_matfree_full on requires compiling with USE_MATFREE=1.");
    }
#endif
    if(petsc_matfree_full && lower_copy(params.solver.linear.backend)!="petsc"){
        throw std::runtime_error("petsc_matfree_full on requires solver_backend petsc.");
    }
    if(petsc_matfree_full && lower_copy(params.solver.linear.petsc_matfree_jvp_mode)=="finite_difference"){
        throw std::runtime_error("petsc_matfree_full forbids finite_difference Jv. Use local_tangent or exact_ad.");
    }
    if(petsc_matfree_full){
#ifdef HIPECS_ENABLE_MATFREE
        const std::string mode=lower_copy(params.solver.linear.petsc_matfree_jvp_mode);
        if(mode!="exact_ad" && mode!="local_tangent") throw std::runtime_error("Unknown matrix-free Jv mode: "+mode);
        if(mode=="exact_ad" && (!supports_constitutive_matfree() || kernel!=compute_constitutive_element_system))
            throw std::runtime_error("exact_ad matrix-free requires a supported model kernel (currently hyperelastic_ad displacement only).");
#endif
        const std::string mf_solver=lower_copy(params.solver.linear.linear_solver);
        if(mf_solver=="direct" || mf_solver=="lu" || mf_solver=="preonly"){
            throw std::runtime_error(
                "petsc_matfree_full requires an iterative KSP such as fgmres/gmres/bicgstab. "
                "direct/preonly/lu cannot solve a MatShell operator."
            );
        }
        const std::string mfpc=lower_copy(params.solver.linear.petsc_matfree_pc.empty()
            ? params.solver.linear.preconditioner
            : params.solver.linear.petsc_matfree_pc);
        if(lower_copy(params.solver.linear.petsc_matfree_jvp_mode)=="exact_ad" && mfpc!="none" && mfpc!="jacobi")
            throw std::runtime_error("exact_ad matrix-free requires petsc_matfree_pc none or jacobi.");
        if(mfpc!="none" && mfpc!="jacobi" && mfpc!="element_block_jacobi" && mfpc!="element_jacobi"){
            throw std::runtime_error("petsc_matfree_full supports petsc_matfree_pc none, jacobi, or element_block_jacobi.");
        }
    }
 
    const auto dof_maps=precompute_global_element_dofs(problem);
    ProblemElementTangentIndexMap idx_maps;
    CSRMatrix matfree_jvp_check_K;
    ProblemElementTangentIndexMap matfree_jvp_check_idx;
    const bool petsc_matfree_assembled_pmat = false;
    if(!petsc_matfree_full){
        result.tangent=build_global_sparsity_pattern(problem,dof_maps);
        idx_maps=precompute_global_tangent_indices(dof_maps,result.tangent);
    } else if(params.solver.linear.matfree_jvp_check){
        matfree_jvp_check_K=build_global_sparsity_pattern(problem,dof_maps);
        matfree_jvp_check_idx=precompute_global_tangent_indices(dof_maps,matfree_jvp_check_K);
    }
 
    MaterialHistoryStore material_history;
    initialize_material_history_store(problem, params, material_history);
    const std::vector<HistorySpec> history_specs = material_history_specs_summary(material_history);
    if(!history_specs.empty()){
        std::cout << "  history variables  :";
        for(const auto& h: history_specs) std::cout << " " << h.name << "(" << h.nvars << ")";
        std::cout << "\n";
    }
 
    double dt_current=initial_time_step(params.solver);
    double dt_previous=dt_current;
 
    Vector x_nm1(ndofs,0.0), x_n(ndofs,0.0), x_trial(ndofs,0.0);
    BoundaryData bc0=build_multidomain_boundary_conditions(problem,params,0.0,active_constitutive_field_names);
    print_shared_boundary_condition_diagnostics(problem,bc0,params);
    apply_dirichlet_values_to_solution(bc0,x_n);
    x_nm1=x_n;
    result.x=x_n;
    reset_hdf5_solution_snapshots(params,ndofs);
    reset_hdf5_solid_gauss_point_snapshots(params,problem,&material_history);
    append_hdf5_solution_snapshot(params,0,0.0,result.x);
    append_hdf5_solid_gauss_point_snapshot(params,problem,0,0.0,result.x,&material_history);
 
    const bool show_fluid_time_details =
        problem_has_fluid_like_domain(problem) || problem.shared_dofs || problem.domains.size() > 1;
    std::cout << "\nTime stepping\n"
              << "  domains            : " << problem.domains.size() << "\n"
              << "  total elements     : " << total_assembled_elements(dof_maps) << "\n"
              << "  total dofs         : " << ndofs << "\n"
              << "  target dt          : " << params.solver.dt << "\n"
              << "  total time         : " << params.solver.total_time << "\n"
              << "  adaptive dt        : " << (params.solver.adaptive_time_step ? "on" : "off") << "\n"
              << "  gauss order        : " << (params.solver.gauss_order > 0 ? std::to_string(params.solver.gauss_order) : std::string("auto")) << "\n"
              << "  PETSc MatFree      : " << (petsc_matfree_full ? "on" : "off")
              << (petsc_matfree_full ? (std::string(" (Jv=") + params.solver.linear.petsc_matfree_jvp_mode + ", pc=" + params.solver.linear.petsc_matfree_pc + ")") : std::string())
              << "\n";
    if(show_fluid_time_details){
        std::cout << "  theta              : " << params.solver.theta
                  << " (expr='" << params.solver.theta_expression << "'"
                  << (params.solver.theta_depends_on_dt ? ", dynamic dt" : ", constant") << ")\n"
                  << "  stabilization      : " << (params.solver.stabilization ? "on" : "off")
                  << " (order=" << params.solver.stabilization_element_order
                  << ", sigma_BDF=" << params.solver.stabilization_time_order
                  << ", Ck=" << params.solver.stabilization_Ck << ")\n"
                  << "  fixed dofs at t=0  : " << bc0.dofs.size() << "\n";
    }
 
    int accepted_steps=0, attempted_steps=0, total_newton=0;
    double assembly_time_total=0.0, linear_solver_time_total=0.0, postprocess_time_total=0.0;
    double time=0.0;
    result.converged=true;
    std::vector<DeferredSnapshot> deferred_snapshots;
    if(need_deferred_snapshot(params,0,0.0)) deferred_snapshots.push_back(DeferredSnapshot{0,0.0,result.x});
 
    BoundaryForcePostprocessState force_pp = initialize_boundary_force_postprocess(problem,params,result.x);
 
    const std::string minj_csv = resolve_output_path(params.solver.minJ_csv, make_minj_csv_path(params));
    const std::string hysteresis_csv = resolve_output_path(params.solver.hysteresis_csv, hysteresis_csv_filename(params));
    const std::string hysteresis_svg = resolve_output_path(params.solver.hysteresis_svg, hysteresis_svg_filename(params));
    const bool minj_csv_enabled = params.solver.debug_csv_output && params.solver.minJ_output && !output_path_disabled(params.solver.minJ_csv);
    const bool hysteresis_svg_enabled = params.solver.hysteresis_output && !output_path_disabled(params.solver.hysteresis_svg);
    const bool hysteresis_csv_enabled = params.solver.hysteresis_output && !output_path_disabled(params.solver.hysteresis_csv)
        && (params.solver.debug_csv_output || (hysteresis_svg_enabled && params.solver.svg_output));
    const bool minj_hdf5_enabled = params.solver.hdf5_output && params.solver.minJ_output;
    const bool hysteresis_hdf5_enabled = params.solver.hdf5_output && params.solver.hysteresis_output;
    bool have_minj_domain = false;
    for(const auto& d:problem.domains){
        if(domain_participates_in_minj_postprocess(problem,params,d)){
            have_minj_domain = true;
            break;
        }
    }
    const bool minj_enabled = have_minj_domain && (minj_csv_enabled || minj_hdf5_enabled);
    HysteresisMonitor hysteresis_monitor;
    if(params.solver.hysteresis_output){
        hysteresis_monitor = make_hysteresis_monitor(problem,params);
        if(hysteresis_csv_enabled){
            reset_hysteresis_csv(hysteresis_csv,hysteresis_monitor);
        }
        if(hysteresis_hdf5_enabled){
            reset_hdf5_table(params,"/history/hysteresis",{"time","x_value","y_value"});
        }
    }
    if(minj_enabled){
        if(minj_csv_enabled){
            ensure_output_parent_directory(minj_csv);
            reset_minj_csv(minj_csv);
        }
        if(minj_hdf5_enabled) reset_hdf5_table(params,"/diagnostics/minJ",{"time","minJ"});
        double minJ0=std::numeric_limits<double>::infinity();
        for(const auto& d:problem.domains){
            if(domain_participates_in_minj_postprocess(problem,params,d)){
                const Vector xd0 = extract_domain_vector(result.x,problem,d);
                minJ0=std::min(minJ0,compute_min_deformation_jacobian(d.mesh,xd0,&params,d.type,d.name));
            }
        }
        if(!std::isfinite(minJ0)) minJ0=0.0;
        if(minj_csv_enabled) append_minj_csv(minj_csv,MinJacobianResult{0.0,minJ0});
        if(minj_hdf5_enabled) append_hdf5_table_row(params,"/diagnostics/minJ",{0.0,minJ0});
        std::cout << "  minJ output        : "
                  << (minj_hdf5_enabled ? params.solver.hdf5_file + ":/diagnostics/minJ" : minj_csv) << "\n";
    }
 
    if(params.solver.vtk_output && params.solver.output_every>0 && !params.solver.postprocess_deferred) {
        const std::string vtk0=write_multidomain_vtk_outputs(problem,result.x,params,0,&material_history);
        launch_step_plot(params,vtk0);
    }
    const int max_attempts=max_time_step_attempts(params.solver);
    const StepControl step_control(params.solver);
 
    while(time < params.solver.total_time - 1.0e-14){
        if(++attempted_steps > max_attempts){ result.converged=false; std::cout << "Exceeded maximum time-step attempts.\n"; break; }
        const StepWindow step_window = step_control.begin_step(time,dt_current,params.solver.total_time,accepted_steps+1);
        const double time_start=step_window.time;
        const double dt_step=step_window.dt;
        const double time_end=step_window.next_time;
        if(dt_step<=0.0) break;
        const auto step_wall_start = std::chrono::steady_clock::now();
        double step_assembly_time = 0.0;
        double step_linear_solver_time = 0.0;
        double step_postprocess_time = 0.0;
 
        ProblemParameters step_params = make_step_params_with_current_dt_theta(params, dt_step);
 
        x_trial=x_n;
        BoundaryData bc=build_multidomain_boundary_conditions(problem,step_params,time_end,active_constitutive_field_names);
        ReducedSystemWorkspace reduced_workspace;
        if(petsc_matfree_full){
#ifdef HIPECS_ENABLE_MATFREE
            reduced_workspace = (params.solver.linear.matfree_jvp_check || petsc_matfree_assembled_pmat)
                ? make_reduced_system_workspace(matfree_jvp_check_K,bc)
                : make_reduced_system_workspace_matfree(ndofs,bc);
#else
            throw std::runtime_error("petsc_matfree_full on requires compiling with USE_MATFREE=1.");
#endif
        } else {
            reduced_workspace = make_reduced_system_workspace(result.tangent,bc);
        }
        apply_dirichlet_values_to_solution(bc,x_trial);
 
        if(params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor)){
            const int step_number = accepted_steps + 1;
            const int step_attempt = attempted_steps - accepted_steps;
            std::cout << "\nStep " << step_number
                      << " attempt " << step_attempt
                      << " | t " << compact_log_number(time_end)
                      << " / " << compact_log_number(params.solver.total_time)
                      << " | dt " << compact_log_number(dt_step) << "\n";
        }
 
        bool step_converged=false;
        std::string reject_reason;
        int iter_used=params.solver.max_iter;
        double final_res=std::numeric_limits<double>::infinity();
        double previous_res=std::numeric_limits<double>::infinity();
        HistoryLocalDiagnostics last_hist_diag;
        const PastDofs past_dofs{x_n,x_nm1};
 
        try{
            for(int iter=0; iter<params.solver.max_iter; ++iter){
                apply_dirichlet_values_to_solution(bc,x_trial);
                HistoryLocalDiagnostics hist_diag;
                const auto assembly_start = std::chrono::steady_clock::now();
                if(petsc_matfree_full && !petsc_matfree_assembled_pmat){
                    assemble_global_residual_only(kernel,problem,step_params,x_trial,past_dofs,dt_step,dt_previous,accepted_steps+1,time_end,
                                                  result.residual,dof_maps,&material_history,&hist_diag);
                } else {
                    assemble_global_system(kernel,problem,step_params,x_trial,past_dofs,dt_step,dt_previous,accepted_steps+1,time_end,
                                           result.residual,result.tangent,dof_maps,idx_maps,
                                           &material_history,&hist_diag);
                }
#ifdef HIPECS_MULTISCALE
                MultiscaleRveDiagnostics ms_diag = consume_multiscale_rve_diagnostics();
#endif
                last_hist_diag = hist_diag;
                const double assembly_elapsed = elapsed_seconds_since(assembly_start);
                step_assembly_time += assembly_elapsed;
                assembly_time_total += assembly_elapsed;
                const double res=residual_norm_free(result.residual,reduced_workspace);
                final_res=res;
                if(iter>0 && params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor)){
                    std::cout << "  Newton Iter " << iter << ": Residual = " << res;
#ifdef HIPECS_MULTISCALE
                    if(ms_diag.active){
                        std::cout << ", RVE Residual: " << ms_diag.max_residual
                                  << ", RVE Iter: " << ms_diag.max_iterations;
                        if(ms_diag.history.active){
                            std::cout << ", RVE history[" << ms_diag.history.name
                                      << "]: Local Residual=" << ms_diag.history.max_residual
                                      << ", Local Iter=" << ms_diag.history.max_iterations;
                        }
                        if(ms_diag.failed_solves>0) std::cout << ", RVE Failed: " << ms_diag.failed_solves;
                    }
#endif
                    if(params.solver.history_diagnostics) print_history_diagnostics_line(hist_diag);
                    std::cout << "\n";
                    if(params.solver.field_diagnostics) print_generic_field_residual_diagnostics(problem,result.residual,bc);
                }
                if(!std::isfinite(res)){ reject_reason="non-finite residual"; break; }
                if(iter>=2 && previous_res < std::numeric_limits<double>::infinity()){
                    const double ratio=res/std::max(previous_res,params.solver.tol);
                    if(ratio>params.solver.divergence_factor){
                        std::ostringstream os; os << "Newton residual diverged: R_k/R_{k-1}=" << ratio;
                        reject_reason=os.str(); break;
                    }
                }
                if(res < params.solver.tol){ step_converged=true; iter_used=iter; break; }
                if(petsc_matfree_full && step_params.solver.linear.matfree_jvp_check){
#ifdef HIPECS_ENABLE_MATFREE
                    const int every=std::max(1,step_params.solver.linear.matfree_jvp_check_every);
                    if(iter==0 || (iter % every)==0){
                        Vector check_R;
                        HistoryLocalDiagnostics check_hist;
                        assemble_global_system(kernel,problem,step_params,x_trial,past_dofs,dt_step,dt_previous,accepted_steps+1,time_end,
                                               check_R,matfree_jvp_check_K,dof_maps,matfree_jvp_check_idx,
                                               &material_history,&check_hist);
                        run_matfree_jvp_check(kernel,problem,step_params,x_trial,past_dofs,dt_step,dt_previous,accepted_steps+1,time_end,
                                              dof_maps,material_history,matfree_jvp_check_K,check_R,result.residual,reduced_workspace,
                                              step_params.solver.linear.matfree_jvp_check_tol);
                    }
#else
                    throw std::runtime_error("matfree_jvp_check requires compiling with USE_MATFREE=1.");
#endif
                }
                LinearSolverSettings ls = make_linear_settings_for_problem(problem, step_params.solver.linear);
                const FieldMesh* schur_mesh = nullptr;
                const Problem* schur_problem = nullptr;
                if(problem.shared_dofs && !problem.domains.empty()){
                    schur_mesh = &problem.domains.front().mesh;
                    schur_problem = &problem;
                } else if(problem.domains.size()==1){
                    schur_mesh = &problem.domains.front().mesh;
                }
                const auto linear_start = std::chrono::steady_clock::now();
                Vector dx;
                if(petsc_matfree_full){
#ifdef HIPECS_ENABLE_MATFREE
                    dx=solve_reduced_matfree_system(kernel,problem,step_params,result.residual,bc,reduced_workspace,ls,
                                                         schur_mesh,schur_problem,dt_step,dt_previous,accepted_steps+1,time_end,
                                                         x_trial,past_dofs,dof_maps,material_history);
#else
                    throw std::runtime_error("petsc_matfree_full on requires compiling with USE_MATFREE=1.");
#endif
                } else {
                    dx=solve_reduced_free_system(result.tangent,result.residual,bc,reduced_workspace,ls,schur_mesh,schur_problem,&step_params,dt_step,x_trial);
                }
                const double linear_elapsed = elapsed_seconds_since(linear_start);
                step_linear_solver_time += linear_elapsed;
                linear_solver_time_total += linear_elapsed;
                if(params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor) && params.solver.field_diagnostics) print_generic_field_update_diagnostics(problem,dx);
                const double alpha=(iter==0) ? params.solver.alpha_first_iter : params.solver.alpha_other_iter;
                for(int i=0;i<ndofs;++i) x_trial[i] += alpha*dx[i];
                apply_dirichlet_values_to_solution(bc,x_trial);
                previous_res=res;
                ++total_newton;
                iter_used=iter+1;
            }
        } catch(const std::exception& e){
            reject_reason=e.what(); step_converged=false;
            if(params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor)) std::cout << "  Trial step failed: " << reject_reason << "\n";
        }
 
        const NewtonStepResult newton_step_result{step_converged,iter_used,final_res,0.0};
        const StepDecision step_decision = step_control.finish_step(newton_step_result,step_window);
 
        if(!step_decision.accepted){
            if(step_decision.retry){
                dt_current = step_decision.next_dt;
                if(params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor)){
                    std::cout << "Rejected time step at t = " << time_start;
                    if(!reject_reason.empty()) std::cout << " because " << reject_reason;
                    std::cout << ". dt -> " << dt_current << "\n";
                }
#ifdef HIPECS_MULTISCALE
                discard_multiscale_rve_trial_cache();
#endif
                continue;
            }
 
            result.converged=false;
            if(params.solver.adaptive_time_step){
                std::cout << "Adaptive time stepping failed: dt < dt_min.\n";
            } else {
                std::cout << "Time step did not converge at t = " << time_start;
                if(!reject_reason.empty()) std::cout << " because " << reject_reason;
                std::cout << "\n";
            }
#ifdef HIPECS_MULTISCALE
            discard_multiscale_rve_trial_cache();
#endif
            break;
        }
 
        x_nm1=x_n; x_n=x_trial; result.x=x_n; commit_material_history_store(material_history); time=time_end; ++accepted_steps;
        append_hdf5_solution_snapshot(params,accepted_steps,time,result.x);
        append_hdf5_solid_gauss_point_snapshot(params,problem,accepted_steps,time,result.x,&material_history);
#ifdef HIPECS_MULTISCALE
        commit_multiscale_rve_history_cache();
#endif
        if(print_summary_newton_monitor(params.solver.residual_monitor)){
            std::cout << "time = " << time << " / " << params.solver.total_time
                      << ", final_residual = " << final_res
                      << ", Newton iterations = " << iter_used;
            if(params.solver.history_diagnostics) print_history_diagnostics_line(last_hist_diag);
            std::cout << "\n";
        } else if(print_progress_newton_monitor(params.solver.residual_monitor)){
            print_progress_bar(time/params.solver.total_time,"Multi-domain solve",final_res,iter_used,last_hist_diag,params.solver.history_diagnostics);
        }
 
        const auto postprocess_start = std::chrono::steady_clock::now();
        if(need_deferred_snapshot(params,accepted_steps,time)){
            deferred_snapshots.push_back(DeferredSnapshot{accepted_steps,time,result.x});
        }
        if(params.solver.vtk_output && params.solver.output_every>0 && !params.solver.postprocess_deferred && (accepted_steps%params.solver.output_every==0 || time>=params.solver.total_time-1.0e-14)){
            const std::string vtk_step=write_multidomain_vtk_outputs(problem,result.x,params,accepted_steps,&material_history);
            launch_step_plot(params,vtk_step);
        }
        BoundaryForceResult force_step;
        bool have_force_step=false;
        if(force_pp.enabled && !force_pp.deferred){
            force_step = append_force_sample(problem,result.x,step_params,force_pp,time);
            have_force_step=true;
        }
        double minJ_step=std::numeric_limits<double>::quiet_NaN();
        if(minj_enabled && !params.solver.postprocess_deferred){
            minJ_step=std::numeric_limits<double>::infinity();
            for(const auto& d:problem.domains){
                if(domain_participates_in_minj_postprocess(problem,params,d)){
                    const Vector xd_step = extract_domain_vector(result.x,problem,d);
                    minJ_step=std::min(minJ_step,compute_min_deformation_jacobian(d.mesh,xd_step,&params,d.type,d.name));
                }
            }
            if(!std::isfinite(minJ_step)) minJ_step=0.0;
            if(minj_csv_enabled) append_minj_csv(minj_csv,MinJacobianResult{time,minJ_step});
            if(minj_hdf5_enabled) append_hdf5_table_row(params,"/diagnostics/minJ",{time,minJ_step});
        }
        if(hysteresis_monitor.enabled && (hysteresis_csv_enabled || hysteresis_hdf5_enabled)){
            const double hx=evaluate_hysteresis_axis(hysteresis_monitor.x_axis,time,result.x,result.residual);
            const double hy=evaluate_hysteresis_axis(hysteresis_monitor.y_axis,time,result.x,result.residual);
            if(hysteresis_csv_enabled) append_hysteresis_csv(hysteresis_csv,time,hx,hy);
            if(hysteresis_hdf5_enabled) append_hdf5_table_row(params,"/history/hysteresis",{time,hx,hy});
 
            // Live hysteresis preview: update the SVG after every accepted step.
            // This keeps the figure visible while the simulation is running,
            // instead of waiting until the final postprocessing stage.
            if(hysteresis_csv_enabled && hysteresis_svg_enabled && params.solver.svg_output){
                write_hysteresis_svg_from_csv(hysteresis_csv,hysteresis_svg,hysteresis_monitor.label,
                                               hysteresis_monitor.x_axis.label,hysteresis_monitor.y_axis.label);
            }
        }
        step_postprocess_time += elapsed_seconds_since(postprocess_start);
        postprocess_time_total += step_postprocess_time;
 
        if(params.solver.newton_log && print_full_newton_monitor(params.solver.residual_monitor) && !print_summary_newton_monitor(params.solver.residual_monitor)){
            std::cout << "Time step " << accepted_steps << " accepted: t = "
                      << std::fixed << std::setprecision(6) << time
                      << ", dt = " << dt_step
                      << ", Newton iters = " << iter_used << std::defaultfloat << "\n";
            if(minj_enabled && !params.solver.postprocess_deferred) std::cout << "  Min J = " << minJ_step << "\n";
            if(have_force_step) std::cout << "  Drag = " << force_step.Drag << ", Lift = " << force_step.Lift << "\n";
        }
 
        const double step_wall_time = elapsed_seconds_since(step_wall_start);
        if(params.solver.timing_diagnostics){
            StepTimingRow timing_row{accepted_steps,time,dt_step,iter_used,step_assembly_time,step_linear_solver_time,step_postprocess_time,step_wall_time};
            append_hdf5_table_row(params,"/diagnostics/timing",{
                static_cast<double>(timing_row.step),timing_row.time,timing_row.dt,
                static_cast<double>(timing_row.newton_iterations),timing_row.assembly_time,
                timing_row.linear_solver_time,timing_row.postprocess_time,timing_row.step_wall_time
            });
            if(params.solver.timing_csv && params.solver.debug_csv_output) append_timing_csv(timing_csv,timing_row);
            if(params.solver.timing_per_step){
                std::cout << "  assembly time      : " << step_assembly_time << " s\n"
                          << "  linear solver time : " << step_linear_solver_time << " s\n"
                          << "  postprocess time   : " << step_postprocess_time << " s\n"
                          << "  step wall time     : " << step_wall_time << " s\n";
            }
        }
 
        dt_previous=dt_step;
        dt_current = step_decision.next_dt;
    }
 
    result.steps=accepted_steps;
    result.iterations=total_newton;
    result.converged = result.converged && (time >= params.solver.total_time - 1.0e-14);
 
    result.timing.accepted_steps = accepted_steps;
    result.timing.newton_iterations = total_newton;
    result.timing.assembly_time_total = assembly_time_total;
    result.timing.linear_solver_time_total = linear_solver_time_total;
    result.timing.postprocess_time_total = postprocess_time_total;
    result.timing.total_wall_time = elapsed_seconds_since(total_wall_start);
 
    std::cout << "\n====================================\n";
    std::cout << "Solve summary\n";
    std::cout << "  converged          : " << (result.converged ? "yes" : "no") << "\n";
    std::cout << "  accepted steps     : " << accepted_steps << "\n";
    std::cout << "  attempted steps    : " << attempted_steps << "\n";
    std::cout << "  total Newton iters : " << total_newton << "\n";
    if(params.solver.postprocess_deferred){
        for(const auto& snap: deferred_snapshots){
            const bool final = snap.time >= params.solver.total_time - 1.0e-14;
            const bool vtk_step = params.solver.vtk_output && params.solver.output_every>0 && (snap.step%params.solver.output_every==0 || final);
            if(vtk_step) {
                const std::string vtk_file=write_multidomain_vtk_outputs(problem,snap.x,params,snap.step);
                launch_step_plot(params,vtk_file);
            }
            if(force_pp.enabled && snap.step>0){
                append_force_sample(problem,snap.x,params,force_pp,snap.time);
            }
            if(minj_enabled && snap.step>0){
                double mj=std::numeric_limits<double>::infinity();
                for(const auto& d:problem.domains){
                    if(domain_participates_in_minj_postprocess(problem,params,d)){
                        const Vector xd = extract_domain_vector(snap.x,problem,d);
                        mj=std::min(mj,compute_min_deformation_jacobian(d.mesh,xd,&params,d.type,d.name));
                    }
                }
                if(!std::isfinite(mj)) mj=0.0;
                if(minj_csv_enabled) append_minj_csv(minj_csv,MinJacobianResult{snap.time,mj});
                if(minj_hdf5_enabled) append_hdf5_table_row(params,"/diagnostics/minJ",{snap.time,mj});
            }
        }
        if(!deferred_snapshots.empty()) std::cout << "  deferred outputs   : wrote " << deferred_snapshots.size() << " stored snapshot(s)\n";
    }
    finalize_force_postprocess(params,force_pp);
    if(hysteresis_monitor.enabled && hysteresis_csv_enabled && hysteresis_svg_enabled && params.solver.svg_output){
        write_hysteresis_svg_from_csv(hysteresis_csv,hysteresis_svg,hysteresis_monitor.label,
                                               hysteresis_monitor.x_axis.label,hysteresis_monitor.y_axis.label);
        std::cout << "  hysteresis figure : " << hysteresis_svg << "\n";
    }
    if(params.solver.timing_diagnostics && params.solver.timing_summary){
        std::cout << "\nTiming summary\n"
                  << "  accepted steps      : " << result.timing.accepted_steps << "\n"
                  << "  Newton iterations   : " << result.timing.newton_iterations << "\n"
                  << "  assembly time total : " << result.timing.assembly_time_total << " s\n"
                  << "  linear solve total  : " << result.timing.linear_solver_time_total << " s\n"
                  << "  postprocess total   : " << result.timing.postprocess_time_total << " s\n"
                  << "  total wall time     : " << result.timing.total_wall_time << " s\n";
        if(params.solver.hdf5_output) std::cout << "  timing hdf5         : " << params.solver.hdf5_file << ":/diagnostics/timing\n";
        if(params.solver.timing_csv && params.solver.debug_csv_output) std::cout << "  timing csv          : " << timing_csv << "\n";
    }
    return result;
}
