void configure_multiscale_rve_context(
    const Problem& micro_problem,
    const ProblemParameters& micro_params,
    const RveSolveOptions& rve_options
) {
    multiscale_context().configured = true;
    multiscale_context().micro_problem = micro_problem;
    multiscale_context().micro_params = micro_params;
    multiscale_context().options = rve_options;
    multiscale_context().options.initial_displacement = nullptr;
    multiscale_context().options.initial_lambda = nullptr;
    multiscale_context().options.initial_history = nullptr;
    multiscale_context().diagnostics = MultiscaleRveDiagnostics{};
    {
        std::lock_guard<std::mutex> lock(warm_start_mutex());
        multiscale_context().accepted_rve_cache.clear();
        multiscale_context().trial_rve_cache.clear();
    }
}
