std::string multiscale_rve_sample_filename(
    const ProblemParameters& params,
    int step,
    const std::string& label
) {
    std::ostringstream os;
    os << params.solver.multiscale_rve_output_dir << "/solution_"
       << std::setw(4) << std::setfill('0') << step
       << "_" << label << "_rve.vtk";
    return os.str();
}
