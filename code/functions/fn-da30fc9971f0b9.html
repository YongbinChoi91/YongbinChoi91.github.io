Vector solve_with_umfpack_backend(const CSRMatrix& A,const Vector& rhs,const LinearSolverSettings& settings){
    (void)settings;
    if(A.n_rows!=A.n_cols || static_cast<int>(rhs.size())!=A.n_rows) throw std::runtime_error("UMFPACK dimension mismatch.");
    std::vector<int> Ap,Ai; std::vector<double> Ax;
    convert_csr_to_csc(A,Ap,Ai,Ax);
 
    struct UmfpackSymbolicCache {
        int n = 0;
        std::vector<int> Ap;
        std::vector<int> Ai;
        void* Symbolic = nullptr;
        ~UmfpackSymbolicCache(){ if(Symbolic) umfpack_di_free_symbolic(&Symbolic); }
        bool same_pattern(int n_in,const std::vector<int>& Ap_in,const std::vector<int>& Ai_in) const {
            return Symbolic && n==n_in && Ap==Ap_in && Ai==Ai_in;
        }
        void reset(int n_in,const std::vector<int>& Ap_in,const std::vector<int>& Ai_in,const std::vector<double>& Ax_in){
            if(Symbolic) umfpack_di_free_symbolic(&Symbolic);
            n=n_in; Ap=Ap_in; Ai=Ai_in; Symbolic=nullptr;
            const int status=umfpack_di_symbolic(n,n,Ap.data(),Ai.data(),Ax_in.data(),&Symbolic,nullptr,nullptr);
            if(status!=UMFPACK_OK) throw std::runtime_error("UMFPACK symbolic factorization failed.");
        }
    };
 
    // The Newton matrix values change every iteration, but its sparsity pattern
    // normally does not.  Reusing the symbolic analysis removes a noticeable
    // fixed cost from every direct solve.
    static thread_local UmfpackSymbolicCache cache;
 
    void* Numeric=nullptr;
    const int n=A.n_rows;
    if(!cache.same_pattern(n,Ap,Ai)) cache.reset(n,Ap,Ai,Ax);
 
    int status=umfpack_di_numeric(Ap.data(),Ai.data(),Ax.data(),cache.Symbolic,&Numeric,nullptr,nullptr);
    if(status!=UMFPACK_OK) throw std::runtime_error("UMFPACK numeric factorization failed.");
    Vector x(n,0.0);
    status=umfpack_di_solve(UMFPACK_A,Ap.data(),Ai.data(),Ax.data(),x.data(),rhs.data(),Numeric,nullptr,nullptr);
    umfpack_di_free_numeric(&Numeric);
    if(status!=UMFPACK_OK) throw std::runtime_error("UMFPACK solve failed.");
    return x;
}
