static std::vector<double> collect_history_gauss_points(const DomainMaterialHistory& domain_history){
    std::vector<double> data;
    for(const ElementMaterialHistory& eh:domain_history.elements){
        for(const MaterialPointHistory& gp:eh.gp){
            const Vector& h = gp.h_n.empty() ? gp.h_trial : gp.h_n;
            data.insert(data.end(), h.begin(), h.end());
        }
    }
    return data;
}
