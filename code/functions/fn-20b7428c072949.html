static std::vector<GaussPoint> gauss_rule_quad_face_assembly(int order){
    const auto qline = gauss_rule_line(order);
    std::vector<GaussPoint> gps;
    gps.reserve(qline.size()*qline.size());
    for(const auto& a: qline){
        for(const auto& b: qline){
            gps.push_back({a.xi,b.xi,0.0,a.weight*b.weight});
        }
    }
    return gps;
}
