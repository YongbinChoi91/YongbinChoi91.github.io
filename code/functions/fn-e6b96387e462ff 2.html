std::array<double,3> edge_reference_point(ElementType type,int edge,double s){
    if(type==ElementType::Q4 || type==ElementType::Q9){
        if(edge==0) return { s,-1.0,0.0};
        if(edge==1) return { 1.0, s,0.0};
        if(edge==2) return {-s, 1.0,0.0};
        if(edge==3) return {-1.0,-s,0.0};
    }
    if(type==ElementType::T3 || type==ElementType::T6){
        const double a=0.5*(s+1.0);
        const double b=0.5*(1.0-s);
        if(edge==0) return {a,0.0,0.0};
        if(edge==1) return {b,a,0.0};
        if(edge==2) return {0.0,b,0.0};
    }
    throw std::runtime_error("Unsupported edge reference point.");
}
