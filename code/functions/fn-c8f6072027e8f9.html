inline Mat3<Scalar> mul3(
    const Mat3<Scalar>& A,
    const Mat3<Scalar>& B
) {
    auto C = zero3<Scalar>();
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) {
            for (int k = 0; k < 3; ++k) C[i][j] += A[i][k] * B[k][j];
        }
    }
    return C;
}
