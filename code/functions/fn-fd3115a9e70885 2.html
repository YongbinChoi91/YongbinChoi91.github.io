ScalarFunction make_scalar_function(const std::string& expr_in){
    ScalarFunction f;
    f.expression = trim_copy(expr_in);
    if(f.expression.empty()) f.expression = "0.0";
    const std::string l = lower_copy(f.expression);
    auto starts=[&](const std::string& p){ return l.rfind(p,0)==0; };
    if(starts("piecewise(") || starts("piecewise_cycle(")){
        f.is_constant=false; f.is_piecewise=true; f.is_cyclic=starts("piecewise_cycle(");
        const auto b=f.expression.find('('), e=f.expression.rfind(')');
        if(b==std::string::npos || e==std::string::npos || e<=b) throw std::runtime_error("Bad piecewise expression.");
        std::string inside=f.expression.substr(b+1,e-b-1);
        std::vector<std::string> items; std::stringstream ss(inside); std::string item;
        while(std::getline(ss,item,';')){ item=trim_copy(item); if(!item.empty()) items.push_back(item); }
        int first=0;
        if(f.is_cyclic){ if(items.size()<3) throw std::runtime_error("piecewise_cycle requires period and two points."); f.period=std::stod(items[0]); first=1; }
        if(static_cast<int>(items.size())-first < 2) throw std::runtime_error("piecewise requires at least two points.");
        for(int i=first;i<static_cast<int>(items.size());++i){
            for(char& c:items[i]) if(c==',') c=' ';
            std::stringstream ps(items[i]);
            double tt,vv; ps>>tt>>vv;
            if(ps.fail()) throw std::runtime_error("Bad piecewise point.");
            f.points.push_back({{tt,vv}});
        }
        std::sort(f.points.begin(), f.points.end(), [](auto a,auto b){return a[0]<b[0];});
        return f;
    }
    char* end=nullptr;
    const double v=std::strtod(f.expression.c_str(), &end);
    while(end && *end && std::isspace(static_cast<unsigned char>(*end))) ++end;
    if(end && *end=='\0'){ f.constant=v; f.is_constant=true; }
    else { f.constant=0.0; f.is_constant=false; }
    return f;
}
