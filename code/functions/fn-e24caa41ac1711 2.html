static void evaluate_velocity_gradient_and_pressure(
    const FieldMesh& mesh,
    const Vector& x,
    const ResolvedBoundaryEdge& r,
    double xi,
    double eta,
    double zeta,
    std::vector<std::vector<double>>& grad_v,
    double& pressure
){
    const int dim=mesh.dim;
    const ElementGeometry geom=make_owner_geometry_element(mesh,r);
    const ElementType vtype = geom.type;
    const ShapeData sv=compute_shape(vtype,xi,eta,zeta);
    const JacobianData J=compute_jacobian_data(geom,sv);
    if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while computing boundary force.");
    const auto dNv=compute_dN_dx(sv,J);
    const ShapeData sp=compute_shape(mesh.pressure_type,xi,eta,zeta);
 
    grad_v.assign(dim,std::vector<double>(dim,0.0));
    const auto& ve=owner_connectivity(mesh,r);
    const int nv_loc=std::min(static_cast<int>(ve.size()),static_cast<int>(sv.N.size()));
    for(int a=0;a<nv_loc;++a){
        const int vnode=ve[a];
        if(vnode<0 || vnode>=static_cast<int>(mesh.velocity_nodes.size())) continue;
        for(int i=0;i<dim;++i){
            const double vi=x[velocity_dof(mesh,vnode,i)];
            for(int j=0;j<dim;++j) grad_v[i][j] += vi*dNv[a][j];
        }
    }
 
    pressure=0.0;
    const std::vector<int> pe=pressure_connectivity_for_owner(mesh,r);
    const int np_loc=std::min(static_cast<int>(pe.size()),static_cast<int>(sp.N.size()));
    for(int a=0;a<np_loc;++a){
        const int pnode=pe[a];
        if(pnode>=0 && pnode<static_cast<int>(mesh.pressure_nodes.size()))
            pressure += sp.N[a]*x[pressure_dof(mesh,pnode)];
    }
}
