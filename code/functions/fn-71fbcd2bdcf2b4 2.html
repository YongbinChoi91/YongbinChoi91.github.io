static std::vector<BoundaryEntity> resolve_neumann_boundary_entities(
    const FieldInfo& field,
    const FieldNeumannBC& bc,
    int dim
){
    BCSelector selector=bc.selector;
    if(selector.kind==BCSelectorKind::BoundaryGroup && selector.group.empty()) selector.group=bc.group;
 
    if(selector.kind==BCSelectorKind::BoundaryGroup){
        auto bit=field.boundaries.find(selector.group);
        if(bit==field.boundaries.end()){
            std::cout << "  Neumann/traction skipped: boundary group '" << selector.group
                      << "' not found for field '" << bc.field << "'.\n";
            return {};
        }
        return bit->second;
    }
 
    if(selector.kind==BCSelectorKind::Point){
        throw std::runtime_error("NEUMANN POINT[...] is not supported. Use a nodal point-load feature instead of surface traction.");
    }
 
    std::vector<BoundaryEntity> out;
    std::set<std::pair<int,std::vector<int>>> seen;
    for(const auto& kv: field.boundaries){
        for(const BoundaryEntity& ent: kv.second){
            if(!boundary_entity_on_line_selector(field,ent,selector,dim)) continue;
            auto key=std::make_pair(ent.element_id,ent.node_ids);
            if(seen.insert(key).second) out.push_back(ent);
        }
    }
    if(out.empty()){
        std::cout << "  Neumann/traction skipped: no boundary entities matched LINE selector '"
                  << selector.group << "' for field '" << bc.field << "'.\n";
    }
    return out;
}
