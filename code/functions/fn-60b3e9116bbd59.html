double evaluate_hysteresis_axis(
    const HysteresisAxisRuntime& axis,
    double time,
    const Vector& x,
    const Vector& residual
){
    double v=0.0;
    const std::string mode=lower_copy(axis.spec.mode);
    if(mode=="time"){
        v=time;
    } else if(mode=="function"){
        v=eval_scalar_function(axis.spec.function,0.0,0.0,0.0,time);
    } else if(mode=="prescribed"){
        v=eval_scalar_function(axis.prescribed_function,0.0,0.0,0.0,time);
    } else if(mode=="reaction" || mode=="residual"){
        for(int dof:axis.dofs){
            if(dof<0 || dof>=static_cast<int>(residual.size())) throw std::runtime_error("Hysteresis DOF out of residual range.");
            v += residual[dof];
        }
        if(mode=="reaction") v = -v;
    } else if(mode=="value"){
        for(int dof:axis.dofs){
            if(dof<0 || dof>=static_cast<int>(x.size())) throw std::runtime_error("Hysteresis DOF out of solution range.");
            v += x[dof];
        }
        const std::string red=lower_copy(axis.spec.reduction);
        if((red.empty() || red=="mean" || red=="average" || red=="avg" || red=="point") && !axis.dofs.empty()){
            v /= static_cast<double>(axis.dofs.size());
        } else if(red=="sum") {
        } else if(!red.empty()) {
            throw std::runtime_error("Unknown hysteresis value reduction: "+axis.spec.reduction);
        }
    } else {
        throw std::runtime_error("Unknown hysteresis axis mode: "+axis.spec.mode);
    }
    return axis.spec.scale * v + axis.spec.offset;
}
