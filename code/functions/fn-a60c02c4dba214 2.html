void write_rve_macro_csv(
    const std::string& filename,
    const RveMacroResponse& response,
    const RveSensitivity* sensitivity
) {
    ensure_output_parent_directory(filename);
    std::ofstream out(filename);
    if (!out) throw std::runtime_error("Could not open RVE CSV output: " + filename);
    out << std::setprecision(16);
    out << "quantity,i,A,a,B,value\n";
    out << "volume,-1,-1,-1,-1," << response.volume << "\n";
    for (int i = 0; i < 3; ++i) {
        for (int A = 0; A < 3; ++A) {
            out << "intP," << i << ',' << A << ",-1,-1," << response.intP[i][A] << "\n";
            out << "Pbar," << i << ',' << A << ",-1,-1," << response.Pbar[i][A] << "\n";
        }
    }
    if (sensitivity) {
        for (int i = 0; i < 3; ++i) {
            for (int A = 0; A < 3; ++A) {
                for (int a = 0; a < 2; ++a) {
                    for (int B = 0; B < 2; ++B) {
                        out << "dintP_dF," << i << ',' << A << ',' << a << ',' << B << ','
                            << sensitivity->C[i][A][a][B] << "\n";
                    }
                }
            }
        }
    }
}
