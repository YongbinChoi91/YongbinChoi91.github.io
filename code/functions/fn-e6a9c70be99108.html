inline Scalar tr(const std::vector<std::vector<Scalar>>& A) {
    const int n = static_cast<int>(A.size());
    if (n <= 0) return Scalar(0.0);
    Scalar tr(0.0);
    for (int i = 0; i < n; ++i) {
        if (static_cast<int>(A[i].size()) != n) throw std::runtime_error("tr expects a square matrix.");
        tr += A[i][i];
    }
    return tr;
}
