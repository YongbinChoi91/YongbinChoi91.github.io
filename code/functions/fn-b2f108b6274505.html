inline auto neo_hookean_energy_lame(const std::array<std::array<Scalar, Dim>, Dim>& F, double mu, double lambda) {
    const auto C = matmul(transpose(F), F);
    const auto J = det(F);
    const auto logJ = hpx::log(J);
    return 0.5 * mu * (trace(C) - static_cast<double>(Dim))
         - mu * logJ
         + 0.25 * lambda * (J * J - 1.0)
         - 0.5 * lambda * logJ;
}
