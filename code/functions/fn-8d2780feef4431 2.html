PetscPCDData& update_pressure_pcd_data_cached(
    PressurePcdCache& cache,
    const std::vector<int>& free_dofs,
    const FieldMesh& mesh,
    const ProblemParameters& params,
    const LinearSolverSettings& settings,
    double dt_for_schur,
    const Vector& x_current
){
    if(static_cast<int>(x_current.size()) < total_dof_count(mesh)) {
        throw std::runtime_error("PCD requested but current solution vector is smaller than the representative mesh DOF count.");
    }
    if(!pressure_pcd_cache_matches(cache,free_dofs,mesh,params,settings,dt_for_schur)){
        rebuild_pressure_pcd_cache(cache,free_dofs,mesh,params,settings,dt_for_schur);
    }
 
    const double eps = 1.0e-14;
    const double theta = std::max(params.solver.theta, eps);
    const double alpha_t_auto = params.solver.stabilization_time_order / (theta * std::max(dt_for_schur, eps));
    const double nu = params.material.mu / std::max(params.material.rho, 1.0e-30);
    const double kp_mass_shift = (settings.pcd_kp_mass_shift<0.0) ? 1.0e-8 : settings.pcd_kp_mass_shift;
    const double ap_mass_w = (settings.pcd_ap_mass_weight<0.0) ? alpha_t_auto : settings.pcd_ap_mass_weight;
    const double ap_lap_w = (settings.pcd_ap_laplacian_weight<0.0) ? nu : settings.pcd_ap_laplacian_weight * nu;
    const double ap_shift = std::max(kp_mass_shift,0.0);
 
    for(int q=0;q<static_cast<int>(cache.data.Ap.values.size());++q){
        cache.data.Ap.values[q] = (ap_mass_w + ap_shift)*cache.data.Mp.values[q] + ap_lap_w*cache.Kp_base.values[q];
    }
 
    const int n_full=total_dof_count(mesh);
    for(int e=0;e<static_cast<int>(mesh.pressure_elements.size());++e){
        const auto& ve=mesh.velocity_elements[e];
        const auto& pe=mesh.pressure_elements[e];
        const ElementGeometry geom=make_velocity_element_geometry(mesh,e);
        for(const auto& gp: gauss_rule(mesh.velocity_type, params.solver.gauss_order)){
            const ShapeData sv=compute_shape(mesh.velocity_type,gp.xi,gp.eta,gp.zeta);
            const JacobianData J=compute_jacobian_data(geom,sv);
            if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while updating PCD pressure operators.");
            const ShapeData sp=compute_shape(mesh.pressure_type,gp.xi,gp.eta,gp.zeta);
            const auto dNp=compute_dN_dx(sp,J);
            const double w=J.detJ*gp.weight;
 
            std::vector<double> v_gp(mesh.dim,0.0);
            const int nv_loc=std::min(static_cast<int>(ve.size()),static_cast<int>(sv.N.size()));
            for(int a=0;a<nv_loc;++a){
                const int vnode=ve[a];
                for(int d=0; d<mesh.dim; ++d){
                    const int I=velocity_dof(mesh,vnode,d);
                    if(I<0 || I>=static_cast<int>(x_current.size())) {
                        throw std::runtime_error("Velocity DOF out of range while updating Python-style PCD Ap.");
                    }
                    v_gp[d] += sv.N[a] * x_current[I];
                }
            }
 
            const int np_loc=std::min(static_cast<int>(pe.size()),static_cast<int>(sp.N.size()));
            std::vector<double> g(np_loc,0.0);
            for(int b=0;b<np_loc;++b){
                for(int d=0; d<mesh.dim; ++d) g[b] += dNp[b][d] * v_gp[d];
            }
 
            for(int a=0;a<np_loc;++a){
                const int Ia_full=pressure_dof(mesh,pe[a]);
                const int Ia=(Ia_full>=0 && Ia_full<n_full) ? cache.full_to_pidx[Ia_full] : -1;
                if(Ia<0) continue;
                for(int b=0;b<np_loc;++b){
                    const int Ib_full=pressure_dof(mesh,pe[b]);
                    const int Ib=(Ib_full>=0 && Ib_full<n_full) ? cache.full_to_pidx[Ib_full] : -1;
                    if(Ib<0) continue;
                    const int idx=pressure_cache_csr_index(cache,Ia,Ib);
                    if(idx<0) throw std::runtime_error("PCD cache CSR entry missing while updating Ap.");
                    cache.data.Ap.values[idx] += sp.N[a]*g[b]*w;
                }
            }
        }
    }
 
    double ap_diag_scale=0.0;
    int ap_diag_count=0;
    const int np_red=cache.data.Ap.n_rows;
    for(int i=0;i<np_red;++i){
        const int idx=pressure_cache_csr_index(cache,i,i);
        if(idx>=0){
            ap_diag_scale += std::abs(cache.data.Ap.values[idx]);
            ++ap_diag_count;
        }
    }
    const double ap_diag_floor = std::max(1.0e-14, 1.0e-12 * (ap_diag_count>0 ? ap_diag_scale/ap_diag_count : 1.0));
    for(int i=0;i<np_red;++i){
        const int idx=pressure_cache_csr_index(cache,i,i);
        if(idx>=0 && std::abs(cache.data.Ap.values[idx]) < ap_diag_floor){
            cache.data.Ap.values[idx] += ap_diag_floor;
        }
    }
    return cache.data;
}
