AugmentedMatrixPattern make_rve_augmented_matrix_pattern(
    const CSRMatrix& Kuu_pattern,
    const RvePbcSystem& pbc,
    int ndofs
) {
    if (Kuu_pattern.n_rows != ndofs || Kuu_pattern.n_cols != ndofs) {
        throw std::runtime_error("RVE augmented matrix pattern received a Kuu size inconsistent with ndofs.");
    }
 
    const int m = static_cast<int>(pbc.constraints.size());
    const int n = ndofs + m;
    std::vector<std::vector<std::pair<int,int>>> rows(n);
 
    for (int i = 0; i < ndofs; ++i) {
        const int row_nnz = Kuu_pattern.row_ptr[i+1] - Kuu_pattern.row_ptr[i];
        rows[i].reserve(static_cast<std::size_t>(row_nnz) + 2);
        for (int p = Kuu_pattern.row_ptr[i]; p < Kuu_pattern.row_ptr[i+1]; ++p) {
            const int j = Kuu_pattern.col_idx[p];
            if (j < 0 || j >= ndofs) {
                throw std::runtime_error("RVE Kuu CSR column index is out of range.");
            }
            rows[i].push_back({j, p});
        }
    }
 
    for (int r = 0; r < m; ++r) {
        const int constraint_row = ndofs + r;
        rows[constraint_row].reserve(pbc.constraints[r].entries.size());
        for (const auto& e : pbc.constraints[r].entries) {
            if (e.dof < 0 || e.dof >= ndofs) {
                throw std::runtime_error("RVE PBC constraint DOF is out of range.");
            }
            rows[constraint_row].push_back({e.dof, -1 - r});
            rows[e.dof].push_back({constraint_row, -1 - r});
        }
    }
 
    AugmentedMatrixPattern pattern;
    CSRMatrix& A = pattern.structure;
    A.n_rows = n;
    A.n_cols = n;
    A.row_ptr.assign(n + 1, 0);
 
    for (int i = 0; i < n; ++i) {
        auto& row = rows[i];
        std::sort(row.begin(), row.end(), [](const auto& a, const auto& b){
            return a.first < b.first;
        });
 
        A.row_ptr[i] = static_cast<int>(A.col_idx.size());
        for (const auto& [col, source] : row) {
            A.col_idx.push_back(col);
            if (source >= 0) {
                pattern.kuu_value_pos.push_back(static_cast<int>(A.values.size()));
                A.values.push_back(0.0);
            } else {
                const int r = -1 - source;
                double value = 0.0;
                if (r >= 0 && r < m) {
                    const int constraint_row = ndofs + r;
                    const auto& c = pbc.constraints[r];
                    for (const auto& e : c.entries) {
                        if ((i == constraint_row && col == e.dof) ||
                            (i == e.dof && col == constraint_row)) {
                            value = e.value;
                            break;
                        }
                    }
                }
                A.values.push_back(value);
            }
        }
    }
    A.row_ptr[n] = static_cast<int>(A.col_idx.size());
    if (static_cast<int>(pattern.kuu_value_pos.size()) != static_cast<int>(Kuu_pattern.values.size())) {
        throw std::runtime_error("RVE augmented matrix pattern lost Kuu entries.");
    }
    return pattern;
}
