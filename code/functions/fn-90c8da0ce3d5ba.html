static LinearSolverSettings make_linear_settings_for_problem(const Problem& problem, LinearSolverSettings s){
    if(problem.domains.empty()) return s;
 
    if(problem.shared_dofs){
        const FieldMesh& rep = problem.domains.front().mesh;
        s.n_velocity_dofs_full = n_velocity_dofs(rep);
        s.n_displacement_dofs_full = n_displacement_dofs(rep);
        s.n_pressure_dofs_full = n_pressure_dofs(rep);
        s.pressure_dofs_full = pressure_full_dofs_for_mesh(rep);
        return s;
    }
 
    // Legacy non-shared multidomain layout: concatenate per-domain blocks.
    // The PETSc Schur split then treats all velocity/displacement blocks as
    // mechanical unknowns and all pressure blocks as pressure unknowns.
    s.n_velocity_dofs_full = 0;
    s.n_displacement_dofs_full = 0;
    s.n_pressure_dofs_full = 0;
    s.pressure_dofs_full.clear();
    for(const auto& d: problem.domains){
        s.n_velocity_dofs_full += n_velocity_dofs(d.mesh);
        s.n_displacement_dofs_full += n_displacement_dofs(d.mesh);
        s.n_pressure_dofs_full += n_pressure_dofs(d.mesh);
        for(int id:pressure_full_dofs_for_mesh(d.mesh)) s.pressure_dofs_full.push_back(d.dof_offset + id);
    }
    return s;
}
