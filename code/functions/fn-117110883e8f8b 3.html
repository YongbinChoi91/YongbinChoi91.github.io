static std::array<double,3> average_selected_field_node_coordinates(
    const FieldInfo& field,
    const std::set<int>& nodes,
    const std::string& field_name
){
    if(nodes.empty()) {
        throw std::runtime_error("Cannot compute auto torsion center for empty node set on field '" + field_name + "'.");
    }
    std::array<double,3> center{{0.0,0.0,0.0}};
    int count = 0;
    for(int node: nodes){
        if(node<0 || node>=static_cast<int>(field.nodes.size())) {
            throw std::runtime_error("Torsion Dirichlet node id out of range for field '" + field_name + "'.");
        }
        for(int d=0; d<3; ++d) center[d] += field.nodes[node].x[d];
        ++count;
    }
    for(int d=0; d<3; ++d) center[d] /= static_cast<double>(count);
    return center;
}
