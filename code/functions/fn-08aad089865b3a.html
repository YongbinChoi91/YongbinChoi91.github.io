void print_dof_layout_summary(const DofLayout& mesh,const std::string& title){
    std::cout << title << "\n";
    std::cout << "  dim   : " << mesh.dim << "\n";
    std::cout << "  fields: " << mesh.fields.size() << "\n";
    for(const auto& f: mesh.fields){
        std::cout << "    - " << f.name
                  << " comp=" << f.components
                  << " nodes=" << f.nodes.size()
                  << " elements=" << f.elements.size()
                  << " type=" << element_type_to_string(f.element_type)
                  << " offset=" << f.offset
                  << " ndofs=" << f.ndofs
                  << " role=" << field_role_to_string(f.role) << "\n";
    }
    std::cout << "  total dofs: " << mesh.total_dofs << "\n";
}
