ConstitutiveMultiscaleResponse compute_rve_constitutive_multiscale_response(
    const FieldMesh& mesh,
    const Vector& displacement,
    const ProblemParameters& params,
    const MaterialHistoryStore* history,
    const ElementState& state
) {
    ConstitutiveMultiscaleResponse response =
        compute_constitutive_multiscale_response(mesh, displacement, params, history, state);
 
    const int ndofs = total_dof_count(mesh);
    if (static_cast<int>(displacement.size()) < ndofs) {
        throw std::runtime_error("RVE multiscale response received a displacement vector shorter than mesh DOFs.");
    }
    if (response.volume <= 0.0) {
        throw std::runtime_error("RVE multiscale response volume is non-positive.");
    }
    if (static_cast<int>(response.dIntPdu.size()) != 9) {
        throw std::runtime_error("RVE multiscale response must provide dIntPdu as 9 x ndofs.");
    }
    for (const auto& row : response.dIntPdu) {
        if (static_cast<int>(row.size()) != ndofs) {
            throw std::runtime_error("RVE multiscale response dIntPdu column count must equal the number of displacement DOFs.");
        }
    }
    return response;
}
