static bool compute_boundary_traction_from_constitutive(
    const FieldMesh& mesh,
    const Vector& x,
    const ProblemParameters& params,
    const ResolvedBoundaryEdge& owner,
    const std::array<double,3>& reference_point,
    const std::array<double,3>& physical_point,
    const std::array<double,3>& normal_measure,
    DomainType domain_type,
    std::array<double,3>& traction
){
    if(owner.use_total_elements) return false;
    bool supported=false;
    for(const auto& spec:active_constitutive_postprocess_outputs()){
        if(spec.location == ConstitutivePostprocessLocation::Boundary && spec.name == "traction"){
            supported=true;
            break;
        }
    }
    if(!supported) return false;
 
    const auto field_names = active_constitutive_field_names(mesh);
    const DofLayout& layout = get_cached_dof_layout(mesh);
    const auto dofs = make_element_dofs_for_fields(layout,owner.elem_id,field_names);
    const Vector x_elem = gather_local_vector(x,dofs);
    const LocalFieldView local_view = make_local_field_view(mesh,owner.elem_id,field_names);
 
    ElementState state;
    state.domain_type = domain_type;
    state.postprocess_location = ConstitutivePostprocessLocation::Boundary;
    state.postprocess_reference_point = reference_point;
    state.postprocess_physical_point = physical_point;
    state.postprocess_normal_measure = normal_measure;
 
    std::vector<ConstitutivePostprocessOutputValues> outputs;
    compute_constitutive_postprocess(mesh,owner.elem_id,x_elem,params,state,{"traction"},outputs);
    for(const auto& out:outputs){
        if(out.location != ConstitutivePostprocessLocation::Boundary) continue;
        if(out.name != "traction" || out.components < 3 || out.values.size() < 3) continue;
        traction = {{out.values[0],out.values[1],out.values[2]}};
        return true;
    }
    return false;
}
