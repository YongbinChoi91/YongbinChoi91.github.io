inline std::vector<std::vector<Scalar>> transpose(
    const std::vector<std::vector<Scalar>>& A
) {
    const int m = static_cast<int>(A.size());
    const int n = m > 0 ? static_cast<int>(A[0].size()) : 0;
    std::vector<std::vector<Scalar>> AT(n, std::vector<Scalar>(m, Scalar(0.0)));
    for (int i = 0; i < m; ++i) {
        if (static_cast<int>(A[i].size()) != n) {
            throw std::runtime_error("Cannot transpose a ragged matrix.");
        }
        for (int j = 0; j < n; ++j) AT[j][i] = A[i][j];
    }
    return AT;
}
