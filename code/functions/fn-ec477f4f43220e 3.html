static void convert_csr_to_csc(const CSRMatrix& A,std::vector<int>& Ap,std::vector<int>& Ai,std::vector<double>& Ax){
    if(A.n_rows!=A.n_cols) throw std::runtime_error("CSR-to-CSC requires square matrix.");
    const int n=A.n_rows, nnz=static_cast<int>(A.values.size());
    Ap.assign(n+1,0); Ai.assign(nnz,0); Ax.assign(nnz,0.0);
    for(int row=0; row<n; ++row){
        for(int p=A.row_ptr[row]; p<A.row_ptr[row+1]; ++p){
            const int col=A.col_idx[p];
            if(col<0 || col>=n) throw std::runtime_error("CSR column out of range.");
            ++Ap[col+1];
        }
    }
    for(int i=0;i<n;++i) Ap[i+1]+=Ap[i];
    std::vector<int> next=Ap;
    for(int row=0; row<n; ++row){
        for(int p=A.row_ptr[row]; p<A.row_ptr[row+1]; ++p){
            const int col=A.col_idx[p], dest=next[col]++;
            Ai[dest]=row; Ax[dest]=A.values[p];
        }
    }
}
