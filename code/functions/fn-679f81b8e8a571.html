inline Mat3<Scalar> sym3(const Mat3<Scalar>& A) {
    auto S = zero3<Scalar>();
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) S[i][j] = Scalar(0.5) * (A[i][j] + A[j][i]);
    }
    return S;
}
