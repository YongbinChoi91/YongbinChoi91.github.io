inline auto grad(F&& f, const std::array<std::array<Scalar, C>, R>& x) {
    auto x_ad = detail::make_active_tensor2(x);
    auto y = f(x_ad);
    if constexpr (is_ad1_v<decltype(y)>) {
        return detail::unflatten_tensor2_gradient<Scalar, R, C>(detail::extract_output_gradient(y));
    } else if constexpr (detail::is_tensor2_v<decltype(y)>) {
        return detail::unflatten_tensor2_output_tensor2_input_gradient<Scalar,
            std::tuple_size_v<std::decay_t<decltype(y)>>,
            std::tuple_size_v<typename std::decay_t<decltype(y)>::value_type>, R, C>(detail::extract_output_gradient(y));
    } else {
        return detail::extract_output_gradient(y);
    }
}
