static void constrain_inactive_constitutive_dofs(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    BoundaryData& bc,
    const std::vector<std::string>& active
){
 
    if(!contains_string(active,"velocity")){
        for(int n=0;n<static_cast<int>(mesh.velocity_nodes.size());++n)
            for(int c=0;c<mesh.dim;++c)
                add_dirichlet_bc(mesh,params,bc,velocity_dof(mesh,n,c),0.0,"INACTIVE_MODEL_VELOCITY_DOF");
    }
 
    if(has_displacement_field(mesh) && !contains_string(active,"displacement")){
        for(int n=0;n<static_cast<int>(mesh.displacement_nodes.size());++n)
            for(int c=0;c<mesh.dim;++c)
                add_dirichlet_bc(mesh,params,bc,displacement_dof(mesh,n,c),0.0,"INACTIVE_MODEL_DISPLACEMENT_DOF");
    }
 
    if(!contains_string(active,"pressure")){
        for(int n=0;n<static_cast<int>(mesh.pressure_nodes.size());++n)
            add_dirichlet_bc(mesh,params,bc,pressure_dof(mesh,n),0.0,"INACTIVE_MODEL_PRESSURE_DOF");
    }
}
