static void apply_field_dirichlet_bcs_to_domain(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    double time,
    BoundaryData& bc
){
    const DofLayout& gmesh = get_cached_dof_layout(mesh);
    for(const auto& rule: params.field_dirichlet_bcs){
        const int fid = find_field_index(gmesh,rule.field,false);
        if(fid<0) continue; // A global BC file may contain fields absent from this mesh/domain.
        const FieldInfo& field = gmesh.fields[fid];
        if(rule.component<0 || rule.component>=field.components){
            std::ostringstream os;
            os << "Dirichlet component " << rule.component << " is incompatible with field '"
               << rule.field << "' which has " << field.components << " component(s).";
            throw std::runtime_error(os.str());
        }
        BCSelector selector = rule.selector;
        if(selector.kind==BCSelectorKind::BoundaryGroup && selector.group.empty()) selector.group=rule.group;
        const std::set<int> nodes = resolve_dirichlet_nodes_from_selector(field,selector,mesh.dim,rule.field);
 
        std::map<int,double> parabolic;
        if(rule.field=="velocity" && selector.kind==BCSelectorKind::BoundaryGroup)
            parabolic = compute_velocity_boundary_parabolic_profile(mesh,selector.group);
 
        add_field_dirichlet_nodes(mesh, gmesh, field, fid, rule, nodes, parabolic,
                                  params, time, 0, bc, "FIELD_DIRICHLET");
    }
}
