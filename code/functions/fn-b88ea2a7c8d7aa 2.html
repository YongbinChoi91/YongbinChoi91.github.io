static void compact_mixed_mesh_to_active_nodes(FieldMesh& mesh){
    const int nv_old=static_cast<int>(mesh.velocity_nodes.size());
    const int np_old=static_cast<int>(mesh.pressure_nodes.size());
    const int nu_old=static_cast<int>(mesh.displacement_nodes.size());
 
    const std::vector<int> vmap=make_used_node_map(nv_old,mesh.velocity_elements);
    const std::vector<int> pmap=make_used_node_map(np_old,mesh.pressure_elements);
 
    mesh.velocity_nodes=compact_nodes_by_map(mesh.velocity_nodes,vmap);
    mesh.pressure_nodes=compact_nodes_by_map(mesh.pressure_nodes,pmap);
    remap_element_connectivity(mesh.velocity_elements,vmap,"velocity");
    remap_element_connectivity(mesh.pressure_elements,pmap,"pressure");
    mesh.velocity_boundaries=remap_boundary_map(mesh.velocity_boundaries,vmap);
    mesh.pressure_boundaries=remap_boundary_map(mesh.pressure_boundaries,pmap);
 
    if(!mesh.displacement_elements.empty() && !mesh.displacement_nodes.empty()){
        const std::vector<int> umap=make_used_node_map(nu_old,mesh.displacement_elements);
        mesh.displacement_nodes=compact_nodes_by_map(mesh.displacement_nodes,umap);
        remap_element_connectivity(mesh.displacement_elements,umap,"displacement");
        mesh.displacement_boundaries=remap_boundary_map(mesh.displacement_boundaries,umap);
    }
}
