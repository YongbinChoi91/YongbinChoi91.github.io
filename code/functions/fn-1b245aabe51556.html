static std::array<double,3> boundary_group_center_from_velocity_nodes(
    const FieldMesh& mesh,
    const std::vector<BoundaryEntity>& entities
){
    std::array<double,3> c{{0.0,0.0,0.0}};
    std::vector<char> used(mesh.velocity_nodes.size(),0);
    int count=0;
    for(const auto& ent:entities){
        for(int n:ent.node_ids){
            if(n<0 || n>=static_cast<int>(mesh.velocity_nodes.size()) || used[n]) continue;
            used[n]=1;
            for(int i=0;i<3;++i) c[i]+=mesh.velocity_nodes[n].x[i];
            ++count;
        }
    }
    if(count>0) for(int i=0;i<3;++i) c[i]/=static_cast<double>(count);
    return c;
}
