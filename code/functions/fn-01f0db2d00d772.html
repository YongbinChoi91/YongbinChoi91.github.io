static void merge_history_diagnostics(HistoryLocalDiagnostics& dst,const HistoryLocalDiagnostics& src){
    if(!src.active) return;
    if(!dst.active){
        dst = src;
        return;
    }
    if(dst.name=="none") dst.name = src.name;
    dst.total_gp_count += src.total_gp_count;
    dst.yielded_gp_count += src.yielded_gp_count;
    dst.local_failure_count += src.local_failure_count;
    if(src.max_residual > dst.max_residual){
        dst.max_residual = src.max_residual;
        dst.worst_domain = src.worst_domain;
        dst.worst_element = src.worst_element;
        dst.worst_gp = src.worst_gp;
    }
    if(src.max_iterations > dst.max_iterations) dst.max_iterations = src.max_iterations;
}
