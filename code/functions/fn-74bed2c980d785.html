inline std::array<std::array<Scalar, 2>, 2> inv(const std::array<std::array<Scalar, 2>, 2>& F) {
    const Scalar J = det(F);
    auto A = zero_tensor2<Scalar, 2, 2>();
    A[0][0] =  F[1][1] / J; A[0][1] = -F[0][1] / J;
    A[1][0] = -F[1][0] / J; A[1][1] =  F[0][0] / J;
    return A;
}
