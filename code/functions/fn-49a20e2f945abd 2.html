static std::vector<AD2> shape_values_Tet10_ad2(const AD2& xi,const AD2& eta,const AD2& zeta) {
    const AD2 L1 = 1.0 - xi - eta - zeta;
    const AD2 L2 = xi;
    const AD2 L3 = eta;
    const AD2 L4 = zeta;
    std::vector<AD2> N(10);
    N[0] = L1*(2.0*L1 - 1.0);
    N[1] = L2*(2.0*L2 - 1.0);
    N[2] = L3*(2.0*L3 - 1.0);
    N[3] = L4*(2.0*L4 - 1.0);
    N[4] = 4.0*L1*L2;
    N[5] = 4.0*L2*L3;
    N[6] = 4.0*L3*L1;
    N[7] = 4.0*L1*L4;
    N[8] = 4.0*L2*L4;
    N[9] = 4.0*L3*L4;
    return N;
}
