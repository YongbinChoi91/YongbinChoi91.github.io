static void apply_generic_field_dirichlet_bcs_shared(
    const Problem& problem,
    const ProblemParameters& params,
    double time,
    BoundaryData& bc
){
    if(problem.domains.empty()) return;
    const FieldMesh& rep=problem.domains.front().mesh;
    const DofLayout& rep_gmesh=get_cached_dof_layout(rep);
 
    auto domain_is_assembled=[](const Domain& d){
        return d.type==DomainType::Fluid || d.type==DomainType::Solid;
    };
 
    for(const auto& rule: params.field_dirichlet_bcs){
        const int fid=find_field_index(rep_gmesh,rule.field,false);
        if(fid<0) continue;
        const FieldInfo& field=rep_gmesh.fields[fid];
        if(rule.component<0 || rule.component>=field.components){
            std::ostringstream os;
            os << "Dirichlet component " << rule.component << " is incompatible with field '"
               << rule.field << "' which has " << field.components << " component(s).";
            throw std::runtime_error(os.str());
        }
 
        BCSelector selector=rule.selector;
        if(selector.kind==BCSelectorKind::BoundaryGroup && selector.group.empty()) selector.group=rule.group;
        std::set<int> nodes;
        bool found=false;
        if(selector.kind==BCSelectorKind::BoundaryGroup){
            for(const auto& domain: problem.domains){
                if(!domain_is_assembled(domain)) continue;
                std::string group;
                const bool qualified=strip_domain_group_prefix(rule.group,domain.name,group);
                if(!qualified) group=rule.group;
                const bool has_boundary = generic_boundary_exists(domain.mesh,rule.field,group);
                if(qualified || has_boundary){
                    const std::size_t before=nodes.size();
                    add_unique_nodes_from_generic_boundary(nodes,domain.mesh,rule.field,group);
                    if(nodes.size()!=before || has_boundary) found=true;
                }
            }
            if(!found) throw std::runtime_error("Unknown Dirichlet boundary group in shared mesh for field '"+rule.field+"': "+rule.group);
        } else {
            nodes = resolve_dirichlet_nodes_from_selector(field,selector,rep.dim,rule.field);
            found = !nodes.empty();
        }
 
        std::map<int,double> parabolic;
        if(rule.field=="velocity") parabolic = compute_shared_velocity_nodes_parabolic_profile(rep,nodes);
 
        add_field_dirichlet_nodes(rep, rep_gmesh, field, fid, rule, nodes, parabolic,
                                  params, time, 0, bc, "SHARED_FIELD_DIRICHLET");
    }
}
