int default_gauss_order(ElementType type){
    switch(type){
        case ElementType::T3:    return 1;
        case ElementType::T6:    return 3;
        case ElementType::Q4:    return 2;
        case ElementType::Q9:    return 3;
        case ElementType::Tet4:  return 1;
        case ElementType::Tet10: return 4;
        case ElementType::H8:    return 2;
        case ElementType::H20:   return 3;
        case ElementType::H27:   return 3;
        default: throw std::runtime_error("Unknown element type in default_gauss_order.");
    }
}
