static FieldComponentTarget parse_field_component_target_tokens(
    const std::string& a,
    const std::string& b,
    const std::string& c,
    bool& explicit_field
){
    const std::string al=lower_copy(a);
    explicit_field = is_known_field_name(al);
    if(explicit_field){
        FieldComponentTarget t;
        t.field = (al=="theta") ? std::string("temperature") : al;
        t.group = b;
        t.selector = parse_bc_selector_token(b);
        t.component = (t.field=="pressure" || t.field=="temperature") ? scalar_component_index_from_string(c) : component_index_from_string(c);
        return t;
    }
    return target_from_alias(a,b);
}
