inline AD1<Scalar, N> sqrt(const AD1<Scalar, N>& a) {
    using std::sqrt;
    const Scalar s = sqrt(a.val);
    AD1<Scalar, N> c(s);
    for (std::size_t i = 0; i < N; ++i) c.grad[i] = Scalar(0.5) * a.grad[i] / s;
    return c;
}
