JacobianData compute_jacobian_data(const ElementGeometry& elem,const ShapeData& s){
    if(elem.coords.size()!=s.N.size()) throw std::runtime_error("Coordinate/shape size mismatch.");
    JacobianData J; J.dim=elem.dim;
    for(size_t a=0;a<elem.coords.size();++a){
        for(int A=0; A<elem.dim; ++A){
            for(int i=0; i<elem.dim; ++i){
                J.J[A][i] += s.dN_dxi[a][A] * elem.coords[a][i];
            }
        }
    }
    if(elem.dim==2){
        const double A=J.J[0][0], B=J.J[0][1], C=J.J[1][0], D=J.J[1][1];
        J.detJ=A*D-B*C;
        if(std::abs(J.detJ)<1.0e-14) throw std::runtime_error("Jacobian determinant is too close to zero.");
        J.J_inv[0][0]= D/J.detJ; J.J_inv[0][1]=-B/J.detJ;
        J.J_inv[1][0]=-C/J.detJ; J.J_inv[1][1]= A/J.detJ;
    } else if(elem.dim==3){
        J.detJ=det3(J.J);
        if(std::abs(J.detJ)<1.0e-14) throw std::runtime_error("Jacobian determinant is too close to zero.");
        const auto& A=J.J; const double id=1.0/J.detJ;
        J.J_inv[0][0]=(A[1][1]*A[2][2]-A[1][2]*A[2][1])*id;
        J.J_inv[0][1]=-(A[0][1]*A[2][2]-A[0][2]*A[2][1])*id;
        J.J_inv[0][2]=(A[0][1]*A[1][2]-A[0][2]*A[1][1])*id;
        J.J_inv[1][0]=-(A[1][0]*A[2][2]-A[1][2]*A[2][0])*id;
        J.J_inv[1][1]=(A[0][0]*A[2][2]-A[0][2]*A[2][0])*id;
        J.J_inv[1][2]=-(A[0][0]*A[1][2]-A[0][2]*A[1][0])*id;
        J.J_inv[2][0]=(A[1][0]*A[2][1]-A[1][1]*A[2][0])*id;
        J.J_inv[2][1]=-(A[0][0]*A[2][1]-A[0][1]*A[2][0])*id;
        J.J_inv[2][2]=(A[0][0]*A[1][1]-A[0][1]*A[1][0])*id;
    } else throw std::runtime_error("Unsupported dimension in Jacobian.");
    return J;
}
