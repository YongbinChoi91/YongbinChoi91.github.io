static void gauss_legendre_1d_interval(int n,double a,double b,std::vector<double>& x,std::vector<double>& w){
    if(n<=0) throw std::runtime_error("Gauss order must be positive.");
    x.assign(n,0.0); w.assign(n,0.0);
    const double eps=1.0e-14;
    const int m=(n+1)/2;
    const double xm=0.5*(a+b);
    const double xl=0.5*(b-a);
    for(int i=0;i<m;++i){
        double z=std::cos(3.141592653589793238462643383279502884*(i+0.75)/(n+0.5));
        double z1=0.0, pp=0.0;
        for(int its=0; its<100; ++its){
            double p1=1.0, p2=0.0;
            for(int j=1;j<=n;++j){
                const double p3=p2;
                p2=p1;
                p1=((2.0*j-1.0)*z*p2-(j-1.0)*p3)/j;
            }
            pp=n*(z*p1-p2)/(z*z-1.0);
            z1=z;
            z=z1-p1/pp;
            if(std::abs(z-z1)<=eps) break;
        }
        x[i]=xm-xl*z;
        x[n-1-i]=xm+xl*z;
        const double wi=2.0*xl/((1.0-z*z)*pp*pp);
        w[i]=wi;
        w[n-1-i]=wi;
    }
}
