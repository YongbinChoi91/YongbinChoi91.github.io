void add_field_dirichlet_nodes(
    const FieldMesh& dof_mesh,
    const DofLayout& dof_layout,
    const FieldInfo& field,
    int fid,
    const FieldDirichletBC& rule,
    const std::set<int>& nodes,
    const std::map<int,double>& parabolic_profile,
    const ProblemParameters& params,
    double time,
    int global_offset,
    BoundaryData& bc,
    const std::string& source_prefix
){
    const std::string source = source_prefix+" field="+rule.field+" group="+rule.group;
    for(int node: nodes){
        if(node<0 || node>=static_cast<int>(field.nodes.size())){
            throw std::runtime_error("Dirichlet boundary node id out of range for field '"+rule.field+"', group '"+rule.group+"'.");
        }
        const auto& X = field.nodes[node].x;
        double value=0.0;
        if(rule.field=="velocity"){
            const auto pit=parabolic_profile.find(node);
            const double profile=(pit==parabolic_profile.end()) ? 0.0 : pit->second;
            value=evaluate_dirichlet_value(rule.value,X,time,profile,params.material.u_bar);
        } else {
            value=eval_scalar_function(rule.value,X[0],X[1],X[2],time);
        }
        add_dirichlet_bc(dof_mesh,params,bc,global_offset + global_field_dof(dof_layout,fid,node,rule.component),value,source);
    }
}
