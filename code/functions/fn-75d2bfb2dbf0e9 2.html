inline std::vector<HistorySpec> problem_history_specs_summary(const Problem& problem) {
    std::vector<HistorySpec> specs;
    for (const auto& domain : problem.domains) {
        merge_history_specs_into(specs, domain.mesh.history_specs);
    }
    return specs;
}
