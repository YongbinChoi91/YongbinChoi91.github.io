static int vtk_plastic_gamma_history_index(const MaterialHistoryStore* history) {
    const int nvars = (history && !history->domains.empty())
        ? history_nvars_or_zero(history->domains.front().specs, "plasticity")
        : 0;
    return nvars >= 7 ? 6 : 4;
}
