    int dof(const std::string& name, int node, int component = 0) const {
        const auto& f = field(name);
        if (node < 0 || node >= f.n_nodes) {
            throw std::runtime_error("Local node index out of range for field " + name + ".");
        }
        if (component < 0 || component >= f.components) {
            throw std::runtime_error("Local component index out of range for field " + name + ".");
        }
        return f.local_offset + f.components * node + component;
    }
