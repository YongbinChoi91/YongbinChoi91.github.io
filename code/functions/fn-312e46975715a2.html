static FieldInfo make_builtin_field_info(const FieldMesh& mesh,const std::string& name){
    FieldInfo f;
    f.name=name;
    if(name=="velocity"){
        f.components=mesh.dim;
        f.role=FieldRole::Mechanical;
        f.element_type=mesh.velocity_type;
        f.nodes=mesh.velocity_nodes;
        f.elements=mesh.velocity_elements;
        f.boundaries=mesh.velocity_boundaries;
        return f;
    }
    if(name=="displacement"){
        if(!has_displacement_field(mesh)) throw std::runtime_error("Field 'displacement' requested in DOFS_info.txt but the mesh has no displacement field.");
        f.components=mesh.dim;
        f.role=FieldRole::Mechanical;
        f.element_type=mesh.displacement_type;
        f.nodes=mesh.displacement_nodes;
        f.elements=mesh.displacement_elements;
        f.boundaries=mesh.displacement_boundaries;
        return f;
    }
    if(name=="pressure"){
        f.components=1;
        f.role=FieldRole::Constraint;
        f.element_type=mesh.pressure_type;
        f.nodes=mesh.pressure_nodes;
        f.elements=mesh.pressure_elements;
        f.boundaries=mesh.pressure_boundaries;
        return f;
    }
    for(const auto& e:mesh.extra_fields){
        if(e.name==name) return e;
    }
 
    // Convenience default for the first thermo-coupled extension: if a user adds
    //   field_key ... temperature ...
    // before exporting explicit TEMPERATURE_NODES/ELEMENTS, use the velocity
    // interpolation space.  Constitutive kernels still have to add the heat
    // residual through L.dof("temperature",...).
    if(name=="temperature"){
        f.components=1;
        f.role=FieldRole::Thermal;
        f.element_type=mesh.velocity_type;
        f.nodes=mesh.velocity_nodes;
        f.elements=mesh.velocity_elements;
        f.boundaries=mesh.velocity_boundaries;
        return f;
    }
 
    throw std::runtime_error("Unknown field in DOFS_info.txt field_key list: "+name);
}
