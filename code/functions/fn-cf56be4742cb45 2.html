Vector solve_linear_system_backend(
    const CSRMatrix& A,
    const Vector& rhs,
    const LinearSolverSettings& settings,
    const CSRMatrix* preconditioner_matrix,
    const PetscPCDData* pcd_data
){
    const std::string b = lower_copy(settings.backend);
    if(b=="umfpack" || b=="suite_sparse" || b=="suitesparse"){
#ifdef USE_UMFPACK
        return solve_with_umfpack_backend(A, rhs, settings);
#else
        (void)A; (void)rhs;
        throw std::runtime_error("UMFPACK requested, but code was not compiled with -DUSE_UMFPACK.");
#endif
    }
    if(b=="petsc"){
#ifdef USE_PETSC
        return solve_with_petsc_backend(A, rhs, settings, preconditioner_matrix, pcd_data);
#else
        (void)A; (void)rhs; (void)preconditioner_matrix; (void)pcd_data;
        throw std::runtime_error("PETSc requested, but code was not compiled with -DUSE_PETSC.");
#endif
    }
    if(b=="pardiso" || b=="mkl_pardiso"){
#ifdef USE_PARDISO
        return solve_with_pardiso_backend(A, rhs, settings);
#else
        (void)A; (void)rhs;
        throw std::runtime_error("PARDISO requested, but code was not compiled with -DUSE_PARDISO.");
#endif
    }
    throw std::runtime_error("Unknown solver_backend: "+settings.backend);
}
