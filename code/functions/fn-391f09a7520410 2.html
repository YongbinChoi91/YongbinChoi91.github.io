CSRMatrix build_global_sparsity_pattern(const Problem& problem,const ProblemElementDofMap& dof_maps){
    const int n=problem.total_dofs;
    CSRMatrix K; K.n_rows=n; K.n_cols=n; K.row_ptr.assign(n+1,0);
 
    // Allocation-friendly path.  The caller can pass precomputed element DOFs so
    // sparsity construction, tangent-index construction, and Newton assembly all
    // reuse the same connectivity map.
    std::vector<std::vector<int>> rows(n);
    for(const auto& domain_maps: dof_maps){
        for(const auto& dofs: domain_maps){
            for(int A:dofs){
                if(A<0 || A>=n) throw std::runtime_error("DOF out of range while building sparsity pattern.");
                auto& row = rows[A];
                row.insert(row.end(), dofs.begin(), dofs.end());
            }
        }
    }
 
    int nnz=0;
    for(int i=0;i<n;++i){
        auto& row = rows[i];
        std::sort(row.begin(), row.end());
        row.erase(std::unique(row.begin(), row.end()), row.end());
        K.row_ptr[i]=nnz;
        nnz+=static_cast<int>(row.size());
    }
    K.row_ptr[n]=nnz;
    K.col_idx.reserve(nnz);
    K.values.assign(nnz,0.0);
    for(int i=0;i<n;++i){
        K.col_idx.insert(K.col_idx.end(), rows[i].begin(), rows[i].end());
    }
    return K;
}
