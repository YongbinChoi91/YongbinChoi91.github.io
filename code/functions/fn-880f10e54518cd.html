inline std::vector<std::vector<Scalar>> embed_3d(
    const std::vector<std::vector<Scalar>>& A,
    int dim
) {
    if (dim == 3) return A;
    if (dim != 2) {
        throw std::runtime_error("3D embedding supports only 2D or 3D inputs.");
    }
    auto A3 = eye<Scalar>(3);
    for (int i = 0; i < 2; ++i) {
        for (int j = 0; j < 2; ++j) {
            A3[i][j] = A[i][j];
        }
    }
    return A3;
}
