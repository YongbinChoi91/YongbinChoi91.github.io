static int recover_neumann_local_edge_2d(
    const FieldInfo& field,
    int elem_id,
    const BoundaryEntity& ent
){
    if(elem_id<0 || elem_id>=static_cast<int>(field.elements.size())) return -1;
    const auto& conn = field.elements[elem_id];
    const auto edges = reference_edges(field.element_type);
 
    std::vector<int> local_nodes;
    local_nodes.reserve(ent.node_ids.size());
    for(int node: ent.node_ids){
        auto it = std::find(conn.begin(), conn.end(), node);
        if(it==conn.end()) return -1;
        local_nodes.push_back(static_cast<int>(it-conn.begin()));
    }
 
    for(int edge=0; edge<static_cast<int>(edges.size()); ++edge){
        bool ok=true;
        for(int ln: local_nodes){
            if(std::find(edges[edge].begin(),edges[edge].end(),ln)==edges[edge].end()){
                ok=false; break;
            }
        }
        if(ok) return edge;
    }
 
    // Robust fallback: if the exporter wrote only corners or used a different
    // midside ordering, choose the edge with the largest overlap.  At least two
    // boundary nodes must match one edge for line traction integration.
    int best=-1, best_count=-1;
    for(int edge=0; edge<static_cast<int>(edges.size()); ++edge){
        int count=0;
        for(int ln: local_nodes){
            if(std::find(edges[edge].begin(),edges[edge].end(),ln)!=edges[edge].end()) ++count;
        }
        if(count>best_count){ best_count=count; best=edge; }
    }
    const int required = std::min<int>(2, static_cast<int>(local_nodes.size()));
    return (best>=0 && best_count>=required) ? best : -1;
}
