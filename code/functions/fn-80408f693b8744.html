static TimeSolverSettings::HysteresisAxisSpec default_hysteresis_y_axis(const ProblemParameters& params){
    TimeSolverSettings::HysteresisAxisSpec a;
    a.enabled=true;
    const std::string sign=lower_copy(params.solver.hysteresis_sign);
    if(sign=="reaction" || sign=="force" || sign=="external") a.mode="reaction";
    else if(sign=="residual" || sign=="internal") a.mode="residual";
    else throw std::runtime_error("Unknown hysteresis_sign: "+params.solver.hysteresis_sign+". Use reaction or residual.");
    a.target=params.solver.hysteresis_target;
    a.scale=1.0;
    a.offset=0.0;
    a.reduction="sum";
    a.label=a.mode + " " + hysteresis_label(a.target);
    return a;
}
