void append_minj_csv(const std::string& filename,const MinJacobianResult& result){
    ensure_output_parent_directory(filename);
    std::ofstream out(filename,std::ios::app);
    if(!out) throw std::runtime_error("Could not open minJ CSV for appending: "+filename);
    out.setf(std::ios::scientific);
    out.precision(17);
    out << result.time << ',' << result.minJ << '\n';
}
