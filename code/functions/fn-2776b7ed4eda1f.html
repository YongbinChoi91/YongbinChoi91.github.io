inline Vec2T<Scalar> linear_projected_scalar_gradient(
    const LinearElementCache& cache,
    const std::vector<Scalar>& elem
) {
    const auto& U = cache.layout.field(cache.field_name);
    if (U.components != 1) {
        throw std::runtime_error("Scalar linear VEM projection requires a one-component field.");
    }
    Vec2T<Scalar> grad{{Scalar(0.0), Scalar(0.0)}};
    for (int ar = 0; ar < cache.n_real; ++ar) {
        const int br = (ar + 1) % cache.n_real;
        const int a_slot = cache.real_slots[ar];
        const int b_slot = cache.real_slots[br];
        const Scalar value_mid = Scalar(0.5) * (
            elem[hpx::local_dof(U, a_slot, 0)] + elem[hpx::local_dof(U, b_slot, 0)]
        );
        for (int A = 0; A < 2; ++A) {
            grad[A] += value_mid * Scalar(cache.edge_normal_scaled[ar][A] / cache.area);
        }
    }
    return grad;
}
