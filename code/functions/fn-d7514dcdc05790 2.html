std::vector<char> make_fixed_dof_mask(int ndofs,const BoundaryData& bc){
    std::vector<char> fixed(ndofs,0);
    for(int d:bc.dofs){
        if(d<0 || d>=ndofs) throw std::runtime_error("Fixed DOF out of range.");
        fixed[d]=1;
    }
    return fixed;
}
