inline auto grad(F&& f, const std::array<Scalar, N>& x) {
    auto x_ad = detail::make_active_vector(x);
    auto y = f(x_ad);
    if constexpr (is_ad1_v<decltype(y)>) {
        return detail::unflatten_vector_gradient(detail::extract_output_gradient(y));
    } else {
        return detail::extract_output_gradient(y);
    }
}
