static ScalarFunction make_torsion_angle_function(const std::string& unit_token,const std::string& expr_in){
    if(expr_in.empty()) throw std::runtime_error("Bad TORSION_DIRICHLET line: missing angle expression.");
    const std::string u = lower_copy(unit_token);
    if(u=="angle_deg" || u=="deg" || u=="degree" || u=="degrees"){
        return make_scalar_function("(" + expr_in + ")*pi/180.0");
    }
    if(u=="angle_rad" || u=="rad" || u=="radian" || u=="radians" || u=="angle"){
        return make_scalar_function(expr_in);
    }
    throw std::runtime_error("Bad TORSION_DIRICHLET angle unit. Use angle_deg or angle_rad.");
}
