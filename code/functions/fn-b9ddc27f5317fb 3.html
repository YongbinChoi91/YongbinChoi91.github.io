static std::vector<GaussPoint> triangle_duffy_rule(int order){
    if(order<=1) return {{1.0/3.0,1.0/3.0,0.0,0.5}};
    std::vector<double> r,wr;
    gauss_legendre_1d_interval(order,0.0,1.0,r,wr);
    std::vector<GaussPoint> gps;
    gps.reserve(order*order);
    for(int i=0;i<order;++i){
        for(int j=0;j<order;++j){
            const double xi=r[i];
            const double eta=(1.0-r[i])*r[j];
            const double wt=wr[i]*wr[j]*(1.0-r[i]);
            gps.push_back({xi,eta,0.0,wt});
        }
    }
    return gps;
}
