static ReducedSystemWorkspace make_reduced_system_workspace(
    const CSRMatrix& K,
    const BoundaryData& bc
){
    const int ndofs = K.n_rows;
    if(K.n_cols != ndofs) throw std::runtime_error("Reduced workspace requires a square matrix.");
 
    ReducedSystemWorkspace w;
    w.fixed_mask.assign(ndofs,0);
    for(int d:bc.dofs){
        if(d<0 || d>=ndofs) throw std::runtime_error("Fixed DOF out of range while building reduced workspace.");
        w.fixed_mask[d]=1;
    }
 
    w.free_dofs.reserve(std::max(0, ndofs-static_cast<int>(bc.dofs.size())));
    std::vector<int> old_to_new(ndofs,-1);
    for(int i=0;i<ndofs;++i){
        if(!w.fixed_mask[i]){
            old_to_new[i] = static_cast<int>(w.free_dofs.size());
            w.free_dofs.push_back(i);
        }
    }
    if(w.free_dofs.empty()) throw std::runtime_error("No free DOFs in reduced workspace.");
 
    const int nf = static_cast<int>(w.free_dofs.size());
    w.Kff.n_rows = nf;
    w.Kff.n_cols = nf;
    w.Kff.row_ptr.assign(nf+1,0);
 
    int nnz=0;
    for(int i=0;i<nf;++i){
        const int row = w.free_dofs[i];
        w.Kff.row_ptr[i]=nnz;
        for(int p=K.row_ptr[row]; p<K.row_ptr[row+1]; ++p){
            const int j = old_to_new[K.col_idx[p]];
            if(j>=0) ++nnz;
        }
    }
    w.Kff.row_ptr[nf]=nnz;
    w.Kff.col_idx.reserve(nnz);
    w.Kff.values.assign(nnz,0.0);
    w.kff_value_from_full.reserve(nnz);
 
    for(int i=0;i<nf;++i){
        const int row = w.free_dofs[i];
        for(int p=K.row_ptr[row]; p<K.row_ptr[row+1]; ++p){
            const int j = old_to_new[K.col_idx[p]];
            if(j>=0){
                w.Kff.col_idx.push_back(j);
                w.kff_value_from_full.push_back(p);
            }
        }
    }
    return w;
}
