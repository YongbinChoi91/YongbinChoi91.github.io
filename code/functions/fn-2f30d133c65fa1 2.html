inline ElementSystem scalar_weak_form_element_system(
    const Vector& x_elem,
    Eval&& eval
) {
    const int n = static_cast<int>(x_elem.size());
    if (n <= 8) return detail::scalar_weak_form_element_system_slfad<8>(n, x_elem, std::forward<Eval>(eval));
    if (n <= 12) return detail::scalar_weak_form_element_system_slfad<12>(n, x_elem, std::forward<Eval>(eval));
    if (n <= 18) return detail::scalar_weak_form_element_system_slfad<18>(n, x_elem, std::forward<Eval>(eval));
    if (n <= 24) return detail::scalar_weak_form_element_system_slfad<24>(n, x_elem, std::forward<Eval>(eval));
    if (n <= 32) return detail::scalar_weak_form_element_system_slfad<32>(n, x_elem, std::forward<Eval>(eval));
    // if (n <= 64) return detail::scalar_weak_form_element_system_slfad<64>(n, x_elem, std::forward<Eval>(eval));
    return detail::scalar_weak_form_element_system_dfad(n, x_elem, std::forward<Eval>(eval));
}
