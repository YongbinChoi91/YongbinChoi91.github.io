static AssemblyMpiInfo get_assembly_mpi_info() {
    AssemblyMpiInfo info;
#ifdef USE_MPI
    int initialized = 0;
    MPI_Initialized(&initialized);
    if (initialized) {
        MPI_Comm_rank(MPI_COMM_WORLD, &info.rank);
        MPI_Comm_size(MPI_COMM_WORLD, &info.size);
        info.active = info.size > 1;
    }
#endif
    return info;
}
