DomainType domain_type_from_string(const std::string& type){
    const std::string k=lower_copy(type);
    if(k=="fluid" || k=="fluid_domain") return DomainType::Fluid;
    if(k=="solid" || k=="solid_domain") return DomainType::Solid;
    if(k=="thermal" || k=="temperature" || k=="thermal_domain") return DomainType::Thermal;
    return DomainType::Unknown;
}
