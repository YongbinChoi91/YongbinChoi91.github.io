static void gather_local_vector_into(const Vector& global,const std::vector<int>& dofs,Vector& local){
    const int n=static_cast<int>(dofs.size());
    // Hot path: keep capacity across elements/Newton iterations.  assign(n,0)
    // repeatedly reallocates and zero-fills; here every entry is overwritten.
    if(static_cast<int>(local.size()) < n) local.resize(n);
    for(int i=0;i<n;++i){
        const int I=dofs[i];
        if(I<0 || I>=static_cast<int>(global.size())) throw std::runtime_error("Local extraction DOF out of range.");
        local[i]=global[I];
    }
    if(static_cast<int>(local.size()) != n) local.resize(n);
}
