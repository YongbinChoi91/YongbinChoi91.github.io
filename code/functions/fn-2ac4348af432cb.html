inline TimeDofStencil BDF(int order, double dt, double dt_previous, int step) {
    if (dt <= 0.0) throw std::runtime_error("Time step dt must be positive.");
    if (order != 2) return BDF(order);
 
    if (step < 2) {
        return {"BDF", 2, {1.0, -1.0, 0.0}};
    }
 
    if (dt_previous <= 0.0) dt_previous = dt;
    const double h = dt;
    const double k = dt_previous;
    return {
        "BDF",
        2,
        {
            (2.0 * h + k) / (h + k),
            -(h + k) / k,
            (h * h) / (k * (h + k))
        }
    };
}
