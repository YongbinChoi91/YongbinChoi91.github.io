inline std::vector<Scalar> field_value_as(
    const LocalFieldView& L,
    const std::string& field_name,
    const Vec& x_elem,
    const std::vector<double>& N
) {
    const auto& Fld = L.field(field_name);
    if (Fld.components <= 0) {
        throw std::runtime_error("Field has non-positive component count: " + field_name);
    }
    if (static_cast<int>(N.size()) < Fld.n_nodes) {
        throw std::runtime_error("Shape-function size is smaller than field node count for " + field_name + ".");
    }
 
    std::vector<Scalar> value(Fld.components, Scalar(0.0));
    for (int a = 0; a < Fld.n_nodes; ++a) {
        const Scalar Na = Scalar(N[a]);
        const int base = Fld.local_offset + Fld.components * a;
        for (int i = 0; i < Fld.components; ++i) {
            value[i] += Na * Scalar(x_elem[base + i]);
        }
    }
    return value;
}
