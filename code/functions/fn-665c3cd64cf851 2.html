static PetscErrorCode destroy_pcd_shell(PC pc){
    PCDShellContext* ctx=nullptr;
    PetscCall(PCShellGetContext(pc,(void**)&ctx));
    if(ctx){
        if(ctx->kspAp) KSPDestroy(&ctx->kspAp);
        if(ctx->kspKp) KSPDestroy(&ctx->kspKp);
        if(ctx->tmp1) VecDestroy(&ctx->tmp1);
        if(ctx->tmp2) VecDestroy(&ctx->tmp2);
        if(ctx->Ap) MatDestroy(&ctx->Ap);
        if(ctx->Kp) MatDestroy(&ctx->Kp);
        delete ctx;
    }
    return 0;
}
