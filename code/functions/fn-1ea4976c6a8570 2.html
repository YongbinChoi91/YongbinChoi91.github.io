BoundaryData build_boundary_conditions(const FieldMesh& mesh,const ProblemParameters& params,double time,const std::vector<std::string>& active_fields,bool add_inactive_and_pin){
    BoundaryData bc;
 
    apply_field_dirichlet_bcs_to_domain(mesh,params,time,bc);
 
    apply_torsion_dirichlet_bcs_to_domain(mesh,params,time,bc);
 
    if(add_inactive_and_pin){
        constrain_unused_mesh_dofs(mesh,params,bc);
        constrain_inactive_constitutive_dofs(mesh,params,bc,active_fields);
 
        if(contains_string(active_fields,"pressure")){
            const int pin=(params.solver.pressure_pin_node>=0)?params.solver.pressure_pin_node:choose_pressure_pin_node(mesh);
            add_dirichlet_bc(mesh,params,bc,pressure_dof(mesh,pin),params.solver.pressure_pin_value,"PRESSURE_PIN");
        }
    }
    return bc;
}
