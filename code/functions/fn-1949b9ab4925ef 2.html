static int component_index_from_string(const std::string& s){
    const std::string k=lower_copy(s);
    if(k=="0" || k=="vx" || k=="ux" || k=="x") return 0;
    if(k=="1" || k=="vy" || k=="uy" || k=="y") return 1;
    if(k=="2" || k=="vz" || k=="uz" || k=="z") return 2;
    throw std::runtime_error("Unknown vector component: "+s);
}
