static void reduce_assembled_system_if_parallel(Vector& R, CSRMatrix& K, const AssemblyMpiInfo& mpi) {
#ifdef USE_MPI
    if (!mpi.active) return;
    if (!R.empty()) {
        MPI_Allreduce(MPI_IN_PLACE, R.data(), static_cast<int>(R.size()), MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
    }
    if (!K.values.empty()) {
        MPI_Allreduce(MPI_IN_PLACE, K.values.data(), static_cast<int>(K.values.size()), MPI_DOUBLE, MPI_SUM, MPI_COMM_WORLD);
    }
#else
    (void)R;
    (void)K;
    (void)mpi;
#endif
}
