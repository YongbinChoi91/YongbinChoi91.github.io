static void constrain_unused_mesh_dofs(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    BoundaryData& bc
){
    // Some shared multi-domain mesh exporters keep the full global node list in
    // every DOMAIN block, even when a domain uses only a small element subset.
    // Those unused local nodes generate global DOFs with no element stiffness
    // row, which makes the reduced Kff exactly singular. They are not physical
    // unknowns, so constrain them to zero automatically.
    std::vector<char> used_v(mesh.velocity_nodes.size(),0);
    for(const auto& e:mesh.velocity_elements) for(int n:e)
        if(n>=0 && n<static_cast<int>(used_v.size())) used_v[n]=1;
 
    for(int n=0;n<static_cast<int>(used_v.size());++n){
        if(!used_v[n]){
            for(int c=0;c<mesh.dim;++c)
                add_dirichlet_bc(mesh,params,bc,velocity_dof(mesh,n,c),0.0,"INACTIVE_VELOCITY_DOF");
        }
    }
 
    if(has_displacement_field(mesh)){
        std::vector<char> used_u(mesh.displacement_nodes.size(),0);
        for(const auto& e:mesh.displacement_elements) for(int n:e)
            if(n>=0 && n<static_cast<int>(used_u.size())) used_u[n]=1;
        for(int n=0;n<static_cast<int>(used_u.size());++n){
            if(!used_u[n]){
                for(int c=0;c<mesh.dim;++c)
                    add_dirichlet_bc(mesh,params,bc,displacement_dof(mesh,n,c),0.0,"INACTIVE_DISPLACEMENT_DOF");
            }
        }
    }
 
    std::vector<char> used_p(mesh.pressure_nodes.size(),0);
    for(const auto& e:mesh.pressure_elements) for(int n:e)
        if(n>=0 && n<static_cast<int>(used_p.size())) used_p[n]=1;
    for(int n=0;n<static_cast<int>(used_p.size());++n){
        if(!used_p[n]) add_dirichlet_bc(mesh,params,bc,pressure_dof(mesh,n),0.0,"INACTIVE_PRESSURE_DOF");
    }
}
