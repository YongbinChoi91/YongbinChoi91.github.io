static void interpolate_pressure_to_velocity_nodes(
    const FieldMesh& mesh,
    const Vector& x,
    std::vector<double>& p_at_v
){
    const int nv=static_cast<int>(mesh.velocity_nodes.size());
    const int np=static_cast<int>(mesh.pressure_nodes.size());
    p_at_v.assign(nv,0.0);
    std::vector<double> weight(nv,0.0);
 
    if(np==0 || mesh.pressure_elements.empty()) return;
 
    if(mesh.velocity_type==ElementType::Polygon && mesh.pressure_type==ElementType::Polygon){
        // Taylor-Hood VEM connectivity is [v0,m0,v1,m1,...] for velocity
        // and [p0,p1,...] for pressure. Linear pressure is exact at the
        // velocity vertices and averages to the quadratic edge midpoint.
        for(int e=0;e<static_cast<int>(mesh.velocity_elements.size());++e){
            const auto& ve=mesh.velocity_elements[e];
            const auto& pe=mesh.pressure_elements[e];
            if(ve.size()!=2*pe.size() || pe.empty()) continue;
            for(int a=0;a<static_cast<int>(pe.size());++a){
                const int vertex=ve[2*a];
                const int midpoint=ve[2*a+1];
                const int pa=pe[a];
                const int pb=pe[(a+1)%pe.size()];
                if(vertex>=0 && vertex<nv && pa>=0 && pa<np){
                    p_at_v[vertex]+=x[pressure_dof(mesh,pa)];
                    weight[vertex]+=1.0;
                }
                if(midpoint>=0 && midpoint<nv && pa>=0 && pa<np && pb>=0 && pb<np){
                    p_at_v[midpoint]+=0.5*(x[pressure_dof(mesh,pa)]+x[pressure_dof(mesh,pb)]);
                    weight[midpoint]+=1.0;
                }
            }
        }
        for(int a=0;a<nv;++a) if(weight[a]>0.0) p_at_v[a]/=weight[a];
        return;
    }
 
    // Primary path: element-wise interpolation from the pressure basis to every
    // velocity interpolation point.  This fixes Q9/Q4 output, where midside and
    // center velocity nodes otherwise received zero pressure when only exact
    // pressure-node coordinates were used.
    for(int e=0;e<static_cast<int>(mesh.velocity_elements.size());++e){
        const auto& ve=mesh.velocity_elements[e];
        const auto& pe=mesh.pressure_elements[e];
        for(int av=0; av<static_cast<int>(ve.size()); ++av){
            const int vnode=ve[av];
            if(vnode<0 || vnode>=nv) continue;
            const auto rc=reference_node_coordinate(mesh.velocity_type,av);
            const ShapeData sp=compute_shape(mesh.pressure_type,rc[0],rc[1],rc[2]);
            double p=0.0;
            const int m=std::min(static_cast<int>(sp.N.size()),static_cast<int>(pe.size()));
            for(int b=0;b<m;++b){
                const int pnode=pe[b];
                if(pnode>=0 && pnode<np) p += sp.N[b] * x[pressure_dof(mesh,pnode)];
            }
            p_at_v[vnode] += p;
            weight[vnode] += 1.0;
        }
    }
 
    for(int a=0;a<nv;++a){
        if(weight[a]>0.0) p_at_v[a] /= weight[a];
    }
 
    // Fallback for any isolated velocity point not touched by elements: map by
    // nearest coincident pressure coordinate with a practical tolerance.
    const double tol2=1.0e-18;
    for(int vnode=0; vnode<nv; ++vnode){
        if(weight[vnode]>0.0) continue;
        const auto& V=mesh.velocity_nodes[vnode].x;
        int best=-1;
        double best_d2=std::numeric_limits<double>::infinity();
        for(int pnode=0;pnode<np;++pnode){
            const auto& P=mesh.pressure_nodes[pnode].x;
            const double dx=P[0]-V[0], dy=P[1]-V[1], dz=P[2]-V[2];
            const double d2=dx*dx+dy*dy+dz*dz;
            if(d2<best_d2){ best_d2=d2; best=pnode; }
        }
        if(best>=0 && best_d2<=tol2) p_at_v[vnode]=x[pressure_dof(mesh,best)];
    }
}
