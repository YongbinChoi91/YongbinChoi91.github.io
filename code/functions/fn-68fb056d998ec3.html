std::vector<GaussPoint> gauss_rule(ElementType type,int order_per_direction){
    const int order = (order_per_direction > 0) ? order_per_direction : default_gauss_order(type);
    if(order<=0) throw std::runtime_error("Gauss order must be positive.");
    switch(type){
        case ElementType::T3:
        case ElementType::T6:
            return triangle_duffy_rule(order);
        case ElementType::Q4:
        case ElementType::Q9:
        case ElementType::H8:
        case ElementType::H20:
        case ElementType::H27:
            return tensor_product_rule(type, order);
        case ElementType::Tet4:
        case ElementType::Tet10:
            return tetra_duffy_rule(order);
        default:
            throw std::runtime_error("Unsupported element type in gauss_rule.");
    }
}
