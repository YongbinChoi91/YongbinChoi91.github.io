void write_vtk(
    const FieldMesh& mesh,
    const Vector& x,
    const std::string& filename,
    const MaterialHistoryStore* history
){
    ensure_output_parent_directory(filename);
    const int dim=mesh.dim;
    if(static_cast<int>(x.size()) != total_dof_count(mesh)) throw std::runtime_error("VTK vector size mismatch.");
 
    const bool write_on_velocity_mesh = !mesh.velocity_nodes.empty() && !mesh.velocity_elements.empty();
    const bool write_on_displacement_mesh = !write_on_velocity_mesh && has_displacement_field(mesh);
    if(!write_on_velocity_mesh && !write_on_displacement_mesh){
        throw std::runtime_error("VTK output needs either a velocity mesh or a displacement mesh.");
    }
 
    const auto& out_nodes = write_on_velocity_mesh ? mesh.velocity_nodes : mesh.displacement_nodes;
    const auto& out_elements = write_on_velocity_mesh ? mesh.velocity_elements : mesh.displacement_elements;
    const ElementType out_type = write_on_velocity_mesh ? mesh.velocity_type : mesh.displacement_type;
    const int nout=static_cast<int>(out_nodes.size());
    std::ofstream out(filename);
    if(!out) throw std::runtime_error("Could not open VTK file: "+filename);
 
    out << "# vtk DataFile Version 3.0\n";
    out << "mixed domain result\n";
    out << "ASCII\n";
    out << "DATASET UNSTRUCTURED_GRID\n";
    out << "POINTS " << nout << " double\n";
    for(const auto& n:out_nodes){
        out << n.x[0] << " " << n.x[1] << " " << (dim==3 ? n.x[2] : 0.0) << "\n";
    }
 
    std::vector<std::vector<int>> vtk_cells;
    vtk_cells.reserve(out_elements.size());
    for(const auto& e:out_elements){
        vtk_cells.push_back(make_vtk_visualization_cell(e,out_nodes,out_type,dim));
        if(static_cast<int>(vtk_cells.back().size())<dim+1){
            throw std::runtime_error("VTK visualization cell has too few nodes.");
        }
    }
 
    int cell_size=0;
    for(const auto& e:vtk_cells) cell_size += 1 + static_cast<int>(e.size());
    out << "CELLS " << vtk_cells.size() << " " << cell_size << "\n";
    for(const auto& e:vtk_cells){
        out << e.size();
        for(int node:e) out << " " << node;
        out << "\n";
    }
 
    out << "CELL_TYPES " << vtk_cells.size() << "\n";
    const int vtk_type=vtk_cell_type(out_type);
    for(size_t i=0;i<out_elements.size();++i) out << vtk_type << "\n";
 
    out << "POINT_DATA " << nout << "\n";
 
    if(write_on_velocity_mesh){
        out << "VECTORS velocity double\n";
        for(int a=0;a<nout;++a){
            const double vx=x[velocity_dof(mesh,a,0)];
            const double vy=(dim>=2)?x[velocity_dof(mesh,a,1)]:0.0;
            const double vz=(dim==3)?x[velocity_dof(mesh,a,2)]:0.0;
            out << vx << " " << vy << " " << vz << "\n";
        }
    }
    if(has_displacement_field(mesh)){
        out << "VECTORS displacement double\n";
        const int nud=static_cast<int>(mesh.displacement_nodes.size());
        for(int a=0;a<nout;++a){
            double ux=0.0, uy=0.0, uz=0.0;
            if(write_on_displacement_mesh){
                if(a<nud){
                    ux=x[displacement_dof(mesh,a,0)];
                    uy=(dim>=2)?x[displacement_dof(mesh,a,1)]:0.0;
                    uz=(dim==3)?x[displacement_dof(mesh,a,2)]:0.0;
                }
            } else if(a<nud){
                ux=x[displacement_dof(mesh,a,0)];
                uy=(dim>=2)?x[displacement_dof(mesh,a,1)]:0.0;
                uz=(dim==3)?x[displacement_dof(mesh,a,2)]:0.0;
            }
            out << ux << " " << uy << " " << uz << "\n";
        }
    }
 
    if(write_on_velocity_mesh){
        std::vector<double> p_at_v;
        interpolate_pressure_to_velocity_nodes(mesh,x,p_at_v);
        if(!p_at_v.empty()){
            out << "SCALARS pressure double 1\n";
            out << "LOOKUP_TABLE default\n";
            for(double p:p_at_v) out << p << "\n";
        }
    }
 
    if (vtk_has_plastic_history_for_cells(history, static_cast<int>(out_elements.size()))) {
        const int gamma_index = vtk_plastic_gamma_history_index(history);
        out << "CELL_DATA " << out_elements.size() << "\n";
        out << "SCALARS plastic_gamma double 1\n";
        out << "LOOKUP_TABLE default\n";
        for (const ElementMaterialHistory& eh : history->domains.front().elements) {
            out << average_gp_history_value(eh, gamma_index) << "\n";
        }
        out << "SCALARS hardening double 1\n";
        out << "LOOKUP_TABLE default\n";
        for (const ElementMaterialHistory& eh : history->domains.front().elements) {
            out << average_gp_history_value(eh, gamma_index) << "\n";
        }
    }
 
    // std::cout << "Wrote VTK file: " << filename << "\n";
}
