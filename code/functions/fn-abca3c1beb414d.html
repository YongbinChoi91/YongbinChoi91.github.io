int element_node_count(ElementType type){
    switch(type){
        case ElementType::T3: return 3;
        case ElementType::T6: return 6;
        case ElementType::Q4: return 4;
        case ElementType::Q9: return 9;
        case ElementType::Tet4: return 4;
        case ElementType::Tet10: return 10;
        case ElementType::H8: return 8;
        case ElementType::H20: return 20;
        case ElementType::H27: return 27;
        case ElementType::Polygon: return -1;
        default: return 0;
    }
}
