inline std::array<std::array<Scalar, C>, R> unflatten_tensor2_gradient(const std::array<Scalar, R * C>& g) {
    std::array<std::array<Scalar, C>, R> G{};
    for (std::size_t i = 0; i < R; ++i) for (std::size_t j = 0; j < C; ++j) G[i][j] = g[i * C + j];
    return G;
}
