static std::vector<AD2> shape_values_H8_ad2(const AD2& xi,const AD2& eta,const AD2& zeta) {
    const int sx[8]={-1, 1, 1,-1,-1, 1, 1,-1};
    const int sy[8]={-1,-1, 1, 1,-1,-1, 1, 1};
    const int sz[8]={-1,-1,-1,-1, 1, 1, 1, 1};
    std::vector<AD2> N(8);
    for(int a=0; a<8; ++a){
        const double X=sx[a], Y=sy[a], Z=sz[a];
        N[a] = 0.125*(1.0+X*xi)*(1.0+Y*eta)*(1.0+Z*zeta);
    }
    return N;
}
