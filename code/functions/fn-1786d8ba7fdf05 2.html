ShapeData compute_shape(ElementType type,double xi,double eta,double zeta){
    // Unified AD shape backend:
    //   N, dN/dxi and d2N/dxi2 are all evaluated from the same polynomial
    //   expression.  J, invJ and detJ are intentionally still computed by the
    //   explicit double-precision geometry routine below.
    const std::vector<AD2> Nad = evaluate_shape_ad2(type, xi, eta, zeta);
 
    ShapeData s;
    s.N.resize(Nad.size());
    s.dN_dxi.assign(Nad.size(), {0.0,0.0,0.0});
    for(std::size_t a=0; a<Nad.size(); ++a){
        s.N[a] = Nad[a].v;
        for(int A=0; A<3; ++A) s.dN_dxi[a][A] = Nad[a].d[A];
    }
    return s;
}
