inline AD1<Scalar, N> exp(const AD1<Scalar, N>& a) {
    using std::exp;
    const Scalar e = exp(a.val);
    AD1<Scalar, N> c(e);
    for (std::size_t i = 0; i < N; ++i) c.grad[i] = e * a.grad[i];
    return c;
}
