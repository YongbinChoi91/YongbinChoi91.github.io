inline Scalar det(const std::array<std::array<Scalar, 3>, 3>& F) {
    return F[0][0] * (F[1][1] * F[2][2] - F[1][2] * F[2][1])
         - F[0][1] * (F[1][0] * F[2][2] - F[1][2] * F[2][0])
         + F[0][2] * (F[1][0] * F[2][1] - F[1][1] * F[2][0]);
}
