static Problem read_shared_multi_domain_problem_from_lines(const std::vector<std::string>& lines,const std::string& filename,bool vem_taylor_hood){
    Problem problem;
    problem.shared_dofs=true;
 
    std::vector<Node> global_v_nodes;
    std::vector<Node> global_u_nodes;
    std::vector<Node> global_p_nodes;
    std::map<std::string,std::vector<BoundaryEntity>> global_v_boundaries;
    std::map<std::string,std::vector<BoundaryEntity>> global_u_boundaries;
    std::map<std::string,std::vector<BoundaryEntity>> global_p_boundaries;
    std::vector<std::vector<int>> global_total_velocity_elements;
 
    std::string section;
    std::string current_boundary;
    Domain current_domain;
    bool inside_domain=false;
    bool have_domain=false;
    int detected_dim=0;
 
    auto output_connectivity=[&](const std::vector<int>& conn){
        return vem_taylor_hood ? conn : reorder_quad_python_to_cpp(conn);
    };
 
    auto assign_node=[&](std::vector<Node>& nodes,int id,const std::vector<double>& vals,const std::string& what){
        if(vals.size()!=2 && vals.size()!=3) throw std::runtime_error(what+" node line needs id x y [z].");
        if(detected_dim==0) detected_dim=static_cast<int>(vals.size());
        if(static_cast<int>(vals.size())!=detected_dim) throw std::runtime_error("Inconsistent node dimensions in shared multi-domain mesh.");
        if(id>=static_cast<int>(nodes.size())) nodes.resize(id+1);
        nodes[id].x={{vals[0],vals[1],vals.size()==3?vals[2]:0.0}};
    };
 
    auto parse_mixed_connectivity_row=[&](const std::string& line,int& id,std::vector<int>& v,std::vector<int>& u,std::vector<int>& p,const std::string& where){
        const auto parts=split_pipe_fields(line);
        if(parts.size()!=3) throw std::runtime_error(where+" row must be 'id velocity | displacement | pressure'.");
        std::stringstream vss(parts[0]);
        read_nonnegative_id(vss,id,where);
        int n;
        while(vss>>n) v.push_back(n);
        u=parse_ints_from_string(parts[1],where+" displacement part");
        p=parse_ints_from_string(parts[2],where+" pressure part");
        if(v.empty()) throw std::runtime_error(where+" velocity part is empty.");
    };
 
    for(const std::string& original:lines){
        std::string clean=original;
        const auto comment=clean.find('#');
        if(comment!=std::string::npos) clean=clean.substr(0,comment);
        clean=trim_copy(clean);
        if(clean.empty()) continue;
 
        std::stringstream head(clean);
        std::string key; head >> key;
 
        if(key=="DIM" && !inside_domain){
            head >> problem.dim;
            if(problem.dim!=2 && problem.dim!=3) throw std::runtime_error("DIM must be 2 or 3 in shared multi-domain mesh.");
            detected_dim=problem.dim;
            continue;
        }
        if(key=="MESH_TYPE" && !inside_domain) continue;
        if(key=="FIELDS") continue;
 
        if(key=="DOMAIN"){
            if(inside_domain) throw std::runtime_error("Nested DOMAIN blocks are not allowed in shared multi-domain mesh.");
            inside_domain=true;
            have_domain=true;
            section.clear();
            current_domain=Domain{};
            head >> current_domain.name;
            if(current_domain.name.empty()) current_domain.name="domain"+std::to_string(problem.domains.size());
            current_domain.mesh.dim = problem.dim;
            current_domain.mesh.velocity_nodes = global_v_nodes;
            current_domain.mesh.displacement_nodes = global_u_nodes;
            current_domain.mesh.pressure_nodes = global_p_nodes;
            current_domain.mesh.velocity_boundaries = global_v_boundaries;
            current_domain.mesh.displacement_boundaries = global_u_boundaries;
            current_domain.mesh.pressure_boundaries = global_p_boundaries;
            current_domain.mesh.total_velocity_elements = global_total_velocity_elements;
            continue;
        }
        if(key=="END_DOMAIN"){
            if(!inside_domain) throw std::runtime_error("END_DOMAIN without DOMAIN in shared multi-domain mesh.");
            if(current_domain.type==DomainType::Unknown){
                current_domain.type = (lower_copy(current_domain.name).find("fluid")!=std::string::npos) ? DomainType::Fluid : DomainType::Unknown;
            }
            if(current_domain.type==DomainType::Unknown) throw std::runtime_error("DOMAIN "+current_domain.name+" has unknown TYPE.");
            validate_and_finalize_mixed_mesh_no_compaction(current_domain.mesh,"DOMAIN "+current_domain.name,vem_taylor_hood);
            current_domain.dof_offset=0;
            current_domain.n_dofs=total_dof_count(current_domain.mesh);
            problem.domains.push_back(std::move(current_domain));
            inside_domain=false;
            section.clear();
            continue;
        }
        if(inside_domain && key=="TYPE"){
            std::string t; head >> t;
            current_domain.type=domain_type_from_string(t);
            if(current_domain.type==DomainType::Unknown) throw std::runtime_error("Unknown DOMAIN TYPE in shared multi-domain mesh: "+t);
            continue;
        }
 
        if(!inside_domain){
            if(key=="VELOCITY_NODES" || key=="DISPLACEMENT_NODES" || key=="PRESSURE_NODES"){
                section=key;
                current_boundary.clear();
                continue;
            }
            if(key=="TOTAL_ELEMENTS" || key=="TOTAL_VELOCITY_ELEMENTS"){
                section="TOTAL_ELEMENTS";
                current_boundary.clear();
                continue;
            }
            if(key.rfind("MIXED_BOUNDARY_",0)==0){
                section="MIXED_BOUNDARY";
                current_boundary=key.substr(std::string("MIXED_BOUNDARY_").size());
                global_v_boundaries[current_boundary];
                global_u_boundaries[current_boundary];
                global_p_boundaries[current_boundary];
                continue;
            }
 
            std::stringstream ss(clean);
            if(section=="VELOCITY_NODES" || section=="DISPLACEMENT_NODES" || section=="PRESSURE_NODES"){
                int id=-1; read_nonnegative_id(ss,id,section);
                std::vector<double> vals; double q; while(ss>>q) vals.push_back(q);
                if(section=="VELOCITY_NODES") assign_node(global_v_nodes,id,vals,"velocity");
                else if(section=="DISPLACEMENT_NODES") assign_node(global_u_nodes,id,vals,"displacement");
                else assign_node(global_p_nodes,id,vals,"pressure");
                continue;
            }
            if(section=="TOTAL_ELEMENTS"){
                std::stringstream ss(clean);
                int id=-1;
                read_nonnegative_id(ss,id,"TOTAL_ELEMENTS");
                std::vector<int> v;
                int n;
                while(ss>>n) v.push_back(n);
                if(v.empty()) throw std::runtime_error("TOTAL_ELEMENTS row has no velocity connectivity.");
                if(id>=static_cast<int>(global_total_velocity_elements.size())) global_total_velocity_elements.resize(id+1);
                global_total_velocity_elements[id]=output_connectivity(v);
                continue;
            }
            if(section=="MIXED_BOUNDARY"){
                int id=-1; std::vector<int> v,u,pn;
                parse_mixed_connectivity_row(clean,id,v,u,pn,"MIXED_BOUNDARY_"+current_boundary);
                BoundaryEntity ev; ev.element_id=id; ev.node_ids=output_connectivity(v);
                BoundaryEntity eu; eu.element_id=id; eu.node_ids=output_connectivity(u);
                BoundaryEntity ep; ep.element_id=id; ep.node_ids=output_connectivity(pn);
                global_v_boundaries[current_boundary].push_back(std::move(ev));
                global_u_boundaries[current_boundary].push_back(std::move(eu));
                global_p_boundaries[current_boundary].push_back(std::move(ep));
                continue;
            }
            throw std::runtime_error("Unknown global shared multi-domain mesh line: "+clean);
        } else {
            if(key=="MIXED_ELEMENTS"){
                section="MIXED_ELEMENTS";
                continue;
            }
            if(key=="FIELDS") continue;
            if(section!="MIXED_ELEMENTS") throw std::runtime_error("Only MIXED_ELEMENTS data are allowed inside shared multi-domain DOMAIN blocks: "+clean);
 
            int id=-1; std::vector<int> v,u,pn;
            parse_mixed_connectivity_row(clean,id,v,u,pn,"DOMAIN "+current_domain.name+" MIXED_ELEMENTS");
            auto& mesh=current_domain.mesh;
            if(id>=static_cast<int>(mesh.velocity_elements.size())){
                mesh.velocity_elements.resize(id+1);
                mesh.displacement_elements.resize(id+1);
                mesh.pressure_elements.resize(id+1);
            }
            mesh.velocity_elements[id]=output_connectivity(v);
            mesh.displacement_elements[id]=output_connectivity(u);
            mesh.pressure_elements[id]=output_connectivity(pn);
            continue;
        }
    }
 
    if(inside_domain) throw std::runtime_error("Unclosed DOMAIN block in shared multi-domain mesh.");
    if(!have_domain || problem.domains.empty()) throw std::runtime_error("Shared multi-domain mesh has no DOMAIN blocks: "+filename);
    if(problem.dim==0){
        if(detected_dim==0) throw std::runtime_error("Shared multi-domain mesh has no DIM.");
        problem.dim=detected_dim;
    }
    if(global_v_nodes.empty() || global_u_nodes.empty() || global_p_nodes.empty())
        throw std::runtime_error("Shared multi-domain mesh requires global VELOCITY_NODES, DISPLACEMENT_NODES, and PRESSURE_NODES.");
 
    // Detect the optional total second-order element type used by drag/lift
    // postprocessing.  The list is deliberately not assembled.
    ElementType global_total_type = ElementType::Unknown;
    for(const auto& e:global_total_velocity_elements){
        if(!e.empty()){
            global_total_type = vem_taylor_hood ? ElementType::Polygon : element_type_from_node_count(static_cast<int>(e.size()),problem.dim);
            break;
        }
    }
 
    // All domains must already have identical shared node tables.  In particular,
    // every pressure id used by MIXED_ELEMENTS or MIXED_BOUNDARY_* must be
    // explicitly listed in PRESSURE_NODES; validation above throws otherwise.
    for(auto& d:problem.domains){
        d.mesh.total_velocity_elements = global_total_velocity_elements;
        d.mesh.total_velocity_type = global_total_type;
        d.n_dofs=total_dof_count(d.mesh);
    }
 
    // All domains share one global DOF pool.  Use the first domain mesh as the
    // representative because every domain carries the same global node arrays.
    problem.total_dofs=total_dof_count(problem.domains.front().mesh);
    for(const auto& d:problem.domains){
        if(total_dof_count(d.mesh)!=problem.total_dofs) throw std::runtime_error("Shared multi-domain domain global node pools are inconsistent.");
    }
 
    return problem;
}
