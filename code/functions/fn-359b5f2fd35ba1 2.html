inline LocalDenseLU factorize_local_dense_lu(
    LocalDenseMatrix A,
    const std::string& context = "Local dense LU solve"
) {
    const int n = static_cast<int>(A.size());
    check_local_dense_system(A, n, context);
 
    LocalDenseLU out;
    out.lu = std::move(A);
    out.pivots.resize(n);
 
    for (int k = 0; k < n; ++k) {
        int piv = k;
        double best = std::abs(out.lu[k][k]);
        for (int i = k + 1; i < n; ++i) {
            const double v = std::abs(out.lu[i][k]);
            if (v > best) { best = v; piv = i; }
        }
        if (best < 1.0e-20) {
            throw std::runtime_error("Singular " + context + ".");
        }
 
        out.pivots[k] = piv;
        if (piv != k) std::swap(out.lu[piv], out.lu[k]);
 
        const double ukk = out.lu[k][k];
        for (int i = k + 1; i < n; ++i) {
            out.lu[i][k] /= ukk;
            const double lik = out.lu[i][k];
            for (int j = k + 1; j < n; ++j) {
                out.lu[i][j] -= lik * out.lu[k][j];
            }
        }
    }
 
    return out;
}
