std::string format_problem_type_label(const Problem& problem) {
    if (problem.domains.size() == 1 && !problem.shared_dofs) {
        const std::string name = problem.domains.front().name.empty() ? "domain" : problem.domains.front().name;
        return "single-domain " + name;
    }
    return "multi-domain";
}
