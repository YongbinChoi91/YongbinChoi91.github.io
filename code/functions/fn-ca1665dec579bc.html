static std::array<double,3> torsion_rotation_displacement(
    const std::array<double,3>& X,
    const std::array<double,3>& center,
    int axis,
    double theta
){
    const double c = std::cos(theta);
    const double s = std::sin(theta);
    const std::array<double,3> r{{
        X[0] - center[0],
        X[1] - center[1],
        X[2] - center[2]
    }};
 
    std::array<double,3> rr = r;
    if(axis == 0){
        rr[1] =  c*r[1] - s*r[2];
        rr[2] =  s*r[1] + c*r[2];
    } else if(axis == 1){
        rr[0] =  c*r[0] + s*r[2];
        rr[2] = -s*r[0] + c*r[2];
    } else if(axis == 2){
        rr[0] =  c*r[0] - s*r[1];
        rr[1] =  s*r[0] + c*r[1];
    } else {
        throw std::runtime_error("Bad torsion axis id.");
    }
 
    return {{rr[0] - r[0], rr[1] - r[1], rr[2] - r[2]}};
}
