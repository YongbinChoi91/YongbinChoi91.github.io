inline std::vector<std::vector<Scalar>> deformation_gradient_as(
    const LocalFieldView& L,
    const Vec& x_elem,
    const std::vector<std::array<double,3>>& dNdx,
    const std::string& displacement_field = "displacement"
) {
    const auto& U = L.field(displacement_field);
    if (U.components != L.dim) {
        throw std::runtime_error("Deformation gradient requires vector field components equal to spatial dimension: " + displacement_field);
    }
    if (static_cast<int>(dNdx.size()) < U.n_nodes) {
        throw std::runtime_error("Gradient shape-function size is smaller than field node count for " + displacement_field + ".");
    }
 
    std::vector<std::vector<Scalar>> F(L.dim, std::vector<Scalar>(L.dim, Scalar(0.0)));
    for (int i = 0; i < L.dim; ++i) F[i][i] = Scalar(1.0);
 
    for (int a = 0; a < U.n_nodes; ++a) {
        const int base = U.local_offset + U.components * a;
        for (int i = 0; i < L.dim; ++i) {
            const Scalar ua = Scalar(x_elem[base + i]);
            for (int A = 0; A < L.dim; ++A) {
                F[i][A] += ua * Scalar(dNdx[a][A]);
            }
        }
    }
    return F;
}
