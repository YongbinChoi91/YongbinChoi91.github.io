inline ElementSystem scalar_weak_form_element_system_slfad(
    int n,
    const Vector& x_elem,
    Eval&& eval
) {
    using InnerFad = Sacado::Fad::SLFad<double, MaxDofs>;
    using OuterFad = Sacado::Fad::SLFad<InnerFad, MaxDofs>;
    return scalar_weak_form_element_system_impl<InnerFad, OuterFad>(
        n,
        x_elem,
        std::forward<Eval>(eval)
    );
}
