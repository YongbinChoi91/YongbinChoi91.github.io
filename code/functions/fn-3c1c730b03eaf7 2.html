void add_dirichlet_bc(const FieldMesh& mesh,
                   const ProblemParameters& params,
                   BoundaryData& bc,
                   int dof,
                   double value,
                   const std::string& source){
    const auto found = bc.dof_to_index.find(dof);
    if(found != bc.dof_to_index.end()){
        const int i = found->second;
        ++bc.duplicate_count;
        const double old_value=bc.values[i];
        const double diff=std::abs(old_value-value);
        if(diff > params.solver.bc_conflict_tol){
            ++bc.conflict_count;
            std::ostringstream os;
            os << "Conflicting Dirichlet values on " << format_dof_label(mesh,dof)
               << ": existing value=" << old_value
               << " from " << bc.sources[i]
               << ", new value=" << value
               << " from " << source
               << ", |diff|=" << diff
               << " > bc_conflict_tol=" << params.solver.bc_conflict_tol
               << ". This usually means overlapping boundary groups such as INLET/WALLS/CIRCLE.";
            throw std::runtime_error(os.str());
        }
        ++bc.compatible_duplicate_count;
        if(params.solver.bc_diagnostics){
            std::cout << "    [BC duplicate compatible] " << format_dof_label(mesh,dof)
                      << ", value=" << old_value
                      << ", old source=" << bc.sources[i]
                      << ", new source=" << source
                      << ", diff=" << diff << "\n";
        }
        // Keep the first value. This avoids order-dependent overwriting.
        return;
    }
    const int idx = static_cast<int>(bc.dofs.size());
    bc.dof_to_index.emplace(dof, idx);
    bc.dofs.push_back(dof);
    bc.values.push_back(value);
    bc.sources.push_back(source);
}
