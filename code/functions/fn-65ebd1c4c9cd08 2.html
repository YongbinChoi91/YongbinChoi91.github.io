bool coordinate_selector_matches_field(
    const FieldInfo& field,
    const BCSelector& selector,
    int dim
){
    if(selector.kind==BCSelectorKind::BoundaryGroup) return false;
    for(const Node& node: field.nodes){
        if(node_matches_selector(node,selector,dim)) return true;
    }
    return false;
}
