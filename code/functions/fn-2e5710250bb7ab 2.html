ProblemElementDofMap precompute_global_element_dofs(const Problem& problem){
    ProblemElementDofMap maps(problem.domains.size());
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain=problem.domains[d];
        if(!is_supported_continuum_domain(domain.type)) throw std::runtime_error("Unsupported domain type in global element-DOF precompute.");
        const FieldMesh& mesh=domain.mesh;
        const DofLayout& gmesh=get_cached_dof_layout(mesh);
        const std::vector<std::string> active_fields=active_constitutive_field_names(mesh);
        const int nelem=count_active_elements(gmesh,active_fields);
        maps[d].resize(nelem);
        for(int e=0;e<nelem;++e){
            maps[d][e]=make_domain_element_dofs(problem,d,e,active_fields);
        }
    }
    return maps;
}
