static std::vector<HistorySpec> merged_history_specs(const std::vector<HistorySpec>& a,const std::vector<HistorySpec>& b){
    std::vector<HistorySpec> out=a;
    for(const auto& h:b){
        bool found=false;
        for(auto& existing:out){
            if(existing.name==h.name){
                existing.nvars=std::max(existing.nvars,h.nvars);
                found=true;
                break;
            }
        }
        if(!found) out.push_back(h);
    }
    return out;
}
