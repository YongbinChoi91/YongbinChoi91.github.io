void add_inactive_shared_dof_constraints(
    BoundaryData& bc,
    const Problem& problem,
    const ProblemParameters& params
){
    if(problem.domains.empty()) return;
    const FieldMesh& rep=problem.domains.front().mesh;
 
    std::vector<char> used_v(rep.velocity_nodes.size(),0);
    std::vector<char> used_u(rep.displacement_nodes.size(),0);
    std::vector<char> used_p(rep.pressure_nodes.size(),0);
 
    for(const auto& domain:problem.domains){
        if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
        const FieldMesh& mesh=domain.mesh;
        for(const auto& e:mesh.velocity_elements) for(int n:e)
            if(n>=0 && n<static_cast<int>(used_v.size())) used_v[n]=1;
        if(has_displacement_field(mesh)){
            for(const auto& e:mesh.displacement_elements) for(int n:e)
                if(n>=0 && n<static_cast<int>(used_u.size())) used_u[n]=1;
        }
        for(const auto& e:mesh.pressure_elements) for(int n:e)
            if(n>=0 && n<static_cast<int>(used_p.size())) used_p[n]=1;
    }
 
    for(int n=0;n<static_cast<int>(used_v.size());++n){
        if(!used_v[n]){
            for(int c=0;c<rep.dim;++c)
                add_dirichlet_bc(rep,params,bc,velocity_dof(rep,n,c),0.0,"INACTIVE_SHARED_VELOCITY_DOF");
        }
    }
    if(has_displacement_field(rep)){
        for(int n=0;n<static_cast<int>(used_u.size());++n){
            if(!used_u[n]){
                for(int c=0;c<rep.dim;++c)
                    add_dirichlet_bc(rep,params,bc,displacement_dof(rep,n,c),0.0,"INACTIVE_SHARED_DISPLACEMENT_DOF");
            }
        }
    }
    for(int n=0;n<static_cast<int>(used_p.size());++n){
        if(!used_p[n]) add_dirichlet_bc(rep,params,bc,pressure_dof(rep,n),0.0,"INACTIVE_SHARED_PRESSURE_DOF");
    }
}
