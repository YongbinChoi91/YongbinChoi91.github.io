std::string field_role_to_string(FieldRole role){
    switch(role){
        case FieldRole::Mechanical: return "mechanical";
        case FieldRole::Constraint: return "constraint";
        case FieldRole::Thermal: return "thermal";
        case FieldRole::Internal: return "internal";
        case FieldRole::Auxiliary: return "auxiliary";
        default: return "unknown";
    }
}
