void append_hysteresis_csv(const std::string& filename,double time,double x,double y){
    std::ofstream out(filename,std::ios::app);
    if(!out) throw std::runtime_error("Could not append hysteresis CSV: "+filename);
    out << std::setprecision(17) << time << "," << x << "," << y << "\n";
}
