static bool is_field_mesh_section(const std::string& key){
    return key=="VELOCITY_NODES" || key=="DISPLACEMENT_NODES" || key=="PRESSURE_NODES" ||
           key=="NODES_V" || key=="NODES_U" || key=="NODES_P" || key=="NODES" ||
           key=="VELOCITY_ELEMENTS" || key=="DISPLACEMENT_ELEMENTS" || key=="PRESSURE_ELEMENTS" ||
           key=="ELEMENTS_V" || key=="ELEMENTS_U" || key=="ELEMENTS_P" || key=="ELEMENTS" ||
           key=="MIXED_ELEMENTS" ||
           key.rfind("VELOCITY_BOUNDARY_",0)==0 || key.rfind("DISPLACEMENT_BOUNDARY_",0)==0 || key.rfind("PRESSURE_BOUNDARY_",0)==0 ||
           key.rfind("V_",0)==0 || key.rfind("U_",0)==0 || key.rfind("P_",0)==0;
}
