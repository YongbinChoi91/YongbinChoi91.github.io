ElementType parse_element_type(const std::string& name){
    const std::string k=lower_topology_name(name);
    if(k=="t3" || k=="tri3") return ElementType::T3;
    if(k=="t6" || k=="tri6") return ElementType::T6;
    if(k=="q4" || k=="quad4") return ElementType::Q4;
    if(k=="q9" || k=="quad9") return ElementType::Q9;
    if(k=="tet4") return ElementType::Tet4;
    if(k=="tet10") return ElementType::Tet10;
    if(k=="h8" || k=="hex8") return ElementType::H8;
    if(k=="h20" || k=="hex20") return ElementType::H20;
    if(k=="h27" || k=="hex27") return ElementType::H27;
    if(k=="polygon" || k=="vem") return ElementType::Polygon;
    return ElementType::Unknown;
}
