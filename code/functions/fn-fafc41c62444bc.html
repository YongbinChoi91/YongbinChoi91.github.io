inline double compute_element_measure(
    const ElementGeometry& geom,
    ElementType type,
    int gauss_order = -1
) {
    double measure = 0.0;
    for (const auto& gp : gauss_rule(type, gauss_order)) {
        const ShapeData s = compute_shape(type, gp.xi, gp.eta, gp.zeta);
        const JacobianData J = compute_jacobian_data(geom, s);
        if (J.detJ <= 0.0) {
            throw std::runtime_error("Negative detJ while computing element measure.");
        }
        measure += J.detJ * gp.weight;
    }
    return std::max(measure, 1.0e-30);
}
