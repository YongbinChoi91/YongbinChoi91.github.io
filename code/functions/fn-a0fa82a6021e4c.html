inline std::vector<std::vector<Scalar>> neo_hookean_first_piola_young(
    const std::vector<std::vector<Scalar>>& F3,
    double youngs_modulus,
    double poisson_ratio
) {
    if (youngs_modulus <= 0.0) {
        throw std::runtime_error("Neo-Hookean helper needs positive youngs_modulus.");
    }
    if (poisson_ratio <= -1.0 || poisson_ratio >= 0.5) {
        throw std::runtime_error("Neo-Hookean helper needs -1 < poisson_ratio < 0.5.");
    }
 
    const double mu = youngs_modulus / (2.0 * (1.0 + poisson_ratio));
    const double lambda = youngs_modulus * poisson_ratio /
                          ((1.0 + poisson_ratio) * (1.0 - 2.0 * poisson_ratio));
    return neo_hookean_first_piola_lame<Scalar>(F3, lambda, mu);
}
