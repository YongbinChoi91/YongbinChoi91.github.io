static bool uses_custom_pressure_schur(const LinearSolverSettings& s){
    // Build a dedicated pressure Schur matrix only when PETSc will actually
    // use PCFieldSplit in Schur mode.  For additive/multiplicative FieldSplit
    // or full-matrix PCs, replacing the pressure block of Pmat is harmful and
    // can make diagnostics misleading.
    const std::string prec = lower_copy(s.preconditioner);
    const std::string fst  = lower_copy(s.fieldsplit_type);
    if(!(prec=="fieldsplit" || prec=="schur" || prec=="block_schur")) return false;
    if(fst!="schur") return false;
 
    const std::string sch=lower_copy(s.schur_precondition);
    return sch=="pressure_mass" || sch=="pressure_laplacian" ||
           sch=="mass_laplacian" || sch=="pressure_mass_laplacian" ||
           sch=="mass+laplacian";
}
