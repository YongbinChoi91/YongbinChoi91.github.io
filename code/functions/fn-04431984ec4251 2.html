Vector extract_domain_vector(const Vector& x,const Domain& domain){
    if(domain.dof_offset<0 || domain.n_dofs<0 || domain.dof_offset+domain.n_dofs>static_cast<int>(x.size()))
        throw std::runtime_error("Bad domain vector slice.");
    return Vector(x.begin()+domain.dof_offset,x.begin()+domain.dof_offset+domain.n_dofs);
}
