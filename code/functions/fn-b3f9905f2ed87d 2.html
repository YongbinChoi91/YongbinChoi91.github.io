bool self_check_scalar_polynomial() {
    const double x = 2.0;
    const auto dydx = grad([](const auto& x_ad) {
        return x_ad * x_ad * x_ad + 2.0 * x_ad;
    }, x);
    return close(dydx, 14.0, 1e-14);
}
