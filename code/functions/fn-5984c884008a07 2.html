static std::vector<std::vector<int>> reference_faces_3d_post(ElementType type){
    if(type==ElementType::H8){
        return {
            {0,1,2,3},
            {4,5,6,7},
            {0,1,5,4},
            {1,2,6,5},
            {3,2,6,7},
            {0,3,7,4}
        };
    }
    if(type==ElementType::H20){
        return {
            {0,1,2,3,8,9,10,11},
            {4,5,6,7,16,17,18,19},
            {0,1,5,4,8,13,16,12},
            {1,2,6,5,9,14,17,13},
            {3,2,6,7,10,14,18,15},
            {0,3,7,4,11,15,19,12}
        };
    }
    if(type==ElementType::H27){
        return {
            {0,1,2,3,8,9,10,11,20},
            {4,5,6,7,16,17,18,19,25},
            {0,1,5,4,8,13,16,12,23},
            {1,2,6,5,9,14,17,13,22},
            {3,2,6,7,10,14,18,15,24},
            {0,3,7,4,11,15,19,12,21}
        };
    }
    if(type==ElementType::Tet4){
        return {
            {0,1,2},
            {0,1,3},
            {0,2,3},
            {1,2,3}
        };
    }
    if(type==ElementType::Tet10){
        return {
            {0,1,2,4,5,6},
            {0,1,3,4,8,7},
            {0,2,3,6,9,7},
            {1,2,3,5,9,8}
        };
    }
    throw std::runtime_error("Reference faces currently support 3D Tet4/Tet10/H8/H20/H27 elements.");
}
