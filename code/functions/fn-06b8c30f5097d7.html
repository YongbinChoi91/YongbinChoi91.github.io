inline AD1<Scalar, N> log(const AD1<Scalar, N>& a) {
    using std::log;
    AD1<Scalar, N> c(log(a.val));
    for (std::size_t i = 0; i < N; ++i) c.grad[i] = a.grad[i] / a.val;
    return c;
}
