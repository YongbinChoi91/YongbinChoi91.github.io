void apply_dofs_info(Problem& problem,const DofsInfo& info){
    if(!info.domain_name.empty() && problem.domains.size()==1 && !problem.shared_dofs){
        problem.domains.front().name=info.domain_name;
    }
    for(auto& d:problem.domains){
        if(!info.field_order.empty()) d.mesh.dof_field_order=info.field_order;
        // Domain-local history is the canonical multidomain path.
        d.mesh.history_specs=history_specs_for_domain(info,d);
        d.mesh.vem_order=info.vem_order;
        d.mesh.vem_velocity_order=info.vem_velocity_order;
        d.mesh.vem_pressure_order=info.vem_pressure_order;
        invalidate_dof_layout_cache(d.mesh);
        (void)get_cached_dof_layout(d.mesh);
        d.n_dofs=total_dof_count(d.mesh);
    }
    if(problem.shared_dofs){
        if(!problem.domains.empty()) problem.total_dofs=total_dof_count(problem.domains.front().mesh);
    } else {
        int off=0;
        for(auto& d:problem.domains){
            d.dof_offset=off;
            d.n_dofs=total_dof_count(d.mesh);
            off += d.n_dofs;
        }
        problem.total_dofs=off;
    }
}
