static const std::vector<std::vector<int>>& force_owner_table(const FieldMesh& mesh){
    // Hydrodynamic drag/lift must be evaluated only from the fluid-side field.
    // Therefore owner recovery deliberately scans the fluid-domain velocity
    // element table.  In shared multi-domain meshes TOTAL_ELEMENTS also contains solid
    // rows; using it here can pick a solid-side/internal surface and then read
    // fluid pressure/velocity with the wrong element connectivity.
    return mesh.velocity_elements;
}
