inline Mat3<Scalar> transpose3(const Mat3<Scalar>& A) {
    auto B = zero3<Scalar>();
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) B[i][j] = A[j][i];
    }
    return B;
}
