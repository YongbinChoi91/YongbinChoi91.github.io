std::vector<std::string> multiscale_rve_sample_labels_for_element(
    const FieldMesh& mesh,
    int elem_id,
    const ProblemParameters& params
) {
    std::vector<std::string> labels;
    if (params.solver.multiscale_plot_points.empty()) return labels;
    for (const auto& pt : params.solver.multiscale_plot_points) {
        if (choose_sample_element(mesh, pt.x, pt.y) == elem_id) labels.push_back(pt.label);
    }
    return labels;
}
