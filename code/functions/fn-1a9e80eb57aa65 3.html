void print_shared_boundary_condition_diagnostics(
    const Problem& problem,
    const BoundaryData& bc,
    const ProblemParameters& params
){
    if(!params.solver.bc_diagnostics || problem.domains.empty() || !problem.shared_dofs) return;
    const FieldMesh& rep=problem.domains.front().mesh;
    int vx=0,vy=0,vz=0, ux=0,uy=0,uz=0, p=0;
    const int nv=n_velocity_dofs(rep);
    const int nu=n_displacement_dofs(rep);
    const int off_u=nv;
    const int off_p=nv+nu;
    for(int dof:bc.dofs){
        if(dof<nv){
            const int c=dof%rep.dim;
            if(c==0) ++vx; else if(c==1) ++vy; else ++vz;
        } else if(dof<off_p){
            const int c=(dof-off_u)%rep.dim;
            if(c==0) ++ux; else if(c==1) ++uy; else ++uz;
        } else {
            ++p;
        }
    }
    std::cout << "  shared fixed dofs by block:"
              << " vx=" << vx << " vy=" << vy;
    if(rep.dim==3) std::cout << " vz=" << vz;
    std::cout << " ux=" << ux << " uy=" << uy;
    if(rep.dim==3) std::cout << " uz=" << uz;
    std::cout << " p=" << p << "\n";
}
