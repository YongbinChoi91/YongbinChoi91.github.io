std::string format_dof_label(const FieldMesh& mesh,int dof){
    const DofLayout& gmesh = get_cached_dof_layout(mesh);
    std::ostringstream os;
    for(const auto& f:gmesh.fields){
        const int begin=f.offset;
        const int end=f.offset+f.ndofs;
        if(dof>=begin && dof<end){
            const int q=dof-begin;
            const int node=q/f.components;
            const int comp=q%f.components;
            os << f.name << "[" << node;
            if(f.components>1) os << "," << comp;
            os << "] global_dof=" << dof;
            return os.str();
        }
    }
    os << "unknown global_dof=" << dof;
    return os.str();
}
