inline std::vector<std::vector<Scalar>> deformation_gradient_2d(
    const std::array<std::array<Scalar,2>,2>& grad_u
) {
    auto F3 = eye<Scalar>(3);
    F3[0][0] = Scalar(1.0) + grad_u[0][0];
    F3[0][1] = grad_u[0][1];
    F3[1][0] = grad_u[1][0];
    F3[1][1] = Scalar(1.0) + grad_u[1][1];
    return F3;
}
