void reset_hdf5_results_file(const ProblemParameters& params,const Problem& problem,const std::string& model_name){
    if(!params.solver.hdf5_output) return;
#ifndef HIPECS_USE_HDF5
    (void)problem; (void)model_name;
    throw std::runtime_error("hdf5_output is on, but this binary was built without HIPECS_USE_HDF5.");
#else
    ensure_output_parent_directory(params.solver.hdf5_file);
    hid_t file=H5Fcreate(params.solver.hdf5_file.c_str(),H5F_ACC_TRUNC,H5P_DEFAULT,H5P_DEFAULT);
    if(file<0) throw std::runtime_error("Could not create HDF5 file: " + params.solver.hdf5_file);
    h5_write_string_attr(file,"format","HiPeCs HDF5 results");
    h5_write_string_attr(file,"model_name",model_name);
    h5_write_string_attr(file,"model_output_name",params.solver.model_output_name);
    h5_write_string_attr(file,"problem_type",format_problem_type_label(problem));
    h5_write_string_attr(file,"schema_version","2");
    h5_write_mesh(file,problem);
    H5Fclose(file);
#endif
}
