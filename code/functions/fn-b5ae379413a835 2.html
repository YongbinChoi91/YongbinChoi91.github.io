inline void initialize_material_history_store(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    MaterialHistoryStore& history
) {
    history.domains.clear();
    if (mesh.history_specs.empty()) return;
 
    const int nvars = primary_history_nvars(mesh.history_specs);
    if (nvars <= 0) return;
 
    const int nelem = history_element_count_for_mesh(mesh);
    const ElementType etype = history_element_type_for_mesh(mesh);
    if (nelem <= 0 || etype == ElementType::Unknown) return;
 
    const auto gps = gauss_rule(etype, params.solver.gauss_order);
    history.domains.resize(1);
    history.domains[0].specs = mesh.history_specs;
    history.domains[0].elements.resize(nelem);
    for (auto& eh : history.domains[0].elements) {
        eh.gp.resize(gps.size());
        for (auto& gh : eh.gp) {
            gh.h_n.assign(nvars, 0.0);
            gh.h_trial.assign(nvars, 0.0);
            gh.state_n = 0.0;
            gh.state_trial = 0.0;
        }
    }
}
