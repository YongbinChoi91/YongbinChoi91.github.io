static std::vector<AD2> shape_values_Q9_ad2(const AD2& xi,const AD2& eta,const AD2&) {
    auto Lm=[](const AD2& r){ return 0.5*r*(r-1.0); };
    auto L0=[](const AD2& r){ return 1.0-r*r; };
    auto Lp=[](const AD2& r){ return 0.5*r*(r+1.0); };
    const AD2 Lx[3]={Lm(xi),L0(xi),Lp(xi)};
    const AD2 Ly[3]={Lm(eta),L0(eta),Lp(eta)};
    const int ix[9]={0,2,2,0,1,2,1,0,1};
    const int iy[9]={0,0,2,2,0,1,2,1,1};
    std::vector<AD2> N(9);
    for(int a=0; a<9; ++a) N[a] = Lx[ix[a]] * Ly[iy[a]];
    return N;
}
