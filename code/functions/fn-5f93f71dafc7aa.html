void write_force_svg_from_csv(const std::string& csv_filename,const std::string& svg_filename){
    std::ifstream in(csv_filename);
    if(!in) throw std::runtime_error("Could not open force CSV for SVG plotting: "+csv_filename);
    std::string line;
    std::getline(in,line); // header
    std::vector<double> t, drag, lift;
    while(std::getline(in,line)){
        if(line.empty()) continue;
        std::stringstream ss(line);
        std::vector<double> vals;
        std::string item;
        while(std::getline(ss,item,',')) vals.push_back(std::stod(item));
        if(vals.size()>=6){ t.push_back(vals[0]); drag.push_back(vals[4]); lift.push_back(vals[5]); }
        else if(vals.size()>=3){ t.push_back(vals[0]); drag.push_back(vals[1]); lift.push_back(vals[2]); }
    }
    if(t.empty()) return;
    write_xy_svg_plot(svg_filename,"Drag / Lift history","time","force",t,{
        {"Drag","#1f77b4",drag},
        {"Lift","#d62728",lift}
    });
}
