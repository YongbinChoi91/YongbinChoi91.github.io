static void validate_global_dofs_size(
    const Vector& current_dofs,
    const PastDofs& past_dofs,
    int ndofs,
    const std::string& context
){
    if(static_cast<int>(current_dofs.size())!=ndofs)
        throw std::runtime_error("Current DOF vector size mismatch in "+context+".");
    for(int i=0;i<static_cast<int>(past_dofs.size());++i){
        if(static_cast<int>(past_dofs[i].size())!=ndofs)
            throw std::runtime_error("Past DOF vector size mismatch in "+context+".");
    }
}
