static ScalarFunction find_hysteresis_x_function(const ProblemParameters& params,const FieldComponentTarget& target){
    for(const auto& r: params.field_dirichlet_bcs){
        if(dirichlet_rule_matches(r,target)) return r.value;
    }
    throw std::runtime_error("hysteresis prescribed axis '"+hysteresis_label(target)+"' does not match any Dirichlet BC.");
}
