const QuadraticElementCache& get_cached_quadratic_element(
    const FieldMesh& mesh,
    int elem_id,
    int q_order,
    const std::string& field_name
) {
    auto& cache = quadratic_element_cache_map();
    const CacheKey key{&mesh, elem_id, q_order, field_name};
    auto it = cache.find(key);
    if (it == cache.end()) {
        it = cache.emplace(key, build_quadratic_element_cache(mesh, elem_id, q_order, field_name)).first;
    }
    return it->second;
}
