void apply_dofs_info(FieldMesh& mesh,const DofsInfo& info){
    if(!info.field_order.empty()) mesh.dof_field_order=info.field_order;
    if(!info.histories.empty() || info.domain_histories.empty()){
        mesh.history_specs=info.histories;
    } else if(info.domain_histories.size()==1){
        mesh.history_specs=info.domain_histories.begin()->second;
    } else {
        mesh.history_specs=all_history_specs(info);
    }
    mesh.vem_order=info.vem_order;
    mesh.vem_velocity_order=info.vem_velocity_order;
    mesh.vem_pressure_order=info.vem_pressure_order;
    invalidate_dof_layout_cache(mesh);
    (void)get_cached_dof_layout(mesh);
}
