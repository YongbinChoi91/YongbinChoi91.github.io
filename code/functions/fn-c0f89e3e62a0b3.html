inline double det_small_raw(const double A[3][3], int dim) {
    if (dim == 2) {
        return A[0][0]*A[1][1] - A[0][1]*A[1][0];
    }
    if (dim == 3) {
        return A[0][0]*(A[1][1]*A[2][2]-A[1][2]*A[2][1])
             - A[0][1]*(A[1][0]*A[2][2]-A[1][2]*A[2][0])
             + A[0][2]*(A[1][0]*A[2][1]-A[1][1]*A[2][0]);
    }
    throw std::runtime_error("det_small_raw supports only 2D/3D.");
}
