RvePbcSystem build_square_rve_pbc_constraints(
    const FieldMesh& mesh,
    const RveSolveOptions& options
) {
    if (mesh.dim != 2) throw std::runtime_error("The current single RVE PBC builder supports 2D only.");
    if (!has_displacement_field(mesh)) throw std::runtime_error("RVE PBC requires a displacement field.");
 
    const int nn = static_cast<int>(mesh.displacement_nodes.size());
    double xmin = std::numeric_limits<double>::infinity();
    double xmax = -std::numeric_limits<double>::infinity();
    double ymin = std::numeric_limits<double>::infinity();
    double ymax = -std::numeric_limits<double>::infinity();
    for (const auto& n : mesh.displacement_nodes) {
        xmin = std::min(xmin, n.x[0]); xmax = std::max(xmax, n.x[0]);
        ymin = std::min(ymin, n.x[1]); ymax = std::max(ymax, n.x[1]);
    }
    const double L = std::max(xmax-xmin, ymax-ymin);
    const double tol = std::max(options.pair_tol, 1.0e-10 * std::max(1.0, L));
 
    std::vector<int> left, right, bottom, top;
    for (int a = 0; a < nn; ++a) {
        const auto& X = mesh.displacement_nodes[a].x;
        if (nearly_equal(X[0], xmin, tol)) left.push_back(a);
        if (nearly_equal(X[0], xmax, tol)) right.push_back(a);
        if (nearly_equal(X[1], ymin, tol)) bottom.push_back(a);
        if (nearly_equal(X[1], ymax, tol)) top.push_back(a);
    }
    left = sort_nodes_by_coordinates(left, mesh, 1, 0);
    right = sort_nodes_by_coordinates(right, mesh, 1, 0);
    bottom = sort_nodes_by_coordinates(bottom, mesh, 0, 1);
    top = sort_nodes_by_coordinates(top, mesh, 0, 1);
 
    if (left.size() != right.size()) throw std::runtime_error("PBC left/right boundary node counts do not match.");
    if (bottom.size() != top.size()) throw std::runtime_error("PBC bottom/top boundary node counts do not match.");
 
    RvePbcSystem pbc;
    for (int k = 0; k < static_cast<int>(left.size()); ++k) {
        const int Lnode = left[k];
        const int Rnode = right[k];
        const double dy = std::abs(mesh.displacement_nodes[Lnode].x[1] - mesh.displacement_nodes[Rnode].x[1]);
        if (dy > tol) throw std::runtime_error("PBC left/right pairing failed by y-coordinate mismatch.");
        for (int c = 0; c < 2; ++c) pbc.constraints.push_back(make_periodic_pair_constraint(mesh, Rnode, Lnode, c, "right-left"));
    }
 
    // Add bottom-top constraints but skip the left-bottom corner pair.  The full
    // set of LR and BT corner equations contains one dependent loop; skipping one
    // BT corner keeps the ACEFEM-style KKT system nonsingular while preserving PBC.
    for (int k = 0; k < static_cast<int>(bottom.size()); ++k) {
        const int Bnode = bottom[k];
        const int Tnode = top[k];
        const double dx = std::abs(mesh.displacement_nodes[Bnode].x[0] - mesh.displacement_nodes[Tnode].x[0]);
        if (dx > tol) throw std::runtime_error("PBC bottom/top pairing failed by x-coordinate mismatch.");
        if (nearly_equal(mesh.displacement_nodes[Bnode].x[0], xmin, tol)) continue;
        for (int c = 0; c < 2; ++c) pbc.constraints.push_back(make_periodic_pair_constraint(mesh, Tnode, Bnode, c, "top-bottom"));
    }
 
    int anchor = -1;
    double best = std::numeric_limits<double>::infinity();
    for (int a = 0; a < nn; ++a) {
        const auto& X = mesh.displacement_nodes[a].x;
        const double d2 = square(X[0]-xmin) + square(X[1]-ymin);
        if (d2 < best) { best = d2; anchor = a; }
    }
    if (anchor < 0) throw std::runtime_error("Could not choose RVE anchor node.");
    pbc.anchor_node = anchor;
    for (int c = 0; c < 2; ++c) pbc.constraints.push_back(make_anchor_constraint(mesh, anchor, c));
 
    return pbc;
}
