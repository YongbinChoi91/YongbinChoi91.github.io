inline void check_local_dense_system(
    const LocalDenseMatrix& A,
    int n,
    const std::string& context
) {
    if (static_cast<int>(A.size()) != n) {
        throw std::runtime_error(context + " received incompatible matrix/vector size.");
    }
    for (int i = 0; i < n; ++i) {
        if (static_cast<int>(A[i].size()) != n) {
            throw std::runtime_error(context + " expects a square matrix.");
        }
    }
}
