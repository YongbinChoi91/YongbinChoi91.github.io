static std::array<double,3> cross3_assembly(
    const std::array<double,3>& a,
    const std::array<double,3>& b
){
    return {{
        a[1]*b[2] - a[2]*b[1],
        a[2]*b[0] - a[0]*b[2],
        a[0]*b[1] - a[1]*b[0]
    }};
}
