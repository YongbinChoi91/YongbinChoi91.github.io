static ResolvedBoundaryEdge choose_boundary_owner_candidate(
    const FieldMesh& mesh,
    const std::vector<ResolvedBoundaryEdge>& matches,
    const std::array<double,3>* boundary_group_center
){
    if(matches.empty()) return ResolvedBoundaryEdge{};
    if(!boundary_group_center) return matches.front();
 
    // If two domain elements contain the same boundary segment, choose the one
    // whose centroid is farther from the closed inner-boundary center.  This
    // matches the Python behavior for an immersed boundary: the surface element
    // is the exterior/fluid-side element touching the surface, not an arbitrary
    // row id from the boundary file.
    int best=0;
    double best_d2=-1.0;
    for(int k=0;k<static_cast<int>(matches.size());++k){
        const auto c=element_centroid_from_owner(mesh,matches[k]);
        const double dx=c[0]-(*boundary_group_center)[0];
        const double dy=c[1]-(*boundary_group_center)[1];
        const double dz=c[2]-(*boundary_group_center)[2];
        const double d2=dx*dx+dy*dy+dz*dz;
        if(d2>best_d2){ best_d2=d2; best=k; }
    }
    return matches[best];
}
