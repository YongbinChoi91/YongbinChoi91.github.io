static ElementGeometry make_field_geometry_element(
    const FieldInfo& field,
    int dim,
    int elem_id
){
    if(elem_id<0 || elem_id>=static_cast<int>(field.elements.size()))
        throw std::runtime_error("Neumann owner element id out of range for field "+field.name+".");
    ElementGeometry geom;
    geom.dim = dim;
    geom.type = field.element_type;
    const auto& conn = field.elements[elem_id];
    geom.coords.resize(conn.size());
    for(int a=0; a<static_cast<int>(conn.size()); ++a){
        const int node = conn[a];
        if(node<0 || node>=static_cast<int>(field.nodes.size()))
            throw std::runtime_error("Neumann owner element node id out of range for field "+field.name+".");
        geom.coords[a] = field.nodes[node].x;
    }
    return geom;
}
