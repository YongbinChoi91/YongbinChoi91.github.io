inline std::array<std::array<Scalar, InN>, OutN> extract_output_gradient(
    const std::array<AD1<Scalar, InN>, OutN>& y
) {
    std::array<std::array<Scalar, InN>, OutN> J{};
    for (std::size_t a = 0; a < OutN; ++a) {
        for (std::size_t i = 0; i < InN; ++i) J[a][i] = y[a].grad[i];
    }
    return J;
}
