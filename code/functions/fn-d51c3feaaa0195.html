void read_postprocessing_parameters(const std::string& postprocess_file,
                                           ProblemParameters& p){
    p.solver.plot_config_file = postprocess_file;
    std::ifstream file(postprocess_file);
    if(!file) throw std::runtime_error("Cannot open postprocessing file: "+postprocess_file);
 
    std::string output_directory;
    std::string output_name;
 
    auto split_prefix = [](const std::string& prefix, std::string& dir, std::string& name){
        const std::size_t slash = prefix.find_last_of("/\\");
        if(slash == std::string::npos){
            dir = ".";
            name = prefix;
        } else {
            dir = prefix.substr(0, slash);
            name = prefix.substr(slash + 1);
            if(dir.empty()) dir = ".";
        }
    };
 
    auto trim_trailing_slashes = [](std::string s){
        while(s.size() > 1 && (s.back() == '/' || s.back() == '\\')) s.pop_back();
        return s;
    };
 
    auto rebuild_output_prefix = [&](){
        std::string dir, name;
        split_prefix(p.solver.output_prefix, dir, name);
        if(!output_directory.empty()) dir = trim_trailing_slashes(output_directory);
        if(!output_name.empty()) name = output_name;
        if(dir.empty() || dir == ".") p.solver.output_prefix = name;
        else p.solver.output_prefix = dir + "/" + name;
    };
 
    auto parse_hysteresis_axis = [&](std::stringstream& line, const std::string& axis_name){
        TimeSolverSettings::HysteresisAxisSpec axis;
        axis.enabled = true;
        std::string mode;
        line >> mode;
        mode = lower_copy(mode);
        if(mode.empty()) throw std::runtime_error("Bad " + axis_name + " line: missing mode.");
        axis.mode = mode;
 
        auto read_scale_offset = [&](){
            if(line >> axis.scale){
                if(line >> axis.offset) {}
            }
        };
 
        auto parse_point_coordinates = [&](const std::string& first_token){
            std::array<double,3> pnt{{0.0,0.0,0.0}};
            std::string token = first_token;
            if(token.find(',') != std::string::npos){
                std::replace(token.begin(), token.end(), ',', ' ');
                std::stringstream ps(token);
                if(!(ps >> pnt[0] >> pnt[1])){
                    throw std::runtime_error("Bad " + axis_name + " value point coordinates. Use x,y[,z].");
                }
                if(!(ps >> pnt[2])) pnt[2] = 0.0;
            } else {
                if(token.empty()) throw std::runtime_error("Bad " + axis_name + " value point line: missing coordinates.");
                pnt[0] = std::stod(token);
                if(!(line >> pnt[1])){
                    throw std::runtime_error("Bad " + axis_name + " value point coordinates. Use x y or x,y[,z].");
                }
            }
            return pnt;
        };
 
        if(mode=="time"){
            read_scale_offset();
            if(axis.label.empty()) axis.label = "time";
        } else if(mode=="function" || mode=="expr" || mode=="expression"){
            std::string expr = rest_trim(line);
            if(expr.empty()) throw std::runtime_error("Bad " + axis_name + " function line: missing expression.");
            axis.mode = "function";
            axis.function = make_scalar_function(expr);
            axis.label = expr;
        } else if(mode=="prescribed" || mode=="bc" || mode=="dirichlet" ||
                  mode=="reaction" || mode=="force" || mode=="external" ||
                  mode=="residual" || mode=="internal"){
            std::string a,b,c;
            line >> a >> b;
            if(a.empty() || b.empty()) throw std::runtime_error("Bad " + axis_name + " line: missing target.");
            if(is_known_field_name(a)){
                line >> c;
                if(c.empty()) throw std::runtime_error("Bad " + axis_name + " line. Use: " + axis_name + " " + mode + " field group component [scale [offset]].");
                bool explicit_field=false;
                axis.target = parse_field_component_target_tokens(a,b,c,explicit_field);
            } else {
                axis.target = target_from_alias(a,b);
            }
            if(mode=="force" || mode=="external") axis.mode="reaction";
            if(mode=="internal") axis.mode="residual";
            read_scale_offset();
            if(axis.label.empty()) axis.label = axis.mode + " " + axis.target.field + " " + axis.target.group + " " + b;
        } else if(mode=="value" || mode=="field" || mode=="dof" || mode=="point"){
            axis.mode = "value";
            std::string field, selector;
            line >> field >> selector;
            if(field.empty() || selector.empty()) throw std::runtime_error("Bad " + axis_name + " value line.");
            axis.target.field = lower_copy(field);
            selector = lower_copy(selector);
            if(selector=="node"){
                std::string comp;
                line >> axis.node >> comp;
                if(axis.node < 0 || comp.empty()) throw std::runtime_error("Bad " + axis_name + " value node line. Use: " + axis_name + " value field node node_id component [scale [offset]].");
                axis.target.group = "";
                axis.target.component = component_index_from_string(comp);
                axis.reduction = "point";
                read_scale_offset();
                axis.label = axis.target.field + " node " + std::to_string(axis.node) + " " + comp;
            } else if(selector=="point" || selector=="coord" || selector=="coordinate" || selector=="coords"){
                std::string coords, comp;
                line >> coords;
                if(coords.empty()) throw std::runtime_error("Bad " + axis_name + " value point line. Use: " + axis_name + " value field point x,y[,z] component [scale [offset]].");
                axis.has_point = true;
                axis.point = parse_point_coordinates(coords);
                line >> comp;
                if(comp.empty()) throw std::runtime_error("Bad " + axis_name + " value point line. Use: " + axis_name + " value field point x,y[,z] component [scale [offset]].");
                axis.target.group = "";
                axis.target.component = component_index_from_string(comp);
                axis.reduction = "point";
                read_scale_offset();
                std::ostringstream label;
                label << axis.target.field << " point "
                      << std::setprecision(17) << axis.point[0] << "," << axis.point[1];
                if(axis.point[2] != 0.0) label << "," << axis.point[2];
                label << " " << comp;
                axis.label = label.str();
            } else if(selector=="boundary" || selector=="group"){
                std::string group, comp, reduction_or_scale;
                line >> group >> comp;
                if(group.empty() || comp.empty()) throw std::runtime_error("Bad " + axis_name + " value boundary line. Use: " + axis_name + " value field boundary group component [mean|sum] [scale [offset]].");
                axis.target.group = group;
                axis.target.component = component_index_from_string(comp);
                std::streampos pos = line.tellg();
                if(line >> reduction_or_scale){
                    const std::string rlow=lower_copy(reduction_or_scale);
                    if(rlow=="sum" || rlow=="mean" || rlow=="average" || rlow=="avg"){
                        axis.reduction = (rlow=="average" || rlow=="avg") ? "mean" : rlow;
                        read_scale_offset();
                    } else {
                        line.clear();
                        if(pos != std::streampos(-1)) line.seekg(pos);
                        read_scale_offset();
                    }
                }
                if(axis.reduction.empty()) axis.reduction = "mean";
                axis.label = axis.target.field + " " + group + " " + comp;
            } else {
                throw std::runtime_error("Bad " + axis_name + " value selector. Use node, point, or boundary.");
            }
        } else {
            throw std::runtime_error("Unknown " + axis_name + " mode: " + mode + ". Use time, function, prescribed, reaction, residual, or value.");
        }
        return axis;
    };
 
    struct BoolKey { std::initializer_list<const char*> aliases; bool* value; };
    const BoolKey bool_keys[] = {
        {{"hdf5_output","h5_output"}, &p.solver.hdf5_output},
        {{"debug_csv_output","csv_output"}, &p.solver.debug_csv_output},
        {{"model_output_subdir"}, &p.solver.model_output_subdir},
        {{"vtk_output"}, &p.solver.vtk_output},
        {{"svg_output"}, &p.solver.svg_output},
        {{"final_dofs_output"}, &p.solver.final_dofs_output},
        {{"plot_window","show_plot_window"}, &p.solver.plot_window},
        {{"plot_each_step","show_plot_each_step","live_plot"}, &p.solver.plot_each_step},
        {{"postprocess_deferred","deferred_postprocess","defer_postprocess"}, &p.solver.postprocess_deferred},
        {{"force_output","drag_lift_output"}, &p.solver.force_output},
        {{"minj_output","mesh_quality_output"}, &p.solver.minJ_output},
        {{"minj_only_fsi"}, &p.solver.minJ_only_fsi},
        {{"hysteresis_output","hysteresis","reaction_output"}, &p.solver.hysteresis_output}
    };
 
    std::string key;
    std::stringstream ss;
    while(next_config_line(file,key,ss)){
        key = lower_copy(key);
        bool handled=false;
        for(const auto& item:bool_keys){
            if(key_in(key,item.aliases)){ *item.value=read_on_off_bool(ss,key); handled=true; break; }
        }
        if(handled) continue;
 
        if(key=="output_prefix"){
            ss >> p.solver.output_prefix;
            // If output_directory/output_name appeared before output_prefix,
            // keep their explicit override semantics.
            rebuild_output_prefix();
        } else if(key=="output_directory" || key=="output_dir" || key=="output_root"){
            ss >> output_directory;
            if(output_directory.empty()) throw std::runtime_error("Bad output_directory line in postprocessing file.");
            output_directory = trim_trailing_slashes(output_directory);
            p.solver.output_root = output_directory;
            p.solver.vtk_dir = output_directory + "/vtk";
            p.solver.csv_dir = output_directory + "/csv";
            p.solver.figure_dir = output_directory + "/figures";
            p.solver.dof_dir = output_directory + "/dofs";
            rebuild_output_prefix();
        } else if(key=="create" || key=="folder_name"){
            if(key=="create"){
                std::string folder, name;
                ss >> folder >> name;
                if(lower_copy(folder)!="folder" || lower_copy(name)!="name")
                    throw std::runtime_error("Expected Create Folder Name in postprocessing file.");
            }
            p.solver.folder_name = rest_trim(ss);
            if(p.solver.folder_name.empty())
                throw std::runtime_error("Missing value for Create Folder Name in postprocessing file.");
        } else if(key=="vtk_dir" || key=="vtk_output_dir"){
            ss >> p.solver.vtk_dir;
        } else if(key=="csv_dir" || key=="csv_output_dir"){
            ss >> p.solver.csv_dir;
        } else if(key=="figure_dir" || key=="figures_dir" || key=="svg_dir"){
            ss >> p.solver.figure_dir;
        } else if(key=="dof_dir" || key=="dofs_dir"){
            ss >> p.solver.dof_dir;
        } else if(key=="hdf5_file" || key=="h5_file" || key=="hdf5_output_file"){
            ss >> p.solver.hdf5_file;
        } else if(key=="save_data" || key=="hdf5_save_data" || key=="gauss_save_data"){
            p.solver.save_data.clear();
            std::string item;
            while(ss >> item){
                item = lower_copy(item);
                std::replace(item.begin(), item.end(), '-', '_');
                p.solver.save_data.push_back(item);
            }
        } else if(key=="plot_script" || key=="plot_window_script"){
            ss >> p.solver.plot_script;
        } else if(key=="plot_quantity" || key=="plot_component"){
            // Consumed by scripts/plot_vtk_result.py through --post.  The C++
            // parser still has to accept the key so Postprocessing.txt remains
            // the single source of truth for the plot window settings.
            std::string ignored;
            ss >> ignored;
            if(ignored.empty()) throw std::runtime_error("Bad " + key + " line in postprocessing file.");
        } else if(key=="multiscale_plot_point"){
            TimeSolverSettings::MultiscalePlotPoint pt;
            ss >> pt.x >> pt.y;
            if(!ss) throw std::runtime_error("Bad multiscale_plot_point line in postprocessing file.");
            ss >> pt.label;
            if(pt.label.empty()) pt.label = "P" + std::to_string(p.solver.multiscale_plot_points.size() + 1);
            p.solver.multiscale_plot_points.push_back(pt);
        } else if(key=="multiscale_rve_output_dir"){
            ss >> p.solver.multiscale_rve_output_dir;
            if(p.solver.multiscale_rve_output_dir.empty()) throw std::runtime_error("Bad multiscale_rve_output_dir line in postprocessing file.");
        } else if(key.rfind("multiscale_",0)==0){
            // Multiscale dashboard options are consumed by the Python plot script.
        } else if(key=="vtk_interval"){
            ss >> p.solver.output_every;
        } else if(key=="svg_interval"){
            int ignored=0; ss >> ignored;
        } else if(key=="output_name" || key=="case_name"){
            ss >> output_name;
            if(output_name.empty()) throw std::runtime_error("Bad output_name line in postprocessing file.");
            rebuild_output_prefix();
        } else if(key=="output_every"){
            ss >> p.solver.output_every;
        } else if(key=="force_boundary" || key=="drag_lift_boundary"){
            ss >> p.solver.force_boundary;
        } else if(key=="force_boundaries" || key=="drag_lift_boundaries"){
            p.solver.force_boundaries.clear();
            std::string boundary;
            while(ss >> boundary) p.solver.force_boundaries.push_back(boundary);
            if(p.solver.force_boundaries.empty()){
                throw std::runtime_error("Bad force_boundaries line: provide at least one boundary name.");
            }
        } else if(key=="force_boundary_check_tol" || key=="boundary_integral_tol"){
            ss >> p.solver.force_boundary_check_tol;
        } else if(key=="force_attack_angle_deg" || key=="angle_of_attack_deg" || key=="aoa_deg"){
            ss >> p.solver.force_attack_angle_deg;
        } else if(key=="force_csv" || key=="drag_lift_csv"){
            ss >> p.solver.force_csv;
        } else if(key=="force_svg" || key=="drag_lift_svg"){
            ss >> p.solver.force_svg;
        } else if(key=="minJ_csv" || key=="minj_csv"){
            ss >> p.solver.minJ_csv;
        } else if(key=="hysteresis_target" || key=="reaction_target"){
            // Accepted forms:
            //   hysteresis_target TOP uy
            //   hysteresis_target displacement TOP y
            std::string a,b,c; ss >> a >> b;
            if(a.empty() || b.empty()) throw std::runtime_error("Bad hysteresis_target line.");
            if(is_known_field_name(a)){
                ss >> c;
                if(c.empty()) throw std::runtime_error("Bad hysteresis_target line. Use: hysteresis_target field group component");
                bool explicit_field=false;
                p.solver.hysteresis_target = parse_field_component_target_tokens(a,b,c,explicit_field);
            } else {
                p.solver.hysteresis_target = target_from_alias(a,b);
            }
        } else if(key=="hysteresis_x_axis" || key=="hysteresis_x" || key=="hysteresis_axis_x"){
            p.solver.hysteresis_x_axis = parse_hysteresis_axis(ss,"hysteresis_x_axis");
        } else if(key=="hysteresis_y_axis" || key=="hysteresis_y" || key=="hysteresis_axis_y"){
            p.solver.hysteresis_y_axis = parse_hysteresis_axis(ss,"hysteresis_y_axis");
        } else if(key=="hysteresis_csv" || key=="reaction_csv"){
            ss >> p.solver.hysteresis_csv;
        } else if(key=="hysteresis_svg" || key=="reaction_svg"){
            ss >> p.solver.hysteresis_svg;
        } else if(key=="hysteresis_sign" || key=="reaction_sign"){
            ss >> p.solver.hysteresis_sign;
            p.solver.hysteresis_sign = lower_copy(p.solver.hysteresis_sign);
        } else {
            throw std::runtime_error("Unknown key in postprocessing file: "+key);
        }
    }
