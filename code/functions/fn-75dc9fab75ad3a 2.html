static std::string pressure_boundary_name_from_section(const std::string& section){
    if(section.rfind("PRESSURE_BOUNDARY_",0)==0) return section.substr(std::string("PRESSURE_BOUNDARY_").size());
    if(section.rfind("P_",0)==0) return section.substr(2);
    return "";
}
