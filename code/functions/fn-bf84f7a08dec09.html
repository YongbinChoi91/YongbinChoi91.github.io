static void collect_hysteresis_dofs_from_mesh(
    const FieldMesh& mesh,
    const FieldComponentTarget& target,
    int global_offset,
    std::set<int>& dofs
){
    const DofLayout& gmesh=get_cached_dof_layout(mesh);
    const int fid=find_field_index(gmesh,target.field,true);
    const FieldInfo& field=gmesh.fields[fid];
    if(target.component<0 || target.component>=field.components){
        std::ostringstream os;
        os << "hysteresis_target component " << target.component << " is incompatible with field '"
           << target.field << "' which has " << field.components << " component(s).";
        throw std::runtime_error(os.str());
    }
    auto it=field.boundaries.find(target.group);
    if(it==field.boundaries.end()) throw std::runtime_error("Unknown hysteresis boundary group for field '"+target.field+"': "+target.group);
    std::set<int> nodes;
    for(const auto& ent: it->second){
        for(int node: ent.node_ids) nodes.insert(node);
    }
    for(int node:nodes){
        if(node<0 || node>=static_cast<int>(field.nodes.size())){
            throw std::runtime_error("hysteresis_target boundary node id out of range for field '"+target.field+"'.");
        }
        dofs.insert(global_offset + global_field_dof(gmesh,fid,node,target.component));
    }
}
