void write_hdf5_final_raw_dofs(const ProblemParameters& params,const Vector& x){
    if(!params.solver.hdf5_output) return;
#ifdef HIPECS_USE_HDF5
    hid_t file=h5_open_file_rw(params);
    const std::string path="/solution/final_raw_dofs";
    h5_ensure_group(file,"/solution");
    if(H5Lexists(file,path.c_str(),H5P_DEFAULT)>0) h5_check(H5Ldelete(file,path.c_str(),H5P_DEFAULT),"deleting dataset " + path);
    hsize_t dims[2]={static_cast<hsize_t>(x.size()),2};
    hid_t space=H5Screate_simple(2,dims,nullptr);
    hid_t dset=H5Dcreate2(file,path.c_str(),H5T_IEEE_F64LE,space,H5P_DEFAULT,H5P_DEFAULT,H5P_DEFAULT);
    if(dset<0) throw std::runtime_error("Could not create HDF5 dataset: " + path);
    std::vector<double> data;
    data.reserve(x.size()*2);
    for(int i=0;i<static_cast<int>(x.size());++i){
        data.push_back(static_cast<double>(i));
        data.push_back(x[i]);
    }
    h5_check(H5Dwrite(dset,H5T_NATIVE_DOUBLE,H5S_ALL,H5S_ALL,H5P_DEFAULT,data.data()),"writing " + path);
    h5_write_string_attr(dset,"columns","global_dof,value");
    H5Dclose(dset); H5Sclose(space); H5Fclose(file);
#else
    reset_hdf5_table(params,"/solution/final_raw_dofs",{"global_dof","value"});
    for(int i=0;i<static_cast<int>(x.size());++i){
        append_hdf5_table_row(params,"/solution/final_raw_dofs",{static_cast<double>(i),x[i]});
    }
#endif
}
