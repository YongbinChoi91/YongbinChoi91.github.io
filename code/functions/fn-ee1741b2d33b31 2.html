static std::string compact_log_number(double value){
    std::ostringstream os;
    os << std::fixed << std::setprecision(6) << value;
    std::string s = os.str();
    while(s.size() > 1 && s.back() == '0') s.pop_back();
    if(!s.empty() && s.back() == '.') s.push_back('0');
    return s;
}
