static void print_progress_bar(double progress,
                               const std::string& label,
                               double final_residual,
                               int newton_iterations,
                               const HistoryLocalDiagnostics& history_diagnostics,
                               bool show_history_diagnostics){
    progress=std::max(0.0,std::min(1.0,progress));
    const int width=50;
    const int filled=static_cast<int>(std::round(progress*width));
    const int percent=static_cast<int>(std::round(progress*100.0));
 
    std::ostringstream os;
    os << "\r\033[K" << label << " [";
    for(int i=0;i<width;++i){
        os << (i<filled ? "█" : "░");
    }
    os << "] " << std::setw(3) << percent << "%"
       << "  Residual=" << std::scientific << std::setprecision(3) << final_residual
       << "  Newton Iter=" << newton_iterations;
 
    if(show_history_diagnostics && history_diagnostics.active){
        os << "  Local Res=" << std::scientific << std::setprecision(3) << history_diagnostics.max_residual
           << "  Local Iter=" << history_diagnostics.max_iterations;
    }
 
    std::cout << os.str() << std::flush;
    if(progress>=1.0-1.0e-14) std::cout << "\n";
}
