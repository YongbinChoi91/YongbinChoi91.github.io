static double residual_norm_free(const Vector& R,const ReducedSystemWorkspace& w){
    if(static_cast<int>(R.size()) != static_cast<int>(w.fixed_mask.size())){
        throw std::runtime_error("Residual/free mask dimension mismatch.");
    }
    double s=0.0;
    int count=0;
    for(int i=0;i<static_cast<int>(R.size());++i){
        if(!w.fixed_mask[i]){
            s += R[i]*R[i];
            ++count;
        }
    }
    return count>0 ? std::sqrt(s) : 0.0;
}
