inline std::vector<std::vector<Scalar>> inv(
    const std::vector<std::vector<Scalar>>& A
) {
    const int n = static_cast<int>(A.size());
    if (n <= 0) throw std::runtime_error("inv received an empty matrix.");
    for (const auto& row : A) {
        if (static_cast<int>(row.size()) != n) {
            throw std::runtime_error("inv expects a square matrix.");
        }
    }
 
    std::vector<std::vector<Scalar>> B(n, std::vector<Scalar>(n, Scalar(0.0)));
    const Scalar detA = det(A);
 
    if (n == 2) {
        B[0][0] =  A[1][1] / detA;
        B[0][1] = -A[0][1] / detA;
        B[1][0] = -A[1][0] / detA;
        B[1][1] =  A[0][0] / detA;
        return B;
    }
 
    if (n == 3) {
        B[0][0] =  (A[1][1]*A[2][2] - A[1][2]*A[2][1]) / detA;
        B[0][1] = -(A[0][1]*A[2][2] - A[0][2]*A[2][1]) / detA;
        B[0][2] =  (A[0][1]*A[1][2] - A[0][2]*A[1][1]) / detA;
 
        B[1][0] = -(A[1][0]*A[2][2] - A[1][2]*A[2][0]) / detA;
        B[1][1] =  (A[0][0]*A[2][2] - A[0][2]*A[2][0]) / detA;
        B[1][2] = -(A[0][0]*A[1][2] - A[0][2]*A[1][0]) / detA;
 
        B[2][0] =  (A[1][0]*A[2][1] - A[1][1]*A[2][0]) / detA;
        B[2][1] = -(A[0][0]*A[2][1] - A[0][1]*A[2][0]) / detA;
        B[2][2] =  (A[0][0]*A[1][1] - A[0][1]*A[1][0]) / detA;
        return B;
    }
 
    throw std::runtime_error("inv supports only 2D/3D matrices.");
}
