Vector solve_linear_system_backend_petsc_matfree(
    const Vector& rhs,
    const LinearSolverSettings& settings,
    const PetscMatFreeOperator& op,
    const CSRMatrix* preconditioner_matrix,
    const PetscPCDData* pcd_data
){
    (void)preconditioner_matrix;
    (void)pcd_data;
    if(op.n<=0 || static_cast<int>(rhs.size())!=op.n) throw std::runtime_error("PETSc MatFree dimension mismatch.");
    if(!op.apply) throw std::runtime_error("PETSc MatFree operator has no callback.");
    if(lower_copy(settings.backend)!="petsc") throw std::runtime_error("petsc_matfree_full requires solver_backend petsc.");
    const std::string mfpc=lower_copy(settings.petsc_matfree_pc.empty()?settings.preconditioner:settings.petsc_matfree_pc);
    if(mfpc!="none" && mfpc!="jacobi" && mfpc!="element_block_jacobi" && mfpc!="element_jacobi"){
        throw std::runtime_error("PETSc MatFree backend supports petsc_matfree_pc none, jacobi, or element_block_jacobi.");
    }
 
    const std::string solver=lower_copy(settings.linear_solver);
    if(solver=="direct" || solver=="lu" || solver=="preonly"){
        throw std::runtime_error(
            "petsc_matfree_full requires an iterative KSP such as fgmres/gmres/bicgstab. "
            "direct/preonly/lu cannot solve a MatShell operator."
        );
    }
 
    auto chk=[](PetscErrorCode ierr,const char* where){ if(ierr) throw std::runtime_error(std::string("PETSc MatFree error at ")+where); };
    int argc=0; char** argv=nullptr; PetscBool initialized=PETSC_FALSE;
    chk(PetscInitialized(&initialized),"PetscInitialized");
    if(!initialized) chk(PetscInitialize(&argc,&argv,nullptr,nullptr),"PetscInitialize");
 
    auto set_ksp_type_from_string=[&](KSP target,const std::string& name){
        const std::string v=lower_copy(name);
        if(v=="preonly") chk(KSPSetType(target,KSPPREONLY),"KSPSetType preonly");
        else if(v=="fgmres") chk(KSPSetType(target,KSPFGMRES),"KSPSetType fgmres");
        else if(v=="gmres") chk(KSPSetType(target,KSPGMRES),"KSPSetType gmres");
        else if(v=="bicgstab" || v=="bcgs") chk(KSPSetType(target,KSPBCGS),"KSPSetType bicgstab");
        else if(v=="cg") chk(KSPSetType(target,KSPCG),"KSPSetType cg");
        else throw std::runtime_error("Unknown PETSc KSP type: "+name);
        if((v=="gmres" || v=="fgmres") && settings.gmres_restart>0)
            chk(KSPGMRESSetRestart(target,settings.gmres_restart),"KSPGMRESSetRestart");
    };
 
    const int n=op.n;
    PetscMatFreeShellContext shell_ctx{&op};
    PetscMatFreeBlockJacobiContext pc_ctx{&op};
    Mat A=nullptr;
    Vec b=nullptr,x=nullptr;
    KSP ksp=nullptr;
    chk(MatCreateShell(PETSC_COMM_SELF,n,n,n,n,&shell_ctx,&A),"MatCreateShell");
    chk(MatShellSetOperation(A,MATOP_MULT,(void(*)(void))apply_petsc_matfree_shell),"MatShellSetOperation MATOP_MULT");
    chk(MatCreateVecs(A,&x,&b),"MatCreateVecs");
    PetscScalar* ba=nullptr;
    chk(VecGetArray(b,&ba),"VecGetArray b");
    for(int i=0;i<n;++i) ba[i]=rhs[i];
    chk(VecRestoreArray(b,&ba),"VecRestoreArray b");
    chk(VecSet(x,0.0),"VecSet x");
 
    chk(KSPCreate(PETSC_COMM_SELF,&ksp),"KSPCreate");
    set_ksp_type_from_string(ksp,solver);
    chk(KSPSetOperators(ksp,A,A),"KSPSetOperators");
    chk(KSPSetTolerances(ksp,settings.tol,settings.atol>=0.0?settings.atol:PETSC_DEFAULT,PETSC_DEFAULT,settings.max_iter),"KSPSetTolerances");
    // CG requires left preconditioning; GMRES/FGMRES keep their existing side.
    chk(KSPSetPCSide(ksp,solver=="cg" ? PC_LEFT : PC_RIGHT),"KSPSetPCSide");
    chk(KSPSetNormType(ksp,KSP_NORM_UNPRECONDITIONED),"KSPSetNormType unpreconditioned");
 
    PC pc=nullptr;
    chk(KSPGetPC(ksp,&pc),"KSPGetPC");
    if(mfpc=="none"){
        chk(PCSetType(pc,PCNONE),"PCSetType none");
    } else {
        chk(PCSetType(pc,PCSHELL),"PCSetType MatFree block Jacobi PCSHELL");
        chk(PCShellSetContext(pc,&pc_ctx),"PCShellSetContext MatFree block Jacobi");
        chk(PCShellSetApply(pc,apply_petsc_matfree_block_jacobi_pc),"PCShellSetApply MatFree block Jacobi");
    }
 
    chk(KSPSetFromOptions(ksp),"KSPSetFromOptions");
 
    if(settings.verbose){
        std::cout << "    PETSc MatFree KSP\n"
                  << "      ksp       : " << solver << "\n"
                  << "      pc        : " << mfpc << "\n"
                  << "      pmat      : none\n"
                  << "      jvp       : " << settings.petsc_matfree_jvp_mode << "\n";
    }
 
    const auto solve_start=std::chrono::steady_clock::now();
    chk(KSPSolve(ksp,b,x),"KSPSolve MatFree");
    const auto solve_end=std::chrono::steady_clock::now();
    KSPConvergedReason reason;
    chk(KSPGetConvergedReason(ksp,&reason),"KSPGetConvergedReason");
    PetscInt its=0;
    chk(KSPGetIterationNumber(ksp,&its),"KSPGetIterationNumber");
    if(settings.verbose){
        PetscReal rnorm=0.0;
        chk(KSPGetResidualNorm(ksp,&rnorm),"KSPGetResidualNorm");
        const double solve_sec=std::chrono::duration<double>(solve_end-solve_start).count();
        std::cout << "    PETSc MatFree KSP result: its=" << static_cast<int>(its)
                  << ", reason=" << static_cast<int>(reason)
                  << ", rnorm=" << static_cast<double>(rnorm)
                  << ", solve_time=" << solve_sec << " s\n";
    }
    if(reason<0) throw std::runtime_error("PETSc MatFree KSP did not converge. Iterations="+std::to_string(static_cast<int>(its))+
                                           ", reason="+std::to_string(static_cast<int>(reason)));
    Vector sol(n,0.0);
    const PetscScalar* xa=nullptr;
    chk(VecGetArrayRead(x,&xa),"VecGetArrayRead x");
    for(int i=0;i<n;++i) sol[i]=static_cast<double>(xa[i]);
    chk(VecRestoreArrayRead(x,&xa),"VecRestoreArrayRead x");
    KSPDestroy(&ksp);
    VecDestroy(&x);
    VecDestroy(&b);
    MatDestroy(&A);
    return sol;
}
