inline Scalar inner(const std::array<std::array<Scalar, C>, R>& A,
                    const std::array<std::array<Scalar, C>, R>& B) {
    Scalar s(0.0);
    for (std::size_t i = 0; i < R; ++i) for (std::size_t j = 0; j < C; ++j) s = s + A[i][j] * B[i][j];
    return s;
}
