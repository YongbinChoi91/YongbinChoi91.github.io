void write_multiscale_rve_samples_for_element(
    const FieldMesh& macro_mesh,
    int elem_id,
    const ProblemParameters& params,
    int step,
    const Vector& rve_displacement,
    const MaterialHistoryStore* rve_history
) {
    const std::vector<std::string> labels =
        multiscale_rve_sample_labels_for_element(macro_mesh, elem_id, params);
    if (labels.empty()) return;
 
    const FieldMesh& micro_mesh = multiscale_rve_micro_problem().domains.front().mesh;
    for (const std::string& label : labels) {
        write_vtk(micro_mesh, rve_displacement, multiscale_rve_sample_filename(params, step, label), rve_history);
    }
}
