inline std::vector<std::vector<Scalar>> mul(
    const std::vector<std::vector<Scalar>>& A,
    const std::vector<std::vector<Scalar>>& B
) {
    const int m = static_cast<int>(A.size());
    const int k = m > 0 ? static_cast<int>(A[0].size()) : 0;
    const int n = static_cast<int>(B.empty() ? 0 : B[0].size());
    if (static_cast<int>(B.size()) != k) throw std::runtime_error("Bad matrix dimensions in mul.");
    std::vector<std::vector<Scalar>> C(m, std::vector<Scalar>(n, Scalar(0.0)));
    for (int i = 0; i < m; ++i) {
        if (static_cast<int>(A[i].size()) != k) throw std::runtime_error("Ragged left matrix in mul.");
        for (int q = 0; q < k; ++q) {
            if (static_cast<int>(B[q].size()) != n) throw std::runtime_error("Ragged right matrix in mul.");
            for (int j = 0; j < n; ++j) C[i][j] += A[i][q] * B[q][j];
        }
    }
    return C;
}
