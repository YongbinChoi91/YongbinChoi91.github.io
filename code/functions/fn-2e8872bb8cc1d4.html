DofLayout build_dof_layout(const FieldMesh& mesh){
    DofLayout g;
    g.dim=mesh.dim;
 
    const std::vector<std::string> order = active_dof_field_names(mesh);
    for(const std::string& name:order){
        if(find_field_index(g,name,false)>=0) throw std::runtime_error("Duplicate field in DOFS_info.txt field_key list: "+name);
        g.fields.push_back(make_builtin_field_info(mesh,name));
    }
 
    assign_field_dof_offsets(g);
    return g;
}
