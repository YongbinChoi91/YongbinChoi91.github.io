void append_force_csv(const std::string& filename,const BoundaryForceResult& f){
    ensure_output_parent_directory(filename);
    std::ofstream out(filename,std::ios::app);
    if(!out) throw std::runtime_error("Could not open force CSV for appending: "+filename);
    out.setf(std::ios::scientific);
    out.precision(17);
    out << f.time << ',' << f.Drag << ',' << f.Lift << '\n';
}
