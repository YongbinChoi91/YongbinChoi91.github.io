inline LinearMultistepCoefficients Adams_Moulton(int num_steps) {
    switch (num_steps) {
    case 1:
        return {"AdamsMoulton", 2, 1, true, {1.0 / 2.0, 1.0 / 2.0}};
    case 2:
        return {"AdamsMoulton", 3, 2, true, {5.0 / 12.0, 8.0 / 12.0, -1.0 / 12.0}};
    case 3:
        return {"AdamsMoulton", 4, 3, true, {9.0 / 24.0, 19.0 / 24.0, -5.0 / 24.0, 1.0 / 24.0}};
    case 4:
        return {"AdamsMoulton", 5, 4, true, {251.0 / 720.0, 646.0 / 720.0, -264.0 / 720.0, 106.0 / 720.0, -19.0 / 720.0}};
    case 5:
        return {"AdamsMoulton", 6, 5, true, {475.0 / 1440.0, 1427.0 / 1440.0, -798.0 / 1440.0, 482.0 / 1440.0, -173.0 / 1440.0, 27.0 / 1440.0}};
    case 6:
        return {"AdamsMoulton", 7, 6, true, {19087.0 / 60480.0, 65112.0 / 60480.0, -46461.0 / 60480.0, 37504.0 / 60480.0, -20211.0 / 60480.0, 6312.0 / 60480.0, -863.0 / 60480.0}};
    default:
        throw std::runtime_error("Unsupported Adams-Moulton step count. Supported steps: 1 to 6.");
    }
}
