static PetscErrorCode apply_pcd_shell(PC pc, Vec x, Vec y){
    PCDShellContext* ctx=nullptr;
    PetscCall(PCShellGetContext(pc,(void**)&ctx));
    PetscInt n=0;
    PetscCall(VecGetSize(x,&n));
 
    // z = Kp_eff^{-1} x
    PetscCall(VecSet(ctx->tmp1,0.0));
    PetscCall(KSPSolve(ctx->kspKp,x,ctx->tmp1));
 
    // w = Mp_lumped z
    const PetscScalar* z=nullptr;
    PetscScalar* w=nullptr;
    PetscCall(VecGetArrayRead(ctx->tmp1,&z));
    PetscCall(VecGetArray(ctx->tmp2,&w));
    for(PetscInt i=0;i<n;++i){
        const double m = (i<static_cast<PetscInt>(ctx->Mdiag.size())) ? ctx->Mdiag[static_cast<std::size_t>(i)] : 1.0;
        w[i] = z[i] * m;
    }
    PetscCall(VecRestoreArray(ctx->tmp2,&w));
    PetscCall(VecRestoreArrayRead(ctx->tmp1,&z));
 
    // y = Ap^{-1} w
    PetscCall(VecSet(y,0.0));
    PetscCall(KSPSolve(ctx->kspAp,ctx->tmp2,y));
    return 0;
}
