inline Scalar det(const std::vector<std::vector<Scalar>>& A) {
    const int n = static_cast<int>(A.size());
    if (n <= 0) throw std::runtime_error("det received an empty matrix.");
    for (const auto& row : A) {
        if (static_cast<int>(row.size()) != n) {
            throw std::runtime_error("det expects a square matrix.");
        }
    }
 
    if (n == 2) {
        return A[0][0] * A[1][1] - A[0][1] * A[1][0];
    }
    if (n == 3) {
        return A[0][0] * (A[1][1] * A[2][2] - A[1][2] * A[2][1])
             - A[0][1] * (A[1][0] * A[2][2] - A[1][2] * A[2][0])
             + A[0][2] * (A[1][0] * A[2][1] - A[1][1] * A[2][0]);
    }
    throw std::runtime_error("det supports only 2D/3D matrices.");
}
