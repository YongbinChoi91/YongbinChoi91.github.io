static std::vector<int> pressure_connectivity_for_owner(
    const FieldMesh& mesh,
    const ResolvedBoundaryEdge& r
){
    // Python Get_Surface_Info uses the recovered Surface element connectivity
    // for all fields.  For Q9/Q4 and T6/T3 mixed elements, pressure is the
    // low-order part of that recovered owner row.  In shared multi-domain this owner row
    // comes from TOTAL_ELEMENTS, so mesh.pressure_elements[r.elem_id] is the
    // wrong table and may even have a different row numbering.
    const auto& ve = owner_connectivity(mesh,r);
 
    if(r.use_total_elements){
        if(mesh.pressure_type==ElementType::Q4 && ve.size()>=4) return {ve[0],ve[1],ve[2],ve[3]};
        if(mesh.pressure_type==ElementType::T3 && ve.size()>=3) return {ve[0],ve[1],ve[2]};
        throw std::runtime_error("Cannot build low-order pressure connectivity from recovered TOTAL surface owner.");
    }
 
    if(r.elem_id<0 || r.elem_id>=static_cast<int>(mesh.pressure_elements.size()))
        throw std::runtime_error("Bad pressure element id while computing boundary force.");
 
    std::vector<int> pe = mesh.pressure_elements[r.elem_id];
    if(pe.empty()){
        if(mesh.pressure_type==ElementType::Q4 && ve.size()>=4) pe={ve[0],ve[1],ve[2],ve[3]};
        else if(mesh.pressure_type==ElementType::T3 && ve.size()>=3) pe={ve[0],ve[1],ve[2]};
        else throw std::runtime_error("Empty pressure connectivity for recovered surface owner.");
    }
    return pe;
}
