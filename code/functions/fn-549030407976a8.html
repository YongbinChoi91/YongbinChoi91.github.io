static const FieldInfo* primary_postprocess_field(
    const FieldMesh& mesh,
    const std::vector<std::string>& field_names
){
    const DofLayout& layout = get_cached_dof_layout(mesh);
    for(const std::string& name:field_names){
        const int fid = find_field_index(layout,name,false);
        if(fid>=0 && !layout.fields[fid].elements.empty()) return &layout.fields[fid];
    }
    for(const FieldInfo& field:layout.fields){
        if(!field.elements.empty()) return &field;
    }
    return nullptr;
}
