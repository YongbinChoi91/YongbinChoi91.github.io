std::string element_type_to_string(ElementType t){
    switch(t){
        case ElementType::T3: return "T3";
        case ElementType::T6: return "T6";
        case ElementType::Q4: return "Q4";
        case ElementType::Q9: return "Q9";
        case ElementType::Tet4: return "Tet4";
        case ElementType::Tet10: return "Tet10";
        case ElementType::H8: return "H8";
        case ElementType::H20: return "H20";
        case ElementType::H27: return "H27";
        case ElementType::Polygon: return "Polygon";
        default: return "Unknown";
    }
}
