static std::vector<int> parse_ints_from_string(const std::string& text,const std::string& where){
    std::stringstream ss(text);
    std::vector<int> out;
    int v;
    while(ss >> v) out.push_back(v);
    if(out.empty()) throw std::runtime_error("Empty integer list while parsing " + where + ".");
    return out;
}
