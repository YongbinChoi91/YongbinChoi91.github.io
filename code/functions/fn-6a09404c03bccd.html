static Vector solve_dense_regularized(std::vector<double> A, Vector b, int n){
    if(static_cast<int>(A.size())!=n*n || static_cast<int>(b.size())!=n) throw std::runtime_error("Bad dense solve size.");
    double scale=0.0;
    for(int i=0;i<n;++i) scale=std::max(scale,std::abs(A[i*n+i]));
    const double shift=std::max(1.0e-12,1.0e-8*std::max(scale,1.0));
    for(int i=0;i<n;++i) A[i*n+i] += (A[i*n+i] < 0.0 ? -shift : shift);
    for(int k=0;k<n;++k){
        int piv=k;
        double best=std::abs(A[k*n+k]);
        for(int i=k+1;i<n;++i){
            const double v=std::abs(A[i*n+k]);
            if(v>best){ best=v; piv=i; }
        }
        if(best<1.0e-20) A[k*n+k] += (A[k*n+k] < 0.0 ? -shift : shift);
        if(piv!=k){
            for(int j=k;j<n;++j) std::swap(A[k*n+j],A[piv*n+j]);
            std::swap(b[k],b[piv]);
        }
        const double akk=A[k*n+k];
        if(std::abs(akk)<1.0e-30) continue;
        for(int i=k+1;i<n;++i){
            const double f=A[i*n+k]/akk;
            A[i*n+k]=0.0;
            for(int j=k+1;j<n;++j) A[i*n+j] -= f*A[k*n+j];
            b[i] -= f*b[k];
        }
    }
    Vector x(n,0.0);
    for(int i=n-1;i>=0;--i){
        double s=b[i];
        for(int j=i+1;j<n;++j) s -= A[i*n+j]*x[j];
        const double aii=A[i*n+i];
        x[i]=(std::abs(aii)>1.0e-30) ? s/aii : 0.0;
    }
    return x;
}
