static bool node_matches_selector(
    const Node& node,
    const BCSelector& selector,
    int dim
){
    const double tol2=selector.tol*selector.tol;
    if(selector.kind==BCSelectorKind::Point){
        double d2=0.0;
        for(int d=0; d<dim; ++d){
            const double diff=node.x[d]-selector.p0[d];
            d2 += diff*diff;
        }
        return d2<=tol2;
    }
    if(selector.kind==BCSelectorKind::Line){
        return point_segment_distance_squared(node.x,selector.p0,selector.p1,dim)<=tol2;
    }
    return false;
}
