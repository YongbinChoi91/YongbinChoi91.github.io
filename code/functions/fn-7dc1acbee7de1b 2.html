LocalFieldView make_local_field_view(
    const FieldMesh& mesh,
    int elem_id,
    const std::vector<std::string>& field_order
) {
    LocalFieldView view;
    view.dim = mesh.dim;
    for (const auto& name : field_order) {
        append_local_field_from_mixed(view, mesh, elem_id, name);
    }
    return view;
}
