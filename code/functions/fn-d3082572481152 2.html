std::string write_multidomain_vtk_outputs(
    const Problem& problem,
    const Vector& x,
    const ProblemParameters& params,
    int step,
    const MaterialHistoryStore* history
){
    std::string first_file;
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const auto& domain=problem.domains[d];
        if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
        const std::string filename = make_solution_vtk_path(params,problem,domain,step);
        if(first_file.empty()) first_file = filename;
        const MaterialHistoryStore* domain_history = nullptr;
        MaterialHistoryStore single_domain_history;
        if(history && d<static_cast<int>(history->domains.size()) && !history->domains[d].elements.empty()){
            single_domain_history.domains.push_back(history->domains[d]);
            domain_history = &single_domain_history;
        }
        if(problem.shared_dofs){
            write_vtk(domain.mesh,x,filename,domain_history);
        } else {
            const Vector xd=extract_domain_vector(x,domain);
            write_vtk(domain.mesh,xd,filename,domain_history);
        }
    }
    return first_file;
}
