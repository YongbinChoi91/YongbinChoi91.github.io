static int integrate_boundary_edges_2d(
    const FieldMesh& mesh,
    const std::string& boundary_name,
    Callback&& callback
){
    if(mesh.dim!=2) throw std::runtime_error("Boundary edge integration currently supports 2D meshes.");
    auto it=mesh.velocity_boundaries.find(boundary_name);
    if(it==mesh.velocity_boundaries.end()){
        throw std::runtime_error("Unknown velocity boundary group for boundary integration: "+boundary_name);
    }
 
    const auto qline=gauss_rule_line(3);
    const auto boundary_center=boundary_group_center_from_velocity_nodes(mesh,it->second);
    int integrated_entities = 0;
 
    for(const auto& ent:it->second){
        if(mesh.velocity_type==ElementType::Polygon){
            ResolvedBoundaryEdge r;
            auto contains_boundary_nodes = [&](int e){
                if(e<0 || e>=static_cast<int>(mesh.velocity_elements.size())) return false;
                const auto& conn=mesh.velocity_elements[e];
                for(int node:ent.node_ids){
                    if(std::find(conn.begin(),conn.end(),node)==conn.end()) return false;
                }
                return true;
            };
            if(contains_boundary_nodes(ent.element_id)) r.elem_id=ent.element_id;
            else {
                for(int e=0;e<static_cast<int>(mesh.velocity_elements.size());++e){
                    if(contains_boundary_nodes(e)){ r.elem_id=e; break; }
                }
            }
            if(r.elem_id<0 || ent.node_ids.size()<2) continue;
            ++integrated_entities;
 
            const auto owner_center=element_centroid_from_owner(mesh,r);
            const auto& X0=mesh.velocity_nodes[ent.node_ids[0]].x;
            const auto& X1=mesh.velocity_nodes[ent.node_ids[1]].x;
            const bool quadratic=(ent.node_ids.size()>=3);
            const auto& Xm=mesh.velocity_nodes[quadratic ? ent.node_ids[2] : ent.node_ids[0]].x;
            for(const auto& q:qline){
                const double s=q.xi;
                std::array<double,3> Xq{{0.0,0.0,0.0}};
                std::array<double,3> tangent{{0.0,0.0,0.0}};
                if(quadratic){
                    const double N0=0.5*s*(s-1.0), N1=0.5*s*(s+1.0), Nm=1.0-s*s;
                    const double dN0=s-0.5, dN1=s+0.5, dNm=-2.0*s;
                    for(int d=0;d<3;++d){
                        Xq[d]=N0*X0[d]+N1*X1[d]+Nm*Xm[d];
                        tangent[d]=dN0*X0[d]+dN1*X1[d]+dNm*Xm[d];
                    }
                } else {
                    for(int d=0;d<3;++d){
                        Xq[d]=0.5*(1.0-s)*X0[d]+0.5*(1.0+s)*X1[d];
                        tangent[d]=0.5*(X1[d]-X0[d]);
                    }
                }
                std::array<double,3> normal{{tangent[1],-tangent[0],0.0}};
                const double outward=(Xq[0]-owner_center[0])*normal[0]
                                    +(Xq[1]-owner_center[1])*normal[1];
                if(outward<0.0){ normal[0]*=-1.0; normal[1]*=-1.0; }
 
                BoundaryQuadraturePoint2D qp;
                qp.entity=&ent;
                qp.owner=r;
                qp.element_type=ElementType::Polygon;
                qp.reference_point={{0.0,0.0,0.0}};
                qp.physical_point=Xq;
                qp.normal_measure=normal;
                qp.measure=std::sqrt(normal[0]*normal[0]+normal[1]*normal[1]);
                qp.weight=q.weight;
                callback(qp);
            }
            continue;
        }
 
        const ResolvedBoundaryEdge r=resolve_boundary_edge(mesh,ent,&boundary_center);
        if(r.elem_id < 0) continue;
        const ElementGeometry geom=make_owner_geometry_element(mesh,r);
        const ElementType vtype = geom.type;
        ++integrated_entities;
 
        for(const auto& q : qline){
            BoundaryQuadraturePoint2D qp;
            qp.entity = &ent;
            qp.owner = r;
            qp.geometry = geom;
            qp.element_type = vtype;
            qp.reference_point = edge_reference_point(vtype,r.edge,q.xi);
            const ShapeData sv=compute_shape(vtype,qp.reference_point[0],qp.reference_point[1],qp.reference_point[2]);
            for(std::size_t a=0;a<sv.N.size() && a<geom.coords.size();++a){
                for(int d=0;d<3;++d) qp.physical_point[d] += sv.N[a]*geom.coords[a][d];
            }
            qp.jacobian=compute_jacobian_data(geom,sv);
            if(qp.jacobian.detJ<=0.0) throw std::runtime_error("Negative detJ during boundary edge integration.");
            qp.fixed_axis=edge_reference_normal_2d(vtype,r.edge);
            qp.normal_measure=surface_area_vector_from_jacobian_2d(qp.jacobian,qp.fixed_axis);
            qp.measure=surface_measure_from_jacobian_2d(qp.jacobian,qp.fixed_axis);
            qp.weight=q.weight;
            callback(qp);
        }
    }
 
    return integrated_entities;
}
