static bool mesh_file_uses_shared_multi_domain(const std::vector<std::string>& lines){
    for(const std::string& original:lines){
        std::string clean=original;
        const auto comment=clean.find('#');
        if(comment!=std::string::npos) clean=clean.substr(0,comment);
        clean=trim_copy(clean);
        if(clean.empty()) continue;
        std::stringstream ss(clean);
        std::string key, value;
        ss >> key;
        if(key=="MESH_TYPE"){
            ss >> value;
            const std::string mesh_type = lower_copy(value);
            if(mesh_type=="multi_domain"){
                return true;
            }
            return false;
        }
    }
    return false;
}
