static void rebuild_pressure_pcd_cache(
    PressurePcdCache& cache,
    const std::vector<int>& free_dofs,
    const FieldMesh& mesh,
    const ProblemParameters& params,
    const LinearSolverSettings& settings,
    double dt_for_schur
){
    const int n_full=total_dof_count(mesh);
    cache=PressurePcdCache();
    cache.valid=true;
    cache.mesh=&mesh;
    cache.free_dofs=free_dofs;
    cache.full_to_pidx.assign(n_full,-1);
    cache.dt=dt_for_schur;
    cache.theta=params.solver.theta;
    cache.rho=params.material.rho;
    cache.mu=params.material.mu;
    cache.kp_mass_shift=settings.pcd_kp_mass_shift;
    cache.ap_mass_weight=settings.pcd_ap_mass_weight;
    cache.ap_laplacian_weight=settings.pcd_ap_laplacian_weight;
    cache.schur_pressure_shift=settings.schur_pressure_shift;
    cache.stabilization_time_order=params.solver.stabilization_time_order;
    cache.gauss_order=params.solver.gauss_order;
 
    int np_red=0;
    for(int rid=0; rid<static_cast<int>(free_dofs.size()); ++rid){
        const int full=free_dofs[rid];
        if(is_pressure_full_dof_for_mesh(mesh,full)) cache.full_to_pidx[full]=np_red++;
    }
    if(np_red<=0) throw std::runtime_error("PCD requested but there are no free pressure DOFs.");
    if(mesh.pressure_elements.empty()) throw std::runtime_error("Cannot build PCD data: no pressure elements.");
    if(mesh.velocity_elements.size() < mesh.pressure_elements.size()) {
        throw std::runtime_error("Cannot build PCD data: velocity/pressure element counts are inconsistent.");
    }
 
    std::vector<std::map<int,char>> pattern_rows(np_red);
    for(const auto& pe:mesh.pressure_elements){
        const int np_loc=static_cast<int>(pe.size());
        for(int a=0;a<np_loc;++a){
            const int Ia_full=pressure_dof(mesh,pe[a]);
            const int Ia=(Ia_full>=0 && Ia_full<n_full) ? cache.full_to_pidx[Ia_full] : -1;
            if(Ia<0) continue;
            for(int b=0;b<np_loc;++b){
                const int Ib_full=pressure_dof(mesh,pe[b]);
                const int Ib=(Ib_full>=0 && Ib_full<n_full) ? cache.full_to_pidx[Ib_full] : -1;
                if(Ib>=0) pattern_rows[Ia][Ib]=1;
            }
        }
    }
 
    // PCD Kp/Ap are used as PETSc inner preconditioner matrices.  PETSc ILU
    // requires a structurally present diagonal in every row, while reduced
    // pressure spaces can contain pinned/isolated pressure rows after boundary
    // reduction.  Complete the pressure CSR pattern here; this changes only the
    // preconditioner operators, not the nonlinear residual or weak form.
    for(int i=0; i<np_red; ++i) pattern_rows[i][i]=1;
 
    CSRMatrix pattern;
    pattern.n_rows=np_red;
    pattern.n_cols=np_red;
    pattern.row_ptr.assign(np_red+1,0);
    int nnz=0;
    for(int i=0;i<np_red;++i){
        pattern.row_ptr[i]=nnz;
        nnz += static_cast<int>(pattern_rows[i].size());
    }
    pattern.row_ptr[np_red]=nnz;
    pattern.col_idx.reserve(nnz);
    pattern.values.assign(nnz,0.0);
    cache.csr_index.assign(np_red,{});
    for(int i=0;i<np_red;++i){
        cache.csr_index[i].reserve(pattern_rows[i].size());
        for(const auto& kv:pattern_rows[i]){
            const int idx=static_cast<int>(pattern.col_idx.size());
            pattern.col_idx.push_back(kv.first);
            cache.csr_index[i].emplace(kv.first,idx);
        }
    }
 
    cache.data.enabled=true;
    cache.data.Mp=make_zero_csr_like(pattern);
    cache.Kp_base=make_zero_csr_like(pattern);
 
    for(int e=0;e<static_cast<int>(mesh.pressure_elements.size());++e){
        const auto& pe=mesh.pressure_elements[e];
        const ElementGeometry geom=make_velocity_element_geometry(mesh,e);
        for(const auto& gp: gauss_rule(mesh.velocity_type, params.solver.gauss_order)){
            const ShapeData sv=compute_shape(mesh.velocity_type,gp.xi,gp.eta,gp.zeta);
            const JacobianData J=compute_jacobian_data(geom,sv);
            if(J.detJ<=0.0) throw std::runtime_error("Negative detJ while assembling PCD pressure operators.");
 
            const ShapeData sp=compute_shape(mesh.pressure_type,gp.xi,gp.eta,gp.zeta);
            const auto dNp=compute_dN_dx(sp,J);
            const double w=J.detJ*gp.weight;
            const int np_loc=std::min(static_cast<int>(pe.size()),static_cast<int>(sp.N.size()));
            for(int a=0;a<np_loc;++a){
                const int Ia_full=pressure_dof(mesh,pe[a]);
                const int Ia=(Ia_full>=0 && Ia_full<n_full) ? cache.full_to_pidx[Ia_full] : -1;
                if(Ia<0) continue;
                for(int b=0;b<np_loc;++b){
                    const int Ib_full=pressure_dof(mesh,pe[b]);
                    const int Ib=(Ib_full>=0 && Ib_full<n_full) ? cache.full_to_pidx[Ib_full] : -1;
                    if(Ib<0) continue;
                    const int idx=pressure_cache_csr_index(cache,Ia,Ib);
                    if(idx<0) throw std::runtime_error("PCD cache CSR entry missing.");
                    double gd=0.0;
                    for(int d=0; d<mesh.dim; ++d) gd += dNp[a][d]*dNp[b][d];
                    cache.data.Mp.values[idx] += sp.N[a]*sp.N[b]*w;
                    cache.Kp_base.values[idx] += gd*w;
                }
            }
        }
    }
 
    const double kp_mass_shift = (settings.pcd_kp_mass_shift<0.0) ? 1.0e-8 : settings.pcd_kp_mass_shift;
    cache.data.Kp=make_zero_csr_like(pattern);
    for(int q=0;q<static_cast<int>(cache.data.Kp.values.size());++q){
        cache.data.Kp.values[q] = cache.Kp_base.values[q] + kp_mass_shift*cache.data.Mp.values[q];
    }
    if(settings.schur_pressure_shift>0.0){
        for(int i=0;i<np_red;++i){
            const int idx=pressure_cache_csr_index(cache,i,i);
            if(idx>=0) cache.data.Kp.values[idx] += settings.schur_pressure_shift;
        }
    }
 
    // Numerical diagonal completion for robust PETSc PC setup.  The diagonal is
    // already present structurally above; here we make sure a completely empty
    // reduced pressure row does not create a zero-pivot preconditioner.
    double kp_diag_scale=0.0;
    int kp_diag_count=0;
    for(int i=0;i<np_red;++i){
        const int idx=pressure_cache_csr_index(cache,i,i);
        if(idx>=0){
            kp_diag_scale += std::abs(cache.data.Kp.values[idx]);
            ++kp_diag_count;
        }
    }
    const double kp_diag_floor = std::max(1.0e-14, 1.0e-12 * (kp_diag_count>0 ? kp_diag_scale/kp_diag_count : 1.0));
    for(int i=0;i<np_red;++i){
        const int idx=pressure_cache_csr_index(cache,i,i);
        if(idx>=0 && std::abs(cache.data.Kp.values[idx]) < kp_diag_floor){
            cache.data.Kp.values[idx] += kp_diag_floor;
        }
    }
 
    cache.data.Ap=make_zero_csr_like(pattern);
    cache.data.Mp_lumped_inv.assign(np_red,1.0);
    for(int i=0;i<np_red;++i){
        double lump=0.0;
        for(int q=cache.data.Mp.row_ptr[i]; q<cache.data.Mp.row_ptr[i+1]; ++q) lump += cache.data.Mp.values[q];
        cache.data.Mp_lumped_inv[i] = (std::abs(lump)>1.0e-30) ? 1.0/lump : 1.0;
    }
}
