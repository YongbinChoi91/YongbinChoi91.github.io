ReducedSystemWorkspace make_reduced_system_workspace_matfree(
    int ndofs,
    const BoundaryData& bc
){
    ReducedSystemWorkspace w;
    w.fixed_mask.assign(ndofs,0);
    for(int d:bc.dofs){
        if(d<0 || d>=ndofs) throw std::runtime_error("Fixed DOF out of range while building matrix-free reduced workspace.");
        w.fixed_mask[d]=1;
    }
    w.free_dofs.reserve(std::max(0,ndofs-static_cast<int>(bc.dofs.size())));
    for(int i=0;i<ndofs;++i) if(!w.fixed_mask[i]) w.free_dofs.push_back(i);
    if(w.free_dofs.empty()) throw std::runtime_error("No free DOFs in matrix-free reduced workspace.");
    return w;
}
