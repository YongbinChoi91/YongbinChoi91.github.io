static std::vector<int> make_domain_element_dofs(
    const Problem& problem,
    int domain_id,
    int elem_id,
    const std::vector<std::string>& active_fields
){
    if(domain_id<0 || domain_id>=static_cast<int>(problem.domains.size()))
        throw std::runtime_error("Domain id out of range in make_domain_element_dofs.");
    const Domain& domain=problem.domains[domain_id];
    const DofLayout& gmesh=get_cached_dof_layout(domain.mesh);
    std::vector<int> dofs=make_element_dofs_for_fields(gmesh,elem_id,active_fields);
    if(!problem.shared_dofs){
        for(int& I:dofs) I += domain.dof_offset;
    }
    return dofs;
}
