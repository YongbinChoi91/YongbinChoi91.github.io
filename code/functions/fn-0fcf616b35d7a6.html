inline Vector solve_local_Gauss_Jordan(
    LocalDenseMatrix A,
    Vector b,
    const std::string& context = "Local dense Gauss-Jordan solve"
) {
    const int n = static_cast<int>(b.size());
    check_local_dense_system(A, n, context);
 
    for (int k = 0; k < n; ++k) {
        int piv = k;
        double best = std::abs(A[k][k]);
        for (int i = k + 1; i < n; ++i) {
            const double v = std::abs(A[i][k]);
            if (v > best) { best = v; piv = i; }
        }
        if (best < 1.0e-20) {
            throw std::runtime_error("Singular " + context + ".");
        }
        if (piv != k) {
            std::swap(A[piv], A[k]);
            std::swap(b[piv], b[k]);
        }
 
        const double akk = A[k][k];
        for (int j = k; j < n; ++j) A[k][j] /= akk;
        b[k] /= akk;
 
        for (int i = 0; i < n; ++i) {
            if (i == k) continue;
            const double f = A[i][k];
            for (int j = k; j < n; ++j) A[i][j] -= f * A[k][j];
            b[i] -= f * b[k];
        }
    }
    return b;
}
