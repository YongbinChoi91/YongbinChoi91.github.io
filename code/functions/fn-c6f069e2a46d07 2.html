inline auto neo_hookean_energy_young(const std::array<std::array<Scalar, Dim>, Dim>& F, double young, double nu) {
    const double mu = young / (2.0 * (1.0 + nu));
    const double lambda = young * nu / ((1.0 + nu) * (1.0 - 2.0 * nu));
    return neo_hookean_energy_lame(F, mu, lambda);
}
