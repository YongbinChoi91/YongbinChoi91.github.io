inline std::array<std::array<std::array<Scalar, InN>, C>, R> extract_output_gradient(
    const std::array<std::array<AD1<Scalar, InN>, C>, R>& y
) {
    std::array<std::array<std::array<Scalar, InN>, C>, R> J{};
    for (std::size_t r = 0; r < R; ++r) {
        for (std::size_t c = 0; c < C; ++c) {
            for (std::size_t i = 0; i < InN; ++i) J[r][c][i] = y[r][c].grad[i];
        }
    }
    return J;
}
