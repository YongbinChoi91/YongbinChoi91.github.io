static void print_generic_field_residual_diagnostics(const Problem& problem,
                                                    const Vector& R,
                                                    const BoundaryData& bc){
    const auto fixed = make_fixed_dof_mask(static_cast<int>(R.size()), bc);
    std::cout << "    residual by field:\n";
    if(problem.shared_dofs){
        const DofLayout& g = get_cached_dof_layout(problem.domains.front().mesh);
        for(const auto& f: g.fields){
            double s=0.0, inf=0.0; int count=0;
            for(int i=f.offset; i<f.offset+f.ndofs && i<static_cast<int>(R.size()); ++i){
                if(fixed[i]) continue;
                const double a=std::abs(R[i]); s += R[i]*R[i]; inf=std::max(inf,a); ++count;
            }
            const double rms = count>0 ? std::sqrt(s/static_cast<double>(count)) : 0.0;
            std::cout << "      " << f.name << "_rms = " << rms << ", " << f.name << "_inf = " << inf << "\n";
        }
        return;
    }
    for(const auto& d: problem.domains){
        const DofLayout& g = get_cached_dof_layout(d.mesh);
        for(const auto& f: g.fields){
            double s=0.0, inf=0.0; int count=0;
            const int begin=d.dof_offset + f.offset;
            const int end=begin + f.ndofs;
            for(int i=begin; i<end && i<static_cast<int>(R.size()); ++i){
                if(fixed[i]) continue;
                const double a=std::abs(R[i]); s += R[i]*R[i]; inf=std::max(inf,a); ++count;
            }
            const double rms = count>0 ? std::sqrt(s/static_cast<double>(count)) : 0.0;
            std::cout << "      " << d.name << "." << f.name << "_rms = " << rms << ", " << d.name << "." << f.name << "_inf = " << inf << "\n";
        }
    }
}
