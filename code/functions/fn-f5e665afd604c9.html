){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in global assembly.");
    // Fallback slow path.  Solver code should call the overload
    // that receives precomputed dof_maps as well as idx_maps.
    const ProblemElementDofMap dof_maps=precompute_global_element_dofs(problem);
    assemble_global_system(kernel,problem,params,current_dofs,past_dofs,dt,dt_previous,step,time,R,K,dof_maps,idx_maps,history,history_diagnostics);
}
