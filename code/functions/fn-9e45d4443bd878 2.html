LocalFieldView make_local_field_view(
    const FieldMesh& mesh,
    int elem_id,
    std::initializer_list<const char*> field_order
) {
    LocalFieldView view;
    view.dim = mesh.dim;
    for (const char* name : field_order) {
        append_local_field_from_mixed(view, mesh, elem_id, std::string(name));
    }
    return view;
}
