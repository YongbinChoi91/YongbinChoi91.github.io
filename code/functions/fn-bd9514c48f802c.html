void reset_hdf5_table(const ProblemParameters& params,const std::string& dataset_path,const std::vector<std::string>& columns){
    if(!params.solver.hdf5_output) return;
#ifdef HIPECS_USE_HDF5
    hid_t file=h5_open_file_rw(params);
    h5_create_resizable_2d(file,dataset_path,static_cast<hsize_t>(columns.size()));
    hid_t dset=H5Dopen2(file,dataset_path.c_str(),H5P_DEFAULT);
    h5_write_string_attr(dset,"columns",join_strings(columns,","));
    H5Dclose(dset); H5Fclose(file);
#else
    (void)dataset_path; (void)columns;
#endif
}
