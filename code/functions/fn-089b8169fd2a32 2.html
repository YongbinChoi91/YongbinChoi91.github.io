std::vector<std::vector<int>> reference_edges(ElementType type){
    if(type==ElementType::Q4) return {{0,1},{1,2},{2,3},{3,0}};
    if(type==ElementType::Q9) return {{0,4,1},{1,5,2},{2,6,3},{3,7,0}};
    if(type==ElementType::T3) return {{0,1},{1,2},{2,0}};
    if(type==ElementType::T6) return {{0,3,1},{1,4,2},{2,5,0}};
    throw std::runtime_error("Reference edges currently support 2D T3/T6/Q4/Q9 elements.");
}
