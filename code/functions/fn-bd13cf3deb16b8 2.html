const LinearElementCache& get_cached_linear_element(
    const FieldMesh& mesh,
    int elem_id,
    int q_order,
    const std::string& field_name
) {
    auto& cache = linear_element_cache_map();
    const CacheKey key{&mesh, elem_id, q_order, field_name};
    auto it = cache.find(key);
    if (it == cache.end()) {
        it = cache.emplace(key, build_linear_element_cache(mesh, elem_id, q_order, field_name)).first;
    }
    return it->second;
}
