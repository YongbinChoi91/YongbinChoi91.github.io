void assemble_global_system(
    ConstitutiveElementKernel kernel,
    const Problem& problem,
    const ProblemParameters& params,
    const Vector& current_dofs,
    const PastDofs& past_dofs,
    double dt,
    double dt_previous,
    int step,
    double time,
    Vector& R,
    CSRMatrix& K,
    const ProblemElementDofMap& dof_maps,
    const ProblemElementTangentIndexMap& idx_maps,
    MaterialHistoryStore* history,
    HistoryLocalDiagnostics* history_diagnostics
){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in global assembly.");
    const int ndofs=problem.total_dofs;
    validate_global_dofs_size(current_dofs,past_dofs,ndofs,"multi-domain assembly");
    if(dof_maps.size()!=idx_maps.size()) throw std::runtime_error("Multi-domain assembly dof/tangent domain-map size mismatch.");
 
    AssemblyMpiInfo mpi = get_assembly_mpi_info();
    if(params.solver.multiscale_rve_assembly_context) {
        mpi.active = false;
        mpi.rank = 0;
        mpi.size = 1;
    }
    R.assign(ndofs,0.0);
    zero_csr_values(K);
 
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain=problem.domains[d];
        if(!is_supported_continuum_domain(domain.type)) throw std::runtime_error("Unsupported domain type in global assembly.");
        const FieldMesh& mesh=domain.mesh;
        if(static_cast<int>(dof_maps[d].size())!=static_cast<int>(idx_maps[d].size())) throw std::runtime_error("Multi-domain assembly dof/tangent element-map size mismatch.");
 
        const int nelem = static_cast<int>(dof_maps[d].size());
        std::vector<ElementSystem> local_systems(nelem);
        std::vector<HistoryLocalDiagnostics> local_history_diagnostics(history_diagnostics ? nelem : 0);
        std::vector<std::string> element_errors(nelem);
 
#ifdef HIPECS_MULTISCALE
        const bool multiscale_batch_handled =
            !mpi.active &&
            try_compute_multiscale_batch_domain_element_systems(
                mesh, domain.type, d, params,
                current_dofs, past_dofs,
                dt, dt_previous, step,
                dof_maps[d], local_systems, element_errors);
        if(multiscale_batch_handled){
            for(int e=0;e<nelem;++e){
                if(!element_errors[e].empty()){
                    std::ostringstream os;
                    os << "Element assembly failed in domain " << d << ", element " << e << ": " << element_errors[e];
                    throw std::runtime_error(os.str());
                }
            }
            for(int e=0;e<nelem;++e){
                assemble_element_system_csr(local_systems[e],dof_maps[d][e],idx_maps[d][e],R,K);
            }
            const int offset = problem.shared_dofs ? 0 : domain.dof_offset;
            assemble_neumann_conditions(mesh, params, time, offset, R);
            continue;
        }
#endif
 
        // Expensive constitutive work is embarrassingly parallel over elements.
        // Do NOT scatter to the global R/K in this loop: neighboring elements share
        // global DOFs and CSR entries, so R[I] += ... and K.values[p] += ... would race.
        // For multiscale runs the solver can choose whether this element loop is
        // parallelized at the outer macro level, inside each RVE solve, or both.
        auto assemble_local_element = [&](int e){
            if(!assembly_rank_owns_element(e, mpi)) return;
            try{
                const auto& global_dofs=dof_maps[d][e];
                Vector xe;
                PastDofs local_past_dofs;
                gather_local_vector_into(current_dofs,global_dofs,xe);
                gather_local_past_dofs_into(past_dofs,global_dofs,local_past_dofs);
 
                ElementState st;
                st.past_dofs = std::move(local_past_dofs);
                st.dt=dt;
                st.dt_previous=dt_previous;
                st.step=step;
                st.domain_name=domain.name;
                st.domain_type=domain.type;
                st.domain_id=d;
                if(history && d < static_cast<int>(history->domains.size()) && e < static_cast<int>(history->domains[d].elements.size())){
                    st.material_history=&history->domains[d].elements[e];
                }
                st.history_diagnostics = history_diagnostics ? &local_history_diagnostics[e] : nullptr;
 
                local_systems[e]=kernel(mesh,e,xe,params,st);
            } catch(const std::exception& ex){
                element_errors[e]=ex.what();
            } catch(...){
                element_errors[e]="Unknown exception during parallel element assembly.";
            }
        };
 
        const bool parallel_element_loop = use_parallel_element_loop(params.solver);
        const int requested_threads = element_loop_thread_count(params.solver);
#ifdef USE_OPENMP
        if(parallel_element_loop && requested_threads > 0){
            #pragma omp parallel for schedule(dynamic) num_threads(requested_threads)
            for(int e=0;e<nelem;++e) assemble_local_element(e);
        } else if(parallel_element_loop){
            #pragma omp parallel for schedule(dynamic)
            for(int e=0;e<nelem;++e) assemble_local_element(e);
        } else
#endif
        {
            for(int e=0;e<nelem;++e) assemble_local_element(e);
        }
 
        int local_error = 0;
        std::string first_local_error;
        for(int e=0;e<nelem;++e){
            if(assembly_rank_owns_element(e, mpi) && !element_errors[e].empty()){
                local_error = 1;
                if(first_local_error.empty()){
                    std::ostringstream os;
                    os << "Element assembly failed in domain " << d << ", element " << e << ": " << element_errors[e];
                    first_local_error = os.str();
                }
            }
        }
        if(any_rank_reported_assembly_error(local_error, mpi)){
            throw std::runtime_error(
                first_local_error.empty()
                    ? std::string("Element assembly failed on another MPI rank.")
                    : first_local_error
            );
        }
 
        // Serial reduction/scatter: safe and deterministic.  This keeps the CSR
        // addition path unchanged while moving the expensive RVE constitutive work
        // to the OpenMP region above.
        for(int e=0;e<nelem;++e){
            if(!assembly_rank_owns_element(e, mpi)) continue;
            if(history_diagnostics) merge_history_diagnostics(*history_diagnostics, local_history_diagnostics[e]);
            assemble_element_system_csr(local_systems[e],dof_maps[d][e],idx_maps[d][e],R,K);
        }
 
        const int offset = problem.shared_dofs ? 0 : domain.dof_offset;
        if(!mpi.active || mpi.rank == 0) assemble_neumann_conditions(mesh, params, time, offset, R);
    }
    reduce_assembled_system_if_parallel(R, K, mpi);
}
