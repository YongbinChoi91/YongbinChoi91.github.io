static bool field_contains_all_nodes_on_element(
    const FieldInfo& field,
    int elem_id,
    const BoundaryEntity& ent
){
    if(elem_id<0 || elem_id>=static_cast<int>(field.elements.size())) return false;
    const auto& conn = field.elements[elem_id];
    for(int node: ent.node_ids){
        if(std::find(conn.begin(),conn.end(),node)==conn.end()) return false;
    }
    return true;
}
