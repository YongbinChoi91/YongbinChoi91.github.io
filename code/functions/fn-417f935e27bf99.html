static BCSelector parse_bc_selector_token(const std::string& token){
    if(starts_with_case_insensitive(token,"POINT[")){
        const std::vector<double> v=parse_selector_numbers(bracket_body_or_throw(token,"POINT"),"POINT");
        if(v.size()!=2 && v.size()!=3){
            throw std::runtime_error("Bad POINT selector. Use POINT[x,y] or POINT[x,y,z].");
        }
        BCSelector s;
        s.kind=BCSelectorKind::Point;
        s.group=token;
        s.p0={{v[0],v[1],v.size()==3?v[2]:0.0}};
        return s;
    }
    if(starts_with_case_insensitive(token,"LINE[")){
        const std::vector<double> v=parse_selector_numbers(bracket_body_or_throw(token,"LINE"),"LINE");
        if(v.size()!=4 && v.size()!=6){
            throw std::runtime_error("Bad LINE selector. Use LINE[(x0,y0),(x1,y1)] or LINE[(x0,y0,z0),(x1,y1,z1)].");
        }
        BCSelector s;
        s.kind=BCSelectorKind::Line;
        s.group=token;
        if(v.size()==4){
            s.p0={{v[0],v[1],0.0}};
            s.p1={{v[2],v[3],0.0}};
        } else {
            s.p0={{v[0],v[1],v[2]}};
            s.p1={{v[3],v[4],v[5]}};
        }
        return s;
    }
    return make_boundary_group_selector(token);
}
