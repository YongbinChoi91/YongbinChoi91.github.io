static std::vector<GaussPoint> tetra_duffy_rule(int order){
    if(order<=1) return {{0.25,0.25,0.25,1.0/6.0}};
    std::vector<double> r,wr;
    gauss_legendre_1d_interval(order,0.0,1.0,r,wr);
    std::vector<GaussPoint> gps;
    gps.reserve(order*order*order);
    for(int i=0;i<order;++i){
        for(int j=0;j<order;++j){
            for(int k=0;k<order;++k){
                const double xi   = r[i];
                const double eta  = (1.0-r[i])*r[j];
                const double zeta = (1.0-r[i])*(1.0-r[j])*r[k];
                const double jac  = (1.0-r[i])*(1.0-r[i])*(1.0-r[j]);
                const double wt   = wr[i]*wr[j]*wr[k]*jac;
                gps.push_back({xi,eta,zeta,wt});
            }
        }
    }
    return gps;
}
