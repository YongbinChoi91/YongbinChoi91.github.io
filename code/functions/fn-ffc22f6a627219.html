static ResolvedBoundaryEdge resolve_boundary_edge(
    const FieldMesh& mesh,
    const BoundaryEntity& ent,
    const std::array<double,3>* boundary_group_center=nullptr
){
    // Python-compatible logic:
    //   1. use the surface-node ids from the boundary row,
    //   2. scan the global TOTAL_ELEMENTS table when available, otherwise the
    //      mesh velocity connectivity table,
    //   3. recover the row whose connectivity contains the surface nodes,
    //   4. classify the local edge from THAT recovered row.
    // We deliberately DO NOT trust ent.element_id here.  In shared multi-domain meshes it
    // can be a boundary-row id; the real owner is obtained from surface-node
    // containment, just like Python's Drag_Lift_connectivity_mask workflow.
    const auto& candidates = force_owner_table(mesh);
    const bool using_total = (&candidates == &mesh.total_velocity_elements);
 
    auto push_best_edge_by_local_overlap = [&](int e,
                                               const std::vector<int>& local_nodes,
                                               bool by_coords,
                                               std::vector<ResolvedBoundaryEdge>& matches)->void{
        if(e<0 || e>=static_cast<int>(candidates.size())) return;
        const auto& conn=candidates[e];
        if(conn.empty()) return;
        const ElementType etype = element_type_from_connectivity(mesh,conn);
        const auto edges = reference_boundary_entities(etype,mesh.dim);
 
        // First try the strict Python-style classification: all boundary local
        // nodes are exactly on one reference edge/face.
        for(int edge=0; edge<static_cast<int>(edges.size()); ++edge){
            if(contains_all_nodes(edges[edge],local_nodes)){
                ResolvedBoundaryEdge r;
                r.elem_id=e;
                r.edge=edge;
                r.by_coordinate_fallback=by_coords;
                r.use_total_elements=using_total;
                matches.push_back(r);
                return;
            }
        }
 
        // Robust fallback for mixed exporters: sometimes the boundary row has
        // corner nodes plus a high-order midside node whose local ordering is not
        // exactly the current C++ reference order.  The recovered ELEMENT ROW is
        // still correct; choose the reference edge with the largest overlap.
        int best_edge=-1;
        int best_count=-1;
        for(int edge=0; edge<static_cast<int>(edges.size()); ++edge){
            int count=0;
            for(int ln:local_nodes){
                if(std::find(edges[edge].begin(),edges[edge].end(),ln)!=edges[edge].end()) ++count;
            }
            if(count>best_count){ best_count=count; best_edge=edge; }
        }
        const int required = std::min<int>(2, static_cast<int>(local_nodes.size()));
        if(best_edge>=0 && best_count>=required){
            ResolvedBoundaryEdge r;
            r.elem_id=e;
            r.edge=best_edge;
            r.by_coordinate_fallback=by_coords;
            r.use_total_elements=using_total;
            matches.push_back(r);
        }
    };
 
    auto collect_by_ids=[&](int e,std::vector<ResolvedBoundaryEdge>& matches)->void{
        if(e<0 || e>=static_cast<int>(candidates.size())) return;
        const auto& conn=candidates[e];
        if(conn.empty()) return;
        std::vector<int> local_nodes;
        local_nodes.reserve(ent.node_ids.size());
        for(int gnode:ent.node_ids){
            auto it=std::find(conn.begin(),conn.end(),gnode);
            if(it==conn.end()) return;
            local_nodes.push_back(static_cast<int>(it-conn.begin()));
        }
        push_best_edge_by_local_overlap(e,local_nodes,false,matches);
    };
 
    std::vector<ResolvedBoundaryEdge> matches;
    const std::vector<int> id_candidates = candidate_owner_elements_by_node_ids(mesh,ent);
    for(int e:id_candidates) collect_by_ids(e,matches);
    if(!matches.empty()) return choose_boundary_owner_candidate(mesh,matches,boundary_group_center);
 
    // Coordinate fallback: still scan the domain element table only, then use
    // the recovered domain row.  This covers exports where surface-node ids and
    // element-node ids differ but coordinates coincide.
    double h2=0.0;
    for(int n:ent.node_ids){
        if(n>=0 && n<static_cast<int>(mesh.velocity_nodes.size())){
            const auto& X=mesh.velocity_nodes[n].x;
            h2=std::max(h2,X[0]*X[0]+X[1]*X[1]+X[2]*X[2]);
        }
    }
    const double tol2=std::max(1.0e-24,1.0e-20*std::max(1.0,h2));
 
    auto collect_by_coords=[&](int e,std::vector<ResolvedBoundaryEdge>& matches)->void{
        if(e<0 || e>=static_cast<int>(candidates.size())) return;
        const auto& conn=candidates[e];
        if(conn.empty()) return;
        const ElementType etype = element_type_from_connectivity(mesh,conn);
        const auto edges=reference_boundary_entities(etype,mesh.dim);
        for(int edge=0; edge<static_cast<int>(edges.size()); ++edge){
            if(edge_coordinates_match_boundary(mesh,conn,edges[edge],ent,tol2)){
                ResolvedBoundaryEdge r;
                r.elem_id=e;
                r.edge=edge;
                r.by_coordinate_fallback=true;
                r.use_total_elements=using_total;
                matches.push_back(r);
                return;
            }
        }
    };
 
    for(int e=0;e<static_cast<int>(candidates.size());++e) collect_by_coords(e,matches);
    if(!matches.empty()) return choose_boundary_owner_candidate(mesh,matches,boundary_group_center);
 
    // Not every named boundary row is necessarily in contact with the fluid in
    // a shared multi-domain mesh.  For example, CIRCLE can include solid-only/internal
    // segments.  Those must be excluded from hydrodynamic drag/lift, so an
    // unresolved owner is represented by elem_id=-1 and skipped by the caller.
    return ResolvedBoundaryEdge{};
}
