Vector solve_reduced_matfree_system(
    ConstitutiveElementKernel kernel,
    const Problem& problem,
    const ProblemParameters& params,
    const Vector& R,
    const BoundaryData& bc,
    ReducedSystemWorkspace& reduced,
    LinearSolverSettings settings,
    const FieldMesh* schur_mesh,
    const Problem* schur_problem,
    double dt_for_schur,
    double dt_previous,
    int step,
    double time,
    const Vector& current_dofs,
    const PastDofs& past_dofs,
    const ProblemElementDofMap& dof_maps,
    const MaterialHistoryStore& material_history
){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in matrix-free reduced solve.");
    const int ndofs=problem.total_dofs;
    if(static_cast<int>(R.size())!=ndofs) throw std::runtime_error("Matrix-free reduced solve residual dimension mismatch.");
    if(static_cast<int>(current_dofs.size())!=ndofs) throw std::runtime_error("Matrix-free reduced solve current DOF dimension mismatch.");
    for(int p=0;p<static_cast<int>(past_dofs.size());++p){
        if(static_cast<int>(past_dofs[p].size())!=ndofs) throw std::runtime_error("Matrix-free reduced solve past DOF dimension mismatch.");
    }
    const std::vector<int>& free_dofs=reduced.free_dofs;
    settings.reduced_to_full_dofs=free_dofs;
 
    Vector rhs_f(free_dofs.size(),0.0);
    for(int i=0;i<static_cast<int>(free_dofs.size());++i) rhs_f[i] = -R[free_dofs[i]];
 
    const std::string mfpc=lower_copy(settings.petsc_matfree_pc.empty() ? settings.preconditioner : settings.petsc_matfree_pc);
    if(mfpc!="none" && mfpc!="jacobi" && mfpc!="element_block_jacobi" && mfpc!="element_jacobi"){
        throw std::runtime_error("Matrix-free reduced solve supports petsc_matfree_pc none, jacobi, or element_block_jacobi.");
    }
 
    if(lower_copy(settings.petsc_matfree_jvp_mode)=="exact_ad") {
        if(mfpc!="none" && mfpc!="jacobi")
            throw std::runtime_error("exact_ad matrix-free uses pc none or jacobi; element blocks require stored tangents.");
        PetscMatFreeOperator op;
        op.n=static_cast<int>(free_dofs.size());
        // KSPSolve is synchronous: references remain valid throughout the solve.
        // Capture no element matrices and no copies of the model/DOF maps.
        op.apply=[&](const Vector& x, Vector& y) {
            if(x.size()!=free_dofs.size()) throw std::runtime_error("Matrix-free direction size mismatch.");
            Vector full_direction(ndofs,0.0), full_action;
            for(std::size_t i=0; i<free_dofs.size(); ++i) full_direction[free_dofs[i]]=x[i];
            apply_global_jacobian_vector_product(kernel,problem,params,current_dofs,past_dofs,
                full_direction,dt_for_schur,dt_previous,step,time,full_action,dof_maps,&material_history);
            y.resize(free_dofs.size());
            for(std::size_t i=0; i<free_dofs.size(); ++i) y[i]=full_action[free_dofs[i]];
        };
        Vector inverse_diagonal;
        if(mfpc=="jacobi") {
            Vector diagonal;
            assemble_global_jacobian_diagonal(kernel,problem,params,current_dofs,past_dofs,
                dt_for_schur,dt_previous,step,time,diagonal,dof_maps,&material_history);
            inverse_diagonal.resize(free_dofs.size());
            for(std::size_t i=0; i<free_dofs.size(); ++i) {
                const double value=diagonal[free_dofs[i]];
                if(!std::isfinite(value) || value<=0.0)
                    throw std::runtime_error("Matrix-free displacement Jacobi requires a positive finite free-DOF diagonal.");
                inverse_diagonal[i]=1.0/value;
            }
            op.apply_pc=[&](const Vector& x, Vector& y) {
                if(x.size()!=inverse_diagonal.size()) throw std::runtime_error("Matrix-free Jacobi input size mismatch.");
                y.resize(x.size());
                for(std::size_t i=0; i<x.size(); ++i) y[i]=inverse_diagonal[i]*x[i];
            };
        }
        if(settings.verbose) std::cout << "    MatFree storage: no element tangents; PC entries=" << inverse_diagonal.size() << "\n";
        const Vector free_increment=solve_linear_system_backend_petsc_matfree(rhs_f,settings,op,nullptr,nullptr);
        if(free_increment.size()!=free_dofs.size()) throw std::runtime_error("Bad matrix-free increment size.");
        Vector increment(ndofs,0.0);
        for(std::size_t i=0; i<free_dofs.size(); ++i) {
            if(!std::isfinite(free_increment[i])) throw std::runtime_error("Non-finite matrix-free increment.");
            increment[free_dofs[i]]=free_increment[i];
        }
        return increment;
    }
    if(mfpc=="jacobi") throw std::runtime_error("Matrix-free jacobi requires exact_ad mode.");
 
    const auto precond_setup_start=std::chrono::steady_clock::now();
    std::vector<int> full_to_free(ndofs,-1);
    for(int i=0;i<static_cast<int>(free_dofs.size());++i) full_to_free[free_dofs[i]]=i;
    MaterialHistoryStore history_copy=material_history;
    MaterialHistoryStore* mutable_history=&history_copy;
 
    std::vector<std::vector<int>> block_dofs;
    std::vector<Vector> block_mats;
    block_dofs.reserve(dof_maps.size());
    block_mats.reserve(dof_maps.size());
    for(int d=0; d<static_cast<int>(problem.domains.size()); ++d){
        const Domain& domain=problem.domains[d];
        const FieldMesh& mesh=domain.mesh;
        for(int e=0; e<static_cast<int>(dof_maps[d].size()); ++e){
            const auto& gdofs=dof_maps[d][e];
            Vector xe(gdofs.size());
            PastDofs local_past_dofs(past_dofs.size(), Vector(gdofs.size()));
            for(int a=0;a<static_cast<int>(gdofs.size());++a){
                const int I=gdofs[a];
                xe[a]=current_dofs[I];
                for(int p=0;p<static_cast<int>(past_dofs.size());++p) local_past_dofs[p][a]=past_dofs[p][I];
            }
            ElementState st;
            st.past_dofs=std::move(local_past_dofs);
            st.dt=dt_for_schur;
            st.dt_previous=dt_previous;
            st.step=step;
            st.domain_name=domain.name;
            st.domain_type=domain.type;
            st.domain_id=d;
            if(d<static_cast<int>(mutable_history->domains.size()) && e<static_cast<int>(mutable_history->domains[d].elements.size())){
                st.material_history=&mutable_history->domains[d].elements[e];
            }
            ElementSystem es=kernel(mesh,e,xe,params,st);
            const int nloc=static_cast<int>(gdofs.size());
            if(static_cast<int>(es.tangent.size())!=nloc*nloc) throw std::runtime_error("Bad element tangent size in matrix-free tangent cache.");
            block_dofs.push_back(gdofs);
            block_mats.push_back(std::move(es.tangent));
        }
    }
    const auto precond_setup_end=std::chrono::steady_clock::now();
    if(settings.verbose){
        const double setup_sec=std::chrono::duration<double>(precond_setup_end-precond_setup_start).count();
        std::cout << "    PETSc MatFree tangent cache setup time: " << setup_sec << " s\n";
    }
 
    PetscMatFreeOperator op;
    op.n=static_cast<int>(free_dofs.size());
    op.apply=[block_dofs,
              block_mats,
              full_to_free,
              nfree=static_cast<int>(free_dofs.size())](const Vector& x_free, Vector& y_free){
        if(static_cast<int>(x_free.size())!=nfree) throw std::runtime_error("Matrix-free callback input size mismatch.");
        y_free.assign(nfree,0.0);
        for(std::size_t b=0;b<block_dofs.size();++b){
            const auto& gdofs=block_dofs[b];
            const int nloc=static_cast<int>(gdofs.size());
            Vector vloc(nloc,0.0), yloc(nloc,0.0);
            for(int a=0;a<nloc;++a){
                const int fi=(gdofs[a]>=0 && gdofs[a]<static_cast<int>(full_to_free.size())) ? full_to_free[gdofs[a]] : -1;
                if(fi>=0) vloc[a]=x_free[fi];
            }
            const Vector& K=block_mats[b];
            for(int a=0;a<nloc;++a){
                double sum=0.0;
                const int row=a*nloc;
                for(int c=0;c<nloc;++c) sum += K[row+c]*vloc[c];
                yloc[a]=sum;
            }
            for(int a=0;a<nloc;++a){
                const int fi=(gdofs[a]>=0 && gdofs[a]<static_cast<int>(full_to_free.size())) ? full_to_free[gdofs[a]] : -1;
                if(fi>=0) y_free[fi] += yloc[a];
            }
        }
    };
    if(mfpc=="element_block_jacobi" || mfpc=="element_jacobi"){
        op.apply_pc=[block_dofs,
                     block_mats,
                     full_to_free,
                     nfree=static_cast<int>(free_dofs.size())](const Vector& x_free, Vector& y_free){
            if(static_cast<int>(x_free.size())!=nfree) throw std::runtime_error("Element block Jacobi PC input size mismatch.");
            y_free.assign(nfree,0.0);
            for(std::size_t b=0;b<block_dofs.size();++b){
                const auto& gdofs=block_dofs[b];
                const int nloc=static_cast<int>(gdofs.size());
                Vector rhs(nloc,0.0);
                for(int a=0;a<nloc;++a){
                    const int fi=(gdofs[a]>=0 && gdofs[a]<static_cast<int>(full_to_free.size())) ? full_to_free[gdofs[a]] : -1;
                    if(fi>=0) rhs[a]=x_free[fi];
                }
                Vector z=solve_dense_regularized(block_mats[b],rhs,nloc);
                for(int a=0;a<nloc;++a){
                    const int fi=(gdofs[a]>=0 && gdofs[a]<static_cast<int>(full_to_free.size())) ? full_to_free[gdofs[a]] : -1;
                    if(fi>=0) y_free[fi] += z[a];
                }
            }
        };
    }
 
    (void)schur_mesh;
    (void)schur_problem;
    Vector dx_f=solve_linear_system_backend_petsc_matfree(rhs_f,settings,op,nullptr,nullptr);
    if(dx_f.size()!=free_dofs.size()) throw std::runtime_error("Matrix-free reduced solve returned wrong vector size.");
    Vector dx(ndofs,0.0);
    for(int i=0;i<static_cast<int>(free_dofs.size());++i){
        if(!std::isfinite(dx_f[i])) throw std::runtime_error("Matrix-free reduced solve produced non-finite increment.");
        dx[free_dofs[i]]=dx_f[i];
    }
    for(int d:bc.dofs) if(d>=0 && d<ndofs) dx[d]=0.0;
    return dx;
}
