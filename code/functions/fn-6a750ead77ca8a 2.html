inline Tensor4<Scalar, OR, OC, R, C> unflatten_tensor2_output_tensor2_input_gradient(
    const std::array<std::array<std::array<Scalar, R * C>, OC>, OR>& flat
) {
    Tensor4<Scalar, OR, OC, R, C> A{};
    for (std::size_t i = 0; i < OR; ++i) {
        for (std::size_t j = 0; j < OC; ++j) {
            for (std::size_t k = 0; k < R; ++k) {
                for (std::size_t l = 0; l < C; ++l) A[i][j][k][l] = flat[i][j][k * C + l];
            }
        }
    }
    return A;
}
