static bool petsc_outer_cache_is_compatible(
    const PetscOuterSolveCache& cache,
    const CSRMatrix& A,
    const CSRMatrix* P_user,
    int p_kind,
    const std::vector<char>& pressure_mask
){
    if(!cache.valid || cache.n!=A.n_rows || cache.p_kind!=p_kind) return false;
    if(cache.row_ptr!=A.row_ptr || cache.col_idx!=A.col_idx) return false;
    if(cache.pressure_mask!=pressure_mask) return false;
    if(p_kind==1){
        if(!P_user) return false;
        return cache.p_row_ptr==P_user->row_ptr && cache.p_col_idx==P_user->col_idx;
    }
    return true;
}
