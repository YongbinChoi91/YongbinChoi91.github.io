inline Mat3<Scalar> scale3(
    const Mat3<Scalar>& A,
    const Scalar& s
) {
    auto C = zero3<Scalar>();
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) C[i][j] = s * A[i][j];
    }
    return C;
}
