static int integrate_boundary_faces_3d(
    const FieldMesh& mesh,
    const std::string& boundary_name,
    Callback&& callback
){
    if(mesh.dim!=3) throw std::runtime_error("Boundary face integration requires a 3D mesh.");
    auto it=mesh.velocity_boundaries.find(boundary_name);
    if(it==mesh.velocity_boundaries.end()){
        throw std::runtime_error("Unknown velocity boundary group for boundary integration: "+boundary_name);
    }
 
    const auto boundary_center=boundary_group_center_from_velocity_nodes(mesh,it->second);
    int integrated_entities=0;
 
    for(const auto& ent:it->second){
        const ResolvedBoundaryEdge r=resolve_boundary_edge(mesh,ent,&boundary_center);
        if(r.elem_id < 0) continue;
        const ElementGeometry geom=make_owner_geometry_element(mesh,r);
        const ElementType vtype=geom.type;
        const bool hex=is_hex_element_type_post(vtype);
        const bool tet=is_tet_element_type_post(vtype);
        if(!hex && !tet) throw std::runtime_error("3D boundary force supports only Tet/Hex owner elements.");
        const auto qface = hex ? gauss_rule_quad_face_post(3) : gauss_rule(ElementType::T3,3);
        ++integrated_entities;
 
        for(const auto& q:qface){
            BoundaryQuadraturePoint3D qp;
            qp.entity=&ent;
            qp.owner=r;
            qp.geometry=geom;
            qp.element_type=vtype;
            qp.reference_point = hex
                ? hex_face_reference_point_post(r.edge,q.xi,q.eta)
                : tet_face_reference_point_post(r.edge,q.xi,q.eta);
            const ShapeData sv=compute_shape(vtype,qp.reference_point[0],qp.reference_point[1],qp.reference_point[2]);
            for(std::size_t a=0;a<sv.N.size() && a<geom.coords.size();++a){
                for(int d=0;d<3;++d) qp.physical_point[d] += sv.N[a]*geom.coords[a][d];
            }
            qp.jacobian=compute_jacobian_data(geom,sv);
            if(qp.jacobian.detJ<=0.0) throw std::runtime_error("Negative detJ during 3D boundary face integration.");
            qp.normal_measure=signed_surface_vec_area_from_J_3d(vtype,r.edge,qp.jacobian);
            qp.measure=norm3_post(qp.normal_measure);
            qp.weight=q.weight;
            callback(qp);
        }
    }
 
    return integrated_entities;
}
