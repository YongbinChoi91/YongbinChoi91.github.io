inline Scalar linear_projected_scalar_value(
    const LinearElementCache& cache,
    const std::vector<Scalar>& elem,
    const std::array<double,2>& x
) {
    const auto& U = cache.layout.field(cache.field_name);
    const auto grad = linear_projected_scalar_gradient(cache, elem);
    Scalar mean_value(0.0);
    std::array<double,2> mean_x{{0.0,0.0}};
    for (int ar = 0; ar < cache.n_real; ++ar) {
        const int slot = cache.real_slots[ar];
        mean_value += elem[hpx::local_dof(U, slot, 0)];
        mean_x[0] += cache.X[ar][0];
        mean_x[1] += cache.X[ar][1];
    }
    const double inv_n = 1.0 / static_cast<double>(cache.n_real);
    mean_value *= Scalar(inv_n);
    mean_x[0] *= inv_n;
    mean_x[1] *= inv_n;
    return mean_value
         + grad[0] * Scalar(x[0] - mean_x[0])
         + grad[1] * Scalar(x[1] - mean_x[1]);
}
