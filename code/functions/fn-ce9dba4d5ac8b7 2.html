void launch_step_plot(const ProblemParameters& params,const std::string& vtk_file){
    if(!params.solver.plot_each_step || params.solver.plot_script.empty() || vtk_file.empty()) return;
    const char* python_env=std::getenv("HIPECS_PYTHON");
    const std::string python=(python_env && python_env[0]!='\0') ? python_env : "python3";
    const std::string command =
        shell_quote_postprocess(python) + " " + shell_quote_postprocess(params.solver.plot_script) +
        " " + shell_quote_postprocess(vtk_file) +
        (params.solver.plot_config_file.empty()
             ? std::string()
             : " --post " + shell_quote_postprocess(params.solver.plot_config_file));
    const int rc=std::system(command.c_str());
    if(rc!=0){
        std::cerr << "Step plot command failed with status " << rc
                  << ". Try running: " << command << "\n";
    }
}
