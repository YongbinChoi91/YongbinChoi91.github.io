static int pressure_cache_csr_index(const PressurePcdCache& cache,int row,int col){
    if(row<0 || row>=static_cast<int>(cache.csr_index.size())) return -1;
    const auto it=cache.csr_index[row].find(col);
    return it==cache.csr_index[row].end() ? -1 : it->second;
}
