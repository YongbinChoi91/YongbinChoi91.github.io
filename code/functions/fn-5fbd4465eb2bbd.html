void write_xy_svg_plot(
    const std::string& svg_filename,
    const std::string& title,
    const std::string& x_label,
    const std::string& y_label,
    const std::vector<double>& x,
    const std::vector<PostprocessSvgSeries>& series
){
    if(x.empty() || series.empty()) return;
 
    double xmin=*std::min_element(x.begin(),x.end());
    double xmax=*std::max_element(x.begin(),x.end());
    double ymin=std::numeric_limits<double>::infinity();
    double ymax=-std::numeric_limits<double>::infinity();
    for(const auto& s:series){
        for(double v:s.values){
            ymin=std::min(ymin,v);
            ymax=std::max(ymax,v);
        }
    }
    if(!std::isfinite(ymin) || !std::isfinite(ymax)) return;
    pad_svg_range(xmin,xmax);
    pad_svg_range(ymin,ymax);
 
    const int W=900,H=520,L=80,R=30,T=40,B=70;
    auto X=[&](double v){ return L + (v-xmin)/(xmax-xmin)*(W-L-R); };
    auto Y=[&](double v){ return H-B - (v-ymin)/(ymax-ymin)*(H-T-B); };
    auto polyline=[&](const std::vector<double>& y){
        std::ostringstream os;
        const std::size_t n=std::min(x.size(),y.size());
        for(std::size_t i=0;i<n;++i){
            if(i) os << ' ';
            os << X(x[i]) << ',' << Y(y[i]);
        }
        return os.str();
    };
 
    ensure_output_parent_directory(svg_filename);
    std::ofstream out(svg_filename);
    if(!out) throw std::runtime_error("Could not open SVG file for writing: "+svg_filename);
    out << "<svg xmlns=\"http://www.w3.org/2000/svg\" width=\"" << W << "\" height=\"" << H << "\" viewBox=\"0 0 " << W << ' ' << H << "\">\n";
    out << "<rect width=\"100%\" height=\"100%\" fill=\"white\"/>\n";
    out << "<line x1=\"" << L << "\" y1=\"" << H-B << "\" x2=\"" << W-R << "\" y2=\"" << H-B << "\" stroke=\"black\"/>\n";
    out << "<line x1=\"" << L << "\" y1=\"" << T << "\" x2=\"" << L << "\" y2=\"" << H-B << "\" stroke=\"black\"/>\n";
    out << "<text x=\"" << W/2 << "\" y=\"25\" text-anchor=\"middle\" font-family=\"sans-serif\" font-size=\"18\">" << xml_escape(title) << "</text>\n";
    out << "<text x=\"" << W/2 << "\" y=\"" << H-20 << "\" text-anchor=\"middle\" font-family=\"sans-serif\" font-size=\"14\">" << xml_escape(x_label) << "</text>\n";
    out << "<text x=\"20\" y=\"" << H/2 << "\" transform=\"rotate(-90 20 " << H/2 << ")\" text-anchor=\"middle\" font-family=\"sans-serif\" font-size=\"14\">" << xml_escape(y_label) << "</text>\n";
    write_svg_axes_grid(out,W,H,L,R,T,B,xmin,xmax,ymin,ymax);
 
    int legend_y=50;
    for(const auto& s:series){
        out << "<polyline fill=\"none\" stroke=\"" << s.color << "\" stroke-width=\"2\" points=\"" << xml_escape(polyline(s.values)) << "\"/>\n";
        const std::size_t n=std::min(x.size(),s.values.size());
        for(std::size_t i=0;i<n;++i){
            out << "<circle cx=\"" << X(x[i]) << "\" cy=\"" << Y(s.values[i])
                << "\" r=\"3\" fill=\"white\" stroke=\"" << s.color << "\" stroke-width=\"1.5\"/>\n";
        }
        out << "<line x1=\"" << W-155 << "\" y1=\"" << legend_y-4 << "\" x2=\"" << W-125 << "\" y2=\"" << legend_y-4
            << "\" stroke=\"" << s.color << "\" stroke-width=\"2\"/>\n";
        out << "<text x=\"" << W-118 << "\" y=\"" << legend_y
            << "\" font-family=\"sans-serif\" font-size=\"12\">" << xml_escape(s.label) << "</text>\n";
        legend_y += 18;
    }
    out << "</svg>\n";
}
