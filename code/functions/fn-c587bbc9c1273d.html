static std::vector<Node> compact_nodes_by_map(const std::vector<Node>& nodes,const std::vector<int>& map){
    int n_new=0;
    for(int v:map) if(v>=0) n_new=std::max(n_new,v+1);
    std::vector<Node> out(n_new);
    for(int old=0; old<static_cast<int>(map.size()); ++old){
        if(map[old]>=0) out[map[old]]=nodes[old];
    }
    return out;
}
