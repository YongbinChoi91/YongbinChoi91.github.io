inline std::vector<std::vector<Scalar>> cauchy_stress(
    const std::vector<std::vector<Scalar>>& F3,
    const std::vector<std::vector<Scalar>>& P
) {
    if (static_cast<int>(F3.size()) != 3 || static_cast<int>(P.size()) != 3) {
        throw std::runtime_error("Cauchy stress helper expects 3x3 F and P.");
    }
    const Scalar J = det(F3);
    std::vector<std::vector<Scalar>> sigma(3, std::vector<Scalar>(3, Scalar(0.0)));
    for (int i = 0; i < 3; ++i) {
        if (static_cast<int>(F3[i].size()) != 3 || static_cast<int>(P[i].size()) != 3) {
            throw std::runtime_error("Cauchy stress helper expects 3x3 F and P.");
        }
        for (int j = 0; j < 3; ++j) {
            for (int A = 0; A < 3; ++A) sigma[i][j] += P[i][A] * F3[j][A] / J;
        }
    }
    return sigma;
}
