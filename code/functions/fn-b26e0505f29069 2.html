inline std::vector<HistorySpec> material_history_specs_summary(const MaterialHistoryStore& history) {
    std::vector<HistorySpec> specs;
    for (const auto& domain : history.domains) {
        merge_history_specs_into(specs, domain.specs);
    }
    return specs;
}
