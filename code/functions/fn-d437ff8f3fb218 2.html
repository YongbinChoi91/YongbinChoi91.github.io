int find_field_index(const DofLayout& mesh,const std::string& field_name,bool required){
    for(int i=0;i<static_cast<int>(mesh.fields.size());++i){
        if(mesh.fields[i].name==field_name) return i;
    }
    if(required) throw std::runtime_error("Generic field not found: "+field_name);
    return -1;
}
