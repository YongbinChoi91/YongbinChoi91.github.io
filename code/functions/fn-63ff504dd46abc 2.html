BoundaryData build_shared_multidomain_boundary_conditions(
    const Problem& problem,
    const ProblemParameters& params,
    double time,
    const std::vector<std::string>& active_fields
){
    BoundaryData bc;
    if(problem.domains.empty()) return bc;
 
    const FieldMesh& rep=problem.domains.front().mesh;
 
    apply_generic_field_dirichlet_bcs_shared(problem,params,time,bc);
    add_inactive_shared_dof_constraints(bc,problem,params);
    add_inactive_shared_constitutive_field_dof_constraints(bc,problem,params,active_fields);
 
    if(has_active_pressure_field_from_names(rep,active_fields)){
        const int pin=(params.solver.pressure_pin_node>=0)?params.solver.pressure_pin_node:choose_pressure_pin_node(rep);
        add_dirichlet_bc(rep,params,bc,pressure_dof(rep,pin),params.solver.pressure_pin_value,"PRESSURE_PIN");
    }
    return bc;
}
