static std::vector<int> make_used_node_map(int n_nodes,const std::vector<std::vector<int>>& elements){
    std::vector<char> used(n_nodes,0);
    for(const auto& e:elements){
        for(int old:e){
            if(old<0 || old>=n_nodes) throw std::runtime_error("Element connectivity node id out of range while compacting mixed mesh.");
            used[old]=1;
        }
    }
    std::vector<int> map(n_nodes,-1);
    int next=0;
    for(int i=0;i<n_nodes;++i){
        if(used[i]) map[i]=next++;
    }
    return map;
}
