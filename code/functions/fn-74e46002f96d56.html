static AD2 operator*(const AD2& a, const AD2& b) {
    AD2 c;
    c.v = a.v * b.v;
    for(int i=0;i<3;++i){
        c.d[i] = a.d[i]*b.v + a.v*b.d[i];
        for(int j=0;j<3;++j){
            c.h[i][j] = a.h[i][j]*b.v + a.d[i]*b.d[j]
                      + a.d[j]*b.d[i] + a.v*b.h[i][j];
        }
    }
    return c;
}
