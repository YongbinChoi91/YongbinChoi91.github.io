int vtk_cell_type(ElementType type){
    switch(type){
        case ElementType::T3: return 5;    // VTK_TRIANGLE
        case ElementType::T6: return 22;   // VTK_QUADRATIC_TRIANGLE
        case ElementType::Q4: return 9;    // VTK_QUAD
        case ElementType::Q9: return 28;   // VTK_BIQUADRATIC_QUAD
        case ElementType::Tet4: return 10; // VTK_TETRA
        case ElementType::Tet10: return 24;// VTK_QUADRATIC_TETRA
        case ElementType::H8: return 12;   // VTK_HEXAHEDRON
        case ElementType::H20: return 25;  // VTK_QUADRATIC_HEXAHEDRON
        case ElementType::H27: return 29;  // VTK_TRIQUADRATIC_HEXAHEDRON
        case ElementType::Polygon: return 7; // VTK_POLYGON
        default: return 9;
    }
}
