static double eval_scalar_function_impl(const ScalarFunction& f,double x,double y,double z,double t,double parabolic_profile,double u_bar){
    if(f.is_constant) return f.constant;
    if(f.is_piecewise){
        double tt=t;
        if(f.is_cyclic){ tt=std::fmod(tt,f.period); if(tt<0) tt+=f.period; }
        if(tt<=f.points.front()[0]) return f.points.front()[1];
        if(tt>=f.points.back()[0]) return f.points.back()[1];
        for(int i=1;i<static_cast<int>(f.points.size());++i){
            const double t0=f.points[i-1][0], v0=f.points[i-1][1];
            const double t1=f.points[i][0], v1=f.points[i][1];
            if(tt<=t1){ const double a=(tt-t0)/(t1-t0); return (1-a)*v0+a*v1; }
        }
        return f.points.back()[1];
    }
    return ScalarExpressionParser(f.expression,x,y,z,t,parabolic_profile,u_bar).parse();
}
