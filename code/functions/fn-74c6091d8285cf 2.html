ElementType element_type_from_node_count(int n,int dim){
    if(dim==2){
        if(n==3) return ElementType::T3;
        if(n==6) return ElementType::T6;
        if(n==4) return ElementType::Q4;
        if(n==9) return ElementType::Q9;
    }
    if(dim==3){
        if(n==4) return ElementType::Tet4;
        if(n==10) return ElementType::Tet10;
        if(n==8) return ElementType::H8;
        if(n==20) return ElementType::H20;
        if(n==27) return ElementType::H27;
    }
    return ElementType::Unknown;
}
