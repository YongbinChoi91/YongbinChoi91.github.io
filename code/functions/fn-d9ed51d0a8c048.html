static TimeSolverSettings::HysteresisAxisSpec default_hysteresis_x_axis(const ProblemParameters& params){
    TimeSolverSettings::HysteresisAxisSpec a;
    a.enabled=true;
    a.mode="prescribed";
    a.target=params.solver.hysteresis_target;
    a.scale=1.0;
    a.offset=0.0;
    a.label="prescribed " + hysteresis_label(a.target);
    return a;
}
