inline std::array<std::array<Scalar, P>, R> matmul(
    const std::array<std::array<Scalar, C>, R>& A,
    const std::array<std::array<Scalar, P>, C>& B
) {
    static_assert(C > 0, "matmul requires a non-empty contraction dimension.");
    std::array<std::array<Scalar, P>, R> M{};
    for (std::size_t i = 0; i < R; ++i) {
        for (std::size_t j = 0; j < P; ++j) {
            M[i][j] = A[i][0] * B[0][j];
            for (std::size_t k = 1; k < C; ++k) M[i][j] = M[i][j] + A[i][k] * B[k][j];
        }
    }
    return M;
}
