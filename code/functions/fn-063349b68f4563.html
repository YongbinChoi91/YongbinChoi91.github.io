static bool strip_domain_group_prefix(const std::string& raw,const std::string& domain_name,std::string& group){
    const std::string dot=domain_name+".";
    const std::string slash=domain_name+"/";
    if(raw.rfind(dot,0)==0){ group=raw.substr(dot.size()); return true; }
    if(raw.rfind(slash,0)==0){ group=raw.substr(slash.size()); return true; }
    group=raw;
    return false;
}
