void write_rve_solution_vtk_if_enabled(
    const FieldMesh& mesh,
    const Vector& x,
    const std::string& file,
    const MaterialHistoryStore* history
) {
    if (file.empty() || file == "none" || file == "off") return;
    ensure_output_parent_directory(file);
    write_vtk(mesh, x, file, history);
}
