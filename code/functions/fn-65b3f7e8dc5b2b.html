std::vector<GaussPoint> gauss_rule_line(int order_per_direction){
    std::vector<double> x,w;
    gauss_legendre_1d_interval(order_per_direction,-1.0,1.0,x,w);
    std::vector<GaussPoint> gps;
    gps.reserve(static_cast<std::size_t>(order_per_direction));
    for(int i=0;i<order_per_direction;++i) gps.push_back({x[i],0.0,0.0,w[i]});
    return gps;
}
