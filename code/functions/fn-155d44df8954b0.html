static double surface_measure_from_J_3d(ElementType type,int face,const JacobianData& J){
    std::array<double,3> a{{0.0,0.0,0.0}};
    std::array<double,3> b{{0.0,0.0,0.0}};
 
    if(is_hex_element_type(type)){
        if(face==0 || face==1){
            a = jacobian_row3_assembly(J,0);
            b = jacobian_row3_assembly(J,1);
        } else if(face==2 || face==4){
            a = jacobian_row3_assembly(J,0);
            b = jacobian_row3_assembly(J,2);
        } else if(face==3 || face==5){
            a = jacobian_row3_assembly(J,1);
            b = jacobian_row3_assembly(J,2);
        } else {
            throw std::runtime_error("Unsupported hex face id in 3D Neumann surface measure.");
        }
    } else if(is_tet_element_type(type)){
        if(face==0){
            a = jacobian_row3_assembly(J,0);
            b = jacobian_row3_assembly(J,1);
        } else if(face==1){
            a = jacobian_row3_assembly(J,0);
            b = jacobian_row3_assembly(J,2);
        } else if(face==2){
            a = jacobian_row3_assembly(J,1);
            b = jacobian_row3_assembly(J,2);
        } else if(face==3){
            const auto j0 = jacobian_row3_assembly(J,0);
            const auto j1 = jacobian_row3_assembly(J,1);
            const auto j2 = jacobian_row3_assembly(J,2);
            for(int d=0; d<3; ++d){
                a[d] = j1[d] - j0[d];
                b[d] = j2[d] - j0[d];
            }
        } else {
            throw std::runtime_error("Unsupported tetrahedron face id in 3D Neumann surface measure.");
        }
    } else {
        throw std::runtime_error("3D Neumann surface measure supports only Tet/Hex elements.");
    }
 
    const double measure = norm3_assembly(cross3_assembly(a,b));
    if(measure<=0.0) throw std::runtime_error("Degenerate 3D Neumann boundary face: zero surface measure.");
    return measure;
}
