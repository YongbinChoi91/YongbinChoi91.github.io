static std::string velocity_boundary_name_from_section(const std::string& section){
    if(section.rfind("VELOCITY_BOUNDARY_",0)==0) return section.substr(std::string("VELOCITY_BOUNDARY_").size());
    if(section.rfind("V_",0)==0) return section.substr(2);
    return "";
}
