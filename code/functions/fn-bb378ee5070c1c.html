void observe_multiscale_rve_diagnostics(
    double residual,
    int iterations,
    bool converged,
    const HistoryLocalDiagnostics* history_diagnostics
) {
    std::lock_guard<std::mutex> lock(diagnostics_mutex());
    multiscale_context().diagnostics.active = true;
    multiscale_context().diagnostics.max_residual = std::max(multiscale_context().diagnostics.max_residual, residual);
    multiscale_context().diagnostics.max_iterations = std::max(multiscale_context().diagnostics.max_iterations, iterations);
    multiscale_context().diagnostics.total_solves += 1;
    if (!converged) multiscale_context().diagnostics.failed_solves += 1;
    if (history_diagnostics && history_diagnostics->active) {
        multiscale_context().diagnostics.history.active = true;
        multiscale_context().diagnostics.history.name = history_diagnostics->name;
        const bool new_worst = history_diagnostics->max_residual > multiscale_context().diagnostics.history.max_residual;
        multiscale_context().diagnostics.history.max_residual = std::max(
            multiscale_context().diagnostics.history.max_residual,
            history_diagnostics->max_residual
        );
        multiscale_context().diagnostics.history.max_iterations = std::max(
            multiscale_context().diagnostics.history.max_iterations,
            history_diagnostics->max_iterations
        );
        multiscale_context().diagnostics.history.local_failure_count += history_diagnostics->local_failure_count;
        multiscale_context().diagnostics.history.yielded_gp_count += history_diagnostics->yielded_gp_count;
        multiscale_context().diagnostics.history.total_gp_count += history_diagnostics->total_gp_count;
        if (new_worst) {
            multiscale_context().diagnostics.history.worst_domain = history_diagnostics->worst_domain;
            multiscale_context().diagnostics.history.worst_element = history_diagnostics->worst_element;
            multiscale_context().diagnostics.history.worst_gp = history_diagnostics->worst_gp;
        }
    }
}
