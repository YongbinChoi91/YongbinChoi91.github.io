static std::vector<double> parse_selector_numbers(std::string body,const std::string& selector){
    for(char& ch:body){
        if(ch==',' || ch==';' || ch=='(' || ch==')') ch=' ';
    }
    std::stringstream ss(body);
    std::vector<double> values;
    double v=0.0;
    while(ss>>v) values.push_back(v);
    if(ss.fail() && !ss.eof()){
        throw std::runtime_error("Bad "+selector+" selector coordinates.");
    }
    return values;
}
