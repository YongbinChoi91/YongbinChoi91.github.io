static std::array<double,3> hex_face_reference_point(int face,double s,double t){
    if(face==0) return { s, t,-1.0};
    if(face==1) return { s, t, 1.0};
    if(face==2) return { s,-1.0,t};
    if(face==3) return { 1.0,s, t};
    if(face==4) return { s, 1.0,t};
    if(face==5) return {-1.0,s, t};
    throw std::runtime_error("Unsupported hex face id.");
}
