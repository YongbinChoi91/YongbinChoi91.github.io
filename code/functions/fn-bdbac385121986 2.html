inline std::array<AD1<Scalar, N>, N> make_active_vector(const std::array<Scalar, N>& x) {
    std::array<AD1<Scalar, N>, N> a{};
    for (std::size_t i = 0; i < N; ++i) a[i] = AD1<Scalar, N>::variable(x[i], i);
    return a;
}
