configure_model_output_paths()

Resolve model output paths using the configured root and optional custom folder name.

Public

Signature

void configure_model_output_paths(ProblemParameters &params, const std::string &model_name)

A nonempty folder_name other than auto overrides the model-based folder name and is sanitized. A custom folder is used even when model_output_subdir is off; otherwise that switch controls whether the sanitized model name is appended. An existing matching final path component is not appended twice. The resolved root contains vtk, debug_csv, figures, and debug_dofs; the HDF5 filename still uses the sanitized model name. This function sets paths without creating directories.

void configure_model_output_paths(ProblemParameters& params,const std::string& model_name){
    const std::string safe=sanitize_output_name(model_name);
    params.solver.model_output_name=safe;
    std::string root=params.solver.output_root.empty() ? std::string("./output") : params.solver.output_root;
    const bool custom_folder = !params.solver.folder_name.empty()
        && lower_copy(params.solver.folder_name) != "auto";
    const std::string folder = custom_folder
        ? sanitize_output_name(params.solver.folder_name) : safe;
    if((custom_folder || params.solver.model_output_subdir) && !path_basename_equals(root,folder)){
        root=path_join_simple(root,folder);
    }
    params.solver.output_root=root;
    params.solver.vtk_dir=path_join_simple(root,"vtk");
    params.solver.csv_dir=path_join_simple(root,"debug_csv");
    params.solver.figure_dir=path_join_simple(root,"figures");
    params.solver.dof_dir=path_join_simple(root,"debug_dofs");
    params.solver.hdf5_file=path_join_simple(root,safe + ".h5");
}
