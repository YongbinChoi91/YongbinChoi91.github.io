void assign_field_dof_offsets(DofLayout& mesh){
    int off=0;
    for(auto& f: mesh.fields){
        if(f.components<=0) throw std::runtime_error("Generic field '"+f.name+"' has non-positive component count.");
        f.offset=off;
        f.ndofs=f.components*static_cast<int>(f.nodes.size());
        off += f.ndofs;
    }
    mesh.total_dofs=off;
}
