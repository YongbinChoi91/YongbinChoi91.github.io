int element_dimension(ElementType type){
    switch(type){
        case ElementType::T3:
        case ElementType::T6:
        case ElementType::Q4:
        case ElementType::Q9:
        case ElementType::Polygon:
            return 2;
        case ElementType::Tet4:
        case ElementType::Tet10:
        case ElementType::H8:
        case ElementType::H20:
        case ElementType::H27:
            return 3;
        default:
            return 0;
    }
}
