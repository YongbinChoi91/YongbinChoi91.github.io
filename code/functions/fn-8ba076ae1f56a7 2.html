inline std::array<std::array<Scalar, 2>, 2> cofactor(const std::array<std::array<Scalar, 2>, 2>& F) {
    auto C = zero_tensor2<Scalar, 2, 2>();
    C[0][0] =  F[1][1]; C[0][1] = -F[1][0];
    C[1][0] = -F[0][1]; C[1][1] =  F[0][0];
    return C;
}
