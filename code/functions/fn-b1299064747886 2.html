RveSolveResult solve_rve_pbc_augmented(
    const Problem& problem,
    const ProblemParameters& params,
    const RveSolveOptions& options
) {
    if (problem.domains.size() != 1) throw std::runtime_error("single RVE solver expects exactly one domain.");
    const FieldMesh& mesh = problem.domains.front().mesh;
    const auto runtime = get_rve_runtime_cache(problem, params, options);
    const int ndofs = runtime->ndofs;
 
    RveSolveResult result;
    result.pbc = runtime->pbc;
    result.Fmacro = options.Fmacro;
 
    const bool has_warm_displacement = options.use_warm_start && options.initial_displacement;
    const bool has_warm_lambda = options.use_warm_start && options.initial_lambda;
    const bool started_from_warm = has_warm_displacement || has_warm_lambda;
 
    RveNewtonWork accepted;
    if (has_warm_displacement) {
        if (static_cast<int>(options.initial_displacement->size()) != ndofs) {
            throw std::runtime_error("RVE warm-start displacement size does not match the current micro DOF count.");
        }
        accepted.displacement = *options.initial_displacement;
    } else {
        accepted.displacement.assign(ndofs, 0.0);
    }
 
    const int n_constraints = static_cast<int>(result.pbc.constraints.size());
    if (has_warm_lambda) {
        if (static_cast<int>(options.initial_lambda->size()) != n_constraints) {
            throw std::runtime_error("RVE warm-start lambda size does not match the current PBC constraint count.");
        }
        accepted.lambda = *options.initial_lambda;
    } else {
        accepted.lambda.assign(n_constraints, 0.0);
    }
 
    accepted.history = options.initial_history ? *options.initial_history : runtime->history_template;
 
    const std::array<std::array<double,2>,2> F_start =
        (started_from_warm && options.initial_Fmacro) ? *options.initial_Fmacro : identity_F2();
 
    double s_accepted = 0.0;
    double ds = std::max(options.min_substep, std::min(1.0, options.initial_substep));
    int accepted_substeps = 0;
    int attempted_substeps = 0;
    RveNewtonWork last_failed;
    bool have_last_failed = false;
    std::string last_failure_error;
    MaterialHistoryStore response_history;
    bool have_response_history = false;
    double response_dt = options.dt;
    double response_dt_previous = options.dt_previous;
    double previous_accepted_dt = 0.0;
    const double cutback_factor = (options.cutback_factor > 0.0 && options.cutback_factor < 1.0)
        ? options.cutback_factor
        : 0.5;
    const double growth_factor = options.growth_factor > 1.0 ? options.growth_factor : 1.0;
    const int max_substeps = std::max(1, options.max_substeps);
    const double min_substep = std::max(1.0e-14, options.min_substep);
 
    if (!options.adaptive_substepping) {
        ds = 1.0;
    }
 
    while (s_accepted < 1.0 - 1.0e-14) {
        if (++attempted_substeps > std::max(1, max_substeps * 8)) break;
        const double s_trial = std::min(1.0, s_accepted + ds);
        const auto F_trial = interpolate_F2(F_start, options.Fmacro, s_trial);
        if (options.verbose && options.adaptive_substepping) {
            std::cout << "  RVE substep " << accepted_substeps + 1
                      << ": s " << s_accepted << " -> " << s_trial
                      << " (ds=" << (s_trial - s_accepted) << ")\n";
        }
 
        RveNewtonWork trial = accepted;
        const RveSolveOptions substep_options =
            make_substep_options(options, s_accepted, s_trial, previous_accepted_dt);
        try {
            solve_rve_increment_newton(
                problem,
                params,
                substep_options,
                *runtime,
                F_trial,
                started_from_warm || accepted_substeps > 0,
                trial
            );
            last_failure_error.clear();
        } catch (const std::exception& e) {
            trial.converged = false;
            last_failure_error = e.what();
        }
        result.iterations += trial.iterations;
 
        if (trial.converged) {
            if (s_trial >= 1.0 - 1.0e-14) {
                response_history = trial.history;
                have_response_history = true;
                response_dt = substep_options.dt;
                response_dt_previous = substep_options.dt_previous;
            }
            accepted = std::move(trial);
            if (material_history_has_specs(accepted.history)) {
                commit_material_history_store(accepted.history);
            }
            s_accepted = s_trial;
            previous_accepted_dt = substep_options.dt;
            ++accepted_substeps;
            if (options.adaptive_substepping && accepted_substeps >= max_substeps && s_accepted < 1.0 - 1.0e-14) {
                break;
            }
            if (options.adaptive_substepping) {
                ds = std::min(1.0 - s_accepted, ds * growth_factor);
                if (ds <= 0.0) ds = 1.0 - s_accepted;
            }
            continue;
        }
 
        last_failed = std::move(trial);
        have_last_failed = last_failed.Aaug.n_rows > 0;
        if (!options.adaptive_substepping) break;
 
        const double next_ds = ds * cutback_factor;
        if (next_ds < min_substep || accepted_substeps >= max_substeps) {
            break;
        }
        ds = next_ds;
        if (options.verbose) {
            std::cout << "  RVE substep cutback: s = " << s_accepted
                      << ", ds -> " << ds;
            if (!last_failure_error.empty()) std::cout << " because " << last_failure_error;
            std::cout << "\n";
        }
    }
 
    result.converged = s_accepted >= 1.0 - 1.0e-14 && accepted.converged;
    if (!result.converged && !last_failure_error.empty() && !have_last_failed) {
        throw std::runtime_error("RVE substepping failed before a tangent was assembled: " + last_failure_error);
    }
    RveNewtonWork& output = (!result.converged && have_last_failed && last_failed.Aaug.n_rows > 0)
        ? last_failed
        : accepted;
    result.displacement = std::move(output.displacement);
    result.lambda = std::move(output.lambda);
    result.tangent = std::move(output.Aaug);
    result.residual = std::move(output.residual);
 
    ElementState response_state;
    response_state.dt = response_dt;
    response_state.dt_previous = response_dt_previous;
    response_state.step = options.step;
    const MaterialHistoryStore* response_history_ptr =
        have_response_history ? &response_history : &output.history;
    const ConstitutiveMultiscaleResponse micro_response =
        compute_rve_constitutive_multiscale_response(mesh, result.displacement, params, response_history_ptr, response_state);
    result.macro = make_rve_macro_response(micro_response);
    const auto du_dF = compute_rve_displacement_sensitivity(result.tangent, mesh, result.pbc, params.solver.linear);
    result.sensitivity = compute_rve_macro_sensitivity_from_piola_du(mesh, micro_response.dIntPdu, du_dF);
    result.history_diagnostics = output.history_diagnostics;
    if (material_history_has_specs(output.history)) {
        result.history = std::move(output.history);
        result.has_history = true;
    }
 
    write_rve_solution_vtk_if_enabled(
        mesh,
        result.displacement,
        options.vtk_output,
        result.has_history ? &result.history : nullptr
    );
    if (!options.csv_output.empty() && options.csv_output != "none" && options.csv_output != "off") {
        write_rve_macro_csv(options.csv_output, result.macro, &result.sensitivity);
    }
    return result;
}
