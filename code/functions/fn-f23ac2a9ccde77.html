static std::vector<double> collect_solid_deformation_gradients(
    const FieldMesh& mesh,
    const ProblemParameters& params,
    const Vector& x
){
    std::vector<double> data;
    if(!has_displacement_field(mesh)) return data;
    const auto gps = gauss_rule(mesh.displacement_type, params.solver.gauss_order);
    const int nelem = static_cast<int>(mesh.displacement_elements.size());
    data.reserve(static_cast<size_t>(nelem) * gps.size() * 9);
    for(int e=0;e<nelem;++e){
        const ElementGeometry geom = make_displacement_element_geometry(mesh,e);
        const auto& conn = mesh.displacement_elements[e];
        Vector u_local(static_cast<size_t>(mesh.dim) * conn.size(), 0.0);
        for(int a=0;a<static_cast<int>(conn.size());++a){
            const int node = conn[a];
            for(int c=0;c<mesh.dim;++c){
                u_local[static_cast<size_t>(mesh.dim) * a + c] = x[displacement_dof(mesh,node,c)];
            }
        }
        for(const auto& gp:gps){
            const ShapeData shape = compute_shape(mesh.displacement_type,gp.xi,gp.eta,gp.zeta);
            const JacobianData J = compute_jacobian_data(geom,shape);
            if(J.detJ <= 0.0) throw std::runtime_error("Negative detJ while exporting Gauss-point deformation gradients.");
            const auto dN = compute_dN_dx(shape,J);
            const auto F = hpx::deformation_gradient_array<double>(mesh.dim,dN,u_local);
            for(int i=0;i<3;++i){
                for(int j=0;j<3;++j) data.push_back(F[i][j]);
            }
        }
    }
    return data;
}
