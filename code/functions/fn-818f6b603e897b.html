static std::array<double,3> tet_face_reference_point(int face,double r,double s){
    if(face==0) return {r,s,0.0};
    if(face==1) return {r,0.0,s};
    if(face==2) return {0.0,r,s};
    if(face==3) return {1.0-r-s,r,s};
    throw std::runtime_error("Unsupported tetrahedron face id.");
}
