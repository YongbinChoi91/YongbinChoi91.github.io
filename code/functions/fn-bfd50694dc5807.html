inline std::array<std::array<Scalar,3>,3> deformation_gradient_2d_plane_strain_array(
    const std::vector<std::array<double,3>>& dNdx,
    const std::vector<Scalar>& u_local_interleaved
) {
    const int nnode = static_cast<int>(dNdx.size());
    if (static_cast<int>(u_local_interleaved.size()) < 2 * nnode) {
        throw std::runtime_error("2D plane-strain deformation gradient received too few displacement DOFs.");
    }
 
    std::array<std::array<Scalar,3>,3> F{};
    for (int i = 0; i < 3; ++i) {
        for (int j = 0; j < 3; ++j) F[i][j] = Scalar(0.0);
        F[i][i] = Scalar(1.0);
    }
 
    for (int a = 0; a < nnode; ++a) {
        const Scalar ux = u_local_interleaved[2 * a + 0];
        const Scalar uy = u_local_interleaved[2 * a + 1];
        F[0][0] += ux * Scalar(dNdx[a][0]);
        F[0][1] += ux * Scalar(dNdx[a][1]);
        F[1][0] += uy * Scalar(dNdx[a][0]);
        F[1][1] += uy * Scalar(dNdx[a][1]);
    }
    return F;
}
