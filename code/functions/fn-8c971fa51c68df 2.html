bool self_check_det2() {
    const std::array<std::array<double, 2>, 2> F{{{{1.2, 0.3}}, {{-0.2, 0.9}}}};
    const auto dJ = grad([](const auto& F_ad) {
        return det(F_ad);
    }, F);
    return close(dJ[0][0], F[1][1], 1e-14)
        && close(dJ[0][1], -F[1][0], 1e-14)
        && close(dJ[1][0], -F[0][1], 1e-14)
        && close(dJ[1][1], F[0][0], 1e-14);
}
