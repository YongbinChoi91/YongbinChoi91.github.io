void append_local_field_from_mixed(
    LocalFieldView& view,
    const FieldMesh& mesh,
    int elem_id,
    const std::string& name
) {
    LocalFieldInfo f;
    f.name = name;
    f.local_offset = view.n_local();
 
    if (name == "velocity") {
        if (elem_id < 0 || elem_id >= static_cast<int>(mesh.velocity_elements.size()))
            throw std::runtime_error("Element id out of range for velocity local field.");
        f.components = mesh.dim;
        f.n_nodes = static_cast<int>(mesh.velocity_elements[elem_id].size());
        f.element_type = mesh.velocity_type;
    } else if (name == "displacement") {
        if (!has_displacement_field(mesh)) return;
        if (elem_id < 0 || elem_id >= static_cast<int>(mesh.displacement_elements.size()))
            throw std::runtime_error("Element id out of range for displacement local field.");
        f.components = mesh.dim;
        f.n_nodes = static_cast<int>(mesh.displacement_elements[elem_id].size());
        f.element_type = mesh.displacement_type;
    } else if (name == "pressure") {
        if (elem_id < 0 || elem_id >= static_cast<int>(mesh.pressure_elements.size()))
            throw std::runtime_error("Element id out of range for pressure local field.");
        f.components = 1;
        f.n_nodes = static_cast<int>(mesh.pressure_elements[elem_id].size());
        f.element_type = mesh.pressure_type;
    } else if (name == "temperature") {
        const FieldInfo* extra = nullptr;
        for (const auto& ef : mesh.extra_fields) {
            if (ef.name == name) { extra = &ef; break; }
        }
        if (extra) {
            if (elem_id < 0 || elem_id >= static_cast<int>(extra->elements.size()))
                throw std::runtime_error("Element id out of range for temperature local field.");
            f.components = extra->components;
            f.n_nodes = static_cast<int>(extra->elements[elem_id].size());
            f.element_type = extra->element_type;
        } else {
            if (elem_id < 0 || elem_id >= static_cast<int>(mesh.velocity_elements.size()))
                throw std::runtime_error("Element id out of range for temperature local field.");
            f.components = 1;
            f.n_nodes = static_cast<int>(mesh.velocity_elements[elem_id].size());
            f.element_type = mesh.velocity_type;
        }
    } else {
        const DofLayout& gmesh = get_cached_dof_layout(mesh);
        const int fid = find_field_index(gmesh, name, true);
        const FieldInfo& gf = gmesh.fields[fid];
        if (elem_id < 0 || elem_id >= static_cast<int>(gf.elements.size()))
            throw std::runtime_error("Element id out of range for local field " + name + ".");
        f.components = gf.components;
        f.n_nodes = static_cast<int>(gf.elements[elem_id].size());
        f.element_type = gf.element_type;
    }
 
    view.fields.push_back(f);
}
