void commit_multiscale_rve_history_cache() {
    std::lock_guard<std::mutex> lock(warm_start_mutex());
    for (auto& kv : multiscale_context().trial_rve_cache) {
        multiscale_context().accepted_rve_cache[kv.first] = std::move(kv.second);
    }
    multiscale_context().trial_rve_cache.clear();
