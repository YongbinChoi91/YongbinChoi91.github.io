static FieldComponentTarget target_from_alias(const std::string& group,const std::string& comp){
    FieldComponentTarget t;
    t.group = group;
    t.selector = parse_bc_selector_token(group);
    const std::string c=lower_copy(comp);
    if(c=="vx" || c=="vy" || c=="vz"){
        t.field="velocity";
        t.component=component_index_from_string(c);
        return t;
    }
    if(c=="ux" || c=="uy" || c=="uz"){
        t.field="displacement";
        t.component=component_index_from_string(c);
        return t;
    }
    if(c=="p" || c=="pressure"){
        t.field="pressure";
        t.component=0;
        return t;
    }
    if(c=="t" || c=="theta" || c=="temperature"){
        t.field="temperature";
        t.component=0;
        return t;
    }
 
    // Shorthand form: DIRICHLET group x/y/z expr targets velocity unless
    // ux/uy/uz is used.
    t.field="velocity";
    t.component=component_index_from_string(c);
    return t;
}
