ElementGeometry make_velocity_element_geometry(const FieldMesh& mesh,int elem_id){
    ElementGeometry elem;
    elem.dim = mesh.dim;
    elem.type = mesh.velocity_type;
    const auto& conn = mesh.velocity_elements[elem_id];
    elem.coords.resize(conn.size());
    for(int a=0;a<static_cast<int>(conn.size());++a) elem.coords[a] = mesh.velocity_nodes[conn[a]].x;
    return elem;
}
