DofsInfo read_dofs_info(const std::string& filename){
    std::ifstream file(filename);
    if(!file) throw std::runtime_error("Could not open DOFS info file: "+filename);
    DofsInfo info;
    std::string current_domain;
    std::string line;
    while(std::getline(file,line)){
        if(!line.empty() && line.back()=='\r') line.pop_back();
        const auto comment=line.find('#');
        if(comment!=std::string::npos) line=line.substr(0,comment);
        line=trim_copy(line);
        if(line.empty()) continue;
        std::stringstream ss(line);
        std::string key; ss >> key;
        key=lower_copy(key);
        if(key=="field_key" || key=="field_keys" || key=="fields"){
            std::string f;
            while(ss >> f){
                f=lower_copy(f);
                if(string_vector_contains(info.field_order,f)) throw std::runtime_error("Duplicate field in DOFS_info.txt: "+f);
                info.field_order.push_back(f);
            }
        } else if(key=="domain" || key=="domain_name"){
            std::string name; ss >> name;
            if(name.empty()) throw std::runtime_error("DOFS_info domain declaration must be: domain <name>.");
            info.domain_name=name;
            current_domain=normalize_domain_key(name);
        } else if(key=="history" || key=="history_variable" || key=="history_variables" || key=="histories"){
            std::string name;
            int nvars=0;
            if(key=="history_variables" || key=="history_variable"){
                // Accept either:
                //   history_variables 5
                //   history_variables plasticity 5
                std::string a,b; ss >> a;
                if(a.empty()) throw std::runtime_error("Bad history_variables line in DOFS_info.txt.");
                std::stringstream as(a);
                if(as >> nvars && as.eof()){
                    name="history";
                } else {
                    name=lower_copy(a);
                    ss >> nvars;
                }
            } else {
                ss >> name >> nvars;
                name=lower_copy(name);
            }
            if(name.empty() || nvars<=0) throw std::runtime_error("History declaration must be: history <name> <positive_nvars>.");
            if(!current_domain.empty()){
                add_history_spec(info.domain_histories[current_domain],{name,nvars},"domain "+current_domain);
            } else {
                add_history_spec(info.histories,{name,nvars},"global scope");
            }
        } else if(key=="vem" || key=="vem_order" || key=="vem_type"){
            std::string order; ss >> order;
            if(order.empty()) throw std::runtime_error("VEM declaration must be: VEM linear|quadratic or VEM_ORDER 1|2.");
            if(lower_copy(order)=="vem"){
                ss >> order;
                if(order.empty()) throw std::runtime_error("VEM declaration must include linear/quadratic or 1/2.");
            }
            info.vem_order=parse_vem_order_token(order);
        } else if(key=="type"){
            std::string family; ss >> family;
            if(lower_copy(family)=="vem"){
                std::string order; ss >> order;
                if(order.empty()) throw std::runtime_error("TYPE VEM declaration must include linear/quadratic or 1/2.");
                info.vem_order=parse_vem_order_token(order);
            }
        }
    }
    if(info.field_order.empty()) throw std::runtime_error("DOFS_info.txt must contain a line like: field_key velocity displacement pressure");
    return info;
}
