void reset_hdf5_solution_snapshots(const ProblemParameters& params,int ndofs){
    if(!params.solver.hdf5_output) return;
    reset_hdf5_table(params,"/time/values",{"time"});
    reset_hdf5_table(params,"/time/steps",{"step"});
    std::vector<std::string> cols;
    cols.reserve(ndofs);
    for(int i=0;i<ndofs;++i) cols.push_back("dof_" + std::to_string(i));
    reset_hdf5_table(params,"/solution/dofs",cols);
}
