Vector solve_with_petsc_backend(const CSRMatrix& A,const Vector& rhs,const LinearSolverSettings& s,const CSRMatrix* P_user,const PetscPCDData* pcd_data){
    if(A.n_rows!=A.n_cols || static_cast<int>(rhs.size())!=A.n_rows) throw std::runtime_error("PETSc dimension mismatch.");
    const int n=A.n_rows;
    auto chk=[](PetscErrorCode ierr,const char* where){ if(ierr) throw std::runtime_error(std::string("PETSc error at ")+where); };
 
    PetscBool init=PETSC_FALSE;
    chk(PetscInitialized(&init),"PetscInitialized");
    if(!init) chk(PetscInitializeNoArguments(),"PetscInitializeNoArguments");
 
    auto set_ksp_type_from_string=[&](KSP ksp,const std::string& name){
        const std::string v=lower_copy(name);
        if(v=="direct" || v=="lu" || v=="preonly") chk(KSPSetType(ksp,KSPPREONLY),"KSPSetType");
        else if(v=="gmres") chk(KSPSetType(ksp,KSPGMRES),"KSPSetType");
        else if(v=="fgmres") chk(KSPSetType(ksp,KSPFGMRES),"KSPSetType");
        else if(v=="bicgstab" || v=="bcgs") chk(KSPSetType(ksp,KSPBCGS),"KSPSetType");
        else if(v=="cg") chk(KSPSetType(ksp,KSPCG),"KSPSetType");
        else throw std::runtime_error("Unknown PETSc KSP type: "+name);
        if((v=="gmres" || v=="fgmres") && s.gmres_restart>0) chk(KSPGMRESSetRestart(ksp,s.gmres_restart),"KSPGMRESSetRestart");
    };
 
    auto set_pc_type_from_string=[&](PC pc,const std::string& name,int ilu_levels=0){
        const std::string v=lower_copy(name);
        if(v=="none") chk(PCSetType(pc,PCNONE),"PCSetType");
        else if(v=="lu") chk(PCSetType(pc,PCLU),"PCSetType");
        else if(v=="jacobi") chk(PCSetType(pc,PCJACOBI),"PCSetType");
        else if(v=="ilu"){
            chk(PCSetType(pc,PCILU),"PCSetType");
            if(ilu_levels>0) chk(PCFactorSetLevels(pc,ilu_levels),"PCFactorSetLevels");
        }
        else if(v=="hypre" || v=="boomeramg"){
            chk(PCSetType(pc,PCHYPRE),"PCSetType");
            chk(PCHYPRESetType(pc,"boomeramg"),"PCHYPRESetType");
        }
        else if(v=="gamg" || v=="amg") chk(PCSetType(pc,PCGAMG),"PCSetType");
        else if(v=="asm") chk(PCSetType(pc,PCASM),"PCSetType");
        else if(v=="sor") chk(PCSetType(pc,PCSOR),"PCSetType");
        else throw std::runtime_error("Unknown PETSc PC type: "+name);
    };
 
 
    auto insert_csr_rows=[&](Mat M,const CSRMatrix& C,const char* where){
        for(int i=0;i<C.n_rows;++i){
            const PetscInt count = static_cast<PetscInt>(C.row_ptr[i+1]-C.row_ptr[i]);
            if(count<=0) continue;
            const PetscInt row = static_cast<PetscInt>(i);
            if constexpr (std::is_same_v<PetscInt,int> && std::is_same_v<PetscScalar,double>){
                chk(MatSetValues(M,1,&row,count,
                                 reinterpret_cast<const PetscInt*>(&C.col_idx[C.row_ptr[i]]),
                                 reinterpret_cast<const PetscScalar*>(&C.values[C.row_ptr[i]]),
                                 INSERT_VALUES),where);
            } else {
                std::vector<PetscInt> cols(count);
                std::vector<PetscScalar> vals(count);
                for(PetscInt q=0;q<count;++q){
                    cols[q]=static_cast<PetscInt>(C.col_idx[C.row_ptr[i]+q]);
                    vals[q]=static_cast<PetscScalar>(C.values[C.row_ptr[i]+q]);
                }
                chk(MatSetValues(M,1,&row,count,cols.data(),vals.data(),INSERT_VALUES),where);
            }
        }
    };
 
    auto assemble_mat=[&](Mat M,const char* where){
        chk(MatAssemblyBegin(M,MAT_FINAL_ASSEMBLY),where);
        chk(MatAssemblyEnd(M,MAT_FINAL_ASSEMBLY),where);
    };
 
    auto csr_to_petsc_mat=[&](const CSRMatrix& C, Mat* M, const char* where){
        if(C.n_rows!=C.n_cols) throw std::runtime_error(std::string(where)+": CSR matrix must be square.");
        const int m=C.n_rows;
        std::vector<PetscInt> nnzC(m);
        for(int i=0;i<m;++i) nnzC[i]=C.row_ptr[i+1]-C.row_ptr[i];
        chk(MatCreateSeqAIJ(PETSC_COMM_SELF,m,m,0,nnzC.data(),M),where);
        insert_csr_rows(*M,C,where);
        assemble_mat(*M,where);
    };
 
    auto set_csr_values=[&](Mat M,const CSRMatrix& C,const char* where,bool assemble=true){
        insert_csr_rows(M,C,where);
        if(assemble) assemble_mat(M,where);
    };
 
    auto set_vec_values=[&](Vec v,const Vector& values,const char* where){
        PetscScalar* arr=nullptr;
        chk(VecGetArray(v,&arr),where);
        for(int i=0;i<static_cast<int>(values.size());++i) arr[i]=static_cast<PetscScalar>(values[i]);
        chk(VecRestoreArray(v,&arr),where);
    };
 
    KSP ksp; PC pc;
    Mat user_schur_pre = nullptr;
    bool owns_user_schur_pre = false;
 
    // Optional preconditioner-only matrix.  The Newton operator K is never
    // changed.  When P_user is supplied, PETSc uses it only for PC setup.
    // This is the path used for custom pressure Schur approximations such as
    // pressure mass, pressure Laplacian, and mass+Laplacian.
    const bool wants_fieldsplit = lower_copy(s.preconditioner)=="fieldsplit" || lower_copy(s.preconditioner)=="schur" || lower_copy(s.preconditioner)=="block_schur";
    const std::vector<char> pressure_mask = wants_fieldsplit ? make_petsc_pressure_mask(s,n) : std::vector<char>();
    const int p_kind = P_user ? 1 : ((wants_fieldsplit && s.schur_pressure_shift>0.0) ? 2 : 0);
 
    static PetscOuterSolveCache cache;
    if(!petsc_outer_cache_is_compatible(cache,A,P_user,p_kind,pressure_mask)){
        destroy_petsc_outer_cache(cache);
        cache.valid=true;
        cache.n=n;
        cache.p_kind=p_kind;
        cache.row_ptr=A.row_ptr;
        cache.col_idx=A.col_idx;
        cache.pressure_mask=pressure_mask;
 
        std::vector<PetscInt> nnz(n);
        for(int i=0;i<n;++i) nnz[i]=A.row_ptr[i+1]-A.row_ptr[i];
        chk(MatCreateSeqAIJ(PETSC_COMM_SELF,n,n,0,nnz.data(),&cache.K),"MatCreateSeqAIJ");
        if(p_kind==1){
            cache.p_row_ptr=P_user->row_ptr;
            cache.p_col_idx=P_user->col_idx;
            std::vector<PetscInt> pnnz(n);
            for(int i=0;i<n;++i) pnnz[i]=P_user->row_ptr[i+1]-P_user->row_ptr[i];
            chk(MatCreateSeqAIJ(PETSC_COMM_SELF,n,n,0,pnnz.data(),&cache.P),"MatCreateSeqAIJ P_user");
        } else if(p_kind==2){
            chk(MatCreateSeqAIJ(PETSC_COMM_SELF,n,n,0,nnz.data(),&cache.P),"MatCreateSeqAIJ shifted P");
        } else {
            cache.P=cache.K;
        }
 
        chk(VecCreateSeq(PETSC_COMM_SELF,n,&cache.b),"VecCreateSeq");
        chk(VecDuplicate(cache.b,&cache.x),"VecDuplicate");
 
        if(wants_fieldsplit){
            if(s.n_velocity_dofs_full<=0 || s.n_pressure_dofs_full<=0 || static_cast<int>(s.reduced_to_full_dofs.size())!=n){
                throw std::runtime_error("PETSc FieldSplit requested, but reduced/full DOF field map was not provided.");
            }
            std::vector<PetscInt> mech, press;
            mech.reserve(n); press.reserve(n);
            for(int i=0;i<n;++i){
                if(pressure_mask[i]) press.push_back(static_cast<PetscInt>(i));
                else mech.push_back(static_cast<PetscInt>(i));
            }
            if(mech.empty() || press.empty()) throw std::runtime_error("PETSc FieldSplit needs non-empty mechanical and pressure splits.");
            chk(ISCreateGeneral(PETSC_COMM_SELF,static_cast<PetscInt>(mech.size()),mech.data(),PETSC_COPY_VALUES,&cache.is_mech),"ISCreateGeneral mechanical");
            chk(ISCreateGeneral(PETSC_COMM_SELF,static_cast<PetscInt>(press.size()),press.data(),PETSC_COPY_VALUES,&cache.is_p),"ISCreateGeneral pressure");
        }
    }
 
    Mat K=cache.K;
    Mat P=cache.P;
    Vec b=cache.b;
    Vec x=cache.x;
 
    chk(MatZeroEntries(K),"MatZeroEntries K");
    set_csr_values(K,A,"MatSetValues K");
 
    if(p_kind==1){
        if(P_user->n_rows!=n || P_user->n_cols!=n) throw std::runtime_error("PETSc preconditioner matrix dimension mismatch.");
        chk(MatZeroEntries(P),"MatZeroEntries P_user");
        set_csr_values(P,*P_user,"MatSetValues P_user");
    } else if(p_kind==2){
        chk(MatZeroEntries(P),"MatZeroEntries shifted P");
        set_csr_values(P,A,"MatSetValues shifted P",false);
        for(int i=0;i<n;++i){
            if(pressure_mask[i]) chk(MatSetValue(P,i,i,s.schur_pressure_shift,ADD_VALUES),"MatSetValue pressure shift");
        }
        chk(MatAssemblyBegin(P,MAT_FINAL_ASSEMBLY),"MatAssemblyBegin shifted P");
        chk(MatAssemblyEnd(P,MAT_FINAL_ASSEMBLY),"MatAssemblyEnd shifted P");
    }
 
    set_vec_values(b,rhs,"VecGetArray rhs");
    chk(VecSet(x,0.0),"VecSet x");
 
    chk(KSPCreate(PETSC_COMM_SELF,&ksp),"KSPCreate");
    chk(KSPSetOperators(ksp,K,P),"KSPSetOperators");
    const double linear_atol = (s.atol >= 0.0) ? s.atol : PETSC_DEFAULT;
    chk(KSPSetTolerances(ksp,s.tol,linear_atol,PETSC_DEFAULT,s.max_iter),"KSPSetTolerances");
    // Match the Python PETSc path: right preconditioning and unpreconditioned
    // residual norm for the outer KSP diagnostics/convergence test.
    chk(KSPSetPCSide(ksp,PC_RIGHT),"KSPSetPCSide RIGHT");
    chk(KSPSetNormType(ksp,KSP_NORM_UNPRECONDITIONED),"KSPSetNormType unpreconditioned");
 
    const std::string ls=lower_copy(s.linear_solver);
    set_ksp_type_from_string(ksp,ls);
 
    chk(KSPGetPC(ksp,&pc),"KSPGetPC");
    const std::string prec=lower_copy(s.preconditioner);
 
    if(wants_fieldsplit){
        if(s.n_velocity_dofs_full<=0 || s.n_pressure_dofs_full<=0 || static_cast<int>(s.reduced_to_full_dofs.size())!=n){
            throw std::runtime_error("PETSc FieldSplit requested, but reduced/full DOF field map was not provided.");
        }
 
        const std::string layout=lower_copy(s.block_layout);
        if(!(layout=="auto" || layout=="vp" || layout=="v_p" || layout=="vup" || layout=="vu_p" || layout=="mechanical_pressure" || layout=="mechanical-pressure")){
            throw std::runtime_error("Unknown PETSc block_layout: "+s.block_layout);
        }
 
        chk(PCSetType(pc,PCFIELDSPLIT),"PCSetType fieldsplit");
        chk(PCFieldSplitSetIS(pc,"mechanical",cache.is_mech),"PCFieldSplitSetIS mechanical");
        chk(PCFieldSplitSetIS(pc,"pressure",cache.is_p),"PCFieldSplitSetIS pressure");
 
        const std::string fst=lower_copy(s.fieldsplit_type);
        if(fst=="schur") chk(PCFieldSplitSetType(pc,PC_COMPOSITE_SCHUR),"PCFieldSplitSetType");
        else if(fst=="additive") chk(PCFieldSplitSetType(pc,PC_COMPOSITE_ADDITIVE),"PCFieldSplitSetType");
        else if(fst=="multiplicative") chk(PCFieldSplitSetType(pc,PC_COMPOSITE_MULTIPLICATIVE),"PCFieldSplitSetType");
        else throw std::runtime_error("Unknown fieldsplit_type: "+s.fieldsplit_type);
 
        const std::string fact=lower_copy(s.schur_factorization);
        if(fst=="schur"){
            if(fact=="diag" || fact=="diagonal") chk(PCFieldSplitSetSchurFactType(pc,PC_FIELDSPLIT_SCHUR_FACT_DIAG),"PCFieldSplitSetSchurFactType");
            else if(fact=="lower") chk(PCFieldSplitSetSchurFactType(pc,PC_FIELDSPLIT_SCHUR_FACT_LOWER),"PCFieldSplitSetSchurFactType");
            else if(fact=="upper") chk(PCFieldSplitSetSchurFactType(pc,PC_FIELDSPLIT_SCHUR_FACT_UPPER),"PCFieldSplitSetSchurFactType");
            else if(fact=="full") chk(PCFieldSplitSetSchurFactType(pc,PC_FIELDSPLIT_SCHUR_FACT_FULL),"PCFieldSplitSetSchurFactType");
            else throw std::runtime_error("Unknown schur_factorization: "+s.schur_factorization);
 
            const std::string sch=lower_copy(s.schur_precondition);
            const bool dedicated_pressure_schur =
                sch=="pressure_mass" || sch=="pressure_laplacian" ||
                sch=="mass_laplacian" || sch=="pressure_mass_laplacian" ||
                sch=="mass+laplacian";
 
            if(dedicated_pressure_schur) {
                if(!P_user) {
                    throw std::runtime_error(
                        "Dedicated pressure Schur preconditioner requested, but no pressure Pmat was supplied."
                    );
                }
 
                // Important: do not rely on A11/SELFP to interpret the full
                // preconditioner matrix.  For pressure_mass / pressure_laplacian /
                // mass_laplacian, extract the pressure-pressure block from P and
                // pass it explicitly as USER Schur preconditioner.  Then PETSc uses
                // this matrix directly for the pressure Schur solve.
                chk(MatCreateSubMatrix(P,cache.is_p,cache.is_p,MAT_INITIAL_MATRIX,&user_schur_pre),
                    "MatCreateSubMatrix pressure Schur USER");
                owns_user_schur_pre = true;
                chk(PCFieldSplitSetSchurPre(pc,PC_FIELDSPLIT_SCHUR_PRE_USER,user_schur_pre),
                    "PCFieldSplitSetSchurPre USER");
            }
            else if(sch=="a11" || sch=="pspg" || sch=="pspg_pressure_block") {
                chk(PCFieldSplitSetSchurPre(pc,PC_FIELDSPLIT_SCHUR_PRE_A11,nullptr),
                    "PCFieldSplitSetSchurPre A11");
            }
            else if(sch=="self" || sch=="selfp") {
                chk(PCFieldSplitSetSchurPre(pc,PC_FIELDSPLIT_SCHUR_PRE_SELFP,nullptr),
                    "PCFieldSplitSetSchurPre SELFP");
            }
            else if(sch=="pcd") {
                // Python-style PCD is not supplied as a USER Schur matrix.
                // Let PETSc build the Schur operator/preconditioning matrix in
                // the standard SELFP way, then replace the pressure split PC by
                // a PCSHELL below.  Without this branch the option was rejected
                // before the PCD pressure PC could be installed.
                chk(PCFieldSplitSetSchurPre(pc,PC_FIELDSPLIT_SCHUR_PRE_SELFP,nullptr),
                    "PCFieldSplitSetSchurPre SELFP for PCD");
            }
            else if(sch=="full") {
                chk(PCFieldSplitSetSchurPre(pc,PC_FIELDSPLIT_SCHUR_PRE_FULL,nullptr),
                    "PCFieldSplitSetSchurPre FULL");
            }
            else throw std::runtime_error("Unknown schur_precondition: "+s.schur_precondition);
        }
    } else if(ls=="direct" || ls=="lu" || prec=="lu"){
        chk(PCSetType(pc,PCLU),"PCSetType");
    } else {
        set_pc_type_from_string(pc,prec,s.ilu_levels);
    }
 
    chk(KSPSetFromOptions(ksp),"KSPSetFromOptions");
 
    // For FieldSplit, configure sub-solvers after PETSc has created them.  This
    // keeps Solver.txt usable while still allowing command-line PETSc overrides.
    if(wants_fieldsplit){
        chk(KSPSetUp(ksp),"KSPSetUp fieldsplit");
        PetscInt nsplit=0;
        KSP* subksp=nullptr;
        chk(PCFieldSplitGetSubKSP(pc,&nsplit,&subksp),"PCFieldSplitGetSubKSP");
        if(nsplit>=1){
            set_ksp_type_from_string(subksp[0],s.sub_linear_solver);
            PC subpc=nullptr; chk(KSPGetPC(subksp[0],&subpc),"KSPGetPC sub mechanical");
            const bool use_mech = s.n_displacement_dofs_full>0;
            set_pc_type_from_string(subpc,use_mech ? s.mechanical_pc : s.velocity_pc,
                                    use_mech ? s.mechanical_ilu_levels : s.velocity_ilu_levels);
            chk(KSPSetTolerances(subksp[0],s.tol,linear_atol,PETSC_DEFAULT,s.max_iter),"KSPSetTolerances sub mechanical");
        }
        if(nsplit>=2){
            set_ksp_type_from_string(subksp[1],s.sub_linear_solver);
            PC subpc=nullptr; chk(KSPGetPC(subksp[1],&subpc),"KSPGetPC sub pressure");
            const std::string sch=lower_copy(s.schur_precondition);
            if(sch=="pcd"){
                if(!pcd_data || !pcd_data->enabled) throw std::runtime_error("schur_precondition pcd requested, but no PCD data was supplied.");
                PCDShellContext* ctx=new PCDShellContext();
                ctx->Mdiag.resize(pcd_data->Mp_lumped_inv.size(),1.0);
                for(std::size_t ii=0; ii<pcd_data->Mp_lumped_inv.size(); ++ii){
                    const double inv = pcd_data->Mp_lumped_inv[ii];
                    ctx->Mdiag[ii] = (std::abs(inv)>1.0e-300) ? 1.0/inv : 1.0;
                }
                csr_to_petsc_mat(pcd_data->Ap,&ctx->Ap,"PCD Ap");
                csr_to_petsc_mat(pcd_data->Kp,&ctx->Kp,"PCD Kp");
                chk(MatCreateVecs(ctx->Ap,&ctx->tmp1,nullptr),"MatCreateVecs PCD tmp1");
                chk(VecDuplicate(ctx->tmp1,&ctx->tmp2),"VecDuplicate PCD tmp2");
 
                // Kp_eff solve in the Python PCD chain.
                chk(KSPCreate(PETSC_COMM_SELF,&ctx->kspKp),"KSPCreate PCD Kp");
                chk(KSPSetOperators(ctx->kspKp,ctx->Kp,ctx->Kp),"KSPSetOperators PCD Kp");
                set_ksp_type_from_string(ctx->kspKp,s.pcd_kp_solver);
                PC kp_pc=nullptr; chk(KSPGetPC(ctx->kspKp,&kp_pc),"KSPGetPC PCD Kp");
                set_pc_type_from_string(kp_pc,s.pcd_kp_pc,s.pcd_kp_ilu_levels);
                chk(KSPSetTolerances(ctx->kspKp,s.pcd_pressure_rtol,s.pcd_pressure_rtol,PETSC_DEFAULT,s.pcd_pressure_max_iter),"KSPSetTolerances PCD Kp");
                chk(KSPSetFromOptions(ctx->kspKp),"KSPSetFromOptions PCD Kp");
                chk(KSPSetUp(ctx->kspKp),"KSPSetUp PCD Kp");
 
                // Ap solve in the Python PCD chain.  The same inner controls are
                // used for now; for LU diagnostics this makes both inner solves exact.
                chk(KSPCreate(PETSC_COMM_SELF,&ctx->kspAp),"KSPCreate PCD Ap");
                chk(KSPSetOperators(ctx->kspAp,ctx->Ap,ctx->Ap),"KSPSetOperators PCD Ap");
                set_ksp_type_from_string(ctx->kspAp,s.pcd_ap_solver);
                PC ap_pc=nullptr; chk(KSPGetPC(ctx->kspAp,&ap_pc),"KSPGetPC PCD Ap");
                set_pc_type_from_string(ap_pc,s.pcd_ap_pc,s.pcd_ap_ilu_levels);
                chk(KSPSetTolerances(ctx->kspAp,s.pcd_pressure_rtol,s.pcd_pressure_rtol,PETSC_DEFAULT,s.pcd_pressure_max_iter),"KSPSetTolerances PCD Ap");
                chk(KSPSetFromOptions(ctx->kspAp),"KSPSetFromOptions PCD Ap");
                chk(KSPSetUp(ctx->kspAp),"KSPSetUp PCD Ap");
 
                // Pressure Schur sub-KSP.  preonly is cheap but too weak for
                // AMG-only inner inverses; fgmres is the no-LU robust mode.
                set_ksp_type_from_string(subksp[1],s.pcd_pressure_solver);
                {
                    const std::string pksp=lower_copy(s.pcd_pressure_solver);
                    if((pksp=="gmres" || pksp=="fgmres") && s.gmres_restart>0)
                        chk(KSPGMRESSetRestart(subksp[1],s.gmres_restart),"KSPGMRESSetRestart pressure PCD");
                }
 
                chk(PCSetType(subpc,PCSHELL),"PCSetType PCD PCSHELL");
                chk(PCShellSetContext(subpc,ctx),"PCShellSetContext PCD");
                chk(PCShellSetApply(subpc,apply_pcd_shell),"PCShellSetApply PCD");
                chk(PCShellSetDestroy(subpc,destroy_pcd_shell),"PCShellSetDestroy PCD");
                chk(KSPSetTolerances(subksp[1],std::max(s.pcd_pressure_rtol,1e-12),std::max(s.pcd_pressure_rtol,1e-12),PETSC_DEFAULT,std::max(1,s.pcd_pressure_max_iter)),"KSPSetTolerances sub pressure PCD");
                // The outer KSP was already set up once so that PETSc could create
                // the FieldSplit sub-KSP objects.  After replacing the pressure
                // sub-PC by a PCSHELL we must explicitly mark/setup the sub-KSP
                // again; otherwise some PETSc builds continue to use the old PC
                // from the first setup.
                chk(KSPSetUp(subksp[1]),"KSPSetUp pressure PCD subKSP");
                // if(s.verbose){
                //     PetscInt mp=0,np=0;
                //     chk(MatGetSize(ctx->Kp,&mp,&np),"MatGetSize PCD Kp");
                //     std::cout << "    PETSc PCD pressure PC installed: n=" << static_cast<int>(mp)
                //               << ", pressure_ksp=" << s.pcd_pressure_solver
                //               << ", Kp=" << s.pcd_kp_solver << "/" << s.pcd_kp_pc
                //               << ", Ap=" << s.pcd_ap_solver << "/" << s.pcd_ap_pc
                //               << ", rtol=" << s.pcd_pressure_rtol
                //               << ", max_it=" << s.pcd_pressure_max_iter << "\n";
                // }
            } else {
                set_pc_type_from_string(subpc,s.pressure_pc,s.pressure_ilu_levels);
                chk(KSPSetTolerances(subksp[1],s.tol,linear_atol,PETSC_DEFAULT,s.max_iter),"KSPSetTolerances sub pressure");
                chk(KSPSetUp(subksp[1]),"KSPSetUp pressure subKSP");
            }
        }
        // Re-run setup after the sub-KSP/PC choices have been changed.  This is
        // especially important for PCSHELL-based PCD because the first KSPSetUp
        // above only existed to materialize the FieldSplit sub-KSPs.
        chk(KSPSetUp(ksp),"KSPSetUp after fieldsplit subsolver configuration");
        chk(PetscFree(subksp),"PetscFree subksp");
    }
 
    chk(KSPSolve(ksp,b,x),"KSPSolve");
    KSPConvergedReason reason; chk(KSPGetConvergedReason(ksp,&reason),"KSPGetConvergedReason");
    PetscInt its=0; chk(KSPGetIterationNumber(ksp,&its),"KSPGetIterationNumber");
    if(s.verbose){
        PetscReal rnorm=0.0; chk(KSPGetResidualNorm(ksp,&rnorm),"KSPGetResidualNorm");
        std::cout << "    PETSc KSP: its=" << its << ", reason=" << static_cast<int>(reason)
                  << ", rnorm=" << static_cast<double>(rnorm) << "\n";
    }
    if(reason<0) throw std::runtime_error("PETSc KSP did not converge. Iterations="+std::to_string(static_cast<int>(its))+", reason="+std::to_string(static_cast<int>(reason)));
 
    Vector sol(n,0.0); const PetscScalar* arr=nullptr;
    chk(VecGetArrayRead(x,&arr),"VecGetArrayRead");
    for(int i=0;i<n;++i) sol[i]=static_cast<double>(arr[i]);
    chk(VecRestoreArrayRead(x,&arr),"VecRestoreArrayRead");
 
    KSPDestroy(&ksp);
    if(owns_user_schur_pre) MatDestroy(&user_schur_pre);
    return sol;
}
