BoundaryData build_multidomain_boundary_conditions(
    const Problem& problem,
    const ProblemParameters& params,
    double time,
    const std::function<std::vector<std::string>(const FieldMesh&)>& active_fields_for_mesh
){
    if(problem.shared_dofs){
        if(problem.domains.empty()) return BoundaryData{};
        return build_shared_multidomain_boundary_conditions(
            problem,params,time,active_fields_for_mesh(problem.domains.front().mesh)
        );
    }
 
    BoundaryData global;
    const bool multiple=problem.domains.size()>1;
    for(const Domain& domain:problem.domains){
        if(domain.type!=DomainType::Fluid && domain.type!=DomainType::Solid) continue;
        ProblemParameters local_params=filter_params_for_domain(params,domain,multiple);
        BoundaryData local=build_boundary_conditions(domain.mesh,local_params,time,active_fields_for_mesh(domain.mesh));
        for(int i=0;i<static_cast<int>(local.dofs.size());++i){
            global.dofs.push_back(domain.dof_offset+local.dofs[i]);
            global.values.push_back(local.values[i]);
            const std::string src=(i<static_cast<int>(local.sources.size())) ? local.sources[i] : std::string("BC");
            global.sources.push_back(domain.name+":"+src);
        }
        global.duplicate_count += local.duplicate_count;
        global.compatible_duplicate_count += local.compatible_duplicate_count;
        global.conflict_count += local.conflict_count;
    }
    return global;
}
