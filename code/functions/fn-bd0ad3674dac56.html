ElementGeometry make_displacement_element_geometry(const FieldMesh& mesh,int elem_id){
    if(!has_displacement_field(mesh)) return make_velocity_element_geometry(mesh,elem_id);
    ElementGeometry elem;
    elem.dim = mesh.dim;
    elem.type = mesh.displacement_type;
    const auto& conn = mesh.displacement_elements[elem_id];
    elem.coords.resize(conn.size());
    for(int a=0;a<static_cast<int>(conn.size());++a) elem.coords[a] = mesh.displacement_nodes[conn[a]].x;
    return elem;
}
