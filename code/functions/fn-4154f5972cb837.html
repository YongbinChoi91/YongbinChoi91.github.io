static void add_polygon_lobatto_segment_neumann_2d(
    const DofLayout& gmesh,
    const FieldInfo& field,
    int fid,
    const FieldNeumannBC& bc,
    double time,
    int global_offset,
    const std::vector<int>& nodes,
    Vector& R
){
    const int m = static_cast<int>(nodes.size());
    if(m != 2 && m != 3){
        throw std::runtime_error("Polygon Lobatto Neumann segment expects two or three nodes for field '"+bc.field+"'.");
    }
    for(int node: nodes){
        if(node<0 || node>=static_cast<int>(field.nodes.size())){
            throw std::runtime_error("Polygon Lobatto Neumann boundary node id out of range for field '"+bc.field+"'.");
        }
    }
 
    // Gauss-Lobatto points on [-1,1].
    // m=2: exact for a constant traction on a linear boundary segment.
    // m=3: Simpson/Lobatto rule for a quadratic boundary segment with midside node.
    const double s2[2] = {-1.0, 1.0};
    const double w2[2] = { 1.0, 1.0};
    const double s3[3] = {-1.0, 0.0, 1.0};
    const double w3[3] = { 1.0/3.0, 4.0/3.0, 1.0/3.0};
 
    const int nq = m;
    for(int q=0; q<nq; ++q){
        const double s = (m==2) ? s2[q] : s3[q];
        const double w = (m==2) ? w2[q] : w3[q];
 
        double N[3]    = {0.0,0.0,0.0};
        double dNds[3] = {0.0,0.0,0.0};
        if(m==2){
            N[0] = 0.5*(1.0-s);
            N[1] = 0.5*(1.0+s);
            dNds[0] = -0.5;
            dNds[1] =  0.5;
        } else {
            N[0] = 0.5*s*(s-1.0);
            N[1] = 1.0 - s*s;
            N[2] = 0.5*s*(s+1.0);
            dNds[0] = s - 0.5;
            dNds[1] = -2.0*s;
            dNds[2] = s + 0.5;
        }
 
        std::array<double,3> Xq{{0.0,0.0,0.0}};
        std::array<double,3> dXds{{0.0,0.0,0.0}};
        for(int a=0; a<m; ++a){
            const auto& Xa = field.nodes[nodes[a]].x;
            for(int d=0; d<3; ++d){
                Xq[d]   += N[a]    * Xa[d];
                dXds[d] += dNds[a] * Xa[d];
            }
        }
 
        const double jac = std::sqrt(dXds[0]*dXds[0] + dXds[1]*dXds[1]);
        if(jac<=0.0){
            throw std::runtime_error("Degenerate polygon Lobatto Neumann boundary segment for field '"+bc.field+"'.");
        }
 
        const double traction = eval_scalar_function(bc.value, Xq[0], Xq[1], Xq[2], time);
        const double dGamma = jac * w;
 
        // Residual convention: R = internal - external.
        for(int a=0; a<m; ++a){
            const int I = global_offset + global_field_dof(gmesh, fid, nodes[a], bc.component);
            if(I<0 || I>=static_cast<int>(R.size())){
                throw std::runtime_error("Global polygon Lobatto Neumann DOF out of range.");
            }
            R[I] -= N[a] * traction * dGamma;
        }
    }
}
