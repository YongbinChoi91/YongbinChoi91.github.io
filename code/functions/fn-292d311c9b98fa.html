inline std::array<std::array<Scalar,NRHS>,N> multiply_double_matrix(
    const std::array<std::array<double,N>,N>& A,
    const std::array<std::array<Scalar,NRHS>,N>& B
) {
    std::array<std::array<Scalar,NRHS>,N> C{};
    for (auto& row : C) for (auto& v : row) v = Scalar(0.0);
    for (int i = 0; i < N; ++i) {
        for (int k = 0; k < N; ++k) {
            const double aik = A[i][k];
            if (aik == 0.0) continue;
            for (int r = 0; r < NRHS; ++r) C[i][r] += Scalar(aik) * B[k][r];
        }
    }
    return C;
}
