static const BoundaryOwnerNodeAdjacency& boundary_owner_node_adjacency(
    const FieldMesh& mesh
){
    static std::unordered_map<const FieldMesh*, BoundaryOwnerNodeAdjacency> cache;
    const auto& candidates = force_owner_table(mesh);
    auto it = cache.find(&mesh);
    if(it!=cache.end() && it->second.candidates==&candidates) return it->second;
 
    BoundaryOwnerNodeAdjacency adj;
    adj.mesh = &mesh;
    adj.candidates = &candidates;
    adj.node_to_elements.assign(mesh.velocity_nodes.size(),{});
    for(int e=0; e<static_cast<int>(candidates.size()); ++e){
        const auto& conn = candidates[e];
        std::vector<int> unique_nodes;
        unique_nodes.reserve(conn.size());
        for(int node:conn){
            if(node<0 || node>=static_cast<int>(mesh.velocity_nodes.size())) continue;
            if(std::find(unique_nodes.begin(),unique_nodes.end(),node)==unique_nodes.end()){
                unique_nodes.push_back(node);
                adj.node_to_elements[node].push_back(e);
            }
        }
    }
    cache[&mesh] = std::move(adj);
    return cache[&mesh];
}
