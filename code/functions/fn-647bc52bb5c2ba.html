inline void update_reduced_matrix_values(
    const CSRMatrix& K,
    ReducedSystemWorkspace& w
){
    if(w.Kff.values.size() != w.kff_value_from_full.size()){
        throw std::runtime_error("Bad reduced workspace value map.");
    }
    for(int q=0; q<static_cast<int>(w.kff_value_from_full.size()); ++q){
        const int p = w.kff_value_from_full[q];
        if(p<0 || p>=static_cast<int>(K.values.size())) throw std::runtime_error("Reduced matrix value map out of range.");
        w.Kff.values[q] = K.values[p];
    }
}
