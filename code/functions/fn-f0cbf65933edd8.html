inline LinearMultistepCoefficients Adams_Bashforth(int num_steps) {
    switch (num_steps) {
    case 1:
        return {"AdamsBashforth", 1, 1, false, {1.0}};
    case 2:
        return {"AdamsBashforth", 2, 2, false, {3.0 / 2.0, -1.0 / 2.0}};
    case 3:
        return {"AdamsBashforth", 3, 3, false, {23.0 / 12.0, -16.0 / 12.0, 5.0 / 12.0}};
    case 4:
        return {"AdamsBashforth", 4, 4, false, {55.0 / 24.0, -59.0 / 24.0, 37.0 / 24.0, -9.0 / 24.0}};
    case 5:
        return {"AdamsBashforth", 5, 5, false, {1901.0 / 720.0, -2774.0 / 720.0, 2616.0 / 720.0, -1274.0 / 720.0, 251.0 / 720.0}};
    case 6:
        return {"AdamsBashforth", 6, 6, false, {4277.0 / 1440.0, -7923.0 / 1440.0, 9982.0 / 1440.0, -7298.0 / 1440.0, 2877.0 / 1440.0, -475.0 / 1440.0}};
    default:
        throw std::runtime_error("Unsupported Adams-Bashforth step count. Supported steps: 1 to 6.");
    }
}
