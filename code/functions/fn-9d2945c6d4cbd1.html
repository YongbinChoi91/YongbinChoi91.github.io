static std::vector<char> make_petsc_pressure_mask(const LinearSolverSettings& s,int n){
    std::vector<char> is_pressure(n,0);
    const int off_p=s.n_velocity_dofs_full+s.n_displacement_dofs_full;
    const bool have_explicit_pressure = !s.pressure_dofs_full.empty();
    int max_full = -1;
    if(have_explicit_pressure){
        for(int full:s.pressure_dofs_full) max_full=std::max(max_full,full);
    }
    std::vector<char> pressure_full_mask;
    if(have_explicit_pressure && max_full>=0){
        pressure_full_mask.assign(max_full+1,0);
        for(int full:s.pressure_dofs_full){
            if(full>=0) pressure_full_mask[full]=1;
        }
    }
 
    for(int i=0;i<n;++i){
        const int full = (static_cast<int>(s.reduced_to_full_dofs.size())==n) ? s.reduced_to_full_dofs[i] : i;
        if(have_explicit_pressure){
            is_pressure[i] = (full>=0 && full<static_cast<int>(pressure_full_mask.size()) && pressure_full_mask[full]) ? 1 : 0;
        } else {
            is_pressure[i] = (full>=off_p) ? 1 : 0;
        }
    }
    return is_pressure;
}
