static bool strip_hysteresis_domain_group_prefix(const std::string& raw,const std::string& domain_name,std::string& group){
    const std::string prefix=domain_name+":";
    if(raw.rfind(prefix,0)==0){
        group=raw.substr(prefix.size());
        return true;
    }
    group=raw;
    return false;
}
