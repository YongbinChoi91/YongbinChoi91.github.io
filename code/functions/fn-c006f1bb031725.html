static void add_unique_nodes_from_generic_boundary(
    std::set<int>& nodes,
    const FieldMesh& mesh,
    const std::string& field_name,
    const std::string& group
){
    const DofLayout& gmesh=get_cached_dof_layout(mesh);
    const int fid=find_field_index(gmesh,field_name,false);
    if(fid<0) return;
    const FieldInfo& field=gmesh.fields[fid];
    auto it=field.boundaries.find(group);
    if(it==field.boundaries.end()) return;
    for(const auto& ent:it->second){
        for(int node:ent.node_ids) nodes.insert(node);
    }
}
