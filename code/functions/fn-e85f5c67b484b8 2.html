inline Coeff6x2T<Scalar> quadratic_projection_coefficients(
    const QuadraticElementCache& cache,
    const std::vector<Scalar>& elem
) {
    const QuadraticGeometry& g = cache.geom;
    const auto& U = cache.layout.field(cache.field_name);
    const QuadraticMomentData& m = cache.moments;
 
    std::array<std::array<Scalar,2>,5> Ipi1{};
    for (auto& row : Ipi1) for (auto& v : row) v = Scalar(0.0);
 
    for (const QuadraticProjectionEdgeContribution& ec : cache.edge_projection) {
        for (int r = 0; r < 5; ++r) {
            const double coeff = ec.coeff[r];
            if (coeff == 0.0) continue;
            for (int c = 0; c < 2; ++c) {
                Ipi1[r][c] += Scalar(coeff) * elem[hpx::local_dof(U, ec.slot, c)];
            }
        }
    }
 
    std::array<std::array<Scalar,2>,6> rhs_ac{};
    for (auto& row : rhs_ac) for (auto& v : row) v = Scalar(0.0);
 
    for (int s = 0; s < g.n_slots; ++s) {
        const auto& p = cache.slot_basis[s];
        for (int a = 0; a < 6; ++a) {
            for (int c = 0; c < 2; ++c) {
                rhs_ac[a][c] += Scalar(p[a]) * elem[hpx::local_dof(U, s, c)];
            }
        }
    }
 
    const auto acT = multiply_double_matrix<Scalar,6,2>(cache.MpI_inv, rhs_ac);
    std::array<Scalar,2> m0{{Scalar(0.0), Scalar(0.0)}};
    for (int c = 0; c < 2; ++c) {
        for (int a = 0; a < 6; ++a) m0[c] += Scalar(m.MpT[a]) * acT[a][c];
    }
 
    std::array<std::array<Scalar,2>,5> Bpi{};
    const double divN[5] = {0.0, 0.0, 2.0, 0.0, 2.0};
    for (int r = 0; r < 5; ++r) {
        for (int c = 0; c < 2; ++c) Bpi[r][c] = Ipi1[r][c] - Scalar(divN[r]) * m0[c];
    }
 
    const auto grad_a = multiply_double_matrix<Scalar,5,2>(cache.Gpi_inv, Bpi);
 
    Coeff6x2T<Scalar> a_pi{};
    for (int c = 0; c < 2; ++c) {
        Scalar moment_grad(0.0);
        for (int r = 0; r < 5; ++r) moment_grad += Scalar(m.MpT[r+1]) * grad_a[r][c];
        a_pi[0][c] = (m0[c] - moment_grad) / Scalar(m.MpT[0]);
        for (int r = 0; r < 5; ++r) a_pi[r+1][c] = grad_a[r][c];
    }
    return a_pi;
}
