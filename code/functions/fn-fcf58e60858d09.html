inline TimeDofStencil BDF(int order) {
    switch (order) {
    case 1:
        return {"BDF", 1, {1.0, -1.0}};
    case 2:
        return {"BDF", 2, {3.0 / 2.0, -2.0, 1.0 / 2.0}};
    case 3:
        return {"BDF", 3, {11.0 / 6.0, -3.0, 3.0 / 2.0, -1.0 / 3.0}};
    case 4:
        return {"BDF", 4, {25.0 / 12.0, -4.0, 3.0, -4.0 / 3.0, 1.0 / 4.0}};
    case 5:
        return {"BDF", 5, {137.0 / 60.0, -5.0, 5.0, -10.0 / 3.0, 5.0 / 4.0, -1.0 / 5.0}};
    case 6:
        return {"BDF", 6, {49.0 / 20.0, -6.0, 15.0 / 2.0, -20.0 / 3.0, 15.0 / 4.0, -6.0 / 5.0, 1.0 / 6.0}};
    default:
        throw std::runtime_error("Unsupported BDF order. Supported orders: 1 to 6.");
    }
}
