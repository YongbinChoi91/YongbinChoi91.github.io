std::array<double,2> edge_reference_normal_2d(ElementType type,int edge){
    if(type==ElementType::Q4 || type==ElementType::Q9){
        return fixed_axis_vector_for_edge_2d(edge);
    }
    if(type==ElementType::T3 || type==ElementType::T6){
        if(edge==0) return { 0.0,-1.0};
        if(edge==1) return { 1.0, 1.0};
        if(edge==2) return {-1.0, 0.0};
    }
    throw std::runtime_error("Unsupported 2D edge reference normal.");
}
