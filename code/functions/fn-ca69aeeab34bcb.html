static std::vector<double> collect_right_cauchy_green_from_F(const std::vector<double>& F_data){
    if(F_data.size() % 9 != 0) throw std::runtime_error("Bad F data size while exporting C.");
    std::vector<double> C_data;
    C_data.reserve(F_data.size());
    for(size_t q=0; q<F_data.size(); q+=9){
        double F[3][3] = {
            {F_data[q+0], F_data[q+1], F_data[q+2]},
            {F_data[q+3], F_data[q+4], F_data[q+5]},
            {F_data[q+6], F_data[q+7], F_data[q+8]}
        };
        for(int i=0;i<3;++i){
            for(int j=0;j<3;++j){
                double cij=0.0;
                for(int k=0;k<3;++k) cij += F[k][i] * F[k][j];
                C_data.push_back(cij);
            }
        }
    }
    return C_data;
}
