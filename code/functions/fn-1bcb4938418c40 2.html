static double norm_l2_range(const Vector& a,int i0,int i1){
    i0=std::max(0,i0);
    i1=std::min(static_cast<int>(a.size()),i1);
    double s=0.0;
    for(int i=i0;i<i1;++i) s+=a[i]*a[i];
    return std::sqrt(s);
}
