bool self_check_tensor_quadratic() {
    const std::array<std::array<double, 2>, 2> F{{{{1.0, 2.0}}, {{3.0, 4.0}}}};
    const auto P = grad([](const auto& F_ad) {
        return 0.5 * inner(F_ad, F_ad);
    }, F);
 
    bool ok = true;
    for (int i = 0; i < 2; ++i) {
        for (int j = 0; j < 2; ++j) ok = ok && close(P[i][j], F[i][j], 1e-14);
    }
    return ok;
}
