inline RveBatchTask make_macro_rve_task(
    int elem_id,
    int gp_index,
    const std::array<std::array<double,2>,2>& Fmacro,
    double dt,
    double dt_previous,
    int step,
    double time = 0.0
) {
    RveBatchTask task;
    task.Fmacro = Fmacro;
    task.dt = dt;
    task.dt_previous = dt_previous;
    task.step = step;
    task.time = time;
 
    const RveWarmStartState warm = get_multiscale_rve_warm_start(elem_id, gp_index);
    if (warm.valid && multiscale_rve_options().use_warm_start) {
        task.initial_displacement = warm.displacement;
        task.initial_lambda = warm.lambda;
        task.has_initial_displacement = true;
        task.has_initial_lambda = true;
        task.initial_Fmacro = warm.Fmacro;
        task.has_initial_Fmacro = true;
        if (warm.has_history) {
            task.initial_history = warm.history;
            task.has_initial_history = true;
        }
    }
    return task;
}
