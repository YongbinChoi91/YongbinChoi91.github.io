static Vector csr_matvec_solver(const CSRMatrix& A,const Vector& x){
    if(A.n_cols!=static_cast<int>(x.size())) throw std::runtime_error("CSR matvec dimension mismatch.");
    Vector y(A.n_rows,0.0);
    for(int i=0;i<A.n_rows;++i){
        double sum=0.0;
        for(int p=A.row_ptr[i]; p<A.row_ptr[i+1]; ++p){
            const int j=A.col_idx[p];
            if(j<0 || j>=static_cast<int>(x.size())) throw std::runtime_error("CSR matvec column out of range.");
            sum += A.values[p]*x[j];
        }
        y[i]=sum;
    }
    return y;
}
