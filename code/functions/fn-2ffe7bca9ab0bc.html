std::vector<int> make_all_field_element_dofs(const DofLayout& mesh,int elem_id){
    std::vector<int> dofs;
    for(int fid=0; fid<static_cast<int>(mesh.fields.size()); ++fid){
        const FieldInfo& f=mesh.fields[fid];
        if(elem_id<0 || elem_id>=static_cast<int>(f.elements.size())) continue;
        const auto& conn=f.elements[elem_id];
        for(int node:conn){
            for(int c=0;c<f.components;++c) dofs.push_back(global_field_dof(mesh,fid,node,c));
        }
    }
    return dofs;
}
