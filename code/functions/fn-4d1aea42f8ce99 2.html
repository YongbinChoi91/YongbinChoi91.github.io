void reset_hysteresis_csv(const std::string& filename,const HysteresisMonitor& h){
    ensure_output_parent_directory(filename);
    std::ofstream out(filename);
    if(!out) throw std::runtime_error("Could not open hysteresis CSV: "+filename);
    out << "# x_axis=" << h.x_axis.label << ", x_scale=" << h.x_axis.spec.scale << ", x_offset=" << h.x_axis.spec.offset << "\n";
    out << "# y_axis=" << h.y_axis.label << ", y_scale=" << h.y_axis.spec.scale << ", y_offset=" << h.y_axis.spec.offset << ", ndofs=" << h.y_axis.dofs.size() << "\n";
    out << "time,x_value,y_value\n";
}
