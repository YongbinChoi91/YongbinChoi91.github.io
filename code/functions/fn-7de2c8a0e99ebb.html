static const std::vector<int>& owner_connectivity(const FieldMesh& mesh,const ResolvedBoundaryEdge& r){
    const auto& elems = r.use_total_elements ? mesh.total_velocity_elements : mesh.velocity_elements;
    if(r.elem_id<0 || r.elem_id>=static_cast<int>(elems.size()) || elems[r.elem_id].empty())
        throw std::runtime_error("Bad resolved surface owner element id.");
    return elems[r.elem_id];
}
