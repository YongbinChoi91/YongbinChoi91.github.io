void run_matfree_jvp_check(
    ConstitutiveElementKernel kernel,
    const Problem& problem,
    const ProblemParameters& params,
    const Vector& current_dofs,
    const PastDofs& past_dofs,
    double dt,
    double dt_previous,
    int step,
    double time,
    const ProblemElementDofMap& dof_maps,
    const MaterialHistoryStore& material_history,
    const CSRMatrix& K_full,
    const Vector& assembled_residual,
    const Vector& matfree_residual,
    ReducedSystemWorkspace& reduced,
    double tolerance
){
    if(!kernel) throw std::runtime_error("Null constitutive element kernel in MatFree Jv check.");
    if(assembled_residual.size()!=matfree_residual.size()) throw std::runtime_error("MatFree residual check size mismatch.");
    Vector residual_difference(assembled_residual.size());
    for(std::size_t i=0; i<assembled_residual.size(); ++i)
        residual_difference[i]=assembled_residual[i]-matfree_residual[i];
    const double residual_error=l2_norm_solver(residual_difference)/std::max(l2_norm_solver(assembled_residual),1.0);
    std::cout << "    MatFree residual check: rel_error=" << residual_error << ", tol=" << tolerance << "\n";
    if(!std::isfinite(residual_error) || residual_error>tolerance)
        throw std::runtime_error("Matrix-free residual differs from assembled residual.");
    update_reduced_matrix_values(K_full,reduced);
    const std::vector<int>& free_dofs = reduced.free_dofs;
    const int ndofs = problem.total_dofs;
    const int nf = static_cast<int>(free_dofs.size());
    if(nf<=0) throw std::runtime_error("MatFree Jv check has no free DOFs.");
 
    Vector v_free(nf,0.0);
    for(int i=0;i<nf;++i){
        v_free[i] = std::sin(0.17320508075688773*static_cast<double>(i+1))
                  + std::cos(0.0816496580927726*static_cast<double>(i+3));
    }
 
    const Vector assembled = csr_matvec_solver(reduced.Kff,v_free);
 
    Vector v_full(ndofs,0.0), y_full(ndofs,0.0), y_free(nf,0.0);
    for(int i=0;i<nf;++i) v_full[free_dofs[i]]=v_free[i];
    apply_global_jacobian_vector_product(kernel,problem,params,current_dofs,past_dofs,v_full,dt,dt_previous,step,time,y_full,dof_maps,&material_history);
    for(int i=0;i<nf;++i) y_free[i]=y_full[free_dofs[i]];
 
    Vector diff(nf,0.0);
    for(int i=0;i<nf;++i) diff[i]=assembled[i]-y_free[i];
    const double denom=std::max(l2_norm_solver(assembled),1.0);
    const double rel=l2_norm_solver(diff)/denom;
    std::cout << "    MatFree Jv check: rel_error=" << rel
              << ", tol=" << tolerance << "\n";
    if(!std::isfinite(rel) || rel>tolerance){
        std::ostringstream os;
        os << "MatFree Jv check failed: ||Kff*v - Jv||/max(||Kff*v||,1)="
           << rel << " > " << tolerance
           << ". Check reduced/free scatter, pressure pin exclusion, domain offsets, or local tangent action.";
        throw std::runtime_error(os.str());
    }
}
