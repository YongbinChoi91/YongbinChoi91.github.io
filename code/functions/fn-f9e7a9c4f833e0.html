static void assemble_polygon_lobatto_neumann_2d(
    const DofLayout& gmesh,
    const FieldInfo& field,
    int fid,
    const FieldNeumannBC& bc,
    const BoundaryEntity& ent,
    double time,
    int global_offset,
    Vector& R
){
    const int m = static_cast<int>(ent.node_ids.size());
    if(m < 2){
        throw std::runtime_error("Polygon Neumann boundary entity needs at least two nodes for field '"+bc.field+"', group '"+bc.group+"'.");
    }
 
    // Direct use of the exported surface entity.  This avoids recovering a
    // polygon reference edge and makes VEM Neumann loads follow the same
    // boundary connectivity used by the mesh/Python exporter.
    if(m == 2 || m == 3){
        add_polygon_lobatto_segment_neumann_2d(gmesh, field, fid, bc, time, global_offset, ent.node_ids, R);
        return;
    }
 
    // For longer polylines, prefer quadratic Lobatto on triples when the node
    // count looks like [v0,m0,v1,m1,...,vk]; otherwise fall back to consecutive
    // two-node Lobatto segments.
    if(m % 2 == 1){
        for(int k=0; k+2<m; k+=2){
            add_polygon_lobatto_segment_neumann_2d(
                gmesh, field, fid, bc, time, global_offset,
                {ent.node_ids[k], ent.node_ids[k+1], ent.node_ids[k+2]}, R
            );
        }
    } else {
        for(int k=0; k+1<m; ++k){
            add_polygon_lobatto_segment_neumann_2d(
                gmesh, field, fid, bc, time, global_offset,
                {ent.node_ids[k], ent.node_ids[k+1]}, R
            );
        }
    }
}
