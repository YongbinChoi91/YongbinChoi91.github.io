static int torsion_axis_index_from_string(const std::string& token){
    const std::string k = lower_copy(token);
    if(k=="x" || k=="ux" || k=="0") return 0;
    if(k=="y" || k=="uy" || k=="1") return 1;
    if(k=="z" || k=="uz" || k=="2") return 2;
    throw std::runtime_error("Bad torsion axis. Use x, y, or z.");
}
