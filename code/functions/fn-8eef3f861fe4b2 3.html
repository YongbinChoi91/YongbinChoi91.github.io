static ProblemParameters filter_params_for_domain(
    const ProblemParameters& base,
    const Domain& domain,
    bool multiple_domains
){
    ProblemParameters p=base;
    p.field_dirichlet_bcs.clear();
 
    const FieldMesh& mesh=domain.mesh;
    const DofLayout& gmesh=get_cached_dof_layout(mesh);
    for(const auto& r:base.field_dirichlet_bcs){
        std::string g;
        const bool qualified=strip_domain_group_prefix(r.group,domain.name,g);
        const int fid=find_field_index(gmesh,r.field,false);
        BCSelector selector=r.selector;
        if(selector.kind==BCSelectorKind::BoundaryGroup && selector.group.empty()) selector.group=r.group;
        const bool coordinate_selector = selector.kind!=BCSelectorKind::BoundaryGroup;
        const bool has_boundary = (fid>=0 && gmesh.fields[fid].boundaries.count(g)>0);
        const bool has_coordinate_match = (fid>=0 && coordinate_selector &&
                                           coordinate_selector_matches_field(gmesh.fields[fid],selector,mesh.dim));
        if((qualified || !multiple_domains || has_boundary) && has_boundary){
            auto rr=r; rr.group=g; rr.selector.group=g; p.field_dirichlet_bcs.push_back(rr);
        } else if(has_coordinate_match){
            p.field_dirichlet_bcs.push_back(r);
        }
    }
    return p;
}
