static std::map<int,double> compute_velocity_boundary_parabolic_profile(
    const FieldMesh& mesh,
    const std::string& group
){
    std::map<int,double> profile;
    auto it=mesh.velocity_boundaries.find(group);
    if(it==mesh.velocity_boundaries.end()) return profile;
 
    std::set<int> nodes;
    for(const auto& ent:it->second){
        for(int n:ent.node_ids){
            if(n>=0 && n<static_cast<int>(mesh.velocity_nodes.size())) nodes.insert(n);
        }
    }
    if(nodes.empty()) return profile;
 
    if(mesh.dim == 2){
        // Python CFD_Module.Parabolic_flow 2D equivalent:
        // u = u_bar * [1.5 * 4*(y-ymin)*(ymax-y)/(ymax-ymin)^2].
        double ymin=std::numeric_limits<double>::infinity();
        double ymax=-std::numeric_limits<double>::infinity();
        for(int n:nodes){
            const double y=mesh.velocity_nodes[n].x[1];
            ymin=std::min(ymin,y);
            ymax=std::max(ymax,y);
        }
        const double H=std::max(ymax-ymin,1.0e-12);
        for(int n:nodes){
            const double yloc=mesh.velocity_nodes[n].x[1]-ymin;
            profile[n]=1.5*(4.0*yloc*(H-yloc))/(H*H);
        }
        return profile;
    }
 
    if(mesh.dim == 3){
        // Python CFD_Module.parabolic_inlet_bc_unified(shape="rect", flow_dir="x") equivalent.
        // For a rectangular inlet cross-section, Python uses
        //   Umax = (9/4)*Umean,
        //   prof = (1 - ((y-yc)/a)^2) * (1 - ((z-zc)/b)^2),
        // and the C++ BC expression multiplies this dimensionless profile by u_bar.
        // Therefore parabolic_profile = (9/4)*prof.
        double ymin=std::numeric_limits<double>::infinity();
        double ymax=-std::numeric_limits<double>::infinity();
        double zmin=std::numeric_limits<double>::infinity();
        double zmax=-std::numeric_limits<double>::infinity();
        for(int n:nodes){
            const auto& X = mesh.velocity_nodes[n].x;
            ymin=std::min(ymin,X[1]);
            ymax=std::max(ymax,X[1]);
            zmin=std::min(zmin,X[2]);
            zmax=std::max(zmax,X[2]);
        }
 
        const double H = std::max(ymax-ymin,1.0e-12);
        const double W = std::max(zmax-zmin,1.0e-12);
        for(int n:nodes){
            const auto& X = mesh.velocity_nodes[n].x;
            const double sy = (X[1]-ymin)/H;
            const double sz = (X[2]-zmin)/W;
            double prof = (4.0*sy*(1.0-sy)) * (4.0*sz*(1.0-sz));
            if(sy<0.0 || sy>1.0 || sz<0.0 || sz>1.0) prof = 0.0;
            profile[n] = 2.25 * std::max(prof,0.0);
        }
        return profile;
    }
 
    throw std::runtime_error("Parabolic velocity profile supports only 2D and 3D meshes.");
}
