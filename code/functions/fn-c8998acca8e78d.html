FieldMesh read_mixed_mesh(const std::string& filename, bool vem_taylor_hood_from_dofs){
    FieldMesh mesh;
    std::ifstream file(filename);
    if(!file) throw std::runtime_error("Could not open mesh file: "+filename);
 
    std::string line, section;
    bool dim_set=false;
    int detected_dim=0;
    bool fields_line_seen=false;
    bool fields_declares_velocity=false;
    bool fields_declares_displacement=false;
    bool fields_declares_pressure=false;
    bool vem_taylor_hood_mesh=vem_taylor_hood_from_dofs;
    while(std::getline(file,line)){
        if(!line.empty() && line.back()=='\r') line.pop_back();
        const auto comment=line.find('#');
        if(comment!=std::string::npos) line=line.substr(0,comment);
        line=trim_copy(line);
        if(line.empty()) continue;
 
        std::stringstream head(line);
        std::string first; head >> first;
 
        if(first=="DIM"){
            head >> mesh.dim;
            if(mesh.dim!=2 && mesh.dim!=3) throw std::runtime_error("DIM must be 2 or 3 in field mesh.");
            detected_dim=mesh.dim;
            dim_set=true;
            continue;
        }
        if(first=="FIELDS"){
            fields_line_seen=true;
            std::string field;
            while(head >> field){
                const std::string f=lower_copy(field);
                if(f=="velocity") fields_declares_velocity=true;
                else if(f=="displacement") fields_declares_displacement=true;
                else if(f=="pressure") fields_declares_pressure=true;
            }
            continue;
        }
        if(first=="MESH_TYPE"){
            std::string mesh_type; head >> mesh_type;
            const std::string mt=lower_copy(mesh_type);
            if(mt=="vem_taylor_hood" || mt=="vem_taylorhood") vem_taylor_hood_mesh=true;
            continue;
        }
        if(first=="VELOCITY_ELEMENT" || first=="DISPLACEMENT_ELEMENT" || first=="PRESSURE_ELEMENT"){
            continue;
        }
        if(is_field_mesh_section(first)){
            section=first;
            const std::string vg=velocity_boundary_name_from_section(section);
            const std::string ug=displacement_boundary_name_from_section(section);
            const std::string pg=pressure_boundary_name_from_section(section);
            if(!vg.empty()) mesh.velocity_boundaries[vg];
            if(!ug.empty()) mesh.displacement_boundaries[ug];
            if(!pg.empty()) mesh.pressure_boundaries[pg];
            continue;
        }
        if(section.empty()) throw std::runtime_error("Data line before any field mesh section: "+line);
 
        std::stringstream ss(line);
        if(section=="VELOCITY_NODES" || section=="NODES_V" || section=="NODES"){
            int id=-1; read_nonnegative_id(ss,id,"velocity node");
            std::vector<double> vals; double q; while(ss>>q) vals.push_back(q);
            if(vals.size()!=2 && vals.size()!=3) throw std::runtime_error("Velocity node line needs id x y [z].");
            if(detected_dim==0) detected_dim=static_cast<int>(vals.size());
            if(static_cast<int>(vals.size())!=detected_dim) throw std::runtime_error("Mixed node dimensions are inconsistent.");
            if(id>=static_cast<int>(mesh.velocity_nodes.size())) mesh.velocity_nodes.resize(id+1);
            mesh.velocity_nodes[id].x={{vals[0],vals[1], vals.size()==3?vals[2]:0.0}};
        } else if(section=="DISPLACEMENT_NODES" || section=="NODES_U"){
            int id=-1; read_nonnegative_id(ss,id,"displacement node");
            std::vector<double> vals; double q; while(ss>>q) vals.push_back(q);
            if(vals.size()!=2 && vals.size()!=3) throw std::runtime_error("Displacement node line needs id x y [z].");
            if(detected_dim==0) detected_dim=static_cast<int>(vals.size());
            if(static_cast<int>(vals.size())!=detected_dim) throw std::runtime_error("Mixed node dimensions are inconsistent.");
            if(id>=static_cast<int>(mesh.displacement_nodes.size())) mesh.displacement_nodes.resize(id+1);
            mesh.displacement_nodes[id].x={{vals[0],vals[1], vals.size()==3?vals[2]:0.0}};
        } else if(section=="PRESSURE_NODES" || section=="NODES_P"){
            int id=-1; read_nonnegative_id(ss,id,"pressure node");
            std::vector<double> vals; double q; while(ss>>q) vals.push_back(q);
            if(vals.size()!=2 && vals.size()!=3) throw std::runtime_error("Pressure node line needs id x y [z].");
            if(detected_dim==0) detected_dim=static_cast<int>(vals.size());
            if(static_cast<int>(vals.size())!=detected_dim) throw std::runtime_error("Mixed node dimensions are inconsistent.");
            if(id>=static_cast<int>(mesh.pressure_nodes.size())) mesh.pressure_nodes.resize(id+1);
            mesh.pressure_nodes[id].x={{vals[0],vals[1], vals.size()==3?vals[2]:0.0}};
        } else if(section=="VELOCITY_ELEMENTS" || section=="ELEMENTS_V" || section=="ELEMENTS"){
            int id=-1; read_nonnegative_id(ss,id,"velocity element");
            std::vector<int> conn; int n; while(ss>>n) conn.push_back(n);
            if(conn.empty()) throw std::runtime_error("Empty velocity element connectivity.");
            if(id>=static_cast<int>(mesh.velocity_elements.size())) mesh.velocity_elements.resize(id+1);
            mesh.velocity_elements[id]=reorder_quad_python_to_cpp(conn);
        } else if(section=="DISPLACEMENT_ELEMENTS" || section=="ELEMENTS_U"){
            int id=-1; read_nonnegative_id(ss,id,"displacement element");
            std::vector<int> conn; int n; while(ss>>n) conn.push_back(n);
            if(conn.empty()) throw std::runtime_error("Empty displacement element connectivity.");
            if(id>=static_cast<int>(mesh.displacement_elements.size())) mesh.displacement_elements.resize(id+1);
            mesh.displacement_elements[id]=reorder_quad_python_to_cpp(conn);
        } else if(section=="PRESSURE_ELEMENTS" || section=="ELEMENTS_P"){
            int id=-1; read_nonnegative_id(ss,id,"pressure element");
            std::vector<int> conn; int n; while(ss>>n) conn.push_back(n);
            if(conn.empty()) throw std::runtime_error("Empty pressure element connectivity.");
            if(id>=static_cast<int>(mesh.pressure_elements.size())) mesh.pressure_elements.resize(id+1);
            mesh.pressure_elements[id]=reorder_quad_python_to_cpp(conn);
        } else if(section=="MIXED_ELEMENTS"){
            std::vector<std::string> parts;
            std::size_t start=0;
            while(true){
                const auto bar=line.find('|',start);
                parts.push_back(trim_copy(line.substr(start, bar==std::string::npos ? std::string::npos : bar-start)));
                if(bar==std::string::npos) break;
                start=bar+1;
            }
 
            const bool displacement_only_mixed =
                (parts.size()==1 && fields_line_seen &&
                 fields_declares_displacement &&
                 !fields_declares_velocity && !fields_declares_pressure);
 
            if(displacement_only_mixed){
                std::stringstream uss(parts[0]);
                int id=-1; read_nonnegative_id(uss,id,"displacement-only mixed element");
                std::vector<int> u_conn; int n;
                while(uss>>n) u_conn.push_back(n);
                if(u_conn.empty()) throw std::runtime_error("Displacement-only MIXED_ELEMENTS row has empty displacement connectivity.");
                if(id>=static_cast<int>(mesh.displacement_elements.size())) mesh.displacement_elements.resize(id+1);
                mesh.displacement_elements[id]=reorder_quad_python_to_cpp(u_conn);
                fields_declares_displacement=true;
            } else {
                if(parts.size()!=2 && parts.size()!=3)
                    throw std::runtime_error("MIXED_ELEMENTS row must be 'id v... | p...', 'id v... | u... | p...', or for FIELDS displacement: 'id u...'.");
                std::stringstream vss(parts[0]);
                int id=-1; read_nonnegative_id(vss,id,"mixed element");
                std::vector<int> v_conn,u_conn,p_conn; int n;
                while(vss>>n) v_conn.push_back(n);
                if(parts.size()==3){
                    std::stringstream uss(parts[1]);
                    while(uss>>n) u_conn.push_back(n);
                    std::stringstream pss(parts[2]);
                    while(pss>>n) p_conn.push_back(n);
                } else {
                    std::stringstream pss(parts[1]);
                    while(pss>>n) p_conn.push_back(n);
                }
                if(v_conn.empty() || p_conn.empty()) throw std::runtime_error("Mixed element row has empty velocity or pressure connectivity.");
                if(parts.size()==3 && u_conn.empty()) throw std::runtime_error("Mixed Hamilton element row has empty displacement connectivity.");
                if(id>=static_cast<int>(mesh.velocity_elements.size())){
                    mesh.velocity_elements.resize(id+1);
                    mesh.pressure_elements.resize(id+1);
                    if(parts.size()==3) mesh.displacement_elements.resize(id+1);
                }
                mesh.velocity_elements[id]=reorder_quad_python_to_cpp(v_conn);
                if(parts.size()==3){
                    if(id>=static_cast<int>(mesh.displacement_elements.size())) mesh.displacement_elements.resize(id+1);
                    mesh.displacement_elements[id]=reorder_quad_python_to_cpp(u_conn);
                    fields_declares_displacement=true;
                }
                mesh.pressure_elements[id]=reorder_quad_python_to_cpp(p_conn);
            }
        } else {
            const std::string vg=velocity_boundary_name_from_section(section);
            const std::string ug=displacement_boundary_name_from_section(section);
            const std::string pg=pressure_boundary_name_from_section(section);
            if(!vg.empty() || !ug.empty() || !pg.empty()){
                BoundaryEntity ent; ss >> ent.element_id;
                if(ss.fail() || ent.element_id<0) throw std::runtime_error("Bad boundary entity id in field mesh.");
                int n; while(ss>>n) ent.node_ids.push_back(n);
                if(ent.node_ids.empty()) throw std::runtime_error("Boundary entity has no nodes in field mesh.");
                if(!vg.empty()) mesh.velocity_boundaries[vg].push_back(ent);
                else if(!ug.empty()) mesh.displacement_boundaries[ug].push_back(ent);
                else mesh.pressure_boundaries[pg].push_back(ent);
            } else {
                throw std::runtime_error("Unknown field mesh section: "+section);
            }
        }
    }
 
    if(!dim_set && detected_dim==0) throw std::runtime_error("No nodes found in field mesh.");
    if(detected_dim!=0) mesh.dim=detected_dim;
    if(mesh.dim!=2 && mesh.dim!=3) throw std::runtime_error("Field mesh dimension must be 2 or 3.");
 
    const bool has_velocity_block = !mesh.velocity_nodes.empty() || !mesh.velocity_elements.empty();
    const bool has_pressure_block = !mesh.pressure_nodes.empty() || !mesh.pressure_elements.empty();
    const bool displacement_only_mesh =
        fields_line_seen && fields_declares_displacement &&
        !fields_declares_velocity && !fields_declares_pressure &&
        !has_velocity_block && !has_pressure_block;
 
    if(!displacement_only_mesh){
        if(mesh.velocity_nodes.empty()) throw std::runtime_error("Field mesh has no velocity nodes.");
        if(mesh.pressure_nodes.empty()) throw std::runtime_error("Field mesh has no pressure nodes.");
        if(mesh.velocity_elements.empty() || mesh.pressure_elements.empty()) throw std::runtime_error("Field mesh needs velocity and pressure elements.");
        if(mesh.velocity_elements.size()!=mesh.pressure_elements.size()) throw std::runtime_error("Velocity/pressure element count mismatch.");
    }
 
    bool has_nonempty_displacement_elements=false;
    for(const auto& e:mesh.displacement_elements){
        if(!e.empty()){ has_nonempty_displacement_elements=true; break; }
    }
 
    // Default behavior:
    //   - v|p meshes run as classical field mesh.
    //   - v|u|p meshes, or FIELDS ... displacement ..., activate Hamilton kinematics.
    //   - orphan displacement nodes are ignored unless displacement was explicitly requested.
    if(!fields_declares_displacement && !has_nonempty_displacement_elements){
        mesh.displacement_nodes.clear();
        mesh.displacement_elements.clear();
        mesh.displacement_boundaries.clear();
        mesh.displacement_type = ElementType::Unknown;
    } else {
        if(mesh.displacement_nodes.empty() || !has_nonempty_displacement_elements)
            throw std::runtime_error("Displacement field is requested, but displacement nodes/elements are incomplete.");
        if(!displacement_only_mesh && mesh.displacement_elements.size()!=mesh.velocity_elements.size())
            throw std::runtime_error("Velocity/displacement element count mismatch.");
        for(const auto& e:mesh.displacement_elements)
            if(e.empty()) throw std::runtime_error("Empty displacement connectivity in Hamilton mixed element.");
    }
 
    if(vem_taylor_hood_mesh){
        constexpr double coordinate_tol2=1.0e-20;
        for(int e=0;e<static_cast<int>(mesh.velocity_elements.size());++e){
            const auto& ve=mesh.velocity_elements[e];
            const auto& pe=mesh.pressure_elements[e];
            if(ve.size()<6 || (ve.size()%2)!=0 || pe.size()!=ve.size()/2){
                throw std::runtime_error(
                    "Invalid VEM Taylor-Hood connectivity at element "+std::to_string(e)
                    +": expected alternating quadratic velocity slots and one pressure node per vertex."
                );
            }
            for(std::size_t a=0;a<pe.size();++a){
                const int vnode=ve[2*a];
                const int pnode=pe[a];
                if(vnode<0 || vnode>=static_cast<int>(mesh.velocity_nodes.size())
                   || pnode<0 || pnode>=static_cast<int>(mesh.pressure_nodes.size())){
                    throw std::runtime_error("Invalid VEM Taylor-Hood node id at element "+std::to_string(e)+".");
                }
                const double dx=mesh.velocity_nodes[vnode].x[0]-mesh.pressure_nodes[pnode].x[0];
                const double dy=mesh.velocity_nodes[vnode].x[1]-mesh.pressure_nodes[pnode].x[1];
                if(dx*dx+dy*dy>coordinate_tol2){
                    std::ostringstream os;
                    os << "Invalid VEM Taylor-Hood mesh at element " << e
                       << ", vertex " << a << ": velocity node " << vnode
                       << " and pressure node " << pnode
                       << " do not have coincident coordinates.";
                    throw std::runtime_error(os.str());
                }
            }
        }
    }
 
    // The Python exporter may keep the full global node table in each DOMAIN
    // while only a subset of those nodes is used by the domain's elements.
    // Compact each domain to the actually active nodes so the reported node/DOF
    // counts and the linear system contain only the needed field unknowns.
    compact_mixed_mesh_to_active_nodes(mesh);
 
    const bool allow_displacement_polygon = displacement_only_mesh && has_displacement_field(mesh);
    if(!mesh.velocity_elements.empty()) mesh.velocity_type = vem_taylor_hood_mesh
        ? ElementType::Polygon
        : infer_field_element_type(mesh.velocity_elements, mesh.dim, false);
    if(has_displacement_field(mesh)) mesh.displacement_type = infer_field_element_type(mesh.displacement_elements, mesh.dim, allow_displacement_polygon);
    if(!mesh.pressure_elements.empty()) mesh.pressure_type = vem_taylor_hood_mesh
        ? ElementType::Polygon
        : infer_field_element_type(mesh.pressure_elements, mesh.dim, false);
    if((!mesh.velocity_elements.empty() && mesh.velocity_type==ElementType::Unknown) ||
       (!mesh.pressure_elements.empty() && mesh.pressure_type==ElementType::Unknown) ||
       (has_displacement_field(mesh) && mesh.displacement_type==ElementType::Unknown)) throw std::runtime_error("Unsupported field mesh element type.");
 
    const bool velocity_polygonal = (mesh.velocity_type==ElementType::Polygon);
    for(const auto& e:mesh.velocity_elements){
        if(!velocity_polygonal && e.size()!=mesh.velocity_elements.front().size()) throw std::runtime_error("Mixed velocity element types are not supported.");
        if(velocity_polygonal && (e.size()<6 || (e.size()%2)!=0)) throw std::runtime_error("Quadratic velocity VEM connectivity must be [v0,m0,v1,m1,...].");
        for(int nid:e) if(nid<0 || nid>=static_cast<int>(mesh.velocity_nodes.size())) throw std::runtime_error("Invalid velocity node id in element connectivity.");
    }
    if(has_displacement_field(mesh)){
        const bool displacement_polygonal = (mesh.displacement_type==ElementType::Polygon);
        for(const auto& e:mesh.displacement_elements){
            if(!displacement_polygonal && e.size()!=mesh.displacement_elements.front().size()) throw std::runtime_error("Mixed displacement element types are not supported.");
            if(displacement_polygonal && e.size()<3) throw std::runtime_error("A polygonal VEM displacement element needs at least three vertices.");
            for(int nid:e) if(nid<0 || nid>=static_cast<int>(mesh.displacement_nodes.size())) throw std::runtime_error("Invalid displacement node id in element connectivity.");
        }
    }
    const bool pressure_polygonal = (mesh.pressure_type==ElementType::Polygon);
    for(const auto& e:mesh.pressure_elements){
        if(!pressure_polygonal && e.size()!=mesh.pressure_elements.front().size()) throw std::runtime_error("Mixed pressure element types are not supported.");
        if(pressure_polygonal && e.size()<3) throw std::runtime_error("Linear pressure VEM polygon needs at least three vertices.");
        for(int nid:e) if(nid<0 || nid>=static_cast<int>(mesh.pressure_nodes.size())) throw std::runtime_error("Invalid pressure node id in element connectivity.");
    }
    for(const auto& kv:mesh.velocity_boundaries){
        for(const auto& ent:kv.second){
            for(int nid:ent.node_ids) if(nid<0 || nid>=static_cast<int>(mesh.velocity_nodes.size())) throw std::runtime_error("Invalid velocity boundary node id in group: "+kv.first);
        }
    }
    for(const auto& kv:mesh.displacement_boundaries){
        for(const auto& ent:kv.second){
            for(int nid:ent.node_ids) if(nid<0 || nid>=static_cast<int>(mesh.displacement_nodes.size())) throw std::runtime_error("Invalid displacement boundary node id in group: "+kv.first);
        }
    }
    for(const auto& kv:mesh.pressure_boundaries){
        for(const auto& ent:kv.second){
            for(int nid:ent.node_ids) if(nid<0 || nid>=static_cast<int>(mesh.pressure_nodes.size())) throw std::runtime_error("Invalid pressure boundary node id in group: "+kv.first);
        }
    }
 
    return mesh;
}
