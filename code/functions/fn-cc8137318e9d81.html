static bool has_variable_element_arity(const std::vector<std::vector<int>>& elems){
    if(elems.empty()) return false;
    const std::size_t n0 = elems.front().size();
    for(const auto& e: elems){
        if(e.size()!=n0) return true;
    }
    return false;
}
