static double evaluate_theta_for_dt(const TimeSolverSettings& solver, double dt_step){
    if(!solver.theta_depends_on_dt) return solver.theta;
 
    std::string expr_eval = solver.theta_expression;
    std::string expr_low = lower_copy(expr_eval);
    std::size_t pos = 0;
    const std::string dt_str = std::to_string(dt_step);
    while((pos = expr_low.find("dt", pos)) != std::string::npos){
        expr_eval.replace(pos, 2, dt_str);
        expr_low.replace(pos, 2, dt_str);
        pos += dt_str.size();
    }
 
    const double theta = eval_scalar_function(make_scalar_function(expr_eval),0.0,0.0,0.0,0.0);
    if(theta < 0.0 || theta > 1.0){
        std::ostringstream os;
        os << "theta expression '" << solver.theta_expression
           << "' evaluated to " << theta
           << " for dt=" << dt_step << ", but theta must be in [0,1].";
        throw std::runtime_error(os.str());
    }
    return theta;
}
