static int recover_neumann_local_face_3d(
    const FieldInfo& field,
    int elem_id,
    const BoundaryEntity& ent
){
    if(elem_id<0 || elem_id>=static_cast<int>(field.elements.size())) return -1;
    const auto& conn = field.elements[elem_id];
    const auto faces = reference_faces_3d(field.element_type);
 
    std::vector<int> local_nodes;
    local_nodes.reserve(ent.node_ids.size());
    for(int node: ent.node_ids){
        auto it = std::find(conn.begin(), conn.end(), node);
        if(it==conn.end()) return -1;
        local_nodes.push_back(static_cast<int>(it-conn.begin()));
    }
 
    for(int face=0; face<static_cast<int>(faces.size()); ++face){
        bool ok=true;
        for(int ln: local_nodes){
            if(std::find(faces[face].begin(),faces[face].end(),ln)==faces[face].end()){
                ok=false; break;
            }
        }
        if(ok) return face;
    }
 
    int best=-1, best_count=-1;
    for(int face=0; face<static_cast<int>(faces.size()); ++face){
        int count=0;
        for(int ln: local_nodes){
            if(std::find(faces[face].begin(),faces[face].end(),ln)!=faces[face].end()) ++count;
        }
        if(count>best_count){ best_count=count; best=face; }
    }
    const int required = std::min<int>(3, static_cast<int>(local_nodes.size()));
    return (best>=0 && best_count>=required) ? best : -1;
}
