static bool boundary_entity_on_line_selector(
    const FieldInfo& field,
    const BoundaryEntity& ent,
    const BCSelector& selector,
    int dim
){
    const double tol2=selector.tol*selector.tol;
    if(ent.node_ids.empty()) return false;
    for(int node: ent.node_ids){
        if(node<0 || node>=static_cast<int>(field.nodes.size())) return false;
        if(point_segment_distance2_assembly(field.nodes[node].x,selector.p0,selector.p1,dim)>tol2) return false;
    }
    return true;
}
