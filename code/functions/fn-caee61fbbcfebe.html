static PetscErrorCode apply_petsc_matfree_block_jacobi_pc(PC pc, Vec x, Vec y){
    PetscMatFreeBlockJacobiContext* ctx=nullptr;
    PetscCall(PCShellGetContext(pc,(void**)&ctx));
    if(!ctx || !ctx->op || !ctx->op->apply_pc) SETERRQ(PETSC_COMM_SELF,PETSC_ERR_ARG_NULL,"Missing PETSc MatFree block Jacobi callback");
    PetscInt n=0;
    PetscCall(VecGetSize(x,&n));
    std::vector<double> xv(static_cast<std::size_t>(n),0.0), yv(static_cast<std::size_t>(n),0.0);
    const PetscScalar* xa=nullptr;
    PetscCall(VecGetArrayRead(x,&xa));
    for(PetscInt i=0;i<n;++i) xv[static_cast<std::size_t>(i)] = static_cast<double>(xa[i]);
    PetscCall(VecRestoreArrayRead(x,&xa));
    ctx->op->apply_pc(xv,yv);
    if(static_cast<PetscInt>(yv.size())!=n) SETERRQ(PETSC_COMM_SELF,PETSC_ERR_ARG_SIZ,"PETSc MatFree block Jacobi callback returned wrong vector size");
    PetscScalar* ya=nullptr;
    PetscCall(VecGetArray(y,&ya));
    for(PetscInt i=0;i<n;++i) ya[i] = yv[static_cast<std::size_t>(i)];
    PetscCall(VecRestoreArray(y,&ya));
    return 0;
}
