static bool vtk_has_plastic_history_for_cells(
    const MaterialHistoryStore* history,
    int cell_count
) {
    if (!history) return false;
    if (history->domains.empty()) return false;
    if (history_nvars_or_zero(history->domains.front().specs, "plasticity") < 5) return false;
    return static_cast<int>(history->domains.front().elements.size()) == cell_count;
}
