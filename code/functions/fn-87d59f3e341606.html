StepDecision StepControl::finish_step(
    const NewtonStepResult& result,
    const StepWindow& step
) const {
    StepDecision d;
    if(result.converged){
        d.accepted = true;
        d.next_dt = update_time_step_after_acceptance(solver_, step.dt, result.iterations);
        return d;
    }
 
    if(!solver_.adaptive_time_step){
        d.stop = true;
        d.next_dt = step.dt;
        return d;
    }
 
    d.retry = true;
    d.next_dt = shrink_time_step_after_rejection(solver_, step.dt);
    if(d.next_dt < solver_.dt_min){
        d.retry = false;
        d.stop = true;
    }
    return d;
}
