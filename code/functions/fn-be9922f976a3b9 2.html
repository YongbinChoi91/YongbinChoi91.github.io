inline void require_converged_macro_rve(
    const RveSolveResult& rve,
    int elem_id,
    int gp_index,
    const std::string& contribution_name,
    const RveSolveOptions& base_opt,
    const char* macro_model_name
) {
    const double rve_res = multiscale_vector_inf_norm(rve.residual);
    observe_multiscale_rve_diagnostics(rve_res, rve.iterations, rve.converged, &rve.history_diagnostics);
    if (!rve.converged) {
        std::ostringstream os;
        os << "RVE PBC solve did not converge at " << macro_model_name
           << " macro element " << elem_id
           << ", contribution " << gp_index << " (" << contribution_name << ")"
           << ": |R|_inf=" << rve_res
           << ", iterations=" << rve.iterations
           << ", max_iter=" << base_opt.max_iter;
        throw std::runtime_error(os.str());
    }
}
