static std::array<double,3> element_centroid_from_owner(
    const FieldMesh& mesh,
    const ResolvedBoundaryEdge& r
){
    std::array<double,3> c{{0.0,0.0,0.0}};
    const auto& conn=owner_connectivity(mesh,r);
    int m=0;
    for(int node:conn){
        if(node<0 || node>=static_cast<int>(mesh.velocity_nodes.size())) continue;
        for(int i=0;i<3;++i) c[i]+=mesh.velocity_nodes[node].x[i];
        ++m;
    }
    if(m>0) for(int i=0;i<3;++i) c[i]/=static_cast<double>(m);
    return c;
}
