static void print_generic_field_update_diagnostics(const Problem& problem, const Vector& dx){
    std::cout << "    update by field:\n";
    if(problem.shared_dofs){
        const DofLayout& g = get_cached_dof_layout(problem.domains.front().mesh);
        for(const auto& f: g.fields){
            const double l2=norm_l2_range(dx,f.offset,f.offset+f.ndofs);
            const double inf=norm_inf_range(dx,f.offset,f.offset+f.ndofs);
            std::cout << "      " << f.name << "_l2 = " << l2 << ", " << f.name << "_inf = " << inf << "\n";
        }
        return;
    }
    for(const auto& d: problem.domains){
        const DofLayout& g = get_cached_dof_layout(d.mesh);
        for(const auto& f: g.fields){
            const int begin=d.dof_offset + f.offset;
            const int end=begin + f.ndofs;
            const double l2=norm_l2_range(dx,begin,end);
            const double inf=norm_inf_range(dx,begin,end);
            std::cout << "      " << d.name << "." << f.name << "_l2 = " << l2 << ", " << d.name << "." << f.name << "_inf = " << inf << "\n";
        }
    }
}
