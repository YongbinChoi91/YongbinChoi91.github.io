static std::map<std::string,std::vector<BoundaryEntity>> remap_boundary_map(
    const std::map<std::string,std::vector<BoundaryEntity>>& in,
    const std::vector<int>& map
){
    std::map<std::string,std::vector<BoundaryEntity>> out;
    for(const auto& kv:in){
        auto& dst=out[kv.first];
        for(const auto& ent:kv.second){
            BoundaryEntity r;
            r.element_id=ent.element_id;
            bool keep=true;
            for(int old:ent.node_ids){
                if(old<0 || old>=static_cast<int>(map.size()) || map[old]<0){
                    keep=false;
                    break;
                }
                r.node_ids.push_back(map[old]);
            }
            if(keep) dst.push_back(std::move(r));
        }
    }
    return out;
}
