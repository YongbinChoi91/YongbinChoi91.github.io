ElementType infer_field_element_type(
    const std::vector<std::vector<int>>& elems,
    int dim,
    bool allow_polygon
){
    if(elems.empty()) return ElementType::Unknown;
    const ElementType first = element_type_from_node_count(static_cast<int>(elems.front().size()), dim);
    const bool variable = has_variable_element_arity(elems);
    if(allow_polygon && dim==2 && (variable || first==ElementType::Unknown)){
        for(const auto& e: elems){
            if(e.size()<3) return ElementType::Unknown;
        }
        return ElementType::Polygon;
    }
    return first;
}
