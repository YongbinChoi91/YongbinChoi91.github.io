std::vector<std::array<double,3>> compute_dN_dx(const ShapeData& s,const JacobianData& J){
    std::vector<std::array<double,3>> out(s.N.size(),{0.0,0.0,0.0});
    for(size_t a=0;a<s.N.size();++a){
        for(int i=0;i<J.dim;++i){
            for(int A=0;A<J.dim;++A) out[a][i] += J.J_inv[i][A] * s.dN_dxi[a][A];
        }
    }
    return out;
}
