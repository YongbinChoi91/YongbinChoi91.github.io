static std::vector<AD2> shape_values_Q4_ad2(const AD2& xi,const AD2& eta,const AD2&) {
    std::vector<AD2> N(4);
    N[0] = 0.25*(1.0-xi)*(1.0-eta);
    N[1] = 0.25*(1.0+xi)*(1.0-eta);
    N[2] = 0.25*(1.0+xi)*(1.0+eta);
    N[3] = 0.25*(1.0-xi)*(1.0+eta);
    return N;
}
