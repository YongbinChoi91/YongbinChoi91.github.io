static bool pressure_pcd_cache_matches(
    const PressurePcdCache& cache,
    const std::vector<int>& free_dofs,
    const FieldMesh& mesh,
    const ProblemParameters& params,
    const LinearSolverSettings& settings,
    double dt_for_schur
){
    return cache.valid &&
           cache.mesh == &mesh &&
           cache.free_dofs == free_dofs &&
           cache.dt == dt_for_schur &&
           cache.theta == params.solver.theta &&
           cache.rho == params.material.rho &&
           cache.mu == params.material.mu &&
           cache.kp_mass_shift == settings.pcd_kp_mass_shift &&
           cache.ap_mass_weight == settings.pcd_ap_mass_weight &&
           cache.ap_laplacian_weight == settings.pcd_ap_laplacian_weight &&
           cache.schur_pressure_shift == settings.schur_pressure_shift &&
           cache.stabilization_time_order == params.solver.stabilization_time_order &&
           cache.gauss_order == params.solver.gauss_order;
}
