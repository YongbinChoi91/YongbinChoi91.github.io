void append_timing_csv(const std::string& filename, const StepTimingRow& row){
    ensure_output_parent_directory(filename);
    std::ofstream out(filename, std::ios::app);
    if(!out) throw std::runtime_error("Could not append timing CSV: " + filename);
    out << row.step << "," << std::setprecision(17)
        << row.time << "," << row.dt << "," << row.newton_iterations << ","
        << row.assembly_time << "," << row.linear_solver_time << ","
        << row.postprocess_time << "," << row.step_wall_time << "\n";
}
