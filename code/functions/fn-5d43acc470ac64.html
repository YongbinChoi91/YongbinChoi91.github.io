static bool edge_coordinates_match_boundary(
    const FieldMesh& mesh,
    const std::vector<int>& elem_conn,
    const std::vector<int>& local_edge,
    const BoundaryEntity& ent,
    double tol2
){
    // Compare as unordered coordinate sets.  This is needed for shared multi-domain
    // exports where boundary rows may use a shared/global boundary-node id
    // convention while a domain element uses a different node id with the same
    // physical coordinate.  The Python postprocess effectively works with the
    // recovered owner surface, so coordinate matching is the safest fallback.
    if(ent.node_ids.size() > local_edge.size()) return false;
    std::vector<char> used(local_edge.size(),0);
    for(int bnode:ent.node_ids){
        if(bnode<0 || bnode>=static_cast<int>(mesh.velocity_nodes.size())) return false;
        const auto& Xb=mesh.velocity_nodes[bnode].x;
        int best=-1;
        double best_d2=std::numeric_limits<double>::infinity();
        for(int k=0;k<static_cast<int>(local_edge.size());++k){
            if(used[k]) continue;
            const int local=local_edge[k];
            if(local<0 || local>=static_cast<int>(elem_conn.size())) continue;
            const int enode=elem_conn[local];
            if(enode<0 || enode>=static_cast<int>(mesh.velocity_nodes.size())) continue;
            const double d2=coord_dist2(Xb,mesh.velocity_nodes[enode].x);
            if(d2<best_d2){ best_d2=d2; best=k; }
        }
        if(best<0 || best_d2>tol2) return false;
        used[best]=1;
    }
    return true;
}
