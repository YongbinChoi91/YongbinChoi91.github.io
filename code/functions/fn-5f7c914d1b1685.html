ProblemParameters read_problem_parameters(const std::string& material_file,
                                      const std::string& boundary_file,
                                      const std::string& solver_file){
    ProblemParameters p;
 
    // ------------------------------------------------------------------------
    // Material.txt
    // ------------------------------------------------------------------------
    {
        std::ifstream file(material_file);
        if(!file) throw std::runtime_error("Cannot open material file: "+material_file);
        struct DoubleKey { std::initializer_list<const char*> aliases; double* value; };
        const DoubleKey double_keys[] = {
            {{"rho","density"}, &p.material.rho},
            {{"mu","viscosity"}, &p.material.mu},
            {{"u_bar","inlet_u_bar","mean_velocity"}, &p.material.u_bar},
            {{"structure_rho","structure_density","solid_density"}, &p.material.structure_rho},
            {{"newmark_beta","beta_newmark"}, &p.material.newmark_beta},
            {{"newmark_gamma","gamma_newmark"}, &p.material.newmark_gamma},
            {{"e","youngs_modulus","young_modulus","structure_youngs_modulus"}, &p.material.youngs_modulus},
            {{"nu","poisson_ratio","structure_poisson_ratio"}, &p.material.poisson_ratio},
            {{"sigma_y","yield_stress","sigma_y0"}, &p.material.sigma_y},
            {{"kf","hardening_kf","hardening_modulus","linear_hardening"}, &p.material.Kf},
            {{"yinf","hardening_yinf","yield_inf","saturation_yield"}, &p.material.Yinf},
            {{"delta","hardening_delta"}, &p.material.delta},
            {{"visco_elastic_eta","viscoelasticity"}, &p.material.visco_elastic_eta},
            {{"local_tol","local_residual_tol","q_tol"}, &p.material.local_tol},
            {{"mesh_motion_coefficient","mesh_motion_coeff","mmpde_coefficient","a_u"}, &p.material.mesh_motion_coefficient},
            {{"mesh_motion_poisson_ratio","mmpde_poisson_ratio"}, &p.material.mesh_motion_poisson_ratio},
            {{"stabilization_parameter","vem_stabilization_parameter","dofi_stabilization_parameter","dofi_factor"}, &p.material.stabilization_parameter}
        };
        struct IntKey { std::initializer_list<const char*> aliases; int* value; };
        const IntKey int_keys[] = {
            {{"local_max_iter","q_max_iter"}, &p.material.local_max_iter}
        };
 
        std::string key; std::stringstream ss;
        while(next_config_line(file,key,ss)){
            key = lower_copy(key);
            bool handled=false;
            for(const auto& item:double_keys){
                if(key_in(key,item.aliases)){ ss >> *item.value; handled=true; break; }
            }
            if(handled) continue;
            for(const auto& item:int_keys){
                if(key_in(key,item.aliases)){ ss >> *item.value; handled=true; break; }
            }
            if(handled) continue;
            if(key=="mesh_motion_type" || key=="mmpde_type" || key=="mmpde"){
                ss >> p.material.mesh_motion_type;
                std::transform(p.material.mesh_motion_type.begin(), p.material.mesh_motion_type.end(),
                               p.material.mesh_motion_type.begin(),
                               [](unsigned char c){ return static_cast<char>(std::toupper(c)); });
            }
            else throw std::runtime_error("Unknown key in material file: "+key);
        }
    }
    // ------------------------------------------------------------------------
    // BC.txt
    // ------------------------------------------------------------------------
    {
        std::ifstream file(boundary_file);
        if(!file) throw std::runtime_error("Cannot open boundary file: "+boundary_file);
        std::string key; std::stringstream ss;
        while(next_config_line(file,key,ss)){
            if(key=="VELOCITY_DIRICHLET" || key=="DIRICHLET_V"){
                std::string g,c; ss>>g>>c; std::string expr=rest_trim(ss);
                if(expr.empty()) throw std::runtime_error("Bad velocity DIRICHLET line.");
                FieldComponentTarget target{"velocity",g,parse_bc_selector_token(g),component_index_from_string(c)};
                append_dirichlet_rule(p,target,make_scalar_function(expr));
            } else if(key=="DISPLACEMENT" || key=="DISPLACEMENT_DIRICHLET" || key=="DIRICHLET_U"){
                std::string g,c; ss>>g>>c; std::string expr=rest_trim(ss);
                if(expr.empty()) throw std::runtime_error("Bad displacement DIRICHLET line.");
                FieldComponentTarget target{"displacement",g,parse_bc_selector_token(g),component_index_from_string(c)};
                append_dirichlet_rule(p,target,make_scalar_function(expr));
            } else if(key=="DIRICHLET" || key=="FIELD_DIRICHLET" || key=="DBC"){
                // Accepted forms:
                //   DIRICHLET TOP uy expr                  (alias shorthand)
                //   DIRICHLET displacement TOP y expr      (explicit generic)
                //   DBC temperature WALL 0 expr
                std::string a,b,c; ss>>a>>b;
                if(a.empty() || b.empty()) throw std::runtime_error("Bad DIRICHLET line.");
                bool explicit_field=false;
                FieldComponentTarget target;
                std::string expr;
                if(is_known_field_name(a)){
                    ss >> c;
                    if(c.empty()) throw std::runtime_error("Bad generic DIRICHLET line. Use: DIRICHLET field group component expression.");
                    target = parse_field_component_target_tokens(a,b,c,explicit_field);
                    expr = rest_trim(ss);
                } else {
                    target = target_from_alias(a,b);
                    expr = rest_trim(ss);
                }
                if(expr.empty()) throw std::runtime_error("Bad DIRICHLET line: missing expression.");
                append_dirichlet_rule(p,target,make_scalar_function(expr));
            } else if(key=="TORSION_DIRICHLET" || key=="DIRICHLET_TORSION" || key=="TORSION_DBC"){
                // Accepted forms:
                //   TORSION_DIRICHLET displacement RIGHT z auto angle_deg 90*(t/4.0)
                //   TORSION_DIRICHLET displacement RIGHT z center 0 0 0 angle_rad pi/2
                //   TORSION_DIRICHLET RIGHT z auto angle_deg 90
                std::string a,b,c; ss >> a >> b >> c;
                if(a.empty() || b.empty() || c.empty()) {
                    throw std::runtime_error("Bad TORSION_DIRICHLET line. Use: TORSION_DIRICHLET [field] group axis auto|center ... angle_deg|angle_rad expression.");
                }
 
                TorsionDirichletBC rule;
                const std::string al = lower_copy(a);
                if(is_known_field_name(al)){
                    rule.field = (al=="theta") ? std::string("temperature") : al;
                    rule.group = b;
                    rule.selector = parse_bc_selector_token(b);
                    rule.axis = torsion_axis_index_from_string(c);
                    ss >> a; // center mode
                } else {
                    rule.field = "displacement";
                    rule.group = a;
                    rule.selector = parse_bc_selector_token(a);
                    rule.axis = torsion_axis_index_from_string(b);
                    a = c; // center mode
                }
 
                const std::string center_mode = lower_copy(a);
                if(center_mode.empty()) {
                    throw std::runtime_error("Bad TORSION_DIRICHLET line: missing auto|center selector.");
                }
                if(center_mode=="auto" || center_mode=="axis_auto" || center_mode=="centroid"){
                    rule.auto_center = true;
                } else if(center_mode=="center" || center_mode=="centre"){
                    rule.auto_center = false;
                    if(!(ss >> rule.center[0] >> rule.center[1] >> rule.center[2])) {
                        throw std::runtime_error("Bad TORSION_DIRICHLET center. Use: center cx cy cz.");
                    }
                } else {
                    throw std::runtime_error("Bad TORSION_DIRICHLET center mode. Use auto or center cx cy cz.");
                }
 
                std::string angle_unit; ss >> angle_unit;
                const std::string angle_expr = rest_trim(ss);
                rule.angle = make_torsion_angle_function(angle_unit, angle_expr);
                p.torsion_dirichlet_bcs.push_back(rule);
            } else if(key=="PRESSURE" || key=="PRESSURE_DIRICHLET" || key=="DIRICHLET_P"){
                std::string g; ss>>g; std::string expr=rest_trim(ss);
                if(expr.empty()) throw std::runtime_error("Bad pressure DIRICHLET line.");
                FieldComponentTarget target{"pressure",g,parse_bc_selector_token(g),0};
                append_dirichlet_rule(p,target,make_scalar_function(expr));
            } else if(key=="NEUMANN" || key=="FIELD_NEUMANN" || key=="TRACTION" || key=="FLUX"){
                // Full generic form:
                //   NEUMANN field group component expression
                //   NEUMANN displacement RIGHT x 1.0
                // Shorthand for solid traction:
                //   TRACTION group component expression
                //   TRACTION RIGHT ux 1.0
                std::string a,b,c; ss>>a>>b>>c;
                if(a.empty() || b.empty() || c.empty()) throw std::runtime_error("Bad NEUMANN line. Use: NEUMANN field group component expression.");
 
                const std::string al=lower_copy(a);
                const bool explicit_field = (al=="velocity" || al=="displacement" || al=="pressure" || al=="temperature");
 
                std::string field = explicit_field ? lower_copy(a) : std::string("displacement");
                std::string group = explicit_field ? b : a;
                std::string comp  = explicit_field ? c : b;
 
                std::string expr;
                if(explicit_field){
                    expr = rest_trim(ss);
                } else {
                    expr = c;
                    std::string tail;
                    std::getline(ss, tail);
                    tail = trim_copy(tail);
                    if(!tail.empty()) expr += " " + tail;
                }
 
                if(expr.empty()) throw std::runtime_error("Bad NEUMANN line: missing expression.");
                p.neumann_bcs.push_back({field,group,parse_bc_selector_token(group),component_index_from_string(comp),make_scalar_function(expr)});
            } else {
                throw std::runtime_error("Unknown key in boundary file: "+key);
            }
        }
    }
    // ------------------------------------------------------------------------
    // Solver.txt
    // ------------------------------------------------------------------------
    {
        std::ifstream file(solver_file);
        if(!file) throw std::runtime_error("Cannot open solver file: "+solver_file);
        struct SolverDoubleKey { std::initializer_list<const char*> aliases; double* value; };
        const SolverDoubleKey double_keys[] = {
            {{"dt","time_step"}, &p.solver.dt},
            {{"total_time","end_time","t_end"}, &p.solver.total_time},
            {{"dt_min","min_dt"}, &p.solver.dt_min},
            {{"dt_max","max_dt"}, &p.solver.dt_max},
            {{"dt_growth"}, &p.solver.dt_growth},
            {{"dt_shrink"}, &p.solver.dt_shrink},
            {{"tol"}, &p.solver.tol},
            {{"alpha_first_iter","newton_alpha_first"}, &p.solver.alpha_first_iter},
            {{"alpha_other_iter","newton_alpha"}, &p.solver.alpha_other_iter},
            {{"substep_min","min_load_increment"}, &p.solver.substep_min},
            {{"substep_max","max_load_increment"}, &p.solver.substep_max},
            {{"substep_growth","load_growth"}, &p.solver.substep_growth},
            {{"substep_shrink","load_shrink"}, &p.solver.substep_shrink},
            {{"rve_initial_substep"}, &p.solver.rve_initial_substep},
            {{"rve_min_substep"}, &p.solver.rve_min_substep},
            {{"rve_cutback_factor"}, &p.solver.rve_cutback_factor},
            {{"rve_growth_factor"}, &p.solver.rve_growth_factor},
            {{"rve_tol"}, &p.solver.rve_tol},
            {{"rve_pair_tol"}, &p.solver.rve_pair_tol},
            {{"pressure_pin_value"}, &p.solver.pressure_pin_value},
            {{"stabilization_ck","c_k"}, &p.solver.stabilization_Ck},
            {{"stabilization_time_order","time_discretized_order"}, &p.solver.stabilization_time_order},
            {{"divergence_factor","newton_divergence_factor"}, &p.solver.divergence_factor},
            {{"bc_conflict_tol"}, &p.solver.bc_conflict_tol},
            {{"force_boundary_check_tol","boundary_integral_tol"}, &p.solver.force_boundary_check_tol},
            {{"force_attack_angle_deg","attack_angle","aoa"}, &p.solver.force_attack_angle_deg},
            {{"linear_tol"}, &p.solver.linear.tol},
            {{"linear_atol"}, &p.solver.linear.atol},
            {{"schur_pressure_mass_weight"}, &p.solver.linear.schur_pressure_mass_weight},
            {{"schur_pressure_laplacian_weight"}, &p.solver.linear.schur_pressure_laplacian_weight},
            {{"schur_pressure_shift"}, &p.solver.linear.schur_pressure_shift},
            {{"pcd_kp_mass_shift"}, &p.solver.linear.pcd_kp_mass_shift},
            {{"pcd_ap_mass_weight"}, &p.solver.linear.pcd_ap_mass_weight},
            {{"pcd_ap_laplacian_weight"}, &p.solver.linear.pcd_ap_laplacian_weight},
            {{"pcd_pressure_rtol"}, &p.solver.linear.pcd_pressure_rtol},
            {{"matfree_jvp_check_tol"}, &p.solver.linear.matfree_jvp_check_tol}
        };
        struct SolverIntKey { std::initializer_list<const char*> aliases; int* value; };
        const SolverIntKey int_keys[] = {
            {{"max_iter"}, &p.solver.max_iter},
            {{"max_substeps","max_load_steps"}, &p.solver.max_substeps},
            {{"pressure_pin_node"}, &p.solver.pressure_pin_node},
            {{"residual_monitor"}, &p.solver.residual_monitor},
            {{"residual_print_limit"}, &p.solver.residual_print_limit},
            {{"stabilization_element_order","elements_order"}, &p.solver.stabilization_element_order},
            {{"multiscale_macro_threads"}, &p.solver.multiscale_macro_threads},
            {{"multiscale_rve_threads"}, &p.solver.multiscale_rve_threads},
            {{"rve_max_iter"}, &p.solver.rve_max_iter},
            {{"rve_max_substeps"}, &p.solver.rve_max_substeps},
            {{"output_every"}, &p.solver.output_every},
            {{"linear_max_iter"}, &p.solver.linear.max_iter},
            {{"gmres_restart"}, &p.solver.linear.gmres_restart},
            {{"ilu_levels"}, &p.solver.linear.ilu_levels},
            {{"pcd_pressure_max_iter"}, &p.solver.linear.pcd_pressure_max_iter},
            {{"pcd_kp_ilu_levels"}, &p.solver.linear.pcd_kp_ilu_levels},
            {{"pcd_ap_ilu_levels"}, &p.solver.linear.pcd_ap_ilu_levels},
            {{"velocity_ilu_levels"}, &p.solver.linear.velocity_ilu_levels},
            {{"mechanical_ilu_levels"}, &p.solver.linear.mechanical_ilu_levels},
            {{"pressure_ilu_levels"}, &p.solver.linear.pressure_ilu_levels},
            {{"matfree_jvp_check_every"}, &p.solver.linear.matfree_jvp_check_every}
        };
        struct SolverBoolKey { std::initializer_list<const char*> aliases; bool* value; };
        const SolverBoolKey bool_keys[] = {
            {{"adaptive_time_step","adaptive_dt"}, &p.solver.adaptive_time_step},
            {{"stabilization","supg"}, &p.solver.stabilization},
            {{"bc_diagnostics"}, &p.solver.bc_diagnostics},
            {{"newton_log"}, &p.solver.newton_log},
            {{"field_diagnostics"}, &p.solver.field_diagnostics},

            {{"rve_adaptive_substep","rve_adaptive_substepping"}, &p.solver.rve_adaptive_substep},
            {{"timing_diagnostics"}, &p.solver.timing_diagnostics},
            {{"timing_per_step"}, &p.solver.timing_per_step},
            {{"timing_csv"}, &p.solver.timing_csv},
            {{"timing_summary"}, &p.solver.timing_summary},
            {{"postprocess_deferred","deferred_postprocess","defer_postprocess"}, &p.solver.postprocess_deferred},
            {{"force_output","drag_lift_output"}, &p.solver.force_output},
            {{"linear_verbose"}, &p.solver.linear.verbose},
            {{"petsc_matfree_full"}, &p.solver.linear.petsc_matfree_full},
            {{"matfree_jvp_check"}, &p.solver.linear.matfree_jvp_check}
        };
        struct SolverStringKey { std::initializer_list<const char*> aliases; std::string* value; };
        const SolverStringKey string_keys[] = {
            {{"output_prefix"}, &p.solver.output_prefix},
            {{"force_boundary","drag_lift_boundary"}, &p.solver.force_boundary},
            {{"solver_backend"}, &p.solver.linear.backend},
            {{"linear_solver"}, &p.solver.linear.linear_solver},
            {{"preconditioner"}, &p.solver.linear.preconditioner},
            {{"block_layout"}, &p.solver.linear.block_layout},
            {{"fieldsplit_type"}, &p.solver.linear.fieldsplit_type},
            {{"schur_factorization"}, &p.solver.linear.schur_factorization},
            {{"schur_precondition","schur_approx"}, &p.solver.linear.schur_precondition},
            {{"velocity_pc"}, &p.solver.linear.velocity_pc},
            {{"mechanical_pc"}, &p.solver.linear.mechanical_pc},
            {{"pressure_pc"}, &p.solver.linear.pressure_pc},
            {{"sub_linear_solver"}, &p.solver.linear.sub_linear_solver},
            {{"pcd_pressure_solver"}, &p.solver.linear.pcd_pressure_solver},
            {{"pcd_kp_solver"}, &p.solver.linear.pcd_kp_solver},
            {{"pcd_kp_pc"}, &p.solver.linear.pcd_kp_pc},
            {{"pcd_ap_solver"}, &p.solver.linear.pcd_ap_solver},
            {{"pcd_ap_pc"}, &p.solver.linear.pcd_ap_pc},
            {{"petsc_matfree_jvp_mode"}, &p.solver.linear.petsc_matfree_jvp_mode},
            {{"petsc_matfree_pc"}, &p.solver.linear.petsc_matfree_pc}
        };
 
        std::string key; std::stringstream ss;
        while(next_config_line(file,key,ss)){
            key = lower_copy(key);
            bool handled=false;
            for(const auto& item:double_keys){
                if(key_in(key,item.aliases)){ ss >> *item.value; handled=true; break; }
            }
            if(handled) continue;
            for(const auto& item:int_keys){
                if(key_in(key,item.aliases)){ ss >> *item.value; handled=true; break; }
            }
            if(handled) continue;
            for(const auto& item:bool_keys){
                if(key_in(key,item.aliases)){ *item.value = read_on_off_bool(ss,key); handled=true; break; }
            }
            if(handled) continue;
            for(const auto& item:string_keys){
                if(key_in(key,item.aliases)){ ss >> *item.value; handled=true; break; }
            }
            if(handled) continue;
 
            if(key=="theta"){
                std::string expr = rest_trim(ss);
                if(expr.empty()) throw std::runtime_error("Bad theta line.");
 
                // Store the original expression.  If it contains dt, the solver
                // must re-evaluate it on every adaptive time-step attempt using
                // the current dt_step.  This matches the Python shifted-CN usage
                // theta = 0.5 + current_dt.
                p.solver.theta_expression = expr;
                p.solver.theta_depends_on_dt = (lower_copy(expr).find("dt") != std::string::npos);
 
                // Evaluate once here with the target input dt for validation and
                // for non-adaptive/static uses.  The time solver overwrites this
                // value per attempt when theta_depends_on_dt is true.
                std::string expr_eval = expr;
                std::string expr_low = lower_copy(expr_eval);
                std::size_t pos = 0;
                const std::string dt_str = std::to_string(p.solver.dt);
                while((pos = expr_low.find("dt", pos)) != std::string::npos){
                    expr_eval.replace(pos, 2, dt_str);
                    expr_low.replace(pos, 2, dt_str);
                    pos += dt_str.size();
                }
                p.solver.theta = eval_scalar_function(make_scalar_function(expr_eval),0.0,0.0,0.0,0.0);
            }
            else if(key=="residual_print_all"){
                p.solver.residual_print_all = read_on_off_bool(ss,key) ? 1 : 0;
            }
            else if(key=="gauss_order" || key=="quadrature_order"){
                std::string v; ss >> v;
                if(v.empty()) throw std::runtime_error("Bad GAUSS_ORDER line.");
                const std::string lv = lower_copy(v);
                if(lv=="auto" || lv=="default") p.solver.gauss_order = -1;
                else {
                    p.solver.gauss_order = std::stoi(v);
                    if(p.solver.gauss_order <= 0) throw std::runtime_error("GAUSS_ORDER must be positive or auto.");
                }
            }
            else if(key=="multiscale_parallel_level"){
                ss >> p.solver.multiscale_parallel_level;
                p.solver.multiscale_parallel_level = lower_copy(p.solver.multiscale_parallel_level);
                if(p.solver.multiscale_parallel_level!="macro" &&
                   p.solver.multiscale_parallel_level!="rve" &&
                   p.solver.multiscale_parallel_level!="hybrid"){
                    throw std::runtime_error("multiscale_parallel_level must be macro, rve, or hybrid.");
                }
            }
            else if(key=="pcd_inner_pc"){
                ss>>p.solver.linear.pcd_inner_pc;
                p.solver.linear.pcd_kp_pc = p.solver.linear.pcd_inner_pc;
                p.solver.linear.pcd_ap_pc = p.solver.linear.pcd_inner_pc;
            }
            else if(key=="pcd_inner_solver"){
                ss>>p.solver.linear.pcd_inner_solver;
                p.solver.linear.pcd_kp_solver = p.solver.linear.pcd_inner_solver;
                p.solver.linear.pcd_ap_solver = p.solver.linear.pcd_inner_solver;
            }
            else throw std::runtime_error("Unknown key in solver file: "+key);
        }
    }
 
    // ------------------------------------------------------------------------
    // Cross-file validation and defaults
    // ------------------------------------------------------------------------
    if(p.material.rho<=0.0) throw std::runtime_error("rho must be positive.");
    if(p.material.mu<0.0) throw std::runtime_error("mu must be non-negative.");
    if(p.material.structure_rho<=0.0) throw std::runtime_error("structure_rho must be positive.");
    if(p.material.youngs_modulus<0.0) throw std::runtime_error("youngs_modulus must be non-negative.");
    if(p.material.local_max_iter<=0) throw std::runtime_error("ep_local_max_iter must be positive.");
    if(p.material.local_tol<=0.0) throw std::runtime_error("ep_local_tol must be positive.");
    if(p.material.poisson_ratio<=-1.0 || p.material.poisson_ratio>=0.5) throw std::runtime_error("poisson_ratio must be in (-1,0.5).");
    if(p.material.mesh_motion_coefficient<0.0) throw std::runtime_error("mesh_motion_coefficient must be non-negative.");
    if(p.material.mesh_motion_poisson_ratio<=-1.0 || p.material.mesh_motion_poisson_ratio>=0.5) throw std::runtime_error("mesh_motion_poisson_ratio must be in (-1,0.5).");
    if(p.material.stabilization_parameter<0.0) throw std::runtime_error("stabilization_parameter must be non-negative.");
    {
        std::transform(p.material.mesh_motion_type.begin(), p.material.mesh_motion_type.end(),
                       p.material.mesh_motion_type.begin(),
                       [](unsigned char c){ return static_cast<char>(std::toupper(c)); });
        if(p.material.mesh_motion_type != "BH" && p.material.mesh_motion_type != "HE")
            throw std::runtime_error("mesh_motion_type must be BH or HE.");
    }
    if(p.solver.theta<0.0 || p.solver.theta>1.0) throw std::runtime_error("theta must be in [0,1].");
    if(p.solver.dt<=0.0) throw std::runtime_error("dt must be positive.");
    if(p.solver.total_time<=0.0) throw std::runtime_error("total_time must be positive.");
    if(p.solver.dt_max<=0.0) p.solver.dt_max = p.solver.dt;
 
    return p;
}
