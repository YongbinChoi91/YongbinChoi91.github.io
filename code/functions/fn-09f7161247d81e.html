static std::vector<int> make_vtk_visualization_cell(
    const std::vector<int>& conn,
    const std::vector<Node>& nodes,
    ElementType type,
    int dim
){
    std::vector<int> cell=conn;
 
    // Keep the solver/space connectivity order untouched.  This conversion is
    // only for ParaView/VTK output: VTK_QUADRATIC_HEXAHEDRON expects the H20
    // midside nodes in a different order from the internal HiPeCs H20 basis.
    //
    // Internal H20 order used by reference_node_coordinate()/compute_shape():
    //   0..7   corners
    //   8..11  bottom-face edge midsides, zeta = -1
    //   12..15 vertical edge midsides
    //   16..19 top-face edge midsides, zeta = +1
    //
    // VTK_QUADRATIC_HEXAHEDRON order:
    //   0..7   corners
    //   8..11  bottom-face edge midsides
    //   12..15 top-face edge midsides
    //   16..19 vertical edge midsides
    if(type==ElementType::H20 && dim==3 && conn.size()==20){
        cell = {
            conn[0],  conn[1],  conn[2],  conn[3],
            conn[4],  conn[5],  conn[6],  conn[7],
            conn[8],  conn[9],  conn[10], conn[11],
            conn[16], conn[17], conn[18], conn[19],
            conn[12], conn[13], conn[14], conn[15]
        };
    }
    if(type==ElementType::H27 && dim==3 && conn.size()==27){
        cell = {
            conn[0],  conn[1],  conn[2],  conn[3],
            conn[4],  conn[5],  conn[6],  conn[7],
            conn[8],  conn[9],  conn[10], conn[11],
            conn[16], conn[17], conn[18], conn[19],
            conn[12], conn[13], conn[14], conn[15],
            conn[20], conn[25], conn[23], conn[22], conn[24], conn[21],
            conn[26]
        };
    }
 
    std::vector<int> cleaned;
    cleaned.reserve(cell.size());
    for(int node: cell){
        if(node<0 || node>=static_cast<int>(nodes.size())) continue;
        bool duplicate=false;
        for(int kept: cleaned){
            if(kept==node){
                duplicate=true;
                break;
            }
        }
        if(!duplicate) cleaned.push_back(node);
    }
 
    return cleaned;
}
