BoundaryForceResult append_force_sample(
    const Problem& problem,
    const Vector& solution,
    const ProblemParameters& params,
    BoundaryForcePostprocessState& state,
    double time
){
    if(!state.enabled){
        throw std::runtime_error("append_force_sample called while boundary force postprocess is disabled.");
    }
    const BoundaryForceResult force = compute_multidomain_boundary_force_integral(
        problem,solution,params,state,time,params.solver.force_attack_angle_deg
    );
    if(state.csv_enabled) append_force_csv(state.force_csv,force);
    if(state.hdf5_enabled) append_hdf5_table_row(params,"/history/force",{force.time,force.Drag,force.Lift});
    return force;
}
